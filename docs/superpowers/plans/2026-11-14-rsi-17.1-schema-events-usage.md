```go
// call_scope 提供每次 LLM 调用的局部身份作用域。
// Python: openjiuwen/core/foundation/llm/call_scope.py
// 三个 contextvar → 三个 context 键；ContextVar token 还原 → done 闭包还原旧值。
package llm

import (
	"context"
	"crypto/rand"
	"encoding/hex"
)

// ctxKey context 键类型（非导出，防外部伪造）
type ctxKey int

const (
	ctxKeyCallID ctxKey = iota // 当前 LLM 调用 ID
	ctxKeyUnifiedCompletion    // 是否期望统一收尾事件
	ctxKeyObservationSuppressed // 观测抑制标志
)

// GetCurrentLLMCallID 返回当前作用域绑定的调用 ID；无作用域返回空串。
func GetCurrentLLMCallID(ctx context.Context) string {
	if v, ok := ctx.Value(ctxKeyCallID).(string); ok {
		return v
	}
	return ""
}

// ExpectsUnifiedLLMCompletion 返回外层 Model 生命周期是否将发出统一收尾事件。
func ExpectsUnifiedLLMCompletion(ctx context.Context) bool {
	if v, ok := ctx.Value(ctxKeyUnifiedCompletion).(bool); ok {
		return v
	}
	return false
}

// IsLLMObservationSuppressed 返回当前是否抑制 LLM 观测。
func IsLLMObservationSuppressed(ctx context.Context) bool {
	if v, ok := ctx.Value(ctxKeyObservationSuppressed).(bool); ok {
		return v
	}
	return false
}

// EnterCallScope 进入新调用作用域：生成 uuid4 风格随机 ID（Python: uuid.uuid4().hex）。
// 返回新 ctx 与退出还原闭包（对齐 __exit__ 的 token 还原语义）。
func EnterCallScope(ctx context.Context, unifiedCompletion bool) (context.Context, func()) {
	var b [16]byte
	_, _ = rand.Read(b[:]) // crypto/rand 失败仅影响唯一性，ID 仍非空可用
	b[6] = (b[6] & 0x0f) | 0x40 // uuid4 版本位
	b[8] = (b[8] & 0x3f) | 0x80 // uuid4 变体位
	return EnterCallScopeWithID(ctx, hex.EncodeToString(b[:]), unifiedCompletion)
}

// EnterCallScopeWithID 以显式 ID 进入作用域（对齐 LlmCallScope(call_id=...)）。
func EnterCallScopeWithID(ctx context.Context, callID string, unifiedCompletion bool) (context.Context, func()) {
	prevID, _ := ctx.Value(ctxKeyCallID).(string)
	prevUnified, _ := ctx.Value(ctxKeyUnifiedCompletion).(bool)
	next := context.WithValue(
		context.WithValue(ctx, ctxKeyCallID, callID),
		ctxKeyUnifiedCompletion, unifiedCompletion,
	)
	return next, func() {
		// 闭包无法改父 ctx：还原通过返回旧值对实现（见还原版 RestoreCallScope）
		_ = prevID
		_ = prevUnified
	}
}

// SuppressLLMObservation 进入观测抑制，返回新 ctx 与还原闭包。
// Python: LlmObservationSuppression.__enter__/__exit__
func SuppressLLMObservation(ctx context.Context) (context.Context, func()) {
	next := context.WithValue(ctx, ctxKeyObservationSuppressed, true)
	return next, func() { _ = ctx.Value(ctxKeyObservationSuppressed) }
}
```

**⚠️ 实现注意（写代码时修正）**：Go 的 context 不可变——`EnterCallScopeWithID` 返回的还原闭包无法把父 ctx"改回去"。正确做法是：调用方在函数体内直接继续用外层 ctx 变量（Go 侧等价于 Python 的作用域收束），还原闭包仅作显式边界标记；闭包内 `_ = prevID` 占位实现仅为让测试通过的最小骨架，**实施时按此语义收敛实现**：`done()` 无操作或仅做断言辅助。测试断言"退出后还原"按"调用方继续使用外层 ctx"改写（上面测试的 `done()` 段落按此语义调整：还原检查对象为外层 ctx 变量本身）。这是 P1（ctx 不可变 vs contextvar token）偏差，spec 已登记。

```go
package rsi

import "context"

// EngineEvent RSI 引擎事件联合类型。
// Python: EngineEvent = EventStatus | EventProgress | EventNode | NodeStageEvent | EventUsage
type EngineEvent interface {
	// EventType 恒定事件名（对齐 Python dataclass 的 event_type 字段）
	EventType() string
}

// OnEvent 事件回调。Python: Callable[[EngineEvent], Awaitable[None]]（错误不吞，由上层决定补偿）
type OnEvent func(ctx context.Context, event EngineEvent) error

// Emit 发出事件：回调为 nil 时跳过；回调错误原样透传。
// Python: openjiuwen/rsi/events.py emit()——引擎须在相关数据持久化后才调用。
func Emit(ctx context.Context, onEvent OnEvent, event EngineEvent) error {
	if onEvent == nil {
		return nil
	}
	return onEvent(ctx, event)
}
```

5 个事件结构体（私有字段 + 访问器 + `NewXxx` 构造，字段照 Python 逐个抄）：
- `EventStatus`：`status RsiStatus`；`EventType() string { return "status" }`。
- `EventProgress`：`iteration,total int; score,baseline *float64; usage *RsiUsage`；`EventType()=="progress"`。
- `EventNode`：`node *RsiTreeNode; artifacts []map[string]string`；`EventType()=="node"`。
- `EventUsage`：`eventID int; taskID,ts,callID string; modelCall *RsiModelCall; nodeRef,stageRef *string`；`Family()=="progress"`、`Kind()=="usage"`、`EventType()=="progress.usage"`（三个恒定值访问器）。
- `NodeStageEvent`：`nodeRef string; stage map[string]any; note *string`；`EventType()=="node.stage"`（stage 语义保留 Python 注释：JSON 透传对象，AgentServer 不解释）。

`EventNodeStage = *NodeStageEvent` 类型别名保留（Python `EventNodeStage: TypeAlias`）。
