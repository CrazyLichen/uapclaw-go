// processGoalCommand 处理非流式 command.goal 请求。
// Python: interface.py:3408-3505 _process_message 的 COMMAND_GOAL 分支
//
// 非流式承担全部五个 action；set/resume 的真正执行仍走流式路径。
func (uc *UapClawSwarm) processGoalCommand(
	ctx context.Context, request *schema.AgentRequest,
) (*schema.AgentResponse, error) {
	params := parseRequestParams(request)
	action := goalActionOf(request)
	sessionID := swruntime.GetSessionID(request.SessionID)
	logger.Info(logComponent).
		Str("event_type", "goal_command").
		Str("request_id", request.RequestID).
		Str("action", action).
		Str("session_id", sessionID).
		Msg("command.goal received")

	a, err := uc.goalAdapter(request)
	if err != nil {
		return nil, err
	}
	control, err := a.DispatchGoalControl(ctx, sessionID, action,
		stringParam(params, "objective"), boolField(params, "overwrite_confirmed"),
		intField(params, "token_budget"), intField(params, "max_attempts"))
	if err != nil {
		// Python: except Exception → ok=False, {"error": f"Goal command error: {exc}"}
		logger.Error(logComponent).Err(err).
			Str("event_type", "GOAL_COMMAND_FAILED").
			Str("request_id", request.RequestID).
			Msg("command.goal failed")
		return schema.NewAgentResponse(request.RequestID, request.ChannelID,
			schema.WithResponseOK(false),
			schema.WithPayload(map[string]any{
				"error": "Goal command error: " + err.Error(),
			})), nil
	}
	if control == nil {
		return schema.NewAgentResponse(request.RequestID, request.ChannelID,
			schema.WithResponseOK(false),
			schema.WithPayload(map[string]any{"error": "Goal command not handled"})), nil
	}
	return goalControlResponse(request, action, control), nil
}

// processGoalCommandStream 流式 command.goal 的 get/pause/clear 单帧短路。
// Python: interface.py:4175-4235 —— action 不在 {set,resume} 时只发一帧即 return，
// 不进入 attach_output；set/resume 由适配器流式主链路接管（attach→control 两段映射）。
func (uc *UapClawSwarm) processGoalCommandStream(
	ctx context.Context, request *schema.AgentRequest,
) ([]*schema.AgentResponseChunk, error) {
	params := parseRequestParams(request)
	action := goalActionOf(request)
	sessionID := swruntime.GetSessionID(request.SessionID)
	a, err := uc.goalAdapter(request)
	if err != nil {
		return nil, err
	}
	control, err := a.DispatchGoalControl(ctx, sessionID, action,
		stringParam(params, "objective"), boolField(params, "overwrite_confirmed"),
		intField(params, "token_budget"), intField(params, "max_attempts"))
	if err != nil {
		logger.Error(logComponent).Err(err).
			Str("event_type", "GOAL_COMMAND_STREAM_FAILED").
			Str("request_id", request.RequestID).
			Msg("streaming command.goal failed")
		return []*schema.AgentResponseChunk{schema.NewAgentResponseChunk(
			request.RequestID, request.ChannelID, map[string]any{
				"event_type": "chat.error",
				"error":      "Goal command error: " + err.Error(),
			}, schema.WithChunkIsComplete(true))}, nil
	}
	if control == nil {
		return []*schema.AgentResponseChunk{schema.NewAgentResponseChunk(
			request.RequestID, request.ChannelID, map[string]any{
				"event_type": "chat.error",
				"error":      "Goal command not handled",
			}, schema.WithChunkIsComplete(true))}, nil
	}
	if resultType, _ := control["result_type"].(string); resultType == goalResultError {
		return []*schema.AgentResponseChunk{schema.NewAgentResponseChunk(
			request.RequestID, request.ChannelID, map[string]any{
				"event_type": "chat.error",
				"code":       control["error_code"],
				"error":      control["error"],
				"goal":       control["goal"],
			}, schema.WithChunkIsComplete(true))}, nil
	}
	return []*schema.AgentResponseChunk{schema.NewAgentResponseChunk(
		request.RequestID, request.ChannelID, map[string]any{
			"event_type":   goalSnapshotEventType,
			"action":       control["action"],
			"goal":         control["goal"],

package adapter

import (
	"context"
	"errors"
	"strings"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/harness/goal"
	"github.com/uapclaw/uapclaw-go/internal/common/logger"

// ──────────────────────────── 导出函数 ────────────────────────────

// dispatchGoalControl 把协议字段映射到独立的 Goal 能力方法。
// Python: interface_deep.py:17764-17930 _dispatch_goal_control
//
// 分流头（根 → child）由 deep_adapter_sessions.go 的 DispatchGoalControl 承担，
// 本方法只跑在 child 上。
func (d *DeepAdapter) dispatchGoalControl(
	ctx context.Context, sessionID, action, objective string,
	overwriteConfirmed bool, tokenBudget, maxAttempts int,
) (map[string]any, error) {
	normalized := strings.ToLower(strings.TrimSpace(action))
	manager := d.goalManager()
	if manager == nil {
		return map[string]any{
			"result_type": goalResultError,
			"action":      normalized,
			"error_code":  goalErrManagerNotStarted,
			"error":       "goal manager is not started",
		}, nil
	}

	switch normalized {
	case "get":
		// 只读查询走无锁 peek：长跑/卡住的 goal 轮持着交互控制锁，
		// await get() 会串行在同一把锁上，把 command.goal get 拖到超时。
		// Python: interface_deep.py:17822-17829
		return d.goalControlResult(normalized, manager.Peek()), nil
	case "set":
		record, err := manager.Set(ctx, objective, goal.SetOptions{
			OverwriteConfirmed: overwriteConfirmed,
			TokenBudget:        optionalPositive(tokenBudget),
			MaxAttempts:        optionalPositive(maxAttempts),
		})
		if err != nil {
			return d.goalErrorResult(normalized, objective, err), nil
		}
		return d.goalControlResult(normalized, record), nil
	case "pause":
		before := manager.Get()
		if before == nil {
			return map[string]any{
				"result_type": goalResultError, "action": normalized,
				"error_code": goalErrNoGoal,
				"error":      "No goal in this session; cannot pause.",
				"goal":       nil,
			}, nil
		}
		beforeStatus := before.Status
		record, err := manager.Pause(ctx)
		if err != nil {
			return d.goalErrorResult(normalized, objective, err), nil
		}
		payload := goalRecordPayload(record)
		if beforeStatus != goal.GoalStatusActive {
			return map[string]any{
				"result_type": goalResultError, "action": normalized,
				"error_code": goalErrInvalidState,
				"error":      "Goal is " + beforeStatus.String() + "; only active goals can be paused.",
				"goal":       payload,
			}, nil
		}
		return map[string]any{
			"result_type": goalResultControl, "action": normalized,
			"goal": payload, "output": "Goal paused.",
		}, nil
	case "resume":
		before := manager.Get()
		if before == nil {
			return map[string]any{
				"result_type": goalResultError, "action": normalized,
				"error_code": goalErrNoGoal,
				"error":      "No goal in this session; cannot resume.",
				"goal":       nil,
			}, nil
		}
		if before.Status == goal.GoalStatusActive {
			return map[string]any{
				"result_type": goalResultControl, "action": normalized,
				"goal": goalRecordPayload(before), "output": "Goal already active.",
			}, nil
		}
		if before.Status != goal.GoalStatusPaused && before.Status != goal.GoalStatusBlocked {
			return map[string]any{
				"result_type": goalResultError, "action": normalized,
				"error_code": goalErrInvalidState,
				"error":      "Goal is " + before.Status.String() + "; only paused/blocked goals can be resumed.",
				"goal":       goalRecordPayload(before),
			}, nil
		}
		record, err := manager.Resume(ctx)
		if err != nil {
			return d.goalErrorResult(normalized, objective, err), nil
		}
		return d.goalControlResult(normalized, record), nil
	case "clear":
		removed, err := manager.Clear(ctx)
		if err != nil {
			return d.goalErrorResult(normalized, objective, err), nil
		}
		if removed == nil {
			return map[string]any{
				"result_type": goalResultError, "action": normalized,
				"error_code": goalErrNoGoal,
				"error":      "No goal in this session; nothing to clear.",
				"goal":       nil, "cleared_goal": nil,
			}, nil
		}
		return map[string]any{
			"result_type": goalResultControl, "action": normalized,
			"goal": nil, "cleared_goal": goalRecordPayload(removed),
			"output": "Goal cleared.",
		}, nil
	default:
		return map[string]any{
			"result_type": goalResultError, "action": normalized,
			"error_code": goalErrInvalidAction,
			"error":      "unsupported goal action: " + action,
		}, nil
	}
}

// goalManager 返回本适配器会话的 GoalManager（未 start_interaction 时为 nil）。
// Python: interface_deep.py:17739 _get_goal_manager
func (d *DeepAdapter) goalManager() *goal.GoalManager {
	if d.instance == nil {

// Python: interface_deep.py:15739-15757
func parseGoalSlashIntent(query string) map[string]any {
	text := strings.TrimSpace(query)
	if !strings.HasPrefix(text, "/goal") {
		return nil
	}
	args := strings.TrimSpace(text[5:])
	if args == "" {
		return map[string]any{"action": "get"}
	}
	lower := strings.ToLower(args)
	if lower == "pause" || lower == "resume" || lower == "clear" {
		return map[string]any{"action": lower}
	}
	if strings.HasPrefix(lower, "set ") {
		return map[string]any{"action": "set", "objective": strings.TrimSpace(args[4:])}
	}
	if lower == "set" {
		return map[string]any{"action": "set", "objective": ""}
	}
	// get / stop 不是用户命令词，保留为 "/goal <objective>" 形式的正文
	return map[string]any{"action": "set", "objective": args}

// attach→control 两段路径的 pending op。
// Python: interface_deep.py:16251-16281

// 普通聊天文本 "/goal set ..." 不走这里——只有显式 command.goal 方法才映射。
func structuredGoalOpFromRequest(req *schema.AgentRequest) map[string]any {
	if req == nil || req.ReqMethod != schema.ReqMethodCommandGoal {
		return nil
	}
	raw := parseRequestParams(req)
	action := strings.ToLower(strings.TrimSpace(stringParam(raw, "action")))
	if action == "" {
		action = "get"
	}
	if action != "set" && action != "resume" {
		return nil
	}
	op := map[string]any{"action": action}
	if action != "set" {
		return op
	}
	objective, _ := raw["objective"].(string)
	op["objective"] = objective
	overwrite, _ := raw["overwrite_confirmed"].(bool)
	op["overwrite_confirmed"] = overwrite
	for _, key := range []string{"token_budget", "max_attempts"} {
		if value, ok := positiveIntField(raw, key); ok {
			op[key] = value
		}
	}
	return op
}

// positiveIntField 取正整数参数（接受 int / 纯数字字符串，拒绝 bool）。
// Python: isinstance(value, int) 或 str.strip().lstrip("-").isdigit()
func positiveIntField(params map[string]any, key string) (int, bool) {
	value, ok := params[key]
	if !ok || value == nil {
		return 0, false
	}
	switch typed := value.(type) {
	case bool:
		return 0, false
	case int:
		return typed, true
	case int64:
		return int(typed), true
	case float64:
		return int(typed), true
	case string:
		text := strings.TrimSpace(typed)
		text = strings.TrimPrefix(text, "-")
		if text == "" || strings.TrimLeft(text, "0123456789") != "" {
			return 0, false
		}
		parsed, err := strconv.Atoi(strings.TrimSpace(typed))
		if err != nil {
			return 0, false
		}
		return parsed, true
	default:
		return 0, false
	}
}

// attachAndSendInputs 抢输出租约并投递输入，返回流句柄与权限是否已派发。
// Python: interface_deep.py:16008-16085 _attach_and_send_inputs
//
// sendWithoutOutput 为真（steer / follow_up / 中断恢复）时即使拿不到租约
// 也必须投递：HITL 答案若不注入，工具会一直阻塞在等待上。
func (d *DeepAdapter) attachAndSendInputs(
	ctx context.Context, req *schema.AgentRequest, inputs map[string]any,
	sendWithoutOutput bool,
) (*hschema.InteractionOutputStream, bool, error) {
	// 普通中断轮先等前一个读者释放：ACK-only 注入到已排队 EOF 的租约上
	// 会丢掉全部恢复输出（见 OutputHandoff 注释）。
	if prev := d.interactionOutputHandoff; prev != nil {
		if d.isInterruptResumeDispatch(req.Params) && !d.goalRecordIsActive() {
			if err := prev.Wait(ctx); err != nil {
				return nil, false, err
			}
		}
	}

	stream, err := d.instance.AttachOutput(ctx)
	if err != nil {
		return nil, false, err
	}
	if stream == nil && !sendWithoutOutput {
		return nil, false, nil
	}
	if stream != nil {
		stream = d.wrapOutputHandoff(stream)
	}

	mode := resolveInputDispatchMode(req.Params)
	sendReq := hschema.SendInputRequest{
		RequestID: req.RequestID,
		Inputs:    inputs,
		Mode:      mode,
	}
	if sendErr := d.instance.SendInput(ctx, sendReq); sendErr != nil {
		if stream != nil {
			_ = stream.Close(ctx, false)
		}
		return nil, false, sendErr
	}
	return stream, false, nil
}

func (d *DeepAdapter) consumeInteractionStream(
	ctx context.Context, req *schema.AgentRequest, stream *hschema.InteractionOutputStream,
) <-chan *schema.AgentResponseChunk {
	outCh := make(chan *schema.AgentResponseChunk, 64)
	go func() {
		defer close(outCh)
		abort := true
		defer func() {
			// 结束时释放租约；abort 决定是否连带取消在飞轮。
			_ = stream.Close(ctx, abort)
		}()
		for frame := range stream.All(ctx) {
			payload, ok := frame.(map[string]any)
			if !ok {
				continue
			}
			d.trackRoundOutputBoundary(payload)
			adapted, keep := d.adaptGoalIntermediateFinal(payload)
			if !keep {
				continue
			}
			boundary := d.beginVisibleChatContent(req, false)
			if boundary != nil {
				outCh <- schema.NewAgentResponseChunk(req.RequestID, req.ChannelID, boundary)

			isComplete := eventType == "chat.final"
			outCh <- schema.NewAgentResponseChunk(req.RequestID, req.ChannelID, adapted,
				schema.WithChunkIsComplete(isComplete))
			if isComplete {
				abort = false
			}
		}
	}()
	return outCh

// ──────────────────────────── 结构体 ────────────────────────────

// outputHandoff 宿主自持的输出排空屏障。
// Python: output_handoff.py OutputHandoff（整文件）
//
// Core 可能在宿主消费完交互卡之前就排队 EOF。向这种 finishing 租约投递答案
// 会丢掉后续输出——所以下一次中断派发必须等这个读者释放所有权，而不是重发
// 输入、抢租约或中止上一轮。
type outputHandoff struct {
	// owner 持有租约的 DeepAgent
	owner *harness.DeepAgent
	// stream 被包裹的输出流
	stream *hschema.InteractionOutputStream
	// drained 老读者是否已排空
	drained chan struct{}
	// once 保证 drained 只关一次
	once sync.Once
	// closeErr 关闭时的错误（供 Wait 上抛）
	closeErr error
	// mu 保护 closeErr
	mu sync.Mutex
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// newOutputHandoff 构造排空屏障。
func newOutputHandoff(owner *harness.DeepAgent, stream *hschema.InteractionOutputStream) *outputHandoff {
	return &outputHandoff{owner: owner, stream: stream, drained: make(chan struct{})}
}

// Drained 老读者是否已排空。
func (h *outputHandoff) Drained() bool {
	select {
	case <-h.drained:
		return true
	default:
		return false
	}
}

// Wait 等待老读者释放所有权。
// Python: output_handoff.py wait（关闭失败时上抛）
func (h *outputHandoff) Wait(ctx context.Context) error {
	select {
	case <-h.drained:
	case <-ctx.Done():
		return ctx.Err()
	}
	h.mu.Lock()
	defer h.mu.Unlock()
	if h.closeErr != nil {
		return fmt.Errorf("上一个输出所有者关闭失败: %w", h.closeErr)
	}
	return nil
}

// markDrained 标记排空并记录关闭错误。
func (h *outputHandoff) markDrained(closeErr error) {
	h.mu.Lock()
	h.closeErr = closeErr
	h.mu.Unlock()
	h.once.Do(func() { close(h.drained) })
}

// Python: interface_deep.py:16283-16310
//
// 绝不在 goal 轮在飞时中止：goal 监督器拥有自己的生命周期，这里 abort

func (d *DeepAdapter) abortSharedAgentIfSafe(ctx context.Context, sessionID, intent string) bool {
	if d.instance == nil {
		return true
	}
	if d.hasActiveGoalRound() {
		logger.Info(logComponent).
			Str("event_type", "GOAL_ABORT_SKIPPED").
			Str("intent", intent).
			Str("session_id", sessionID).
			Msg("active goal round: skip instance abort (goal controlled via /goal pause|clear)")
		return false
	}
	if d.otherActiveSessions(sessionID) > 0 {
		logger.Info(logComponent).
			Str("event_type", "GOAL_ABORT_SKIPPED_OTHER_SESSIONS").
			Str("intent", intent).
			Str("session_id", sessionID).
			Msg("other sessions still active: skip shared instance abort")
		return false
	}
	_ = d.instance.CancelRound(ctx, intent)
	return true
}

func (d *DeepAdapter) adaptGoalIntermediateFinal(chunk map[string]any) (map[string]any, bool) {

		return chunk, true

	adapted[goalIntermediateField] = true

	}
	if kind := d.streamContentRunKind; kind != "" {
		return kind == "goal"

// beginVisibleChatContent 给内容来源盖章，并在 user→goal 切换处注入分气泡 final。
// Python: interface_deep.py:15574-15617
//
// 共享流从 user 轮文本切到 goal 轮文本且中间没有终止 chat.final 时，补一个
// 空 final，让前端 stopStreaming 并开新气泡。
//
// userOriginated 标记「这是普通用户聊天消费者」（不是 command.goal / attach
// goal 的流）。这种流可能在 user 轮吐出第一个可见 token 之前就被 goal 轮
// 劫持（慢首 token + 早到的 goal 插入），此时 prev 仍是空但边界 final 仍然
// 需要——否则 goal 答案会并进那个（空的）user 气泡。非用户来源的流保留严格
// prev=="user" 规则，纯 goal 流不会在自己头部凭空多一个 final。
func (d *DeepAdapter) beginVisibleChatContent(
	req *schema.AgentRequest, userOriginated bool,
) map[string]any {
	kind := d.consumedRoundRunKind()
	prev := d.streamContentRunKind
	var boundary map[string]any

	switchedFromUser := prev == "user" && kind == "goal"
	hijackedBeforeUserToken := userOriginated && kind == "goal" && prev == ""

	if kind == "goal" && req != nil {
		sessionID := ""
		if req.SessionID != nil {
			sessionID = *req.SessionID
		}
		d.flushPendingGoalObjectiveHistory(sessionID)
	}
	if switchedFromUser || hijackedBeforeUserToken {
		boundary = map[string]any{"event_type": "chat.final", "content": ""}
		d.streamContentRunKind = ""
	}
	if kind != "" {
		d.streamContentRunKind = kind
	}
	return boundary
}

// consumedRoundRunKind 产出当前 chunk 的那一轮的 run kind。
// Python: interface_deep.py:15510-15521
func (d *DeepAdapter) consumedRoundRunKind() string {
	if d.streamRoundKindLatch != "" {
		return d.streamRoundKindLatch
	}
	return d.currentInteractionRunKind()
}

// pendingGoalHistory 挂起的 goal objective 历史记录。
type pendingGoalHistory struct {
	// sessionID 会话标识
	sessionID string
	// requestID 请求标识
	requestID string
	// channelID 渠道标识
	channelID string
	// content 目标正文
	content string
	// channelMetadata 渠道元数据
	channelMetadata map[string]any
	// mode 运行模式
	mode string
	// goalID 目标标识
	goalID string

// pendingGoalObjectives 进程级挂起表：sessionID → 待落盘记录。
// Python: interface_deep.py:_pending_goal_objective_history（模块级 dict）
var (
	pendingGoalObjectivesMu sync.Mutex
	pendingGoalObjectives   = map[string]*pendingGoalHistory{}
)

// recordGoalSetHistoryIfNeeded 成功 set 后写入 objective 用户历史。
// Python: interface_deep.py:17610-17654
//
// 忙碌时（上一轮 user 仍在，或同会话另有并发任务）推迟到轮边界再写，
// 否则重载后的时间线顺序会错。
func (d *DeepAdapter) recordGoalSetHistoryIfNeeded(
	req *schema.AgentRequest, action, resultType string, goalPayload map[string]any,
	deferWrite *bool,
) {
	if action != "set" {
		return
	}
	if resultType == goalResultError || resultType == goalResultConfirmRequired || resultType == "" {
		return
	}
	if goalPayload == nil {
		return
	}
	objective, _ := goalPayload["objective"].(string)
	if objective == "" {
		return
	}
	params := parseParams(req.Params)
	goalID, _ := goalPayload["goal_id"].(string)
	sessionID := "default"
	if req.SessionID != nil && *req.SessionID != "" {
		sessionID = *req.SessionID
	}
	entry := &pendingGoalHistory{
		sessionID:       sessionID,
		requestID:       req.RequestID,
		channelID:       req.ChannelID,
		content:         objective,
		channelMetadata: req.Metadata,
		mode:            stringParam(params, "mode"),
		goalID:          goalID,
	}
	if entry.mode == "" {
		entry.mode = "unknown"
	}

	shouldDefer := d.shouldDeferGoalObjectiveHistory(sessionID)
	if deferWrite != nil {
		shouldDefer = *deferWrite
	}
	if shouldDefer {
		pendingGoalObjectivesMu.Lock()
		pendingGoalObjectives[sessionID] = entry
		pendingGoalObjectivesMu.Unlock()
		return
	}
	pendingGoalObjectivesMu.Lock()
	delete(pendingGoalObjectives, sessionID)
	pendingGoalObjectivesMu.Unlock()
	appendGoalObjectiveHistory(entry)
}

// shouldDeferGoalObjectiveHistory 是否需要把 objective 历史推迟到轮边界。
// Python: interface_deep.py:17584-17608
func (d *DeepAdapter) shouldDeferGoalObjectiveHistory(sessionID string) bool {
	if d.currentInteractionRunKind() == "user" {
		return true
	}
	if d.streamContentRunKind == "user" || d.streamRoundKindLatch == "user" {
		return true
	}
	// 本 stream 已 mark 过；count>1 表示同会话还有别的请求在跑
	if d.activeSessionCount(sessionID) > 1 {
		return true
	}
	return d.sessionHasOtherRunningAgentTasks(sessionID)
}

// flushPendingGoalObjectiveHistory 把挂起的 goal 用户历史落到磁盘（无挂起则 noop）。
// Python: interface_deep.py:17597-17608
func (d *DeepAdapter) flushPendingGoalObjectiveHistory(sessionID string) {
	pendingGoalObjectivesMu.Lock()
	entry := pendingGoalObjectives[sessionID]
	delete(pendingGoalObjectives, sessionID)
	pendingGoalObjectivesMu.Unlock()
	if entry == nil {
		return
	}
	appendGoalObjectiveHistory(entry)
}

// pendingGoalObjective 返回挂起记录（仅测试与调试用）。
func (d *DeepAdapter) pendingGoalObjective(sessionID string) *pendingGoalHistory {
	pendingGoalObjectivesMu.Lock()
	defer pendingGoalObjectivesMu.Unlock()
	return pendingGoalObjectives[sessionID]
}

// appendGoalObjectiveHistory 真正落盘一条 objective 用户历史。
// Python: append_history_record(role="user", extra={goal_id, is_goal_objective_message})
func appendGoalObjectiveHistory(entry *pendingGoalHistory) {
	sessionpkg.AppendHistoryRecord(
		entry.sessionID, entry.requestID, entry.channelID,
		"user", entry.content, float64(time.Now().UnixMilli())/1000,
		"", map[string]any{
			"goal_id":                   entry.goalID,
			"is_goal_objective_message": true,
		},
		entry.channelMetadata, entry.mode,
	)
}
