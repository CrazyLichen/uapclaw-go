```go
package common

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"regexp"
	"strings"
)

// ──────────────────────────── 结构体 ────────────────────────────

// InvalidProtocolId 外部输入的协议标识不符合线协议约定。
// Python: gateway/common/protocol_ids.py InvalidProtocolId
//
// 消息刻意不含原始值：调用方可以安全返回或记录日志，即使输入极大。
type InvalidProtocolId struct {
	// Message 不含原值的错误说明
	Message string
}

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// sessionIDPattern 会话 ID 白名单：ASCII 字母数字下划线，点与横线只允许出现在中间。
	// Python: protocol_ids.py _SESSION_ID_PATTERN
	sessionIDPattern = regexp.MustCompile(`^[A-Za-z0-9_](?:[A-Za-z0-9_.-]{0,78}[A-Za-z0-9_])?$`)
)

// ──────────────────────────── 常量 ────────────────────────────

const (
	// SessionIDMaxLen 会话 ID 最大长度（与持久会话契约一致）
	SessionIDMaxLen = 80
	// WorkflowRunIDMaxLen workflow run ID 最大长度
	WorkflowRunIDMaxLen = 256
)

// ──────────────────────────── 导出函数 ────────────────────────────

// Error 实现 error 接口。
func (e *InvalidProtocolId) Error() string { return e.Message }

// ValidateSessionID 校验会话 ID 并返回错误。
// Python: protocol_ids.py validate_session_id
// 四条判据：是字符串 / 非空且无首尾空白 / 长度上限 / 正则全串匹配。
func ValidateSessionID(value string) error {
	if value == "" || value != strings.TrimSpace(value) {
		return &InvalidProtocolId{Message: "invalid session_id"}
	}
	if len(value) > SessionIDMaxLen {
		return &InvalidProtocolId{
			Message: fmt.Sprintf("session_id exceeds maximum length %d", SessionIDMaxLen),
		}
	}
	if !sessionIDPattern.MatchString(value) {
		return &InvalidProtocolId{Message: "invalid session_id"}
	}
	return nil
}

// IsValidSessionID 判断会话 ID 是否满足共享契约。
// Python: protocol_ids.py is_valid_session_id
func IsValidSessionID(value string) bool {
	return ValidateSessionID(value) == nil
}

// ValidateWorkflowRunID 校验 workflow run ID。
// Python: protocol_ids.py validate_workflow_run_id
func ValidateWorkflowRunID(value string) error {
	if value == "" || value != strings.TrimSpace(value) {
		return &InvalidProtocolId{Message: "invalid run_id"}
	}
	if len(value) > WorkflowRunIDMaxLen {
		return &InvalidProtocolId{
			Message: fmt.Sprintf("run_id exceeds maximum length %d", WorkflowRunIDMaxLen),
		}
	}
	for _, char := range value {
		if char < 32 || char == 127 {
			return &InvalidProtocolId{Message: "invalid run_id"}
		}
	}
	return nil
}

// ResolveSessionDir 安全解析会话目录，返回目录路径与是否存在。
// Python: server/runtime/session/session_history.py resolve_session_dir
//
// 两步防御：
//  1. 白名单校验（不合法时根本不拼路径）
//  2. Clean 后确认结果仍在 sessions 根内（白名单被绕过的兜底）
//
// 额外支持归档目录回退：active 目录不存在时再看 sessions_archived/<id>。
func ResolveSessionDir(sessionID, sessionsRoot string) (string, bool, error) {
	if err := ValidateSessionID(sessionID); err != nil {
		return "", false, err
	}
	active := filepath.Join(sessionsRoot, sessionID)
	// 纵深防御必须在任何磁盘操作之前
	cleanedRoot := filepath.Clean(sessionsRoot)
	cleanedActive := filepath.Clean(active)
	if cleanedActive != cleanedRoot && !strings.HasPrefix(cleanedActive, cleanedRoot+string(os.PathSeparator)) {
		return "", false, &InvalidProtocolId{Message: "invalid session_id"}
	}
	if info, err := os.Stat(cleanedActive); err == nil && info.IsDir() {
		return cleanedActive, true, nil
	}
	archived := filepath.Clean(filepath.Join(filepath.Dir(cleanedRoot), "sessions_archived", sessionID))
	if info, err := os.Stat(archived); err == nil && info.IsDir() {
		return archived, true, nil
	}
	return cleanedActive, false, nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// errInvalidProtocolId 供内部判等用。
var errInvalidProtocolId = errors.New("invalid protocol id")

// _ 保证 errInvalidProtocolId 被使用（避免 unused 告警）。
var _ = errInvalidProtocolId

package runtime

import "strings"

// ──────────────────────────── 常量 ────────────────────────────

const (
	// ClientToolInterruptSource 客户端自己执行了工具、回来交结果的 chat.send 来源。
	// Python: gateway/common/interrupt_resume.py CLIENT_TOOL_INTERRUPT_SOURCE
	ClientToolInterruptSource = "client_tool_interrupt"
)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// InterruptResumeSources 回答被暂停工具调用的 chat.send 来源集合。
	// Python: interrupt_resume.py INTERRUPT_RESUME_SOURCES
	InterruptResumeSources = map[string]struct{}{
		"permission_interrupt": {},
		"confirm_interrupt":    {},
		"ask_user_interrupt":   {},
		"evolution_interrupt":  {},
		ClientToolInterruptSource: {},
	}
)

// ──────────────────────────── 导出函数 ────────────────────────────

// IsClientToolResume chat.send 是否携带客户端执行工具的结果。
// Python: interrupt_resume.py is_client_tool_resume
func IsClientToolResume(params map[string]any) bool {
	if params == nil {
		return false
	}
	source, _ := params["source"].(string)
	return source == ClientToolInterruptSource
}

// AnswersPausedCalls chat.send 是否在回答一个被暂停的轮次。
// Python: interrupt_resume.py answers_paused_calls
//
// 判据：source 在 resume 集合内，answers 是非空列表；
// 且（client_tool resume 例外）request_id 非空——客户端工具结果不指名问题，
// 它的每个答案指名自己跑过的调用。
func AnswersPausedCalls(params map[string]any) bool {
	if params == nil {
		return false
	}
	source, _ := params["source"].(string)
	source = strings.TrimSpace(source)
	if _, ok := InterruptResumeSources[source]; !ok {
		return false
	}
	answers, ok := params["answers"].([]any)
	if !ok || len(answers) == 0 {
		return false
	}
	if source == ClientToolInterruptSource {
		return true
	}
	requestID, _ := params["request_id"].(string)
	return strings.TrimSpace(requestID) != ""
}

package runtime

import "strings"

// ──────────────────────────── 常量 ────────────────────────────

const (
	// LegacySkillEvolutionApprovalSource 老式技能演进审批的 source 取值。
	// Python: agents/harness/common/rails/interrupt/interrupt_helpers.py
	//         LEGACY_SKILL_EVOLUTION_APPROVAL_SOURCE（:57）
	LegacySkillEvolutionApprovalSource = "skill_evolution_approval"
)

// ──────────────────────────── 导出函数 ────────────────────────────

// HasInterruptResumePayload 载荷是否带 request_id 与非空 answers。
// Python: interrupt_helpers.py has_interrupt_resume_payload（:151）
func HasInterruptResumePayload(params map[string]any) bool {
	if params == nil {
		return false
	}
	requestID, _ := params["request_id"].(string)
	if strings.TrimSpace(requestID) == "" {
		return false
	}
	answers, ok := params["answers"].([]any)
	return ok && len(answers) > 0
}

// IsInterruptResumePayload 载荷是否在回答一个被暂停的轮次。
// Python: interrupt_helpers.py is_interrupt_resume_payload（:160）
//
// 三条路径，顺序不得调整：
//  1. answers_paused_calls 命中即为 true
//  2. 不是标准 resume 载荷则 false
//  3. source 必须是 legacy 技能演进审批，且 evolution_meta.approval_transport == "interrupt"
func IsInterruptResumePayload(params map[string]any) bool {
	if AnswersPausedCalls(params) {
		return true
	}
	if !HasInterruptResumePayload(params) {
		return false
	}
	source, _ := params["source"].(string)
	if strings.TrimSpace(source) != LegacySkillEvolutionApprovalSource {
		return false
	}
	evolutionMeta, ok := params["evolution_meta"].(map[string]any)
	if !ok {
		return false
	}
	transport, _ := evolutionMeta["approval_transport"].(string)
	return transport == "interrupt"
}

// IsInterruptResumeRequest 请求是否为中断续答。
// Python: runtime/service.py _is_interrupt_resume_request（:3601）
//
// 保持上游的一行转调形状：ask_user / permission 的答案以 chat.send 运输，
// 原轮次仍持有会话准入，若让答案再等一次准入会死锁交互。
func IsInterruptResumeRequest(params map[string]any) bool {
	return IsInterruptResumePayload(params)
}

const (

// ResolveSessionInputMode 归一化 input_mode 与其遗留别名 runtime_mode。
// Python: session_input.py resolve_session_input_mode
//
// 未知值保留普通发送行为（返回 nil）；交互应答在别处单独分类。
// 主字段与别名冲突时返回 nil（调用方若需报错走 Checked 版本）。
func ResolveSessionInputMode(params map[string]any) *SessionInputMode {
	mode, _ := ResolveSessionInputModeChecked(params)
	return mode
}

// ResolveSessionInputModeChecked 同 ResolveSessionInputMode，但冲突时报错。
// Python 侧冲突抛 ValueError。
func ResolveSessionInputModeChecked(params map[string]any) (*SessionInputMode, error) {
	if params == nil {
		return nil, nil
	}
	primary, hasPrimary := params["input_mode"]
	alias, hasAlias := params["runtime_mode"]
	raw := ""
	if hasPrimary {
		raw = normalizeSessionInputRaw(primary)
	}
	if hasAlias {
		aliasValue := normalizeSessionInputRaw(alias)
		if hasPrimary && raw != "" && aliasValue != "" && raw != aliasValue {
			return nil, errors.New("input_mode and runtime_mode must agree")
		}
		if raw == "" {
			raw = aliasValue
		}
	}
	switch raw {
	case string(SessionInputModeSteer):
		mode := SessionInputModeSteer
		return &mode, nil
	case string(SessionInputModeFollowUp):
		mode := SessionInputModeFollowUp
		return &mode, nil
	default:
		return nil, nil
	}
}

// ValidateSteerImageReferences 校验 steer 的"仅引用"线格式，不碰文件系统。
// Python: session_input.py validate_steer_image_references
func ValidateSteerImageReferences(params map[string]any) ([]map[string]any, error) {
	for _, key := range []string{
		"images", "image_files", "files", "attachments", "audio_files", "video_files",
	} {
		if value, ok := params[key]; ok && value != nil && !isEmptySlice(value) {
			return nil, errors.New(
				"steering accepts uploaded image references only; send other attachments as a queued task")
		}
	}
	rawItems, ok := params["media_items"]
	if !ok || rawItems == nil {
		return nil, nil
	}
	items, ok := rawItems.([]any)
	if !ok {
		return nil, errors.New("steer media_items must be a list of uploaded image references")
	}
	out := make([]map[string]any, 0, len(items))
	for _, item := range items {
		entry, ok := item.(map[string]any)
		if !ok {
			return nil, errors.New("steer images require uploaded file references without inline data")
		}
		for key := range entry {
			if _, allowed := steerImageFields[key]; !allowed {
				return nil, errors.New("steer images require uploaded file references without inline data")
			}
		}
		if entry["type"] != "image" {
			return nil, errors.New("steering accepts uploaded images only")
		}
		for _, key := range []string{"path", "filename", "mime_type"} {
			text, _ := entry[key].(string)
			if strings.TrimSpace(text) == "" {
				return nil, fmt.Errorf("steer image %s must be a non-empty string", key)
			}
		}
		mimeType, _ := entry["mime_type"].(string)
		if _, allowed := steerImageMIMETypes[mimeType]; !allowed {
			return nil, errors.New("unsupported steer image MIME type")
		}
		out = append(out, entry)
	}
	return out, nil
}

// ValidateSessionInput 校验会话补充输入的整体形状。
// Python: session_input.py validate_session_input
func ValidateSessionInput(params map[string]any) error {
	mode := ResolveSessionInputMode(params)
	if target, ok := params["expected_execution_id"]; ok {
		targetText, isString := target.(string)
		if !isString || strings.TrimSpace(targetText) == "" {
			return errors.New("expected_execution_id must be a non-empty string")
		}
		if mode == nil || *mode != SessionInputModeSteer {
			return errors.New("expected_execution_id requires steer mode")
		}
	}
	var imageReferences []map[string]any
	if mode != nil && *mode == SessionInputModeSteer {
		refs, err := ValidateSteerImageReferences(params)
		if err != nil {
			return err
		}
		imageReferences = refs
	}
	query, hasQuery := params["query"]
	if len(imageReferences) > 0 && (!hasQuery || query == nil || isStringValue(query)) {
		return nil
	}
	queryText, isString := query.(string)
	if !isString || strings.TrimSpace(queryText) == "" {
		return errors.New("session input requires non-empty text")
	}
	return nil
}

// normalizeSessionInputRaw 归一化输入意图取值。
// Python: session_input.py normalize（内部闭包）
func normalizeSessionInputRaw(value any) string {
	if mode, ok := value.(SessionInputMode); ok {
		return string(mode)
	}
	text, _ := value.(string)
	return strings.ToLower(strings.TrimSpace(text))
}

// isStringValue 判断是否为字符串。
func isStringValue(value any) bool {
	_, ok := value.(string)
	return ok
}

// isEmptySlice 判断是否为空切片（nil 或长度 0）。
func isEmptySlice(value any) bool {
	items, ok := value.([]any)
	if !ok {
		return false
	}
	return len(items) == 0
}

var (
	// chatTurnMethods 属于一次聊天轮次的方法。
	// Python: runtime/request.py CHAT_TURN_METHODS
	chatTurnMethods = map[schema.ReqMethod]struct{}{
		schema.ReqMethodChatSend:   {},
		schema.ReqMethodChatResume: {},
		schema.ReqMethodChatAnswer: {},
	}

// SessionWorkKind 在 Runtime 边界给一次会话工作分类。
// Python: runtime/service.py AgentRuntime.session_work_kind（:3868-3921）
//
// 返回 nil 表示"不归 Runtime 管"，调用方走门面 SessionManager 队列。
func SessionWorkKind(request *schema.AgentRequest, background bool) *sessionpkg.SessionWorkKind {
	if request == nil || request.SessionID == nil || strings.TrimSpace(*request.SessionID) == "" {
		return nil
	}
	params := parseWorkKindParams(request)

	// 步骤 1：团队判定优先
	if requestTargetsTeam(request, params) {
		if _, isChatTurn := chatTurnMethods[request.ReqMethod]; !isChatTurn {
			return nil
		}
		if teamBackgroundTrigger(params) != "" {
			return kindPtr(sessionpkg.SessionWorkKindTeamBackground)
		}
		if IsInterruptResumeRequest(params) {
			return kindPtr(sessionpkg.SessionWorkKindTeamControl)
		}
		return kindPtr(sessionpkg.SessionWorkKindTeamRequest)
	}

	// 步骤 2：后台分支
	if background {
		if _, isChatTurn := chatTurnMethods[request.ReqMethod]; isChatTurn &&
			isSessionMessageParams(params) && isSingleAgentSessionMode(params) {
			return kindPtr(sessionpkg.SessionWorkKindSessionMessage)
		}
		return nil
	}

	// 步骤 3：单 agent 会话模式门卫
	if !isSingleAgentSessionMode(params) {
		return nil
	}

	// 步骤 4：中断续答
	if IsInterruptResumeRequest(params) {
		return kindPtr(sessionpkg.SessionWorkKindControlInput)
	}

	// 步骤 5：目标命令
	if request.ReqMethod == schema.ReqMethodCommandGoal {
		action := strings.ToLower(strings.TrimSpace(stringParam(params, "action")))
		if action == "" {
			action = "get"
		}
		if action == "set" || action == "resume" {
			return kindPtr(sessionpkg.SessionWorkKindGoalStream)
		}
		return kindPtr(sessionpkg.SessionWorkKindGoalControl)
	}

	// 步骤 6：非聊天轮次
	if _, isChatTurn := chatTurnMethods[request.ReqMethod]; !isChatTurn {
		return nil
	}

	// 步骤 7：目标挂载
	if attachGoal, _ := params["attach_goal"].(bool); attachGoal {
		return kindPtr(sessionpkg.SessionWorkKindGoalAttach)
	}

	// 步骤 8：补充输入
	if ResolveSessionInputMode(params) != nil {
		return kindPtr(sessionpkg.SessionWorkKindSessionInput)
	}

	// 步骤 9：兜底
	if request.IsStream {
		return kindPtr(sessionpkg.SessionWorkKindChatStream)
	}
	return kindPtr(sessionpkg.SessionWorkKindChatUnary)
}

// UsesSessionRuntime 这次执行是否由 Session Runtime 持有。
// Python: runtime/service.py uses_session_runtime（:3843）
func UsesSessionRuntime(request *schema.AgentRequest, background bool) bool {
	return SessionWorkKind(request, background) != nil
}

// parseWorkKindParams 解析请求参数（解析失败按空 map）。
func parseWorkKindParams(request *schema.AgentRequest) map[string]any {
	if request == nil || len(request.Params) == 0 {
		return map[string]any{}
	}
	out := map[string]any{}
	if err := json.Unmarshal(request.Params, &out); err != nil {
		return map[string]any{}
	}
	return out
}

func stringParam(params map[string]any, key string) string {
	value, _ := params[key].(string)
	return value
}

// requestTargetsTeam 解析团队准入：请求参数无 mode 时回退到会话元数据。
// Python: runtime/service.py _request_targets_team（:3753）
func requestTargetsTeam(request *schema.AgentRequest, params map[string]any) bool {
	rawMode := stringParam(params, "mode")
	workMode := stringParam(params, "work_mode")
	if strings.TrimSpace(rawMode) == "" {
		if request.SessionID != nil {
			sessionID := strings.TrimSpace(*request.SessionID)
			if sessionID != "" {
				metadata := swarmsession.GetSessionMetadata(sessionID)
				if metadata != nil {
					if mode, ok := metadata["mode"].(string); ok {
						rawMode = mode
					}
					if stored, ok := metadata["work_mode"].(string); ok && stored != "" {
						workMode = stored
					}
				} else {
					logger.Warn(logComponent).Str("session_id", sessionID).
						Msg("Runtime admission could not read session metadata")
				}
			}
		}
	}
	canonical := resolveWorkKindCanonicalMode(rawMode, workMode)
	return schema.IsTeamMode(canonical)
}

// resolveWorkKindCanonicalMode 解析请求的运行模式为 canonical 形式。
// Python: runtime/request.py resolve_agent_request_mode（返回三元组，此处只取 canonical）
func resolveWorkKindCanonicalMode(rawMode, workMode string) string {
	raw := strings.TrimSpace(rawMode)
	if raw == "" {
		raw = "agent"
	}
	if strings.HasPrefix(raw, "agent") || strings.HasPrefix(raw, "team") || strings.HasPrefix(raw, "code") {
		return schema.CanonicalizeMode(raw)
	}
	// 仅给 work_mode 时按单 agent 处理
	if strings.TrimSpace(workMode) != "" {
		return schema.CanonicalizeMode("agent." + workMode)
	}
	return schema.CanonicalizeMode(raw)
}

// isSingleAgentSessionMode 是否为单 agent 会话模式。
// Python: runtime/service.py _is_single_agent_session_mode
func isSingleAgentSessionMode(params map[string]any) bool {
	canonical := resolveWorkKindCanonicalMode(
		stringParam(params, "mode"), stringParam(params, "work_mode"))
	return schema.IsSingleAgentMode(canonical)
}

// isSessionMessageParams 参数里是否带跨会话消息载荷。
// Python: params.get(SESSION_MESSAGE_INTERNAL_KEY) 是 dict
func isSessionMessageParams(params map[string]any) bool {
	_, ok := params[sessionMessageInternalKey].(map[string]any)
	return ok
}

// teamBackgroundTrigger 团队自动化触发类型（heartbeat / cron / 空串）。
// Python: runtime/service.py _team_background_trigger
func teamBackgroundTrigger(params map[string]any) string {
	if automation, ok := params["automation"].(map[string]any); ok {
		if automation["kind"] == "heartbeat" {
			return "heartbeat"
		}
	}
	if _, ok := params["cron"].(map[string]any); ok {
		return "cron"
	}
	return ""
}

// SessionMessageInternalKey 跨会话消息在 params 中的内部键。
// Python: gateway/common/... SESSION_MESSAGE_INTERNAL_KEY
const sessionMessageInternalKey = "_session_message"

// CloseSession 停止并释放指定会话这一代的任务处理器，返回是否有运行时被清理。
// Python: session_manager.py close_session
//
// 返回值语义与协调器 CloseSession 不同：本方法回答"清没清到东西"。
func (m *SessionManager) CloseSession(
	ctx context.Context, sessionID string, waitTimeout *float64,
) (bool, error) {
	m.mu.Lock()
	closing := m.closingTasks[sessionID]
	processor, hadProcessor := m.sessionProcessors[sessionID]
	task := m.sessionTasks[sessionID]
	queue := m.sessionQueues[sessionID]
	_, hadPriority := m.sessionPriorities[sessionID]
	hadRuntime := len(closing) > 0 || task != nil || hadPriority || queue != nil || hadProcessor

	delete(m.sessionProcessors, sessionID)
	delete(m.sessionQueues, sessionID)
	delete(m.sessionPriorities, sessionID)
	delete(m.sessionTasks, sessionID)
	if queue != nil {
		queue.drainCancelled()
	}
	m.mu.Unlock()

	if processor != nil {
		close(processor)
	}
	if task != nil && task.cancel != nil {
		task.cancel()
	}
	if processor != nil || task != nil {
		timeout := defaultCloseWaitTimeout
		if waitTimeout != nil {
			timeout = *waitTimeout
		}
		timer := time.NewTimer(time.Duration(timeout * float64(time.Second)))
		defer timer.Stop()
		select {
		case <-timer.C:
			logger.Warn(logComponent).Str("session_id", sessionID).
				Msg("[SessionManager] close session wait timeout")
		case <-ctx.Done():
		default:
		}
	}
	return hadRuntime, nil
}

// EnsureSessionProcessor 启动会话的任务处理器（幂等）。
// Python: session_manager.py ensure_session_processor
func (m *SessionManager) EnsureSessionProcessor(ctx context.Context, sessionID string) {
	m.mu.Lock()
	if _, exists := m.sessionProcessors[sessionID]; exists {
		m.mu.Unlock()
		return
	}
	queue, ok := m.sessionQueues[sessionID]
	if !ok {
		queue = &taskQueue{}
		heap.Init(queue)
		m.sessionQueues[sessionID] = queue
	}
	done := make(chan struct{})
	m.sessionProcessors[sessionID] = done
	m.mu.Unlock()

	go m.runProcessor(ctx, sessionID, queue, done)
}

// SubmitTask 把任务排入会话队列，不等待结果。
// Python: session_manager.py submit_task
func (m *SessionManager) SubmitTask(
	ctx context.Context, sessionID string, fn func(context.Context) (any, error), priority *int64,
) {
	m.EnsureSessionProcessor(ctx, sessionID)
	m.mu.Lock()
	value := m.nextPriorityLocked(sessionID)
	if priority != nil {
		value = *priority
	}
	queue := m.sessionQueues[sessionID]
	task := &sessionTask{fn: fn, ctx: ctx, result: make(chan taskResult, 1), priority: value}
	m.sessionTasks[sessionID] = task
	if queue != nil {
		heap.Push(queue, task)
	}
	done := m.sessionProcessors[sessionID]
	m.mu.Unlock()
	_ = done
}

// SubmitAndWait 把任务排入会话队列并等待结果。
// Python: session_manager.py submit_and_wait
func (m *SessionManager) SubmitAndWait(
	ctx context.Context, sessionID string, fn func(context.Context) (any, error),
) (any, error) {
	m.EnsureSessionProcessor(ctx, sessionID)
	m.mu.Lock()
	priority := m.nextPriorityLocked(sessionID)
	queue := m.sessionQueues[sessionID]
	task := &sessionTask{fn: fn, ctx: ctx, result: make(chan taskResult, 1), priority: priority}
	m.sessionTasks[sessionID] = task
	if queue != nil {
		heap.Push(queue, task)
	}
	m.mu.Unlock()

	var result taskResult
	select {
	case result = <-task.result:
	case <-ctx.Done():
		result = taskResult{err: ctx.Err()}
	}
	// 一次性会话：结束后投哨兵让 processor 退出，避免协程泄漏
	if isOneShotSession(sessionID) {
		m.pushOneShotSentinel(sessionID)
	}
	return result.value, result.err
}

// HasSessionRuntime 会话是否还保留队列或任务状态；nil 查全表。
// Python: session_manager.py has_session_runtime
func (m *SessionManager) HasSessionRuntime(sessionID *string) bool {
	m.mu.Lock()
	defer m.mu.Unlock()
	if sessionID != nil {
		id := *sessionID
		return m.sessionTasks[id] != nil ||
			m.hasPriorityLocked(id) ||
			m.sessionQueues[id] != nil ||
			m.sessionProcessors[id] != nil ||
			len(m.closingTasks[id]) > 0
	}
	return len(m.sessionTasks) > 0 ||
		len(m.sessionPriorities) > 0 ||
		len(m.sessionQueues) > 0 ||
		len(m.sessionProcessors) > 0 ||
		len(m.closingTasks) > 0
}

// HasActiveTasks 是否有活跃的会话任务（dreaming busy_checker 用）。

func (m *SessionManager) HasActiveTasks() bool {
	m.mu.Lock()
	defer m.mu.Unlock()
	for _, task := range m.sessionTasks {
		if task != nil && !task.done() {
			return true
		}
	}
	return false

// ──────────────────────────── 非导出函数 ────────────────────────────

// runProcessor 会话处理器：串行取任务执行，收到哨兵或关闭信号退出。
// Python: session_manager.py ensure_session_processor 内的 processor 循环
func (m *SessionManager) runProcessor(
	parentCtx context.Context, sessionID string, queue *taskQueue, done chan struct{},
) {
	defer close(done)
	for {
		select {
		case <-parentCtx.Done():
			return
		default:
		}
		task := m.popTask(sessionID, queue)
		if task == nil {
			return
		}
		if task.fn == nil {
			// 一次性会话哨兵：退出并清理
			m.cleanupProcessor(sessionID, done)
			return
		}
		taskCtx, cancel := context.WithCancel(task.ctx)
		m.mu.Lock()
		task.cancel = cancel
		m.mu.Unlock()

		value, err := safeInvoke(taskCtx, task.fn)
		cancel()
		task.result <- taskResult{value: value, err: err}
	}
}

// safeInvoke 执行任务并把 panic 转为错误。
func safeInvoke(ctx context.Context, fn func(context.Context) (any, error)) (value any, err error) {
	defer func() {
		if recovered := recover(); recovered != nil {
			logger.Error(logComponent).Any("panic", recovered).Msg("session task panicked")
			err = errSessionTaskPanicked
		}
	}()
	return fn(ctx)
}

// popTask 取出下一个待执行任务（阻塞等待队列非空或处理器被关闭）。
func (m *SessionManager) popTask(sessionID string, queue *taskQueue) *sessionTask {
	for {
		m.mu.Lock()
		closed := m.sessionProcessors[sessionID] == nil
		if !closed && queue.Len() > 0 {
			task := heap.Pop(queue).(*sessionTask)
			m.mu.Unlock()
			return task
		}
		m.mu.Unlock()
		if closed {
			return nil
		}
		time.Sleep(time.Millisecond)
	}
}

// cleanupProcessor 一次性会话结束后回收该会话的全部状态。
func (m *SessionManager) cleanupProcessor(sessionID string, done chan struct{}) {
	m.mu.Lock()
	defer m.mu.Unlock()
	if m.sessionProcessors[sessionID] == done {
		delete(m.sessionProcessors, sessionID)
	}
	delete(m.sessionQueues, sessionID)
	delete(m.sessionPriorities, sessionID)
	delete(m.sessionTasks, sessionID)
}

// pushOneShotSentinel 给一次性会话投退出哨兵。
func (m *SessionManager) pushOneShotSentinel(sessionID string) {
	m.mu.Lock()
	queue := m.sessionQueues[sessionID]
	if queue == nil {
		m.mu.Unlock()
		return
	}
	heap.Push(queue, &sessionTask{priority: oneShotSentinelPriority})
	m.mu.Unlock()
}

// nextPriorityLocked 取下一个优先级（递减，实现同会话 LIFO）。调用方须持锁。
func (m *SessionManager) nextPriorityLocked(sessionID string) int64 {
	current, ok := m.sessionPriorities[sessionID]
	if !ok {
		current = 0
	}
	current--
	m.sessionPriorities[sessionID] = current
	return current
}

// hasPriorityLocked 会话是否留有优先级记录。调用方须持锁。
func (m *SessionManager) hasPriorityLocked(sessionID string) bool {
	_, ok := m.sessionPriorities[sessionID]
	return ok
}

// waitTaskDone 等待任务结束（带超时）。
func (m *SessionManager) waitTaskDone(
	ctx context.Context, sessionID string, task *sessionTask, waitTimeout *float64,
) {
	timeout := defaultCloseWaitTimeout
	if waitTimeout != nil {
		timeout = *waitTimeout
	}
	timer := time.NewTimer(time.Duration(timeout * float64(time.Second)))
	defer timer.Stop()
	select {
	case <-timer.C:
		logger.Warn(logComponent).Str("session_id", sessionID).Msg("cancel session task timeout")
	case <-ctx.Done():
	}
}

// isOneShotSession 是否为一次性会话（心跳/定时任务），其 ID 永不复用。
// Python: session_manager.py _is_oneshot_session
func isOneShotSession(sessionID string) bool {
	return strings.HasPrefix(sessionID, "health_check_") ||
		strings.HasPrefix(sessionID, "heartbeat") ||
		strings.HasPrefix(sessionID, "cron")
}

// done 任务是否已结束。
func (t *sessionTask) done() bool {
	return t == nil || t.cancel == nil && t.result == nil
}

// Len 实现 heap.Interface。
func (q *taskQueue) Len() int { return len(q.items) }

// Less 实现 heap.Interface（优先级小的先出，等值按入队顺序稳定）。
func (q *taskQueue) Less(i, j int) bool {
	return q.items[i].priority < q.items[j].priority
}

// Swap 实现 heap.Interface。
func (q *taskQueue) Swap(i, j int) {
	q.items[i], q.items[j] = q.items[j], q.items[i]
}

// Push 实现 heap.Interface。
func (q *taskQueue) Push(value any) {
	q.items = append(q.items, value.(*sessionTask))
}

// Pop 实现 heap.Interface。
func (q *taskQueue) Pop() any {
	old := q.items
	n := len(old)
	item := old[n-1]
	q.items = old[:n-1]
	return item
}

// drainCancelled 取消队列中残留任务的结果通道。
func (q *taskQueue) drainCancelled() {
	for _, task := range q.items {
		if task != nil && task.result != nil {
			select {
			case task.result <- taskResult{err: context.Canceled}:
			default:
			}
		}
	}
	q.items = nil
}

// StreamSessionInputOptions StreamSessionInput 的调用参数。
type StreamSessionInputOptions struct {
	// SessionID 会话标识
	SessionID string
	// RequestID 请求标识
	RequestID string
	// Operation 有父执行时运行的操作（接收会话渠道标识）
	Operation func(ctx context.Context, channelID string) (<-chan any, error)
	// IdleOperation 无父执行时的回落操作
	IdleOperation func(ctx context.Context) (<-chan any, error)
	// InteractionIDs 从元素提取交互标识
	InteractionIDs func(any) map[string]struct{}
	// ExpectedExecutionID 调用方期望的目标执行（nil 表示不校验）
	ExpectedExecutionID *string
}

// StreamSessionInput 跟踪一次补充输入，不占用活跃任务的调度通道。
// Python: coordinator.py stream_session_input（:576-624）
//
// 输入不恢复、不完成、不替换等待中的交互；未绑定父执行时回落 idle 分支。
// 传 expected_execution_id 时父执行必须仍然匹配：该输入不能自起一个聊天轮。
func (c *RuntimeSessionCoordinator) StreamSessionInput(
	ctx context.Context,
	opts StreamSessionInputOptions,
	emit func(any) bool,
) error {
	// 1. 会话必须打开
	record, err := c.requireOpenSession(opts.SessionID)
	if err != nil {
		return err
	}

	// 2. 取当前代次全部活跃执行
	generation := record.generation
	active := c.registry.Select(SelectOptions{
		SessionID:  opts.SessionID,
		Generation: &generation,
		ActiveOnly: true,
	})

	// 3. 任一执行在等交互应答 → 拒收
	for _, handle := range active {
		if len(c.registry.WaitingControlIDsOf(handle.ExecutionID)) > 0 {
			return &SessionInputRejectedError{
				Message: "session is waiting for an interaction answer; supplemental input was not sent",
			}
		}
	}

	// 4. 收集可作为父执行的候选
	parents := make([]*SessionExecutionHandle, 0, len(active))
	for _, handle := range active {
		if c.registry.StateOf(handle.ExecutionID) != SessionExecutionStateRunning {
			continue
		}
		switch handle.WorkKind {
		case SessionWorkKindSessionInput, SessionWorkKindGoalControl, SessionWorkKindGoalAttach:
			continue
		}
		parents = append(parents, handle)
	}

	// 5. 父执行正在被取消 → 拒收
	for _, handle := range parents {
		if c.registry.IsCancellationRequested(handle.ExecutionID) {
			return &SessionInputRejectedError{
				Message: "session execution is being cancelled",
			}
		}
	}

	// 6. 取最新一个作为父执行
	var parent *SessionExecutionHandle
	if len(parents) > 0 {
		parent = parents[len(parents)-1]
	}

	// 7. 目标执行校验
	if opts.ExpectedExecutionID != nil {
		if parent == nil || parent.ExecutionID != *opts.ExpectedExecutionID {
			return &SessionInputTargetError{
				Message: "the targeted execution has ended or changed; supplemental input was not sent",
			}
		}
	}

	// 8. 分流：有父执行走 session_input，否则走普通 chat_stream
	workKind := SessionWorkKindChatStream
	parentExecutionID := ""
	if parent != nil {
		workKind = SessionWorkKindSessionInput
		parentExecutionID = parent.ExecutionID
	}
	channelID := record.channelID
	var operation func(context.Context) (<-chan any, error)
	if parent != nil {
		operation = func(opCtx context.Context) (<-chan any, error) {
			return opts.Operation(opCtx, channelID)
		}
	} else {
		operation = opts.IdleOperation
	}
	handle, err := c.newExecution(record, opts.RequestID, workKind, parentExecutionID, nil)
	if err != nil {
		return err
	}
	return c.runStream(ctx, record, handle, operation, opts.InteractionIDs, emit)
}
```

// CleanupSessionRuntime 释放某会话在既有渠道 agent 上的内存运行时。
// Python: agent_manager.py cleanup_session_runtime（:863）
//
// 遍历该渠道缓存的每个 agent，调它的 cleanupSessionRuntime；TUI 渠道的根 agent
// 若清理后已空闲（无 pin、无借用、无残留会话运行时），一并退役，避免短命 TUI
// 进程按项目累积根 agent。
//
// 单个 agent 失败不中断其余（对齐上游聚合后统一报错的语义，但 Go 侧只告警）。
func (am *AgentManager) CleanupSessionRuntime(
	ctx context.Context, channelID, sessionID string,
) (bool, error) {
	sid := strings.TrimSpace(sessionID)
	if sid == "" {
		return false, nil
	}
	channelKey := normalizeChannelID(channelID)

	am.mu.Lock()
	channelAgents := make([]*UapClawSwarm, 0)
	if cached, ok := am.agents[channelKey]; ok {
		for _, entry := range cached {
			if entry != nil && entry.agent != nil {
				channelAgents = append(channelAgents, entry.agent)
			}
		}
	}
	am.mu.Unlock()

	cleaned := false
	for _, agent := range channelAgents {
		if agent.HasSessionRuntime(sid) {
			if err := agent.CleanupSessionRuntime(ctx, sid); err != nil {
				logger.Warn(logComponent).Err(err).
					Str("channel_id", channelKey).Str("session_id", sid).
					Msg("[AgentManager] cleanup_session_runtime failed")
				continue
			}
			cleaned = true
		}
	}
	return cleaned, nil
}

import (

// 本文件承载会话级 adapter 分层：Python 的 JiuWenSwarmDeepAdapter 按会话分身
// （_session_adapters: dict[str, JiuWenSwarmDeepAdapter]），每个子适配器一份
// DeepAgent。Go 侧此前只有单个 instance，会话仅作运行参数，导致会话删除时
// 「要释放的那一份」无处可寻。本文件提供等价的分身缓存。
//
// 对应 Python: server/runtime/agent_adapter/interface_deep.py:2126 起的
// _session_adapters / _session_adapter_locks / _session_adapter_last_used。

// ──────────────────────────── 结构体 ────────────────────────────

// sessionAdapterCache 会话级子适配器缓存。
// Python: interface_deep.py _session_adapters + _session_adapter_last_used
type sessionAdapterCache struct {
	// mu 保护以下字段
	mu sync.RWMutex
	// entries 会话标识 → 子适配器
	entries map[string]map[string]any
	// lastUsed 会话标识 → 最后使用时刻
	lastUsed map[string]time.Time
}

// ──────────────────────────── 导出函数 ────────────────────────────

// newSessionAdapterCache 创建会话级缓存。
func newSessionAdapterCache() *sessionAdapterCache {
	return &sessionAdapterCache{
		entries:  map[string]map[string]any{},
		lastUsed: map[string]time.Time{},
	}
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// remember 记住某会话的子适配器。
func (c *sessionAdapterCache) remember(sessionID string, adapter map[string]any) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.entries[sessionID] = adapter
	c.lastUsed[sessionID] = time.Now()
}

// get 取某会话的子适配器（不存在返回 nil）。
func (c *sessionAdapterCache) get(sessionID string) map[string]any {
	c.mu.RLock()
	defer c.mu.RUnlock()
	return c.entries[sessionID]
}

// has 判断某会话是否有子适配器。
func (c *sessionAdapterCache) has(sessionID string) bool {
	c.mu.RLock()
	defer c.mu.RUnlock()
	_, ok := c.entries[sessionID]
	return ok
}

// touch 刷新最后使用时刻。
func (c *sessionAdapterCache) touch(sessionID string) {
	c.mu.Lock()
	defer c.mu.Unlock()
	if _, ok := c.entries[sessionID]; ok {
		c.lastUsed[sessionID] = time.Now()
	}
}

// forget 移除某会话的子适配器。
func (c *sessionAdapterCache) forget(sessionID string) {
	c.mu.Lock()
	defer c.mu.Unlock()
	delete(c.entries, sessionID)
	delete(c.lastUsed, sessionID)
}

// cleanup 移除并报告是否清到了东西。
// Python: interface_deep.py cleanup_session_adapter 的缓存清理部分
func (c *sessionAdapterCache) cleanup(sessionID string) bool {
	c.mu.Lock()
	defer c.mu.Unlock()
	_, ok := c.entries[sessionID]
	delete(c.entries, sessionID)
	delete(c.lastUsed, sessionID)
	return ok
}

// count 当前缓存的会话数。
func (c *sessionAdapterCache) count() int {
	c.mu.RLock()
	defer c.mu.RUnlock()
	return len(c.entries)
}

// fileLock 跨进程文件锁。
// Python: lifecycle.py resource_lock（fcntl.flock）
type fileLock struct {
	// path 锁文件路径
	path string
	// file 打开的锁文件句柄
	file *os.File
}

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// processLocks 进程内锁表（键为锁文件路径）
	processLocks sync.Map
)

// ──────────────────────────── 导出函数 ────────────────────────────

// ResourceLock 获取资源跨进程锁，返回释放函数。
// Python: lifecycle.py resource_lock
func ResourceLock(kind, resourceID string) (func(), error) {
	lockPath := filepath.Join(GetAgentRootDir(), "lifecycle", "locks", kind+"_"+resourceID+".lock")
	if err := os.MkdirAll(filepath.Dir(lockPath), 0o755); err != nil {
		return nil, &LifecycleError{Code: "INTERNAL_ERROR", Message: "cannot create lock directory"}
	}
	value, _ := processLocks.LoadOrStore(lockPath, &sync.Mutex{})
	processMutex := value.(*sync.Mutex)
	processMutex.Lock()

	handle, err := os.OpenFile(lockPath, os.O_CREATE|os.O_RDWR, 0o644)
	if err != nil {
		processMutex.Unlock()
		return nil, &LifecycleError{Code: "INTERNAL_ERROR", Message: "cannot open lock file"}
	}
	if err := lockFile(handle); err != nil {
		_ = handle.Close()
		processMutex.Unlock()
		return nil, err
	}
	return func() {
		unlockFile(handle)
		_ = handle.Close()
		processMutex.Unlock()
	}, nil
}

// ReadJSON 读 JSON 文件（不存在或非法返回空 map）。
// Python: lifecycle.py read_json
func ReadJSON(path string) map[string]any {
	raw, err := os.ReadFile(path)
	if err != nil {
		return map[string]any{}
	}
	out := map[string]any{}
	if err := json.Unmarshal(raw, &out); err != nil {
		return map[string]any{}
	}
	return out
}

// AtomicJSON 原子写 JSON（临时文件 + rename）。
// Python: lifecycle.py atomic_json
func AtomicJSON(path string, value map[string]any) error {
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return err
	}
	raw, err := json.Marshal(value)
	if err != nil {
		return err
	}
	temp, err := os.CreateTemp(filepath.Dir(path), ".lifecycle-*.tmp")
	if err != nil {
		return err
	}
	tempName := temp.Name()
	if _, err := temp.Write(raw); err != nil {
		_ = temp.Close()
		_ = os.Remove(tempName)
		return err
	}
	if err := temp.Close(); err != nil {
		_ = os.Remove(tempName)
		return err
	}
	return os.Rename(tempName, path)
}

// State 读资源操作记录（返回 operation 段与顶层字段的合并视图）。

	raw := ReadJSON(ResourcePath(kind, resourceID))
	merged := map[string]any{}
	for key, value := range raw {
		merged[key] = value
	}
	if operation, ok := raw["operation"].(map[string]any); ok {
		for key, value := range operation {
			merged[key] = value
		}
	}
	return merged
}

// SessionDeleteRuntime 本服务依赖的 Runtime 删除边界。
// Python: session_archive.py self.runtime.delete_session
type SessionDeleteRuntime interface {
	// DeleteSession 删除一个持久会话
	DeleteSession(ctx context.Context, channelID, sessionID string) error
}

// Service 会话归档与删除服务。
// Python: session_archive.py SessionArchiveService
type Service struct {
	// mu 保护以下字段
	mu sync.Mutex
	// runtime Runtime 删除边界
	runtime SessionDeleteRuntime
	// finalizing 会话 → 在飞收尾取消函数
	finalizing map[string]context.CancelFunc
	// ownerID 本进程的操作 owner 标识
	ownerID string
	// finalizeSem 并行收尾名额
	finalizeSem chan struct{}
}

// ──────────────────────────── 常量 ────────────────────────────

const (
	// MaxCleanupAttempts 收尾最大尝试次数
	// Python: session_archive.py _MAX_CLEANUP_ATTEMPTS
	MaxCleanupAttempts = 5
	// MaxParallelFinalizers 并行收尾上限
	// Python: session_archive.py _MAX_PARALLEL_FINALIZERS
	MaxParallelFinalizers = 4
)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// cleanupBackoffSeconds 退避序列（秒）
	// Python: session_archive.py _CLEANUP_BACKOFF_SECONDS
	cleanupBackoffSeconds = []float64{5.0, 15.0, 60.0, 300.0}
	// errStubFailure 测试用失败
	errStubFailure = errors.New("stub failure")
)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewService 创建归档服务。
// Python: SessionArchiveService.__init__
func NewService(runtime SessionDeleteRuntime) *Service {
	return &Service{
		runtime:     runtime,
		finalizing:  map[string]context.CancelFunc{},
		finalizeSem: make(chan struct{}, MaxParallelFinalizers),
	}
}

// CleanupBackoff 第 attempts 次失败后的退避秒数。
// Python: session_archive.py _cleanup_backoff
func CleanupBackoff(attempts int) float64 {
	index := attempts - 1
	if index < 0 {
		index = 0
	}
	if index >= len(cleanupBackoffSeconds) {
		index = len(cleanupBackoffSeconds) - 1
	}
	return cleanupBackoffSeconds[index]
}

// Close 停止全部在飞收尾与恢复循环。
// Python: SessionArchiveService.close
func (s *Service) Close() {
	s.mu.Lock()
	cancels := make([]context.CancelFunc, 0, len(s.finalizing))
	for _, cancel := range s.finalizing {
		cancels = append(cancels, cancel)
	}
	s.finalizing = map[string]context.CancelFunc{}
	s.mu.Unlock()
	for _, cancel := range cancels {
		cancel()
	}
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// spawnFinalizer 启动（或复用）某会话的后台收尾。
// Python: session_archive.py _spawn_finalizer
func (s *Service) spawnFinalizer(
	parentCtx context.Context, sessionID, action, channelID, operationID string,
) {
	s.mu.Lock()
	if _, exists := s.finalizing[sessionID]; exists {
		s.mu.Unlock()
		return
	}
	ctx, cancel := context.WithCancel(parentCtx)
	s.finalizing[sessionID] = cancel
	s.mu.Unlock()

	go func() {
		defer func() {
			s.mu.Lock()
			delete(s.finalizing, sessionID)
			s.mu.Unlock()
			cancel()
		}()
		s.finalizeSem <- struct{}{}
		defer func() { <-s.finalizeSem }()
		_ = s.finalizeWithRetry(ctx, sessionID, action, channelID, operationID)
	}()
}

// finalizeWithRetry 有界重试收尾。
// Python: session_archive.py _finalize_session
func (s *Service) finalizeWithRetry(
	ctx context.Context, sessionID, action, channelID, operationID string,
) error {
	var lastErr error
	for attempt := 1; attempt <= MaxCleanupAttempts; attempt++ {
		lifecycle.Update("session", sessionID, map[string]any{
			"cleanup_attempts": attempt,
			"phase":            "stop_sessions",
		})
		lastErr = s.FinalizeSession(ctx, sessionID, action, channelID, operationID)
		if lastErr == nil {
			return nil
		}
		if ctx.Err() != nil {
			s.saveCleanupFailure(sessionID, operationID, "operation cancelled", attempt, 0, true)
			return ctx.Err()
		}
		nextAt := float64(nowSeconds()) + CleanupBackoff(attempt)
		s.saveCleanupFailure(sessionID, operationID, lastErr.Error(), attempt, nextAt, false)
	}
	s.abandonCleanup(sessionID, operationID)
	return lastErr
}

import (
	"context"

// ──────────────────────────── 导出函数 ────────────────────────────

// FinalizeSession 完成一次会话收尾（一次尝试）。
// Python: session_archive.py _finalize_session_body 的 delete 分支
//
// 步骤：挂写围栏 → 调 Runtime 删除 → 落 runtime_deleted 边界。
func (s *Service) FinalizeSession(
	ctx context.Context, sessionID, action, channelID, operationID string,
) error {
	if action != "delete" {
		return nil
	}
	if err := lifecycle.FenceWrites("session", sessionID); err != nil {
		return err
	}
	lifecycle.Update("session", sessionID, map[string]any{"phase": "delete_directory"})
	if s.runtime == nil {
		return fmt.Errorf("session runtime is unavailable")
	}
	if err := s.runtime.DeleteSession(ctx, channelID, sessionID); err != nil {
		return err
	}
	// 先落这个边界，好让删除后重试不再重做破坏性工作
	lifecycle.Update("session", sessionID, map[string]any{
		"phase":           "cleanup_mailbox",
		"runtime_deleted": true,
		"stop_pending":    false,
	})
	lifecycle.Complete("session", sessionID, 0, false, map[string]any{
		"session_id": sessionID,
		"ok":         true,
	})
	return nil
}

// AcceptDelete 接受一次删除请求并立刻返回。
// Python: session_archive.py _accept_session_cleanup
//
// 返回的 payload 是给 Gateway 的完成契约：accepted/finalizing/stop_pending
// 让调用方能区分"已接受"与"已删完"。
func (s *Service) AcceptDelete(
	ctx context.Context, sessionID, channelID, projectID string,
) map[string]any {
	operation := lifecycle.Begin("session", "delete", sessionID, 0)
	_ = operation
	op := lifecycle.State("session", sessionID)
	operationID, _ := op["operation_id"].(string)
	lifecycle.Update("session", sessionID, map[string]any{
		"channel_id": channelID,
		"project_id": projectID,
	})
	lifecycle.MarkFinalizing("session", sessionID, true)
	// 非阻塞围栏：等消费者退出放到后台 finalizer
	_ = lifecycle.FenceWrites("session", sessionID)
	s.spawnFinalizer(ctx, sessionID, "delete", channelID, operationID)
	return map[string]any{
		"session_id":   sessionID,
		"ok":           true,
		"accepted":     true,
		"finalizing":   true,
		"stop_pending": true,
		"project_id":   projectID,
	}
}

// 每轮扫 lifecycle/resources，凡未完成、可重试且退避已到的会话重新拉起收尾。
func (s *Service) StartRecovery(ctx context.Context) {
	go func() {
		ticker := time.NewTicker(recoverIdleInterval)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				s.recoverOnce(ctx)
			}
		}
	}()
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// recoverOnce 执行一轮恢复扫描。
func (s *Service) recoverOnce(ctx context.Context) {
	for _, snapshot := range lifecycle.EventSnapshots() {
		sessionID, _ := snapshot["session_id"].(string)
		operation, _ := snapshot["operation"].(map[string]any)
		if sessionID == "" || operation == nil {
			continue
		}
		status, _ := operation["status"].(string)
		if status == "completed" || status == "abandoned" {
			continue
		}
		if retryable, ok := operation["retryable"].(bool); ok && !retryable {
			continue
		}
		if nextAt, ok := operation["cleanup_next_at"].(float64); ok && nowSeconds() < nextAt {
			continue
		}
		s.mu.Lock()
		_, running := s.finalizing[sessionID]
		s.mu.Unlock()
		if running {
			continue
		}
		kind, _ := operation["kind"].(string)
		if kind != "archive" && kind != "delete" {
			continue
		}
		channelID, _ := operation["channel_id"].(string)
		operationID, _ := operation["operation_id"].(string)
		logger.Info(logComponent).Str("session_id", sessionID).
			Msg("session lifecycle recovery resumed")
		s.spawnFinalizer(ctx, sessionID, kind, channelID, operationID)
	}
}

// saveCleanupFailure 落一次可重试失败。
// Python: session_archive.py _save_cleanup_failure
func (s *Service) saveCleanupFailure(
	sessionID, operationID, reason string, attempts int, nextAt float64, exhausted bool,
) {
	status := "failed"
	if exhausted {
		status = "abandoned"
	}
	lifecycle.Update("session", sessionID, map[string]any{
		"status":          status,
		"retryable":       !exhausted,
		"cleanup_next_at": nextAt,
		"errors":          []string{reason},
		"cleanup_attempts": attempts,
	})
}

// abandonCleanup 放弃收尾（达到尝试上限）。
// Python: session_archive.py _abandon_cleanup
func (s *Service) abandonCleanup(sessionID, operationID string) {
	lifecycle.Update("session", sessionID, map[string]any{
		"status":    "abandoned",
		"retryable": false,
	})
	logger.Warn(logComponent).Str("session_id", sessionID).
		Msg("session cleanup abandoned after attempt limit")
}

	_, archivedRoot := lifecycle.SessionPaths("placeholder")

		}

		left, _ := items[i]["session_id"].(string)

		return left < right
	})
