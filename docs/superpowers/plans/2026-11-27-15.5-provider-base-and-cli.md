## Architecture

- **三个调用方向，三套机制**（spec §1.2）：
  - 宿主 → harness：`harnessprotocol.HarnessProtocol` 接口（15.1 已定义，14 方法）
  - harness → provider 钩子：`providerHooks` 接口字段（本节新建，构造期注入，动态派发）
  - provider → harness 能力：**嵌入** `*SerializedTurnHarness`（静态方法提升）
- **为什么第二套必须存在**：Go 的方法提升只能从外向内。基座方法体内 `h.ExecuteTurn(...)` 会静态绑定到 `*SerializedTurnHarness` 的版本，provider 覆盖永不生效——编译通过、无告警、功能静默失效（`go-embedding-virtual-dispatch-trap`，2026-11-08 实施 25.0c 时踩到，`claim()` 永远拿到基类空实现）。因此基座内部不能持「基座类型的 self」，必须持接口字段。
- **provider 的典型形态**（15.6-15.9 会照此写）：

```go
type NativeHarness struct {
```

- **锁序原则**（spec Q10/D4）：`commandLock` / `lifecycleLock` 的临界区**只包状态字段读写**，绝不包 `Transition` / `Emit` / `hooks.ExecuteTurn` 等可能阻塞的调用。Python 的 `async with lock:` 内 `await` 依赖「asyncio.Lock 是协程级、等待挂起协程而非线程」；Go 的 `sync.Mutex` 是线程级，持锁等待必然死锁。

```go
// ParseApprovalText 从人打的字里读出（批准、记住、反馈）三元组。
// Python: answers.py:70-80 _parse_approval_text
//
// 逐字对齐：先 TrimSpace（Python 的 text.strip()），再转小写匹配；
// 匹配失败时把**去空白后的原文**当反馈返回（不是小写形式）。
func ParseApprovalText(text string) (approved bool, remember bool, feedback string) {
	word := strings.TrimSpace(text)
	lowered := strings.ToLower(word)
	if alwaysWords[lowered] {
		return true, true, ""
	}
	if allowWords[lowered] {
		return true, false, ""
	}
	if denyWords[lowered] {
		return false, false, ""
	}
	return false, false, word
}

// Python: answers.py:83-128 interrupt_answer
//
// 四步（顺序不可调，对齐上游）：
//  1. approved 给了但 kind 不是审批类 → error
//  2. approved 与 text 都没给 → error
//  3. kind 不是审批类 → 直接返回 text
//  4. approved 没给 → 用 text 解析出 approved/remember/feedback
//
// 然后按 kind 分发两种形状：skill_evolution_approval 的 {"action": ...}
// 与其余审批类的 {"approved", "feedback", "auto_confirm"}。
func InterruptAnswerFor(kind string, opts InterruptAnswerOptions) (any, error) {
	approved, remember, feedback := opts.Approved, opts.Remember, opts.Feedback

	// 步骤 1（answers.py:108-109）
	if approved != nil && !IsApprovalKind(kind) {
		return nil, &AnswerShapeError{Kind: kind, Reason: "interrupts of this kind are not approvals; answer them with text"}
	}
	// 步骤 2（answers.py:110-111）
	if approved == nil && opts.Text == nil {
		return nil, &AnswerShapeError{Kind: kind, Reason: "pass approved or text"}
	}
	// 步骤 3（answers.py:112-113）
	if !IsApprovalKind(kind) {
		return *opts.Text, nil
	}
	// 步骤 4（answers.py:114-116）：从文本解析
	if approved == nil {
		parsedApproved, parsedRemember, parsedFeedback := ParseApprovalText(*opts.Text)
		approved = &parsedApproved
		remember = parsedRemember
		if feedback == "" {
			feedback = parsedFeedback
		}
	}
	// skill_evolution_approval 的 {"action": ...}（answers.py:117-120）
	if kind == saschema.InterruptKindSkillEvolutionApproval {
		if *approved {
			action := "allow_once"
			if remember {
				action = "allow_always"
			}
			return map[string]any{"action": action}, nil
		}
		return map[string]any{"action": "reject", "feedback": feedback}, nil
	}
	// 其余审批类的 {"approved", "feedback", "auto_confirm"}（answers.py:121-128）
	answer := map[string]any{
		"approved":     *approved,
		"feedback":     feedbackIfApproved(*approved, feedback),
		"auto_confirm": *approved && remember,
	}
	if *approved && remember {
		answer["persist_allow"] = false
	}
	return answer, nil
}

// Python: answers.py:123 `"feedback": "" if approved else feedback`
func feedbackIfApproved(approved bool, feedback string) string {
	if approved {
		return ""
	}
	return feedback
}
```

```go
// 把协议输入渲染成大多数 SDK 接受的一段纯文本。

// ──────────────────────────── 导出函数 ────────────────────────────

// HarnessInputText 把 HarnessInput 渲染成一个提示词字符串。
// Python: inputs.py:14-27 harness_input_text
//
// 三分支（顺序不可调）：
//  1. 字符串直通
//  2. 非空的 {"type":"text","text":...} 块列表 → 空行拼接
//  3. 其余 JSON 值 → 紧凑序列化（结构化输入不得静默退化成 str(dict)）
func HarnessInputText(content harnessprotocol.HarnessInput) string {
	value := harnessprotocol.JsonValueToBuiltin(content.Content)
	if text, ok := value.(string); ok {
		return text
	}
	if list, ok := value.([]any); ok && len(list) > 0 && allTextBlocks(list) {
		parts := make([]string, 0, len(list))
		for _, block := range list {
			parts = append(parts, block.(map[string]any)["text"].(string))
		}
		return strings.Join(parts, "\n\n")
	}
	// Python: json.dumps(ensure_ascii=False, separators=(",", ":"))
	return marshalCompactNoEscape(value)
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// allTextBlocks 每个块都是 {"type":"text","text":<string>}。
// Python: inputs.py:30-31 _is_text_block
func allTextBlocks(list []any) bool {
	for _, block := range list {
		mapping, ok := block.(map[string]any)
		if !ok {
			return false
		}
		if mapping["type"] != "text" {
			return false
		}
		if _, ok := mapping["text"].(string); !ok {
			return false
		}
	}
	return true
}
```

```go
// marshalCompactNoEscape 紧凑 JSON 序列化，不转义 HTML 字符。
// 对应 Python json.dumps(ensure_ascii=False, separators=(",", ":"))：
// 无空格分隔、中文与 <>& 原样输出。
func marshalCompactNoEscape(value any) string {
	var buf bytes.Buffer
	encoder := json.NewEncoder(&buf)
	encoder.SetEscapeHTML(false)
	if err := encoder.Encode(value); err != nil {
		return ""
	}
	return strings.TrimSuffix(buf.String(), "\n")
}
```

```go
package harness_providers

import (
	"sort"
	"strings"
)

// 本文件对应 Python openjiuwen/harness_providers/mcp_naming.py（49 行）：
// provider 声明「裸工具名如何解析到它的 MCP 工具」。
//
// 宿主按自己注册时的名字写提示词，而 CLI 不按那些名字显示工具——
// 它把每个 MCP 服务器放进自己的命名空间，还可能自带一个名字相似的内置工具。
// 提示词里的裸名字于是有两种读法，小模型会挑错。
//
// 只有 provider 知道自己的 CLI 怎么拼写工具（Claude Code 的
// mcp__<server>__<tool> 与 Codex 的 mcp__<server>.<tool> 连服务器名都不一致），
// 所以由 provider 声明它，紧挨着它刚注册的服务器。宿主声明自己的裸名字属于
// 哪个服务器；两边在服务器名上会合，谁都不用知道对方的规则。

// ──────────────────────────── 常量 ────────────────────────────

// ToolPlaceholder 一条服务器声明里工具自身名字的占位符。
// Python: mcp_naming.py:24 TOOL_PLACEHOLDER
const ToolPlaceholder = "{tool}"

// ──────────────────────────── 导出函数 ────────────────────────────

// McpToolNamingPreamble 为一个会话的 MCP 服务器渲染命名声明。
// Python: mcp_naming.py:27-46 mcp_tool_naming_preamble
//
// patterns: 服务器名 → 模型调用其工具时用的名字模式，每项在工具自身名字处
// 携带 ToolPlaceholder。
//
// 返回声明块；未注册任何 MCP 服务器时返回空串。
//
// 与上游的差异（Go 特有）：Python 的 dict 保持插入顺序，Go 的 map 遍历无序，
// 故此处按服务器名升序排序后渲染，保证输出确定（否则测试与缓存不可复现）。
func McpToolNamingPreamble(patterns map[string]string) string {
	if len(patterns) == 0 {
		return ""
	}
	names := make([]string, 0, len(patterns))
	for name := range patterns {
		names = append(names, name)
	}
	sort.Strings(names)

	lines := make([]string, 0, len(names))
	for _, name := range names {
		lines = append(lines, `<server name="`+name+`" tool-name="`+patterns[name]+`"/>`)
	}
	return "<mcp-tools>\n" + strings.Join(lines, "\n") + "\n</mcp-tools>\n" +
		"Each line above is how you address that server's tools: put the tool's own name where " +
		ToolPlaceholder + " is. Where an instruction names one of those tools without that prefix, " +
		"it means the tool of that server, never a built-in tool of yours whose name resembles it."
}
```

```go
import (
	"context"
	"errors"
	"sync"

	harnessprotocol "github.com/uapclaw/uapclaw-go/internal/agentcore/harness_protocol"
)

// 本文件对应 Python openjiuwen/harness_providers/stream.py（168 行）：
// provider 共享的、有界单消费者观测流。
//
// 15.2 只落了协议侧的 HarnessEventCursor 接口（harness_protocol/stream.go），
// 并发状态机在本文件实现。

// ──────────────────────────── 结构体 ────────────────────────────

// BoundedEventBuffer 可关闭的 FIFO，必需事件用阻塞背压。
// Python: stream.py:25-84 BoundedEventBuffer
type BoundedEventBuffer struct {
	// mu 保护全部字段；cond 挂在 mu 上
	mu   sync.Mutex
	cond *sync.Cond
	// items 事件队列。用切片 + 头部前移（items = items[1:]）而非环形缓冲：
	// 容量上限固定（上游默认 1024），前移只移动 slice header，摊销 O(1)，
	// 且 Peek 就是 items[0]，最直白。
	items []harnessprotocol.HarnessEvent
	// capacity 队列容量（上游默认 1024）
	capacity int
	// closed 生产者侧是否已关闭
	closed bool
	// consumerActive 单消费者租约是否已被占用
	consumerActive bool
}

// ──────────────────────────── 全局变量 ────────────────────────────

// ErrEventBufferClosed 缓冲已关闭时 Put 的返回值。
// Python: stream.py:21-22 EventBufferClosed（RuntimeError）
//
// Python 侧全仓零捕获点（grep -rn "EventBufferClosed" 只命中 stream.py 自身），
// 实际靠 _run_provider_turn 的 except Exception 兜成 FAILED 轮次结果。
// Go 侧同样由 RunProviderTurn 兜底。
var ErrEventBufferClosed = errors.New("harness event buffer is closed")

// ──────────────────────────── 导出函数 ────────────────────────────

// NewBoundedEventBuffer 创建一个容量为 capacity 的有界缓冲。
// Python: stream.py:28-33 BoundedEventBuffer.__init__
func NewBoundedEventBuffer(capacity int) *BoundedEventBuffer {
	buffer := &BoundedEventBuffer{
		capacity: capacity,
		items:    make([]harnessprotocol.HarnessEvent, 0, capacity),
	}
	buffer.cond = sync.NewCond(&buffer.mu)
	return buffer
}

// Put 追加一条事件；缓冲满时阻塞。
// Python: stream.py:35-44 put
//
// ctx 保留位：sync.Cond.Wait() 不响应取消，关闭靠 Close 的 Broadcast 唤醒。
// 与上游一致（asyncio 侧同样靠 close() 而非 task cancel）。
func (b *BoundedEventBuffer) Put(ctx context.Context, event harnessprotocol.HarnessEvent) error {
	_ = ctx // 保留位：cond 不响应取消，见上面注释
	b.mu.Lock()
	defer b.mu.Unlock()
	for len(b.items) >= b.capacity && !b.closed {
		b.cond.Wait()
	}
	if b.closed {
		return ErrEventBufferClosed
	}
	b.items = append(b.items, event)
	b.cond.Broadcast()
	return nil
}

// Get 取出下一条事件；关闭且已排空时返回 ErrEventStreamDone。
// Python: stream.py:46-56 get（返回 HarnessEvent | None）
func (b *BoundedEventBuffer) Get(ctx context.Context) (*harnessprotocol.HarnessEvent, error) {
	_ = ctx
	b.mu.Lock()
	defer b.mu.Unlock()
	for len(b.items) == 0 && !b.closed {
		b.cond.Wait()
	}
	if len(b.items) == 0 {
		return nil, harnessprotocol.ErrEventStreamDone
	}
	event := b.items[0]
	// 清零已消费格子：items = items[1:] 只移动 header，底层数组第 0 格仍持有
	// 该事件的引用，不清零会让其 payload 一直不可回收（Go 切片前移的标准陷阱）。
	// 标准库 container/ring、net/http 的 buffer 同款做法。
	b.items[0] = harnessprotocol.HarnessEvent{}
	b.items = b.items[1:]
	b.cond.Broadcast()
	return &event, nil
}

// Peek 查看下一条事件但不释放生产者容量。
// Python: stream.py:58-64 peek
func (b *BoundedEventBuffer) Peek(ctx context.Context) (*harnessprotocol.HarnessEvent, error) {
	_ = ctx
	b.mu.Lock()
	defer b.mu.Unlock()
	for len(b.items) == 0 && !b.closed {
		b.cond.Wait()
	}
	if len(b.items) == 0 {
		return nil, harnessprotocol.ErrEventStreamDone
	}
	return &b.items[0], nil
}

// Close 关闭生产者侧并唤醒全部阻塞的读写者。幂等。
// Python: stream.py:66-71 close
func (b *BoundedEventBuffer) Close(ctx context.Context) error {
	_ = ctx // Close 本身不阻塞，ctx 仅为与其他方法签名统一
	b.mu.Lock()
	defer b.mu.Unlock()
	b.closed = true
	b.cond.Broadcast()
	return nil
}
```

```go
// Python: stream.py:73-79 cursor
//
// turnID 为 nil 表示周期级游标（perTurn=false 时忽略）；perTurn=true 时
// 表示期望的轮次 ID（nil 表示「下一轮未消费的轮次」）。
func (b *BoundedEventBuffer) Cursor(turnID *string, perTurn bool) (harnessprotocol.HarnessEventCursor, error) {
	b.mu.Lock()
	defer b.mu.Unlock()
	if b.consumerActive {
		return nil, &harnessprotocol.HarnessStateError{
			Msg: "the harness observation stream already has an active consumer",
		}
	}
	b.consumerActive = true
	return &BufferedEventCursor{
		buffer:         b,
		expectedTurnID: turnID,
		perTurn:        perTurn,
	}, nil
}

// releaseConsumer 释放消费者租约。幂等。
// Python: stream.py:81-84 release_consumer
func (b *BoundedEventBuffer) releaseConsumer() {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.consumerActive = false
}

// BufferedEventCursor 一个 BoundedEventBuffer 上的周期级或有限轮次视图。
// Python: stream.py:87-165 BufferedEventCursor
type BufferedEventCursor struct {
	// buffer 底层缓冲
	buffer *BoundedEventBuffer
	// expectedTurnID 期望的轮次 ID（nil 表示下一轮）
	expectedTurnID *string
	// perTurn 是否单轮视图（true）还是周期级（false）
	perTurn bool
	// selectedTurnID 已选定的轮次 ID（perTurn 专用）
	selectedTurnID *string
	// closed 游标是否已关闭
	closed bool
}

// Iter 返回本游标（对齐 __aiter__）。
func (c *BufferedEventCursor) Iter() harnessprotocol.HarnessEventCursor {
	return c
}

// Close 幂等释放观测消费者租约。
// Python: stream.py:161-165 aclose
//
// **与上游的偏差（spec D3）**：此处额外 Broadcast 一次，让阻塞在 Next 里的
// 协程立刻醒来并返回 ErrEventStreamDone。上游 aclose 只释放租约不唤醒，
// 阻塞者必须等 buffer.close()；这在 asyncio 里有 task 取消兜底，在 Go 里
// 是真实 goroutine 泄漏（-race 的泄漏检测会抓到），故此处修正。
func (c *BufferedEventCursor) Close(ctx context.Context) error {
	_ = ctx
	c.buffer.mu.Lock()
	if c.closed {
		c.buffer.mu.Unlock()
		return nil
	}
	c.closed = true
	c.buffer.consumerActive = false
	c.buffer.mu.Unlock()
	// 唤醒阻塞在 Get/Peek 的消费者（偏差 D3）
	c.buffer.mu.Lock()
	c.buffer.cond.Broadcast()
	c.buffer.mu.Unlock()
	return nil
}
```

- [ ] **Step 6: 实现 `Next`**

```go
// Next 返回下一条有序事件；耗尽时返回 ErrEventStreamDone。
// Python: stream.py:106-159 __anext__
//
// 周期级游标（perTurn=false）直通，逐条返回。
//
// 单轮游标（perTurn=true）的 5 步：
//  1. 未选定轮次时用 Peek 验证队首，避免 pop-then-push 与生产者的竞态
//     （若先 Get 再放回，生产者可能立刻填满释放的槽位，丢掉 STARTED）
//  2. 非 STARTED 的事件直接 Get 丢弃，继续找
//  3. 期望轮次与队首不符 → 关闭并报 HarnessStateError
//  4. 兑现 Peek：Get 真正消费掉 STARTED（期间流变化则报 HarnessProtocolError）
//  5. 事件属于已选定轮次且是终态 → 自动关闭（释放租约）
func (c *BufferedEventCursor) Next(ctx context.Context) (*harnessprotocol.HarnessEvent, error) {
	if c.isClosed() {
		return nil, harnessprotocol.ErrEventStreamDone
	}

	for {
		var event *harnessprotocol.HarnessEvent
		var err error
		if c.perTurn && c.selectedTurn() == nil {
			event, err = c.buffer.Peek(ctx)
		} else {
			event, err = c.buffer.Get(ctx)
		}
		// 耗尽
		if errors.Is(err, harnessprotocol.ErrEventStreamDone) || (err == nil && event == nil) {
			selected := c.selectedTurn()
			_ = c.Close(ctx)
			if c.perTurn && selected != nil {
				return nil, &harnessprotocol.HarnessProtocolError{
					Msg: fmt.Sprintf("harness event stream closed before turn %q terminated", *selected),
				}
			}
			return nil, harnessprotocol.ErrEventStreamDone
		}
		if err != nil {
			return nil, err
		}

		if !c.perTurn {
			return event, nil
		}

		// 步骤 1-2：未选定轮次
		if c.selectedTurn() == nil {
			lifecycle, isLifecycle := event.Event.(*harnessprotocol.TurnLifecycleEvent)
			if !isLifecycle || lifecycle.Kind != harnessprotocol.TurnEventKindStarted {
				// 非 STARTED：真正消费掉（Peek 时未删除）并继续
				if _, dropErr := c.buffer.Get(ctx); dropErr != nil {
					_ = c.Close(ctx)
					return nil, dropErr
				}
				continue
			}
			// 步骤 3：期望轮次比对
			if c.expectedTurnID != nil && event.TurnID != *c.expectedTurnID {
				_ = c.Close(ctx)
				return nil, &harnessprotocol.HarnessStateError{
					Msg: fmt.Sprintf(
						"requested turn is not the next unconsumed turn: expected %q, found %q",
						*c.expectedTurnID, event.TurnID,
					),
				}
			}
			// 步骤 4：兑现 Peek
			consumed, getErr := c.buffer.Get(ctx)
			if getErr != nil || consumed == nil {
				_ = c.Close(ctx)
				return nil, &harnessprotocol.HarnessProtocolError{
					Msg: "harness event stream changed while selecting a turn",
				}
			}
			event = consumed
			turnID := event.TurnID
			c.setSelectedTurn(&turnID)
		}

		// 步骤 5：终态自动关闭
		selected := c.selectedTurn()
		if selected != nil && event.TurnID == *selected {
			if lifecycle, isLifecycle := event.Event.(*harnessprotocol.TurnLifecycleEvent); isLifecycle &&
				harnessprotocol.IsTerminalTurnEventKind(lifecycle.Kind) {
				_ = c.Close(ctx)
			}
		}
		return event, nil
	}
}
```

```go
// EmitOptions Emit 的可选关联字段。
// Python: base.py:673-694 _emit 的关键字参数
type EmitOptions struct {
	// Turn 事件所属的轮次（nil 表示不属于任何轮次）
	Turn *PendingTurn
	// ItemID 事件关于的 provider item
	ItemID string
	// ProviderSessionID 覆盖信封的 provider 会话标识（空串表示用 h.sessionID）
	ProviderSessionID string
	// CausationIDs 轮次输入消息之外的额外原因（如产生工具项的模型请求）
	CausationIDs []string
	// Timestamp 事件被观测的时刻（供 provider 为保因果序而延迟发送的事件）；
	// nil 表示用当前时间
	Timestamp *float64
}

// Emit 把一条事件放进观测流。
// Python: base.py:673-716 _emit
//
// 三步：信封组装（sequence 自增 / correlationID=turn.MessageID /
// causationIDs 去重）→ NewHarnessEvent 校验 → buffer.Put（满则阻塞背压）。
func (h *SerializedTurnHarness) Emit(ctx context.Context,
	payload harnessprotocol.HarnessEventPayload, opts EmitOptions) error {
	harnessContext := h.harnessContext
	buffer := h.eventBuffer
	if harnessContext == nil || buffer == nil {
		return &harnessprotocol.HarnessProtocolError{
			Msg: "cannot emit a " + h.card.Name + " event outside an active cycle",
		}
	}
	h.sequence++

	// causes = (turn.message_id,) + 去重后的 causation_ids
	// Python: base.py:700-702
	var causes []string
	if opts.Turn != nil {
		causes = append(causes, opts.Turn.MessageID)
	}
	for _, cause := range opts.CausationIDs {
		if opts.Turn != nil && cause == opts.Turn.MessageID {
			continue
		}
		causes = append(causes, cause)
	}

	timestamp := float64(time.Now().UnixNano()) / 1e9
	if opts.Timestamp != nil {
		timestamp = *opts.Timestamp
	}

	var turnID *string
	var correlationID *string
	if opts.Turn != nil {
		turnIDValue := opts.Turn.TurnID
		turnID = &turnIDValue
		messageIDValue := opts.Turn.MessageID
		correlationID = &messageIDValue
	}

	event, err := harnessprotocol.NewHarnessEvent(harnessprotocol.HarnessEventOptions{
		Sequence:          int(h.sequence),
		Timestamp:         timestamp,
		Event:             payload,
		HostSessionID:     harnessContext.HostSessionID,
		AgentID:           harnessContext.AgentID,
		ProviderSessionID: emitProviderSession(opts.ProviderSessionID, h.sessionID),
		TurnID:            turnID,
		ItemID:            opts.ItemID,
		CorrelationID:     correlationID,
		CausationIDs:      causes,
	})
	if err != nil {
		return err
	}
	return buffer.Put(ctx, event)
}

// transition 相位变更并发射 StateChangedEvent；同相位不发射。
// Python: base.py:666-671 _transition
//
// ⚠️ 不持 commandLock：emit 可能阻塞在满缓冲上，持锁会与监督协程死锁
// （偏差 D4 的通用原则）。
func (h *SerializedTurnHarness) transition(ctx context.Context, newState harnessprotocol.HarnessState) error {
	oldState := h.State()
	if oldState == newState {
		return nil
	}
	stored := newState
	h.state.Store(&stored)
	payload, err := harnessprotocol.NewStateChangedEvent(oldState, newState)
	if err != nil {
		return err
	}
	return h.Emit(ctx, payload, EmitOptions{})
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// emitProviderSession 解析信封的 provider 会话标识。
// Python: base.py:710 `provider_session_id or self._session_id`
//
// Python 的 or 是真值判断：空串也 fallback。Go 若只用 nil 判断，
// 传入 ptr("") 不会 fallback，与上游不一致。
func emitProviderSession(override string, sessionID *string) string {
	if override != "" {
		return override
	}
	if sessionID != nil {
		return *sessionID
	}
	return ""
}
```

```go
func (h *SerializedTurnHarness) Start(ctx context.Context, harnessContext harnessprotocol.HarnessContext) error {

	h.pending = nil

	h.supervisor = nil
	h.stopHandle = nil
	h.stopping = false

	h.latestCheckpoint = nil
	h.checkpointSequence = 0
```

```go
// superviseTurns 串行执行待决轮次。
// Python: base.py:487-531 _supervise_turns
//
// ⚠️ 偏差 D4：Python 在 base.py:526 持命令锁 await _transition(IDLE)。
// Go 的 sync.Mutex 是线程级的，持锁等 Emit → buffer.Put 会与消费者死锁。
// 因此把 IDLE 转移移到锁外：锁内只算 needIdle，出锁后发射。
func (h *SerializedTurnHarness) superviseTurns(ctx context.Context, handle *supervisorHandle) {
	defer close(handle.done)
	defer h.clearSupervisor(handle)

	for {
		h.commandLock.Lock()
		var active *PendingTurn
		var queued []*PendingTurn
		if h.stopping {
			queued = h.pending
			h.pending = nil
			active = nil
		} else if len(h.pending) == 0 {
			h.commandLock.Unlock()
			return
		} else {
			active = h.pending[0]
			h.pending = h.pending[1:]
			h.activeTurn = active
		}
		selection := h.pendingModelSelection
		h.pendingModelSelection = nil
		h.commandLock.Unlock()

		if active == nil {
			h.abortQueuedTurns(ctx, queued)
			return
		}

		if err := h.transition(ctx, harnessprotocol.HarnessStateRunning); err != nil {
			logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
				Msg("failed to transition to running")
		}
		started, err := harnessprotocol.NewTurnLifecycleEvent(harnessprotocol.TurnEventKindStarted, nil)
		if err == nil {
			_ = h.Emit(ctx, started, EmitOptions{Turn: active})
		}
		if selection != nil {
			h.applyDeferredModelSelection(ctx, selection, active)
		}

		terminalKind, result := h.runProviderTurn(ctx, active)
		if terminalKind == harnessprotocol.TurnEventKindFailed && h.tryAuthFallback(ctx, result.Error, active) {
			terminalKind, result = h.runProviderTurn(ctx, active)
		}
		terminal, err := harnessprotocol.NewTurnLifecycleEvent(terminalKind, &result)
		if err == nil {
			_ = h.Emit(ctx, terminal, EmitOptions{Turn: active})
		}

		h.commandLock.Lock()
		h.activeTurn = nil
		needIdle := false
		queued = nil
		if h.stopping {
			queued = h.pending
			h.pending = nil
		} else if len(h.pending) == 0 {
			needIdle = true
		}
		h.commandLock.Unlock()

		// 锁外发事件（偏差 D4）
		if needIdle {
			if err := h.transition(ctx, harnessprotocol.HarnessStateIdle); err != nil {
				logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
					Msg("failed to transition to idle")
			}
		}
		if len(queued) > 0 {
			h.abortQueuedTurns(ctx, queued)
			return
		}
		if h.stopping {
			return
		}
	}
}

		}

		harnessprotocol.TurnStatusFailed,
		harnessprotocol.ResultOptions{Error: &turnError})
	if buildErr != nil {
		return harnessprotocol.TurnEventKindFailed, harnessprotocol.TurnResult{}
	}
	return harnessprotocol.TurnEventKindFailed, result
```

```go
// doStop teardown 的 9 个步骤，顺序不可调。
// Python: base.py:327-353 _do_stop
func (h *SerializedTurnHarness) doStop(ctx context.Context) {
	// 步骤 1：取消全部挂起交互
	h.cancelPendingInteractionsSilently(ctx, harnessprotocol.InteractionCancelReasonHarnessStopped)
	// 步骤 2：关 provider 会话（失败仅记录，继续）
	if err := h.hooks.CloseSession(ctx); err != nil {
		logger.Error(logComponent).Str("provider", h.card.Name).Err(err).
			Msg("provider session close failed during stop")
	}
	// 步骤 3：等监督协程收敛
	h.commandLock.Lock()
	supervisor := h.supervisor
	h.commandLock.Unlock()
	if supervisor != nil {
		<-supervisor.done
	}
	// 步骤 4：取走排队轮次并清空状态
	h.commandLock.Lock()
	queued := h.pending
	h.pending = nil
	h.activeTurn = nil
	h.supervisor = nil
	h.commandLock.Unlock()
	// 步骤 5：中止排队轮次
	if len(queued) > 0 {
		h.abortQueuedTurns(ctx, queued)
	}
	// 步骤 6：转 TERMINATED
	if err := h.transition(ctx, harnessprotocol.HarnessStateTerminated); err != nil {
		logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
			Msg("failed to transition to terminated")
	}
	// 步骤 7：关事件缓冲（唤醒全部消费协程）
	if buffer := h.eventBuffer; buffer != nil {
		_ = buffer.Close(ctx)
	}
	// 步骤 8-9：清上下文与周期标记
	h.harnessContext = nil
	h.cycleStarted = false
}
```

```go
// activateAuthFallback 登录失败时把会话切到回退模型一次。
// Python: base.py:552-612 _activate_auth_fallback
//
// 门卫 1：只有 auth_required 类别才进入（请求没到达模型，重放不会重复工作）
// 门卫 2：已激活过或轮次已请求中止 → 不进入
// 门卫 3：未配置回退 → 不进入
// 步骤 1：重连到回退（失败 → 保留原失败）
// 步骤 2：宿主批准（异常视为拒绝）
// 步骤 3：宿主拒绝 → 恢复主模型（成员绝不在未记录的模型上运行）
// 步骤 4-6：置标志 + 记录端点变更 + 发 ProviderEvent + 返回 true
func (h *SerializedTurnHarness) activateAuthFallback(ctx context.Context,
	turnError *harnessprotocol.TurnError, turn *PendingTurn) (bool, error) {
	if turnError == nil || turnError.Category != "auth_required" {
		return false, nil
	}
	if h.fallbackActivated || turn.abortRequested() {
		return false, nil
	}
	payload, ok := h.hooks.AuthFallbackPayload()
	if !ok {
		return false, nil
	}
	if err := h.hooks.ReconnectForAuth(ctx, true); err != nil {
		logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
			Msg("authentication fallback activation failed")
		return false, nil
	}
	ratified, err := h.confirmProviderExtension(ctx, AuthFallbackRequestType, payload)
	if err != nil {
		// 抛异常或越协议应答的处理器没有持久化这次切换：当作拒绝
		logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
			Msg("host failed to ratify the authentication fallback")
		ratified = false
	}
	if !ratified {
		logger.Warn(logComponent).Str("provider", h.card.Name).
			Msg("host declined the authentication fallback; restoring the primary model")
		if err := h.hooks.ReconnectForAuth(ctx, false); err != nil {
			logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
				Msg("restoring the primary model failed")
			return false, nil
		}
		_ = h.hooks.OnAuthEndpointChanged(ctx, false)
		return false, nil
	}
	h.fallbackActivated = true
	if err := h.hooks.OnAuthEndpointChanged(ctx, true); err != nil {
		// 宿主已持久化切换且会话已在回退上；记账失败不应让重放落空
		logger.Warn(logComponent).Str("provider", h.card.Name).Err(err).
			Msg("recording the authentication fallback failed")
	}
	event, buildErr := harnessprotocol.NewProviderEvent(h.card.Name, AuthFallbackEventType, "1", payload)
	if buildErr == nil {
		_ = h.Emit(ctx, event, EmitOptions{Turn: turn})
	}
	return true, nil
}
```

```go
// requestInteraction 等待宿主对 request 的应答；无处理器时返回 nil。
// Python: base.py:726-738 _request_interaction
func (h *SerializedTurnHarness) requestInteraction(ctx context.Context,
	request harnessprotocol.HarnessInteractionRequest) (harnessprotocol.HarnessInteractionResponse, error) {
	harnessContext := h.harnessContext
	if harnessContext == nil || harnessContext.Interactions == nil {
		return nil, nil
	}
	handler := harnessContext.Interactions
	h.commandLock.Lock()
	h.pendingInteractions[requestIDOf(request)] = handler
	h.commandLock.Unlock()
	defer func() {
		h.commandLock.Lock()
		delete(h.pendingInteractions, requestIDOf(request))
		h.commandLock.Unlock()
	}()
	response, err := handler.Handle(ctx, request)
	if err != nil {
		return nil, err
	}
	return harnessprotocol.ValidateInteractionResponse(request, response)
}

// confirmProviderExtension 请宿主批准一个 provider 特定决定。
// Python: base.py:740-765 _confirm_provider_extension
//
// 宿主声明 PROVIDER_INTERACTION 且提供了处理器时发 ProviderInteractionRequest；
// true 表示宿主完成了请求。没有这样的宿主时决定权在 provider 自己，返回 true。
func (h *SerializedTurnHarness) confirmProviderExtension(ctx context.Context,
	requestType string, payload harnessprotocol.JsonObject) (bool, error) {
	harnessContext := h.harnessContext
	if harnessContext == nil || harnessContext.Interactions == nil {
		return true, nil
	}
	if !harnessContext.HostCapabilities[harnessprotocol.HostCapabilityProviderInteraction] {
		return true, nil
	}
	h.commandLock.Lock()
	active := h.activeTurn
	h.commandLock.Unlock()

	var turnID *string
	if active != nil {
		turnIDValue := active.TurnID
		turnID = &turnIDValue
	}
	request, err := harnessprotocol.NewProviderInteractionRequest(
		h.card.Name+":"+requestType+":"+uuid.NewString(),
		h.card.Name, requestType, "1",
		harnessprotocol.ProviderInteractionRequestOptions{
			Payload:           payload,
			ProviderSessionID: h.sessionID,
			TurnID:            turnID,
		})
	if err != nil {
		return false, err
	}
	var response harnessprotocol.HarnessInteractionResponse
	response, err = h.requestInteraction(ctx, &request)
	if err != nil {
		return false, err
	}
	if response == nil {
		return false, nil
	}
	return responseStatusOf(response) == harnessprotocol.InteractionResponseStatusCompleted, nil
}

// cancelPendingInteractions 幂等地批量撤销挂起交互。
// Python: base.py:767-774 _cancel_pending_interactions
func (h *SerializedTurnHarness) cancelPendingInteractions(ctx context.Context,
	reason harnessprotocol.InteractionCancelReason) error {
	h.commandLock.Lock()
	pending := make(map[string]harnessprotocol.HarnessInteractionHandler, len(h.pendingInteractions))
	for requestID, handler := range h.pendingInteractions {
		pending[requestID] = handler
	}
	h.pendingInteractions = make(map[string]harnessprotocol.HarnessInteractionHandler)
	h.commandLock.Unlock()

	var firstErr error
	for requestID, handler := range pending {
		option := harnessprotocol.CancelOption(func(target *harnessprotocol.InteractionCancelReason) {
			*target = reason
		})
		if err := handler.Cancel(ctx, requestID, option); err != nil {
			logger.Error(logComponent).Str("provider", h.card.Name).Str("request_id", requestID).Err(err).
				Msg("failed to cancel interaction")
			if firstErr == nil {
				firstErr = err
			}
		}
	}
	return firstErr
}

// PublishCheckpoint 记录最新快照并尽力推送给宿主 sink。
// Python: base.py:780-820 _publish_checkpoint
//
// 关键：第 2 步先保存 latestCheckpoint 再推送——即使推送失败，
// ExportCheckpoint 仍能返回最新快照。
func (h *SerializedTurnHarness) PublishCheckpoint(ctx context.Context,
	data harnessprotocol.JsonObject, reason harnessprotocol.CheckpointReason,
	schemaVersion string) (*harnessprotocol.HarnessCheckpoint, error) {
	harnessContext := h.harnessContext
	if harnessContext == nil {
		return nil, nil
	}
	h.checkpointSequence++
	checkpointSequence := h.checkpointSequence
	checkpoint, err := harnessprotocol.NewHarnessCheckpoint(harnessprotocol.CheckpointOptions{
		Provider:          h.card.Name,
		SchemaVersion:     schemaVersion,
		AgentID:           harnessContext.AgentID,
		HostSessionID:     harnessContext.HostSessionID,
		CheckpointID:      uuid.NewString(),
		Sequence:          checkpointSequence,
		Data:              data,
		ProviderSessionID: h.sessionID,
	})
	if err != nil {
		return nil, err
	}
	h.latestCheckpoint = &checkpoint

	sink := harnessContext.CheckpointSink
	if sink == nil {
		return &checkpoint, nil
	}
	receipt, saveErr := sink.Save(ctx, checkpoint, reason, h.checkpointStorageRevision)
	if saveErr != nil {
		if isCheckpointConflict(saveErr) {
			logger.Warn(logComponent).Str("provider", h.card.Name).
				Str("checkpoint_id", checkpoint.CheckpointID).
				Msg("checkpoint rejected as stale")
			return &checkpoint, nil
		}
		logger.Error(logComponent).Str("provider", h.card.Name).Err(saveErr).
			Msg("checkpoint sink failed")
		return &checkpoint, nil
	}
	revision := receipt.StorageRevision
	h.checkpointStorageRevision = &revision
	return &checkpoint, nil
}

// RestoredCheckpointData 返回属于本 provider 的 checkpoint 数据，并校验作用域。
// Python: base.py:822-836 _restored_checkpoint_data
func (h *SerializedTurnHarness) RestoredCheckpointData(harnessContext harnessprotocol.HarnessContext) (harnessprotocol.JsonObject, error) {
	checkpoint := harnessContext.Checkpoint
	if checkpoint == nil {
		return nil, nil
	}
	if checkpoint.Provider != h.card.Name {
		return nil, &harnessprotocol.HarnessProtocolError{
			Msg: "checkpoint provider " + checkpoint.Provider + " does not belong to " + h.card.Name,
		}
	}
	if checkpoint.AgentID != harnessContext.AgentID || checkpoint.HostSessionID != harnessContext.HostSessionID {

		}

	if checkpoint.Sequence > h.checkpointSequence {
		h.checkpointSequence = checkpoint.Sequence
	}
	revision := checkpoint.Revision
	h.checkpointStorageRevision = &revision
	return checkpoint.Data, nil
}
```

```go
		if _, statErr := os.Stat(candidate); statErr != nil {

func LoadAgentTemplatePackage(path string) (hschema.AgentTemplateSpec, error) {
	content, err := os.ReadFile(path)
	if err != nil {
		return hschema.AgentTemplateSpec{}, err
	}
	var envelope struct {
		PackageType string                   `json:"package_type"`
		Spec        hschema.AgentTemplateSpec `json:"-"`
	}
	if err := json.Unmarshal(content, &envelope); err != nil {
		return hschema.AgentTemplateSpec{}, err
	}
	if envelope.PackageType != agentTemplatePackageType {
		return hschema.AgentTemplateSpec{}, fmt.Errorf(
			"manifest %q has package_type %q, want %q",
			path, envelope.PackageType, agentTemplatePackageType)
	}
	var spec hschema.AgentTemplateSpec
	if err := json.Unmarshal(content, &spec); err != nil {
		return hschema.AgentTemplateSpec{}, err
	}
	return spec, nil
```

```go
// ResolveProvider 返回注册在 name 下的 provider SPI 实现。
// Python: factory.py:112-134 resolve_provider
//
// 解析顺序：内置名 → 进程内注册 → 错误。
// 内置名已声明但尚未链接（15.6-15.9 未落地）时返回明确错误，
// 让调用方能区分「名字拼错」与「章节未实现」。
func ResolveProvider(name string) (harnessprotocol.HarnessProvider, error) {
	registeredProvidersMu.RLock()
	factory, registered := registeredProviders[name]
	registeredProvidersMu.RUnlock()

	if slices.Contains(ProviderNames, name) {
		if !registered {
			return nil, fmt.Errorf(
				"harness provider %q is declared but not yet linked", name)
		}
		return asProvider(name, factory)
	}
	if registered {
		return asProvider(name, factory)
	}
	return nil, fmt.Errorf("unknown harness provider %q; expected one of %s",
		name, strings.Join(AvailableProviders(), ", "))
}

// StdioCommand 解析包相对路径的 stdio 脚本参数。
// Python: factory.py:266-278 _stdio_command
//
// cwd 非空且参数是相对路径、且拼接后确实是文件时，解析成绝对路径；
// 否则参数原样保留。
func StdioCommand(spec hschema.McpServerSpec) ([]string, error) {
	var cwd string
	if spec.Cwd != "" {
		expanded, err := expandHome(spec.Cwd)
		if err != nil {
			return nil, err
		}
		cwd = expanded
	}
	args := make([]string, 0, len(spec.Args))
	for _, arg := range spec.Args {
		if cwd != "" && !filepath.IsAbs(arg) {
			located := filepath.Join(cwd, arg)
			if info, err := os.Stat(located); err == nil && !info.IsDir() {
				absolute, absErr := filepath.Abs(located)
				if absErr != nil {
					return nil, absErr
				}
				args = append(args, absolute)
				continue
			}
		}
		args = append(args, arg)
	}
	return append([]string{spec.Command}, args...), nil
}

// mcpServerConfig 把一条 MCP 声明转成协议配置。
// Python: factory.py:243-263 _mcp_server_config
func mcpServerConfig(spec hschema.McpServerSpec) (harnessprotocol.McpServerConfig, error) {
	name := spec.ServerName
	if name == "" {
		name = spec.ServerID
	}
	if name == "" {
		return harnessprotocol.McpServerConfig{}, errors.New("manifest MCP server requires server_name")
	}
	if spec.Type == "stdio" {
		if spec.Command == "" {
			return harnessprotocol.McpServerConfig{}, fmt.Errorf(
				"manifest stdio MCP server %q requires command", name)
		}
		command, err := StdioCommand(spec)
		if err != nil {
			return harnessprotocol.McpServerConfig{}, err
		}
		return harnessprotocol.McpServerConfig{
			Name:      name,
			Transport: harnessprotocol.McpTransportStdio,
			Command:   command,
			Env:       envOrNil(spec.Env),
		}, nil
	}
	if spec.URL == "" {
		return harnessprotocol.McpServerConfig{}, fmt.Errorf(
			"manifest %s MCP server %q requires url", spec.Type, name)
	}
	return harnessprotocol.McpServerConfig{
		Name:      name,
		Transport: harnessprotocol.McpTransportHTTP,
		URL:       spec.URL,
		Headers:   envOrNil(spec.AuthHeaders),
	}, nil
}

// expandHome 把前导 ~ 展开成用户主目录（对应 os.path.expanduser）。
func expandHome(path string) (string, error) {
	if !strings.HasPrefix(path, "~") {
		return path, nil
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return path, nil // 无法展开时原样返回，与 Python 的宽松语义一致
	}
	if path == "~" {
		return home, nil
	}
	if strings.HasPrefix(path, "~/") {
		return filepath.Join(home, path[2:]), nil
	}
	return path, nil
}

// envOrNil 空 map 转 nil（JSON 序列化时省略字段）。
func envOrNil(values map[string]string) map[string]string {
	if len(values) == 0 {
		return nil
	}
	return values
}
```

```go
// 进 context；对第三方 provider：persona 提示词段落渲染进 systemPrompt，
// 清单的 MCP 服务器变成 mcpServers。
func BuildHarnessContext(manifest hschema.AgentTemplateSpec,
	opts BuildHarnessContextOptions) (harnessprotocol.HarnessContext, error) {
	template, err := LoadManifest(manifest)
	if err != nil {
		return harnessprotocol.HarnessContext{}, err
	}
	card := template.AgentCard
	var promptParts []string
	var mcpServers []harnessprotocol.McpServerConfig

	if !nativeProviders[opts.Provider] {
		language := opts.Language
		if language == "" {
			language = "cn"
		}
		if rendered := hresources.RenderAgentTemplateSystemPrompt(template, language); rendered != "" {
			promptParts = append(promptParts, rendered)
		}
		mcpServers, err = ManifestMcpServers(template)
		if err != nil {
			return harnessprotocol.HarnessContext{}, err
		}
	}
	if opts.ExtraSystemPrompt != "" {
		promptParts = append(promptParts, opts.ExtraSystemPrompt)
	}

	capabilities := make(map[harnessprotocol.HostCapability]bool, len(opts.HostCapabilities)+2)
	for capability, declared := range opts.HostCapabilities {
		capabilities[capability] = declared
	}
	if opts.CheckpointSink != nil {
		capabilities[harnessprotocol.HostCapabilityCheckpointSink] = true
	}
	if len(mcpServers) > 0 {
		capabilities[harnessprotocol.HostCapabilityMcpServers] = true
	}

	agentName := opts.AgentName
	if agentName == "" {
		agentName = card.Name
	}
	agentID := opts.AgentID
	if agentID == "" {
		agentID = card.ID
	}
	if agentID == "" {
		agentID = card.Name
	}
	resumePolicy := opts.ResumePolicy
	if resumePolicy == "" {
		resumePolicy = harnessprotocol.ResumePolicyNew
	}

	context := harnessprotocol.HarnessContext{
		AgentName:        agentName,
		AgentID:          agentID,
		HostSessionID:    opts.HostSessionID,
		SystemPrompt:     strings.Join(promptParts, "\n\n"),
		HostCapabilities: capabilities,
		ResumePolicy:     resumePolicy,
		Env:              opts.Env,
		Checkpoint:       opts.Checkpoint,
		CheckpointSink:   opts.CheckpointSink,
		McpServers:       mcpServers,
		Interactions:     opts.Interactions,
		Metadata:         opts.Metadata,
	}
	if opts.Cwd != "" {
		cwd := opts.Cwd
		context.Cwd = &cwd
	}
	if context.Env == nil {
		context.Env = map[string]string{}
	}
	if context.Metadata == nil {
		context.Metadata = harnessprotocol.JsonObject{}
	}
	return context, nil
}

// ProviderModelConfig 通过 provider 的 modelConfig 把模型选择映射到其配置。
// Python: factory.py:337-366 provider_model_config
func ProviderModelConfig(provider string, choice harnessprotocol.ModelChoice,
	fallback *harnessprotocol.ModelChoice) (map[string]any, error) {
	resolved, err := ResolveProvider(provider)
	if err != nil {
		return nil, err
	}
	modelConfigProvider, ok := resolved.(harnessprotocol.ModelConfigProvider)
	if !ok {
		return nil, fmt.Errorf("harness provider %q does not configure models", provider)
	}
	if fallback != nil && !resolved.Card().Supports(harnessprotocol.HarnessCapabilityAuthFallback) {
		return nil, fmt.Errorf("harness provider %q has no authentication fallback", provider)
	}
	usesEndpoint := choice.Endpoint != nil || (fallback != nil && fallback.Endpoint != nil)
	if usesEndpoint && !resolved.Card().Supports(harnessprotocol.HarnessCapabilityEndpointModels) {
		return nil, fmt.Errorf("harness provider %q does not take endpoint models", provider)
	}
	fragment, err := modelConfigProvider.ModelConfig(choice, fallback)
	if err != nil {
		return nil, err
	}
	values := make(map[string]any, len(fragment))
	for key, value := range fragment {
		values[key] = value
	}
	return values, nil
}

// manifestModelChoice 返回清单的模型；命名了 API base 时由该端点服务。
// Python: factory.py:318-334 _manifest_model_choice
func manifestModelChoice(manifest hschema.AgentTemplateSpec) (harnessprotocol.ModelChoice, bool) {
	model := manifest.Model
	if model == nil {
		return harnessprotocol.ModelChoice{}, false
	}
	var modelName string
	if model.ModelRequestConfig != nil && model.ModelRequestConfig.ModelName != "" {
		modelName = model.ModelRequestConfig.ModelName
	}
	if model.ModelClientConfig == nil || model.ModelClientConfig.APIBase == "" {
		choice := harnessprotocol.ModelChoice{}
		if modelName != "" {
			choice.Model = &modelName
		}
		return choice, true
	}
	client := model.ModelClientConfig
	endpoint := &harnessprotocol.ModelEndpoint{APIBase: client.APIBase}
	if client.APIKey != "" {
		apiKey := client.APIKey
		endpoint.APIKey = &apiKey
	}
	if providerName := providerLabel(client.ClientProvider); providerName != "" {
		endpoint.Provider = &providerName
	}
	choice := harnessprotocol.ModelChoice{Endpoint: endpoint}
	if modelName != "" {
		choice.Model = &modelName
	}
	return choice, true
}

// providerLabel 取 client_provider 的字符串值。
// Python: factory.py:332 `str(getattr(raw_provider, "value", raw_provider))`
//
// Python 的枚举取 .value，否则 str() 原值。Go 侧 JSON 解出来是 string 或
// 自定义类型；统一走 fmt 的字符串形式。
func providerLabel(raw any) string {
	switch typed := raw.(type) {
	case nil:
		return ""
	case string:
		return typed
	case interface{ String() string }:
		return typed.String()
	default:
		return fmt.Sprintf("%v", raw)
	}
}
```

```go
func isTerminal(r io.Reader) bool {
	file, ok := r.(*os.File)
	if !ok {
		return false
	}
	info, err := file.Stat()
	if err != nil {
		return false
	}
	// 必须是字符设备**且**不是 /dev/null
	if info.Mode()&os.ModeCharDevice == 0 {
		return false
	}
	return file.Name() != os.DevNull
}
```
