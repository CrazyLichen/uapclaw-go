// pdfObject PDF 对象条目。
type pdfObject struct {
	num    int      // 对象号
	stream []byte   // 流数据（nil 表示纯字典对象）
	dict   string   // 字典内容（含流对象的额外条目）

// buildPDF 合成最小合法 PDF 字节。
//
// pages：每页一项（页文本）；imgOnPage：页号→是否嵌图（嵌入 DCTDecode XObject）。
// 固定对象布局：1=Catalog 2=Pages 3=Font 4=Image XObject（仅需要时）
// 5..=每页 Page 与 Contents。图片绘制走内容流 Do 操作符 + Resources XObject 声明。
func buildPDF(pages []string, imgOnPage map[int]bool) []byte {
	var objs []pdfObject

	// 计算对象号分配
	imgObj := 0
	hasImage := false
	for _, has := range imgOnPage {
		if has {
			hasImage = true
		}
	}
	if hasImage {
		imgObj = 4
	}

	pageStart := 5
	if !hasImage {
		pageStart = 4
	}

	// 对象 1：Catalog
	objs = append(objs, pdfObject{num: 1, dict: "<< /Type /Catalog /Pages 2 0 R >>"})

	// 对象 2：Pages
	kidRefs := ""
	for i := range pages {
		kidRefs += fmt.Sprintf("%d 0 R ", pageStart+2*i)
	}
	objs = append(objs, pdfObject{num: 2, dict: fmt.Sprintf(
		"<< /Type /Pages /Count %d /Kids [ %s] >>", len(pages), kidRefs)})

	// 对象 3：Font（Helvetica 内置）
	objs = append(objs, pdfObject{num: 3, dict: "<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>"})

	// 对象 4（可选）：Image XObject
	if hasImage {
		objs = append(objs, pdfObject{num: imgObj, dict: fmt.Sprintf(
			"<< /Type /XObject /Subtype /Image /Width 1 /Height 1 /ColorSpace /DeviceGray /BitsPerComponent 8 /Filter /DCTDecode /Length %d >>",
			len(tinyJPEGBytes)), stream: tinyJPEGBytes})
	}

	// 每页：Page 对象 + Contents 流对象
	for i, text := range pages {
		pageNum := pageStart + 2*i
		contentsNum := pageNum + 1

		resources := "<< /Font << /F1 3 0 R >>"
		content := fmt.Sprintf("BT /F1 12 Tf 72 720 Td (%s) Tj ET", escapePDFText(text))
		if imgOnPage[i+1] {
			resources += fmt.Sprintf(" /XObject << /Im0 %d 0 R >>", imgObj)
			content = fmt.Sprintf("q 100 0 0 100 72 500 cm /Im0 Do Q BT /F1 12 Tf 72 720 Td (%s) Tj ET", escapePDFText(text))
		}
		resources += " >>"

		objs = append(objs, pdfObject{num: pageNum, dict: fmt.Sprintf(
			"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources %s /Contents %d 0 R >>",
			resources, contentsNum)})
		objs = append(objs, pdfObject{num: contentsNum, dict: fmt.Sprintf("<< /Length %d >>", len(content)), stream: []byte(content)})
	}

	// 序列化：头 + 对象 + xref + trailer
	var buf bytes.Buffer
	buf.WriteString("%PDF-1.4\n")
	offsets := map[int]int{}
	for _, o := range objs {
		offsets[o.num] = buf.Len()
		buf.WriteString(fmt.Sprintf("%d 0 obj\n%s\n", o.num, o.dict))
		if o.stream != nil {
			buf.WriteString("stream\n")
			buf.Write(o.stream)
			buf.WriteString("\nendstream\n")
		}
		buf.WriteString("endobj\n")
	}
	xrefPos := buf.Len()
	maxNum := 0
	for n := range offsets {
		if n > maxNum {
			maxNum = n
		}
	}
	buf.WriteString(fmt.Sprintf("xref\n0 %d\n", maxNum+1))
	buf.WriteString("0000000000 65535 f \n")
	for i := 1; i <= maxNum; i++ {
		if off, ok := offsets[i]; ok {
			buf.WriteString(fmt.Sprintf("%010d 00000 n \n", off))
		} else {
			buf.WriteString("0000000000 65535 f \n")
		}
	}
	buf.WriteString(fmt.Sprintf("trailer << /Size %d /Root 1 0 R >>\nstartxref\n%d\n%%%%EOF\n", maxNum+1, xrefPos))
	return buf.Bytes()
}

// escapePDFText 转义文本操作数中的特殊字符。
func escapePDFText(s string) string {
	var b bytes.Buffer
	for _, r := range s {
		switch r {
		case '(', ')', '\\':
			b.WriteByte('\\')
			b.WriteRune(r)
		default:
			b.WriteRune(r)
		}
	}
	return b.String()
}

func writeTempPDF(t *testing.T, name string, pages []string, imgOnPage map[int]bool) string {
	t.Helper()

	if err := os.WriteFile(path, buildPDF(pages, imgOnPage), 0o644); err != nil {
		t.Fatalf("写入测试 PDF 失败: %v", err)
	}
	return path
}

type PDFParser struct {

func NewPDFParser() (*PDFParser, error) {

func extractImagesByPage(ctx context.Context, pdfBytes []byte, outputDir, filename string) map[int][]string {
	imgByPage := map[int][]string{}

		if err != nil {
			return fmt.Errorf("读取图片流失败: %w", err)
		}
		// 落盘目录（对齐 Python os.makedirs(output_dir, exist_ok=True)）
		if err := os.MkdirAll(outputDir, 0o755); err != nil {
			return fmt.Errorf("创建图片目录失败: %w", err)
		}
		// 页内序号 = 该页已落盘张数（回调页内顺序确定性：对象号升序）
		imgIndex := len(imgByPage[img.PageNr])
		ext := img.FileType
		if ext == "" {
			ext = "png" // 防御：无扩展名回落 png
		}
		imagePath := filepath.Join(outputDir, fmt.Sprintf("%s__page_%d__img_%d.%s", filename, img.PageNr, imgIndex, ext))
		if err := os.WriteFile(imagePath, data, 0o644); err != nil {
			return fmt.Errorf("落盘图片失败: %w", err)
		}
		imgByPage[img.PageNr] = append(imgByPage[img.PageNr], imagePath)
		return nil
	}, nil); err != nil {

		// 此处 err 为文档级失败（损坏/加密等）
		logger.Error(logComponent).
			Err(err).
			Msg("failed to extract images from pdf")
		return map[int][]string{}
	}
	return imgByPage
}

// ParseContent 解析 PDF：双通道（对齐 Python _parse(file_path, llm_client)）。
//
// 流程（对齐 Python _async_parse_pdf 主循环）：
//  1. 读入字节（双通道共享——图片通道 io.ReadSeeker / 文本通道 ReaderAt）
//  2. extractImagesByPage 一次性提图分桶（独立软失败）
//  3. 逐页 GetPlainText（fonts 跨页累积——read_file.go 先例）；
//     非空文本入 content
//  4. 该页图片 caption 经 CaptionImages 生成；非空 caption 紧随该页
//     文本入 content（Python 同页+追加序关联语义）
//  5. 过滤空行后 "\n".join；空串返回（对齐 "result if result else None"）
//
// 整体软失败：任一通道 err → Error 日志 + 返回 ("", nil)
// （对齐 Python except → logger.error → return None）。
//
// Python: pdf_parser.py PDFParser._parse
func (p *PDFParser) ParseContent(ctx context.Context, doc string, llmClient *llm.Model) (string, error) {
	pdfBytes, err := os.ReadFile(doc)
	if err != nil {
		logger.Error(logComponent).
			Str("file_path", doc).
			Err(err).
			Msg("failed to parse pdf")
		return "", nil
	}

	imgByPage := extractImagesByPage(ctx, pdfBytes, savedImageDir, filepath.Base(doc))

	// 文本通道：ledongthuc 逐页（ReadFile err 之外错误在此统一软失败）
	content, err := p.parsePages(ctx, pdfBytes, imgByPage, llmClient)
	if err != nil {
		// 对齐 Python: logger.error(f"Failed to parse PDF {file_path}: {e}")
		logger.Error(logComponent).
			Str("file_path", doc).
			Err(err).
			Msg("failed to parse pdf")
		return "", nil
	}

	result := joinNonBlank(content)
	if result == "" {
		return "", nil // 对齐 Python "result if result else None"
	}
	return result, nil
}

// parsePages 文本通道逐页循环 + caption 并入。
//
// Python: pdf_parser.py _async_parse_pdf 循环体
func (p *PDFParser) parsePages(ctx context.Context, pdfBytes []byte, imgByPage map[int][]string, llmClient *llm.Model) ([]string, error) {
	r, err := pdf.NewReader(bytes.NewReader(pdfBytes), int64(len(pdfBytes)))
	if err != nil {
		return nil, err
	}

	// 单次构造 captioner（Python 每页重建但无状态副作用，等价——偏差 §8-6）
	captioner := NewImageCaptioner(llmClient)

	fonts := make(map[string]*pdf.Font)
	content := []string{}
	for pageNr := 1; pageNr <= r.NumPage(); pageNr++ {
		page := r.Page(pageNr)
		if page.V.IsNull() {
			continue // 空页跳过（Python extract_text 返回 None → ""）
		}
		// 缓存当前页字体（跨页共享——read_file.go 先例）
		for _, name := range page.Fonts() {
			if _, ok := fonts[name]; !ok {
				f := page.Font(name)
				fonts[name] = &f
			}
		}
		pageText, err := page.GetPlainText(fonts)
		if err != nil {
			return nil, fmt.Errorf("第 %d 页文本抽取失败: %w", pageNr, err)
		}
		if pageText != "" {
			content = append(content, pageText)
		}

		// 该页图片 caption（对齐 Python captions 紧随该页文本）
		imgs := imgByPage[pageNr]
		if len(imgs) > 0 {
			captions := captioner.CaptionImages(ctx, imgs)
			for _, c := range captions {
				if c != "" {
					content = append(content, c)
				}
			}
		}
	}
	return content, nil
}

// joinNonBlank 过滤空行后拼接（对齐 Python "[line for line in content if line.strip()]"）。
func joinNonBlank(lines []string) string {
	var kept []string
	for _, line := range lines {
		if strings.TrimSpace(line) != "" {
			kept = append(kept, line)
		}
	}
	return strings.Join(kept, "\n")
}
