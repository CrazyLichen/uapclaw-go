```go
// parseTodosFile 解析 markdown 待办文件（frontmatter+body）。
// Python: _parse_todos_file()——lookahead 正则 RE2 不支持，改逐行状态机（偏差#2，产出一致）。
func parseTodosFile(filePath string) []*TodoItem {
	data, err := os.ReadFile(filePath)
	if err != nil {
		// Python: os.path.exists → False 时返回 []；读失败同样容错为空
		return nil
	}
	lines := strings.Split(string(data), "\n")
	var todos []*TodoItem
	var current map[string]string
	var body []string
	inFrontmatter := false
	flush := func() {
		if current != nil && current["id"] != "" {
			if len(body) > 0 {
				current["description"] = strings.TrimSpace(strings.Join(body, "\n"))
			}
			if t, ok := todoItemFromDict(current); ok {
				todos = append(todos, t)
			} else {
				logger.Warn(logComponent).Msg("Failed to parse todo item")
			}
		}
		current = nil
		body = nil
	}
	for _, line := range lines {
		trimmed := strings.TrimRight(line, " \t\r")
		if trimmed == "---" {
			if inFrontmatter {
				// frontmatter 结束，进入 body 收集
				inFrontmatter = false
				continue
			}
			// 新条目开始（Python 正则按 ^---\n 分隔）
			flush()
			current = map[string]string{}
			inFrontmatter = true
			continue
		}
		if inFrontmatter && current != nil {
			if k, v, ok := strings.Cut(trimmed, ":"); ok {
				current[strings.TrimSpace(k)] = strings.TrimSpace(v)
			}
		} else if current != nil {
			// body 区：到下一个 --- 前全部累积（对齐 (?=^---\s*\n|\Z)）
			body = append(body, line)
		}
	}
	flush()
	return todos
}

// todoItemFromDict 从解析 map 构造 TodoItem（容错默认值）。
// Python: TodoItem.from_dict()
func todoItemFromDict(data map[string]string) (*TodoItem, bool) {
	if data["id"] == "" {
		return nil, false
	}
	now := time.Now().UTC().Format(time.RFC3339)
	status := TodoStatusPending
	if s := data["status"]; s != "" {
		status = TodoStatus(s)
	}
	priority := TodoPriorityMedium
	if p := data["priority"]; p != "" {
		priority = TodoPriority(p)
	}
	return &TodoItem{
		ID:          data["id"],
		Title:       data["title"],
		Status:      status,
		Priority:    priority,
		DueAt:       data["due_at"],
		RemindAt:    data["remind_at"],
		CreateAt:    strOr(data["create_at"], now),
		UpdateAt:    strOr(data["update_at"], now),
		CreatedBy:   data["created_by"],
		Description: data["description"],
	}, true
}

// strOr 空串回退。
func strOr(s, fallback string) string {
	if s == "" {
		return fallback
	}
	return s
}

// serializeTodoItem 序列化单条待办为 markdown frontmatter 块。
// Python: _serialize_todo_item()——逐行 key: value，空可选字段输出 "key: "
func serializeTodoItem(todo *TodoItem) string {
	var sb strings.Builder
	sb.WriteString("---\n")
	sb.WriteString("id: " + todo.ID + "\n")
	sb.WriteString("title: " + todo.Title + "\n")
	sb.WriteString("status: " + string(todo.Status) + "\n")
	sb.WriteString("priority: " + string(todo.Priority) + "\n")
	sb.WriteString("due_at: " + todo.DueAt + "\n")
	sb.WriteString("remind_at: " + todo.RemindAt + "\n")
	sb.WriteString("create_at: " + todo.CreateAt + "\n")
	sb.WriteString("update_at: " + todo.UpdateAt + "\n")
	sb.WriteString("created_by: " + todo.CreatedBy + "\n")
	sb.WriteString("---\n")
	if todo.Description != "" {
		sb.WriteString(todo.Description + "\n")
	}
	sb.WriteString("\n")
	return sb.String()
}

// writeTodosFile 写回待办文件。
```

```go
}

// handleUserTodos 处理用户待办六 action。
// Python: _handle_user_todos()——整体 try/except → {"success": false, "error": ...}
func handleUserTodos(ctx context.Context, inputs map[string]any) (result map[string]any, err error) {
	defer func() {
		if r := recover(); r != nil {
			logger.Error(logComponent).Any("panic", r).Msg("user_todos failed")
			result = map[string]any{"success": false, "error": fmt.Sprintf("%v", r)}
			err = nil
		}
	}()
	fail := func(msg string) map[string]any {
		return map[string]any{"success": false, "error": msg}
	}

	cid := resolveUserTodoChannel(ctx, todoStr(inputs, "channel_id"))
	filePath := getTodosFilePath(cid)
	action := todoStr(inputs, "action")

	switch action {
	case "list":
		todos := parseTodosFile(filePath)
		out := make([]map[string]any, 0, len(todos))
		for _, t := range todos {
			out = append(out, todoToDict(t))
		}
		return map[string]any{"success": true, "channel_id": cid, "todos": out, "count": len(todos)}, nil

	case "get":
		todoID := todoStr(inputs, "todo_id")
		if todoID == "" {
			return fail("todo_id is required for get action"), nil
		}
		for _, t := range parseTodosFile(filePath) {
			if t.ID == todoID {
				return map[string]any{"success": true, "todo": todoToDict(t)}, nil
			}
		}
		return fail("Todo not found: " + todoID), nil

	case "create":
		title := todoStr(inputs, "title")
		if title == "" {
			return fail("title is required for create action"), nil
		}
		todos := parseTodosFile(filePath)
		// remind_at 推算：due_at 且无 remind_at → due-5min；解析失败 → ""
		finalRemindAt := todoStr(inputs, "remind_at")
		dueAt := todoStr(inputs, "due_at")
		if dueAt != "" && finalRemindAt == "" {
			if due, perr := time.Parse(time.RFC3339, dueAt); perr == nil {
				finalRemindAt = due.Add(-5 * time.Minute).Format(time.RFC3339)
			} else {
				finalRemindAt = ""
			}
		}
		status := TodoStatusPending
		if s := todoStr(inputs, "status"); s != "" {
			status = TodoStatus(s)
		}
		priority := TodoPriorityMedium
		if p := todoStr(inputs, "priority"); p != "" {
			priority = TodoPriority(p)
		}
		now := time.Now().UTC().Format(time.RFC3339)
		newTodo := &TodoItem{
			ID:          uuidShort8(),
			Title:       title,
			Status:      status,
			Priority:    priority,
			DueAt:       dueAt,
			RemindAt:    finalRemindAt,
			CreateAt:    now,
			UpdateAt:    now,
			CreatedBy:   resolveUserTodoCreatedBy(ctx, todoStr(inputs, "created_by")),
			Description: todoStr(inputs, "description"),
		}
		todos = append(todos, newTodo)
		writeTodosFile(filePath, todos)
		logger.Info(logComponent).Str("todo_id", newTodo.ID).Str("channel", cid).Msg("Created todo")
		return map[string]any{"success": true, "todo": todoToDict(newTodo), "message": "Todo created: " + title}, nil

	case "update":
		todoID := todoStr(inputs, "todo_id")
		if todoID == "" {
			return fail("todo_id is required for update action"), nil
		}
		todos := parseTodosFile(filePath)
		var found *TodoItem
		for _, t := range todos {
			if t.ID == todoID {
				found = t
				break
			}
		}
		if found == nil {
			return fail("Todo not found: " + todoID), nil
		}
		if v := todoStr(inputs, "title"); v != "" {
			found.Title = v
		}
		if v := todoStr(inputs, "status"); v != "" {
			found.Status = TodoStatus(v)
		}
		if v := todoStr(inputs, "priority"); v != "" {
			found.Priority = TodoPriority(v)
		}
		if v := todoStr(inputs, "due_at"); v != "" {
			found.DueAt = v
		}
		if v := todoStr(inputs, "remind_at"); v != "" {
			found.RemindAt = v
		}
		if v := todoStr(inputs, "created_by"); v != "" {
			found.CreatedBy = v
		}
		if v := todoStr(inputs, "description"); v != "" {
			found.Description = v
		}
		found.UpdateAt = time.Now().UTC().Format(time.RFC3339)
		writeTodosFile(filePath, todos)
		logger.Info(logComponent).Str("todo_id", todoID).Str("channel", cid).Msg("Updated todo")
		return map[string]any{"success": true, "todo": todoToDict(found), "message": "Todo updated: " + found.Title}, nil

	case "delete":
		todoID := todoStr(inputs, "todo_id")
		if todoID == "" {
			return fail("todo_id is required for delete action"), nil
		}
		todos := parseTodosFile(filePath)
		var found *TodoItem
		for _, t := range todos {
			if t.ID == todoID {
				found = t
				break
			}
		}
		if found == nil {
			return fail("Todo not found: " + todoID), nil
		}
		kept := todos[:0]
		for _, t := range todos {
			if t.ID != todoID {
				kept = append(kept, t)
			}
		}
		writeTodosFile(filePath, kept)
		logger.Info(logComponent).Str("todo_id", todoID).Str("channel", cid).Msg("Deleted todo")
		return map[string]any{"success": true, "message": "Todo deleted: " + found.Title}, nil

	case "search":
		query := todoStr(inputs, "query")
		if query == "" {
			return fail("query is required for search action"), nil
		}
		q := strings.ToLower(query)
		var matched []*TodoItem
		for _, t := range parseTodosFile(filePath) {
			if strings.Contains(strings.ToLower(t.Title), q) || strings.Contains(strings.ToLower(t.Description), q) {
				matched = append(matched, t)
			}
		}
		out := make([]map[string]any, 0, len(matched))
		for _, t := range matched {
			out = append(out, todoToDict(t))
		}
		return map[string]any{"success": true, "channel_id": cid, "query": query, "todos": out, "count": len(matched)}, nil

	default:
		return fail("Unknown action: " + action + ". Valid actions: list, get, create, update, delete, search"), nil
	}
}

		"due_at": t.DueAt, "remind_at": t.RemindAt, "create_at": t.CreateAt, "update_at": t.UpdateAt,
		"created_by": t.CreatedBy, "description": t.Description,
	}
}
```

```go
}

// CancelSession 按 sessionID 取消运行中会话。
// Python: cancel_session()
func (t *MultiSessionToolkit) CancelSession(ctx context.Context, inputs map[string]any) (map[string]any, error) {

		Str("parent_session_id", t.sessionID).

		Msg("[MultiSessionToolkit] cancel_session request")

	t.mu.Unlock()

		// 未在 tasks 中：从未创建或已结束（Python task is None / task.done() 同文案）

}

// toStringSlice 宽容转字符串切片。
func toStringSlice(v any) []string {
	switch arr := v.(type) {
	case []string:
		return arr
	case []any:
		out := make([]string, 0, len(arr))
		for _, item := range arr {
			if s, ok := item.(string); ok {
				out = append(out, s)
			}
		}
		return out
	case string:
		return []string{arr}
	}
	return nil
}
```

```go
// runAndNotify 运行子代理并在结束（成功/取消/异常）时通知。
// Python: _run_and_notify()
func (t *MultiSessionToolkit) runAndNotify(ctx context.Context, sessionID, description string, agent agentinterfaces.BaseAgent, inputs map[string]any) {
	defer func() {
		if r := recover(); r != nil {
			// Python except Exception → ERROR + notify + re-raise（Task 记录异常）
			errStr := fmt.Sprintf("%v", r)
			logger.Error(logComponent).
				Str("session_id", sessionID).Str("error", errStr).
				Msg("[MultiSessionToolkit] coroutine panic")
			t.updateSession(sessionID, SessionStatusError, errStr)
			t.notify(ctx, sessionID, SessionStatusError, "", errStr)
		}
		// Python finally: self._tasks.pop(session_id, None)
		t.mu.Lock()
		delete(t.tasks, sessionID)
		t.mu.Unlock()
		logger.Debug(logComponent).
			Str("session_id", sessionID).
			Int("remaining", len(t.tasks)).
			Msg("[MultiSessionToolkit] run_and_notify exit")
	}()

	logger.Debug(logComponent).
		Str("session_id", sessionID).Str("description", truncate(description, 80)).
		Msg("[MultiSessionToolkit] run_and_notify start")
	t.mu.Lock()
	t.sessions = append(t.sessions, SessionTask{
		SessionID: sessionID, Description: description, Status: SessionStatusRunning, Result: "",
	})
	t.mu.Unlock()

	result, err := runner.RunAgent(ctx, runner.ByAgent(agent), inputs, runner.SessionRef{}, nil, nil)
	if err == nil {
		resultStr := ""
		if output, ok := result["output"].(string); ok {
			resultStr = output
		} else if result != nil {
			resultStr = fmt.Sprintf("%v", result)
		}
		logger.Info(logComponent).
			Str("session_id", sessionID).
			Int("result_len", len(resultStr)).
			Msg("[MultiSessionToolkit] coroutine completed")
		t.updateSession(sessionID, SessionStatusCompleted, resultStr)
		t.notify(ctx, sessionID, SessionStatusCompleted, resultStr, "")
		return
	}
	if errors.Is(err, context.Canceled) {
		logger.Info(logComponent).Str("session_id", sessionID).Msg("[MultiSessionToolkit] coroutine cancelled")
		t.updateSession(sessionID, SessionStatusCancelled, "任务已取消")
		t.notify(ctx, sessionID, SessionStatusCancelled, "", "")
		return
	}
	errStr := err.Error()
	logger.Error(logComponent).
		Str("session_id", sessionID).Str("error", errStr).
		Msg("[MultiSessionToolkit] coroutine error")
	t.updateSession(sessionID, SessionStatusError, errStr)
	t.notify(ctx, sessionID, SessionStatusError, "", errStr)
}

// notify 推送子任务更新；全部完成时触发汇总链。
// Python: notify()——send_push(chat.session_result) + all_tasks_done → 汇总 → chat.final
func (t *MultiSessionToolkit) notify(ctx context.Context, sessionID string, status SessionStatus, result, errStr string) {
	pushFunc := getSessionPushFunc()
	if pushFunc == nil {
		// Python: MessageHandler 未初始化 → warn 跳过
		logger.Warn(logComponent).
			Str("session_id", sessionID).
			Msg("[MultiSessionToolkit] pushFunc not injected, skip notify")
		return
	}

	t.mu.Lock()
	description := ""
	index := 0
	for i, s := range t.sessions {
		if s.SessionID == sessionID {
			description = s.Description
			index = i + 1
			break
		}
	}
	total := len(t.sessions)
	allDone := true
	for _, s := range t.sessions {
		if s.Status != SessionStatusCompleted && s.Status != SessionStatusError {
			allDone = false
			break
		}
	}
	t.mu.Unlock()

	// Python: cancelled → error 映射（前端 SubtaskStatus 只有两态）
	payloadStatus := "error"
	message := errStr
	if status == SessionStatusCompleted {
		payloadStatus = "completed"
		message = result
	} else if status == SessionStatusCancelled {
		message = strOr(errStr, "任务已取消")
	}
	msg := map[string]any{
		"request_id": t.requestID,
		"channel_id": t.channelID,
		"session_id": t.sessionID,
		"payload": map[string]any{
			"event_type":   "chat.session_result",
			"session_id":   sessionID,
			"description":  description,
			"status":       payloadStatus,
			"index":        index,
			"total":        total,
			"result":       message,
			"is_parallel":  true,
		},
		"is_complete": false,
	}
	logger.Debug(logComponent).
		Str("session_id", sessionID).Str("status", payloadStatus).
		Int("index", index).Int("total", total).
		Msg("[MultiSessionToolkit] notify subtask_update")
	if err := pushFunc(ctx, msg); err != nil {
		logger.Warn(logComponent).Err(err).Msg("[MultiSessionToolkit] push session_result failed")
	}

	if !allDone {
		return
	}
	// 汇总链：仅一个完成者进入（sync.Once，Python GIL 原子性等价）
	t.sumOnce.Do(func() {
		t.runSummary(ctx, pushFunc)
	})
}

// runSummary 全部完成后的汇总链。
// Python: notify() 尾部——汇总文本 + run_agent_streaming + chat.final
func (t *MultiSessionToolkit) runSummary(ctx context.Context, pushFunc func(context.Context, map[string]any) error) {
	// 偏差#4：汇总独立于父请求取消（Python 协程天然独立）
	summaryCtx := context.WithoutCancel(ctx)

	provider := getSessionAgentProvider()
	if provider == nil {
		logger.Error(logComponent).Msg("[MultiSessionToolkit] agentProvider not injected, skip summary")
		return
	}
	agent := provider()
	if agent == nil {
		logger.Error(logComponent).Msg("[MultiSessionToolkit] get_agent returned nil, skip summary")
		return
	}

	var sb strings.Builder
	sb.WriteString("后台会话任务均已完成：\n")
	t.mu.Lock()
	for _, st := range t.sessions {
		sb.WriteString(fmt.Sprintf("\nsession_id: %s\ndescription: %s\nresult: %s\n", st.SessionID, st.Description, st.Result))
	}
	t.mu.Unlock()

	queryBytes, _ := json.Marshal(map[string]any{
		"source":  "system",
		"content": sb.String(),
		"type":    "notify",
	})
	inputs := map[string]any{
		"conversation_id": t.sessionID,
		"query":           string(queryBytes),
	}

	// Python 注释：用 streaming 而非 run_agent 以确保 session.post_run() 持久化
	ch, err := runner.RunAgentStreaming(summaryCtx, runner.ByAgent(agent), inputs, runner.SessionRef{}, nil, nil, nil)
	if err != nil {
		logger.Warn(logComponent).Err(err).Msg("[MultiSessionToolkit] summary streaming failed")
		return
	}
	var accumulated []string
	var finalOutput string
	for chunk := range ch {
		// 偏差#1：stream.Schema 接口三态适配（Python hasattr 判定）
		var chunkType string
		var payload any
		switch s := chunk.(type) {
		case *stream.OutputSchema:
			chunkType = s.Type
			payload = s.Payload
		case stream.OutputSchema:
			chunkType = s.Type
			payload = s.Payload
		case *stream.CustomSchema:
			chunkType = s.Type
			payload = s.Data
		case stream.CustomSchema:
			chunkType = s.Type
			payload = s.Data
		default:
			continue // TraceSchema 等无 type 语义 → skip
		}
		payloadMap, _ := payload.(map[string]any)
		switch chunkType {
		case "content_chunk":
			if c, _ := payloadMap["content"].(string); c != "" {
				accumulated = append(accumulated, c)
			}
		case "answer":
			switch out := payloadMap["output"].(type) {
			case map[string]any:
				temp := ""
				if s, _ := out["output"].(string); s != "" {
					temp = s
				} else if len(accumulated) > 0 {
					temp = strings.Join(accumulated, "")
				}
				if temp != "" {
					finalOutput = temp
				}
			case string:
				if out != "" {
					finalOutput = out
				}
			default:
				if len(accumulated) > 0 {
					finalOutput = strings.Join(accumulated, "")
				}
			}
		}
	}
	if finalOutput == "" {
		finalOutput = strings.Join(accumulated, "")
	}
	result := map[string]any{
		"output":      finalOutput,
		"result_type": "answer",
	}
	finalMsg := map[string]any{
		"request_id": t.requestID,
		"channel_id": t.channelID,
		"session_id": t.sessionID,
		"payload": map[string]any{
			"event_type": "chat.final",
			"task_id":    t.sessionID,
			"content":    result,
		},
		"is_complete": true,
	}
	if err := pushFunc(summaryCtx, finalMsg); err != nil {
		logger.Warn(logComponent).Err(err).Msg("[MultiSessionToolkit] push chat.final failed")
	}
}
```
