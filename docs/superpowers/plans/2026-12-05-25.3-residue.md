```go
// MarkSessionWaitingForInput 把空闲团队代次的持久响应 owner 全部停泊。
// Python: RuntimeSessionCoordinator.mark_session_waiting_for_input（coordinator.py:961-979）
func (c *RuntimeSessionCoordinator) MarkSessionWaitingForInput(sessionID string, generation int) {
	for _, h := range c.registry.Select(SelectOptions{
		SessionID: sessionID, Generation: &generation, ActiveOnly: true,
	}) {
		if h.Team == nil || h.Team.WaiterKind != TeamWaiterKindPersistent {
			continue
		}
		if h.State == SessionExecutionStateRunning {
			h.State = SessionExecutionStateWaitingForInput
		}
	}
	c.mu.Lock()
	record := c.sessions[sessionID]
	if record != nil && record.generation == generation {
		c.refreshSessionStateLocked(record)
	}
	c.mu.Unlock()
}

// ResumeWaitingForInput 把新输入代次的等待持久 owner 置回运行。
// Python: RuntimeSessionCoordinator.resume_waiting_for_input（coordinator.py:982-991）
func (c *RuntimeSessionCoordinator) ResumeWaitingForInput(sessionID string, generation int) {
	for _, h := range c.registry.Select(SelectOptions{
		SessionID: sessionID, Generation: &generation, ActiveOnly: true,
	}) {
		if h.State == SessionExecutionStateWaitingForInput {
			h.State = SessionExecutionStateRunning
		}
	}
	c.mu.Lock()
	record := c.sessions[sessionID]
	if record != nil && record.generation == generation {
		c.refreshSessionStateLocked(record)
	}
	c.mu.Unlock()
}
```

注意：`c.registry`/`c.sessions`/`record.generation` 的实际字段名以 coordinator.go 现有代码为准（grep `c.registry.Select`、`sessions[` 校对）；registry 句柄 State 写回沿用文件内既有「锁内写 State」模式（参考 MarkAwaitingControl 附近），不得锁外直写（-race）。
