`internal/agent_teams/harness/state.go`：

```go
package harness

import (
	"sync"
	"sync/atomic"
)

// ──────────────────────────── 结构体 ────────────────────────────

// ActiveRound 一次外圈执行（一轮）。
//
// Python: ActiveRound (agent_teams/harness/state.py)
// 本阶段只含"驱动一轮"所需字段；快照与 turn 字段由 25.1c / 25.1e 追加。
type ActiveRound struct {
	// RoundID 轮次序号（自 1 递增）
	RoundID int
	// TaskID 提交给任务循环内核的任务标识（立即中断取消它）
	TaskID string
	// Query 本轮原始查询
	Query any
	// Cancel 取消本轮的上下文取消函数
	Cancel func()
	// Done 本轮结束信号
	Done chan struct{}
}

// InboxMessage 排队等待驱动的输入。
type InboxMessage struct {
	// Content 输入内容
	Content any
	// Immediate true 表示转向当前轮，false 表示等本轮结束后再跑
	Immediate bool
}

// HarnessInternalState 全部可变状态容器；supervisor 是唯一写者。
//
// Python: HarnessInternalState (state.py)。相位用原子量保存，
// 因为 State() 会被外部协程并发读取（Python 靠 GIL 串行，Go 必须显式同步）。
type HarnessInternalState struct {
	// phase 当前生命周期相位
	phase atomic.Value
	// mu 保护 active 与 roundSeq（仅 supervisor 写入，但 State 之外的读取路径也在锁内）
	mu sync.RWMutex
	// active 当前活动轮次，无则为 nil
	active *ActiveRound
	// roundSeq 轮次序号自增源
	roundSeq int
	// queue 待驱动输入队列
	queue []InboxMessage
}

// ──────────────────────────── 枚举 ────────────────────────────

// HarnessState 生命周期相位。
//
// Python: HarnessState (harness_protocol/state.py)，5 态。
// 注意：DeepAgent 内部的 InteractionPhase 只有 idle/running/terminated 三态，
// 暂停是 L2 独有的概念。
type HarnessState string

const (
	// HarnessStateIdle 空闲（无在飞轮次）
	HarnessStateIdle HarnessState = "idle"
	// HarnessStateRunning 有轮次在执行
	HarnessStateRunning HarnessState = "running"
	// HarnessStatePausing 已接受暂停、尚未抵达可恢复边界（25.1c 启用）
	HarnessStatePausing HarnessState = "pausing"
	// HarnessStatePaused 已暂停在可恢复边界（25.1c 启用）
	HarnessStatePaused HarnessState = "paused"
	// HarnessStateTerminated 已终止（不可逆）
	HarnessStateTerminated HarnessState = "terminated"
)

// ──────────────────────────── 非导出函数 ────────────────────────────

// newInternalState 创建内部状态，相位初始化为 idle。
func newInternalState() *HarnessInternalState {
	st := &HarnessInternalState{queue: make([]InboxMessage, 0, 4)}
	st.phase.Store(HarnessStateIdle)
	return st
}

// Phase 返回当前相位。
func (s *HarnessInternalState) Phase() HarnessState {
	if v, ok := s.phase.Load().(HarnessState); ok {
		return v
	}
	return HarnessStateIdle
}

// setPhase 更新相位（仅 supervisor 调用）。
func (s *HarnessInternalState) setPhase(p HarnessState) {
	s.phase.Store(p)
}

// Active 返回当前活动轮次。
func (s *HarnessInternalState) Active() *ActiveRound {
	s.mu.RLock()
	defer s.mu.RUnlock()
	return s.active
}

// SetActive 设置当前活动轮次。
func (s *HarnessInternalState) SetActive(r *ActiveRound) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.active = r
}

// NextRoundID 分配下一个轮次序号。
func (s *HarnessInternalState) NextRoundID() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.roundSeq++
	return s.roundSeq
}
```

> 说明：`state.go` 同时含声明与实现（不是纯声明文件），因此不触发"纯声明文件覆盖率"问题。

package harness

import (
	"context"
	"sync/atomic"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/session/stream"
)

// ──────────────────────────── 结构体 ────────────────────────────

// OutputIterator 输出块迭代器。
//
// Python: _OutputIterator (agent_teams/harness/outputs.py)，44 行。
// 契约：单消费者——END 哨兵只会被第一个迭代器读到，之后所有读取返回 false。
type OutputIterator struct {
	// ch 输出块通道（由 NativeHarness 持有写端）
	ch <-chan stream.Schema
	// consumed 是否已完成（读到 END 或上下文取消）
	consumed atomic.Bool
}

// ──────────────────────────── 全局变量 ────────────────────────────

// endSentinel 输出终止哨兵。
//
// Python: _END (outputs.py)。用**指针身份**判定终止：forwarder 投递的是同一个
// 指针实例，因此 `chunk == endSentinel` 是接口值比较（同指针即相等）。
var endSentinel = &stream.OutputSchema{Type: "__harness_end__"}

// ──────────────────────────── 导出函数 ────────────────────────────

// Next 取下一个输出块；返回 false 表示流已终止。
func (it *OutputIterator) Next(ctx context.Context) (stream.Schema, bool) {
	if it.consumed.Load() {
		return nil, false
	}
	select {
	case <-ctx.Done():
		it.consumed.Store(true)
		return nil, false
	case chunk, ok := <-it.ch:
		if !ok {
			it.consumed.Store(true)
			return nil, false
		}
		if chunk == endSentinel {
			it.consumed.Store(true)
			return nil, false
		}
		return chunk, true
	}
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// newOutputIterator 创建输出迭代器。
func newOutputIterator(ch <-chan stream.Schema) *OutputIterator {
	return &OutputIterator{ch: ch}
}
```

> 说明：`agentcore/session/stream` 包没有 `New*Schema` 构造助手，一律用 `&stream.OutputSchema{...}` 字面量；终止哨兵靠**指针身份**比较（forwarder 投递同一指针实例）。

package harness

// ──────────────────────────── 结构体 ────────────────────────────

// controlCmd 控制命令。
//
// Python: _CmdSend/_CmdAbort/_CmdPause/_CmdResume/_CmdRoundFinished/_CmdStop
// (agent_teams/harness/control.py)。Go 用接口 + 各自结构体表达，
// 每条命令携带应答通道，等价 Python 的 ack: asyncio.Future。
type controlCmd interface {
	// ack 返回应答通道
	ack() chan error
}

// cmdSend 投递一条输入。
type cmdSend struct {
	// content 输入内容（string 或 InteractiveInput）
	content any
	// immediate true 转向当前轮，false 排队等本轮结束
	immediate bool
	// reply 应答通道
	reply chan error
}

// ack 实现 controlCmd。
func (c *cmdSend) ack() chan error { return c.reply }

// cmdAbort 中断。
type cmdAbort struct {
	// immediate true 立即取消调度任务，false 让本轮自然收尾
	immediate bool
	// reply 应答通道
	reply chan error
}

// ack 实现 controlCmd。
func (c *cmdAbort) ack() chan error { return c.reply }

// cmdPause 暂停（25.1c 前恒返回不支持）。
type cmdPause struct {
	// reply 应答通道
	reply chan error
}

// ack 实现 controlCmd。
func (c *cmdPause) ack() chan error { return c.reply }

// cmdResume 继续（25.1c 前恒返回不支持）。
type cmdResume struct {
	// query 继续时可选拼接的新输入
	query any
	// reply 应答通道
	reply chan error
}

// ack 实现 controlCmd。
func (c *cmdResume) ack() chan error { return c.reply }

// cmdRoundFinished 一轮结束（由轮次协程投递）。
type cmdRoundFinished struct {
	// roundID 轮次序号
	roundID int
	// result 本轮结果
	result map[string]any
	// err 本轮错误
	err error
	// reply 应答通道
	reply chan error
}

// ack 实现 controlCmd。
func (c *cmdRoundFinished) ack() chan error { return c.reply }

// cmdStop 停机。
type cmdStop struct {
	// reply 应答通道
	reply chan error
}

// ack 实现 controlCmd。
func (c *cmdStop) ack() chan error { return c.reply }

// ──────────────────────────── 非导出函数 ────────────────────────────

// ack 非阻塞回写应答；通道满则丢弃，保证 supervisor 崩溃路径不卡死。
//
// Python: NativeHarness._ack
func ack(ch chan error, err error) {
	if ch == nil {
		return
	}
	select {
	case ch <- err:
	default:
	}
}
```

import (
	"context"
	"errors"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/session"

// HarnessProtocol 并发安全的多轮交互契约。
//
// Python: HarnessProtocol (agent_teams/harness/protocol.py，10 方法 + 2 属性)。
// 契约（写进接口而非实现）：
//  1. 所有方法可从任意 goroutine 调用，调用方无需外部加锁；
//  2. 外部只"入队并等应答"，状态只由 supervisor 单写者修改；
//  3. 订阅回调在 supervisor 协程内同步执行，必须便宜且不可阻塞。
type HarnessProtocol interface {
	// State 返回当前生命周期相位
	State() HarnessState
	// SessionID 返回拥有的会话标识（start 前为空串）
	SessionID() string
	// Start 初始化并启动 supervisor
	Start(ctx context.Context, sess *session.Session) error
	// Stop 停止 supervisor、取消在飞轮次、关闭输出
	Stop(ctx context.Context) error
	// Outputs 返回输出迭代器（单消费者）
	Outputs() *OutputIterator
	// Send 投递输入；immediate=true 转向当前轮，false 排队等本轮结束
	Send(ctx context.Context, content any, immediate bool) error
	// Abort 中断；immediate=true 立即取消调度任务，false 让本轮自然收尾
	Abort(ctx context.Context, immediate bool) error
	// Pause 暂停到可恢复边界（25.1c 前返回 ErrPauseUnsupported）
	Pause(ctx context.Context) error
	// Resume 从暂停处继续（25.1c 前返回 ErrPauseUnsupported）
	Resume(ctx context.Context, query any) error
	// Subscribe 订阅生命周期事件
	Subscribe(ctx context.Context, onState StateHandler, onRound RoundHandler) error
}

// HarnessStateEvent 相位变化事件。
//
// Python: harness.state 事件载荷（kwargs old/new/session_id）
type HarnessStateEvent struct {

	Old HarnessState
	// New 变化后相位
	New HarnessState
	// SessionID 所属会话
	SessionID string
}

	// Kind 取值 started/finished/aborted/paused/failed
	Kind string
	// RoundID 轮次序号
	RoundID int
	// Result 本轮结果（可为 nil）
	Result map[string]any
}

// StateHandler 相位事件回调。

// RoundHandler 轮次事件回调。
type RoundHandler func(ctx context.Context, ev HarnessRoundEvent)

// eventBus 生命周期事件总线。
//
// Python: self._events = AsyncCallbackFramework(enable_metrics=False, enable_logging=False)
// Go 复用 agentcore/runner/callback.CallbackFramework 的自定义事件通道
// （OnCustom/TriggerCustom），事件名用 "harness.state"/"harness.round"，
// 命名空间 native_harness。上游的 metrics/logging 噪声开关在 Go 框架中不存在。
type eventBus struct {
	// fw 底层回调框架
	fw *callback.CallbackFramework
}

// ──────────────────────────── 枚举 ────────────────────────────

// 事件名常量。

	// eventNameState 相位事件名
	eventNameState = "harness.state"
	// eventNameRound 轮次事件名
	eventNameRound = "harness.round"
	// eventNamespace 事件命名空间
	eventNamespace = "native_harness"
)

var ErrPauseUnsupported = errors.New("harness: pause/resume 尚未实现（25.1c）")

// ──────────────────────────── 非导出函数 ────────────────────────────

// newEventBus 创建事件总线。
func newEventBus() *eventBus {
	return &eventBus{fw: callback.NewCallbackFramework()}
}

// subscribe 注册事件回调；回调在触发方协程内同步执行。
func (b *eventBus) subscribe(_ context.Context, onState StateHandler, onRound RoundHandler) error {
	if onState != nil {
		b.fw.OnCustom(eventNameState, func(ctx context.Context, data map[string]any) any {
			onState(ctx, HarnessStateEvent{
				Old:       toHarnessState(data["old"]),
				New:       toHarnessState(data["new"]),
				SessionID: toString(data["session_id"]),
			})
			return nil
		}, callback.WithNamespace(eventNamespace))
	}
	if onRound != nil {
		b.fw.OnCustom(eventNameRound, func(ctx context.Context, data map[string]any) any {
			onRound(ctx, HarnessRoundEvent{
				Kind:    toString(data["kind"]),
				RoundID: toInt(data["round_id"]),
				Result:  toMap(data["result"]),
			})
			return nil
		}, callback.WithNamespace(eventNamespace))
	}
	return nil
}

// emitState 触发相位事件。
func (b *eventBus) emitState(ctx context.Context, ev HarnessStateEvent) {
	b.fw.TriggerCustom(ctx, eventNameState, map[string]any{
		"old":        string(ev.Old),
		"new":        string(ev.New),
		"session_id": ev.SessionID,
	})
}

// emitRound 触发轮次事件。
func (b *eventBus) emitRound(ctx context.Context, ev HarnessRoundEvent) {
	b.fw.TriggerCustom(ctx, eventNameRound, map[string]any{
		"kind":     ev.Kind,
		"round_id": ev.RoundID,
		"result":   ev.Result,
	})
}

// toHarnessState 载荷转相位。
func toHarnessState(v any) HarnessState {
	if s, ok := v.(string); ok {
		return HarnessState(s)
	}
	if st, ok := v.(HarnessState); ok {
		return st
	}
	return HarnessStateIdle
}

// toString 载荷转字符串。
func toString(v any) string {
	if s, ok := v.(string); ok {
		return s
	}
	return ""
}

// toInt 载荷转整数。
func toInt(v any) int {
	switch n := v.(type) {
	case int:
		return n
	case int64:
		return int(n)
	case float64:
		return int(n)
	}
	return 0
}

// toMap 载荷转 map。
func toMap(v any) map[string]any {
	if m, ok := v.(map[string]any); ok {
		return m
	}
	return nil
}

// NativeHarness 团队成员的多轮交互层，组合一台 DeepAgent。

// Start 初始化内核并启动 supervisor 与 forwarder。
//
// Python: NativeHarness.start (native_harness.py:348)
func (h *NativeHarness) Start(ctx context.Context, sess *session.Session) error {
	h.startMu.Lock()
	defer h.startMu.Unlock()

	if h.started {
		if h.sess != nil && sess != nil && h.sess.GetSessionID() != sess.GetSessionID() {
			return fmt.Errorf("harness: 交互循环已绑定会话 %s，无法改绑 %s",
				h.sess.GetSessionID(), sess.GetSessionID())
		}
		return nil
	}

	// 步骤 2：初始化 rails（模板热加载在 25.1b 随 spec 驱动一起落）
	if err := h.agent.EnsureInitialized(ctx); err != nil {
		return fmt.Errorf("harness: 初始化失败: %w", err)
	}

	// 步骤 3：定会话
	if sess == nil {
		return fmt.Errorf("harness: Start 需要显式会话")
	}
	h.sess = sess
	h.ownSession = false

	// 步骤 4：建任务循环内核
	coord, ctrl, err := h.agent.PrepareInteractionTaskLoop(ctx, sess)
	if err != nil {
		return fmt.Errorf("harness: 准备任务循环内核失败: %w", err)
	}
	h.coordinator = coord
	h.controller = ctrl
	coord.Reset()

	// 步骤 5：起两条协程
	h.supervisorDone = make(chan struct{})
	h.forwarderDone = make(chan struct{})
	h.ready = make(chan struct{}, 1)
	h.started = true
	h.transition(ctx, HarnessStateRunning)

	go h.supervisorLoop(ctx)
	go h.forwardOutputs(ctx)

	// 步骤 6：等 supervisor 抵达首个等待点
	select {
	case <-h.ready:
	case <-ctx.Done():
		return ctx.Err()
	}
	return nil
}

// Stop 取消在飞轮次、停 supervisor、关闭输出。
//
// Python: NativeHarness.stop (native_harness.py:393)
func (h *NativeHarness) Stop(ctx context.Context) error {
	h.startMu.Lock()
	if !h.started || h.stopping {
		h.startMu.Unlock()
		return nil
	}
	h.stopping = true
	done := h.supervisorDone
	h.startMu.Unlock()

	// 入队停机命令；队列满或已停时降级为直接取消（由 supervisor 退出兜底）
	reply := make(chan error, 1)
	select {
	case h.cmd <- &cmdStop{reply: reply}:
		select {
		case <-reply:
		case <-ctx.Done():
		}
	case <-ctx.Done():
	}

	if done != nil {
		select {
		case <-done:
		case <-ctx.Done():
		}
	}
	return nil
}

// transition 切相位并发事件（仅 supervisor 或 Start 内调用）。
func (h *NativeHarness) transition(ctx context.Context, next HarnessState) {
	old := h.st.Phase()
	if old == next {
		return
	}
	h.st.setPhase(next)
	h.events.emitState(ctx, HarnessStateEvent{
		Old:       old,
		New:       next,
		SessionID: h.SessionID(),
	})
}

// supervisorLoop 控制命令唯一消费者与状态唯一写者。
//
// Python: NativeHarness._supervisor (native_harness.py:777)
func (h *NativeHarness) supervisorLoop(ctx context.Context) {
	defer close(h.supervisorDone)
	defer h.failRemainingCommands(ctx)

	signaled := false
	signalReady := func() {
		if signaled {
			return
		}
		signaled = true
		select {
		case h.ready <- struct{}{}:
		default:
		}
	}

	for {
		if ctx.Err() != nil {
			return
		}
		select {
		case <-ctx.Done():
			return
		case c := <-h.cmd:
			if stop := h.dispatch(ctx, c); stop {
				return
			}
			signalReady()
		}
	}
}

// forwardOutputs 把会话流泵入输出队列，结束时投 END。
//
// Python: NativeHarness._forward_outputs (native_harness.py:1658)
func (h *NativeHarness) forwardOutputs(ctx context.Context) {
	defer close(h.forwarderDone)
	defer func() {
		// 结束后投一次 END，让 outputs() 终止（C17）
		select {
		case h.queue <- endSentinel:
		default:
		}
	}()

	ch := h.sess.StreamIterator()
	for {
		select {
		case <-ctx.Done():
			return
		case chunk, ok := <-ch:
			if !ok {
				return
			}
			select {
			case h.queue <- chunk:
			case <-ctx.Done():
				return
			}
		}
	}
}

// dispatch 处理一条控制命令；返回 true 表示应退出 supervisor。
func (h *NativeHarness) dispatch(_ context.Context, c controlCmd) bool {
	switch cmd := c.(type) {
	case *cmdStop:
		h.stopping = true
		h.transition(context.Background(), HarnessStateTerminated)
		h.abortActiveRound()
		ack(cmd.reply, nil)
		return true
	case *cmdPause:
		ack(cmd.reply, ErrPauseUnsupported)
	case *cmdResume:
		ack(cmd.reply, ErrPauseUnsupported)
	default:
		ack(c.ack(), nil)
	}
	return false
}

// abortActiveRound 取消当前活动轮次。
func (h *NativeHarness) abortActiveRound() {
	if active := h.st.Active(); active != nil && active.Cancel != nil {
		active.Cancel()
	}
}

// failRemainingCommands supervisor 退出时把队列中剩余命令统一回错误，
// 避免调用方永久阻塞（对齐 Python _fail_remaining_commands）。
func (h *NativeHarness) failRemainingCommands(_ context.Context) {
	for {
		select {
		case c := <-h.cmd:
			ack(c.ack(), fmt.Errorf("harness: supervisor 已退出"))
		default:
			return
		}
	}
}

// Python: NativeHarness.send (native_harness.py:562)
func (h *NativeHarness) Send(ctx context.Context, content any, immediate bool) error {
	reply := make(chan error, 1)
	if err := h.enqueue(ctx, &cmdSend{content: content, immediate: immediate, reply: reply}); err != nil {
		return err
	}
	return waitAck(ctx, reply)
}

// enqueue 把命令放入控制队列，队列满或上下文取消时返回错误。
func (h *NativeHarness) enqueue(ctx context.Context, c controlCmd) error {
	select {
	case h.cmd <- c:
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
}

// waitAck 等待命令应答。
func waitAck(ctx context.Context, reply chan error) error {
	select {
	case err := <-reply:
		return err
	case <-ctx.Done():
		return ctx.Err()
	}
}

// onSend 处理投递命令。
//
// Python: NativeHarness._on_send (native_harness.py:867)
func (h *NativeHarness) onSend(ctx context.Context, cmd *cmdSend) error {
	if h.st.Phase() == HarnessStateRunning && h.st.Active() != nil {
		if cmd.immediate {
			// 转向：交给当前轮的内核 steer 通道
			return h.steerActive(ctx, cmd.content)
		}
		h.enqueueInbox(InboxMessage{Content: cmd.content, Immediate: false})
		return nil
	}
	return h.startRound(ctx, cmd.content, false)
}

// enqueueInbox 追加待驱动输入。
func (h *NativeHarness) enqueueInbox(m InboxMessage) {
	h.st.mu.Lock()
	defer h.st.mu.Unlock()
	h.st.queue = append(h.st.queue, m)
}

// startRound 起一轮执行。
//
// Python: NativeHarness._start_round / _run_round（本阶段不含快照与 turn）
func (h *NativeHarness) startRound(ctx context.Context, query any, isFollowUp bool) error {
	roundID := h.st.NextRoundID()
	roundCtx, cancel := context.WithCancel(ctx)
	active := &ActiveRound{
		RoundID: roundID,
		TaskID:  fmt.Sprintf("round-%d", roundID),
		Query:   query,
		Cancel:  cancel,
		Done:    make(chan struct{}),
	}
	h.st.SetActive(active)
	h.transition(ctx, HarnessStateRunning)
	h.events.emitRound(ctx, HarnessRoundEvent{Kind: "started", RoundID: roundID})

	go h.runRound(roundCtx, active, isFollowUp)
	return nil
}

// steerActive 把转向输入交给当前轮。
func (h *NativeHarness) steerActive(ctx context.Context, content any) error {
	text, ok := content.(string)
	if !ok {
		return fmt.Errorf("harness: 转向输入必须是字符串，实际 %T", content)
	}
	return h.controller.EnqueueSteer(ctx, text)
}

// runRound 驱动一轮并回报结果。
func (h *NativeHarness) runRound(ctx context.Context, active *ActiveRound, isFollowUp bool) {
	defer close(active.Done)

	h.st.mu.Lock()
	ctrl := h.controller
	h.st.mu.Unlock()

	query, _ := active.Query.(string)
	err := ctrl.SubmitRound(ctx, h.sess, query, isFollowUp, false,
		agentinterfaces.RunKindUser, nil, task_loop.WithTaskID(active.TaskID))
	if err == nil {
		_ = ctrl.WaitRoundCompletion(ctx, nil)
	}

	reply := make(chan error, 1)
	select {
	case h.cmd <- &cmdRoundFinished{roundID: active.RoundID, err: err, reply: reply}:
		_ = waitAck(context.WithoutCancel(ctx), reply)
	case <-ctx.Done():
	}
}

// settleRoundDone 结算一轮：发事件、清活动轮、决定是否继续。
//
// Python: NativeHarness._settle_round_done (native_harness.py:1098)
func (h *NativeHarness) settleRoundDone(ctx context.Context, cmd *cmdRoundFinished) {
	kind := "finished"
	if cmd.err != nil {
		kind = "failed"
	}
	h.st.SetActive(nil)
	h.events.emitRound(ctx, HarnessRoundEvent{Kind: kind, RoundID: cmd.roundID})

	// 队列里还有输入 → 立刻起下一轮
	h.st.mu.Lock()
	var next *InboxMessage
	if len(h.st.queue) > 0 {
		m := h.st.queue[0]
		h.st.queue = h.st.queue[1:]
		next = &m
	}
	h.st.mu.Unlock()

	if next != nil {
		_ = h.startRound(ctx, next.Content, true)
		return
	}
	if kind == "finished" {
		h.transition(ctx, HarnessStateIdle)
	}
}

// RunOnce 执行一次非流式调用，绕过 supervisor。
//
// Python: NativeHarness.run_once (native_harness.py:301)
// 消费者：scheduler reviewer（25.3）、swarmflow worker、TinyAgent 单次。
func (h *NativeHarness) RunOnce(ctx context.Context, content any, sess *session.Session) (map[string]any, error) {
	h.startMu.Lock()
	started := h.started
	h.startMu.Unlock()
	if started {
		return nil, fmt.Errorf("harness: supervisor 运行中不允许 run_once（请用 Send）")
	}

	own := false
	if sess == nil {
		sess = session.CreateAgentSession("default", h.agent.Card(), nil)
		if err := sess.PreRun(ctx, map[string]any{}); err != nil {
			return nil, err
		}
		own = true
	}
	if own {
		defer func() { _ = sess.PostRun(ctx) }()
	}
	defer func() { _ = h.agent.AbilityManager().TeardownTools() }()

	return h.agent.Invoke(ctx, map[string]any{"query": content})
}

// Dispose 永久收尾：停机 + 关闭内存管理器。
//
// Python: NativeHarness.dispose / _close_memory_managers (native_harness.py:464/499)
func (h *NativeHarness) Dispose(ctx context.Context) error {
	if err := h.Stop(ctx); err != nil {
		return err
	}
	_ = h.agent.AbilityManager().TeardownTools()
	return nil
}

		return false
	}
	taskErr, ok := err.(*taskFailedError)
	if !ok {
		return false
	}
	code := taskErr.Code()
	text := taskErr.Text()
	if !isRetryableErrorCode(code) || sc.retryAttempt >= maxRetryAttempts {
		logger.Error(logComponent).Int("error_code", code).
			Int("attempts", sc.retryAttempt).Str("error_text", text).
			Msg("DeepAgent 轮次失败")
		return false
	}

	sc.retryAttempt++
	logger.Warn(logComponent).Int("error_code", code).
		Int("attempt", sc.retryAttempt).Int("max_attempts", maxRetryAttempts).
		Str("error_text", text).Msg("DeepAgent 轮次瞬时错误")

	sc.swallowFailedRound = true
	sc.emitRetryOutput(ctx, code, text)
	if h := sc.resources.Harness; h != nil {
		if sendErr := h.Send(ctx, retryQuery, false); sendErr != nil {
			logger.Warn(logComponent).Err(sendErr).Msg("重试指令投递失败")
		}
	}
	return true
}

// emitRetryOutput 把重试提示作为可见的 llm_output 吐出。
//
// Python: StreamController._emit_retry_output (stream_controller.py:371)
func (sc *StreamController) emitRetryOutput(ctx context.Context, code int, text string) {
	content := fmt.Sprintf("\n\n[Retry %d/%d] %s\n\n", sc.retryAttempt, maxRetryAttempts, text)
	tagged := sc.tagChunk(&stream.OutputSchema{
		Type: "llm_output",
		Index: 0,
		Payload: map[string]any{
			"content":      content,
			"retrying":     true,
			"attempt":      sc.retryAttempt,
			"max_attempts": maxRetryAttempts,
			"error_code":   code,
		},
	})
	if sc.streamQueue != nil {
		select {
		case sc.streamQueue <- tagged:
		case <-ctx.Done():
		}
	}
	sc.fanOutToObservers(ctx, tagged)
}
