// BuildAcpSessionUpdate 构造 session/update。
// Python: build_acp_session_update (L101-147)。
// 分流：delta/reasoning → 文本 chunk（IsReasoningEvent 复用）；tool_call → BuildAcpToolCallUpdate；
// tool_update/tool_result → BuildAcpToolResultUpdate；subtask_update → plan 原样；
// todo_updated → BuildAcpTodoUpdate；processing_status → session_info_update；其余 nil。
func BuildAcpSessionUpdate(msg *schema.Message, payload map[string]any, state *AcpSessionUpdateState) map[string]any {
	switch msg.EventType {
	case schema.EventTypeChatDelta, schema.EventTypeChatReasoning:
		text := strAny(payload["content"])
		if text == "" {
			return nil
		}
		if IsReasoningEvent(msg.EventType, payload) {
			state.ThoughtText += text
			return map[string]any{
				"sessionUpdate": "agent_thought_chunk",
				"messageId":     ensureThoughtMessageID(state),
				"content":       map[string]any{"type": "text", "text": text},
			}
		}
		state.AssistantText += text
		return map[string]any{
			"sessionUpdate": "agent_message_chunk",
			"messageId":     ensureAssistantMessageID(state),
			"content":       map[string]any{"type": "text", "text": text},
		}
	case schema.EventTypeChatToolCall:
		return BuildAcpToolCallUpdate(payload, ensureToolCallCache(state))
	case schema.EventTypeChatToolUpdate, schema.EventTypeChatToolResult:
		return BuildAcpToolResultUpdate(payload, ensureToolCallCache(state))
	case schema.EventTypeChatSubtaskUpdate:
		return map[string]any{
			"sessionUpdate": "plan",
			"plan":          cpMap(payload),
		}
	case schema.EventTypeTodoUpdated:
		return BuildAcpTodoUpdate(payload)
	case schema.EventTypeChatProcessingStatus:

type buildIncrementalTextUpdateKind int

const (

	thoughtTextKind

// buildIncrementalTextUpdate 增量三态构造。
// Python: _build_incremental_text_update (L59-98)：
// 空 text → nil；existing 空 → 直发+ensure；text == existing → nil；
// 前缀扩展 → delta+ensure；回退 → 全文+reset。
// Go 形态差异（对应 Python 传 _ensure/_reset 函数对）：以 kind 枚举 + 延迟调用
// ensureXxxID/resetXxxID 保证 reset 不被急切求值（否则会无条件换 ID，破坏前缀
// 扩展复用 messageId 的语义）。
func buildIncrementalTextUpdate(state *AcpSessionUpdateState, text string, kind buildIncrementalTextUpdateKind, updateKind string) map[string]any {
	if text == "" {
		return nil
	}
	var existing string
	if kind == thoughtTextKind {
		existing = state.ThoughtText
	} else {
		existing = state.AssistantText
	}
	if existing != "" {
		if text == existing {
			return nil
		}
		var delta string
		if strings.HasPrefix(text, existing) {
			delta = text[len(existing):]
		}
		if delta != "" {
			if kind == thoughtTextKind {
				state.ThoughtText = existing + delta
			} else {
				state.AssistantText = existing + delta
			}
			return map[string]any{
				"sessionUpdate": updateKind,
				"messageId":     ensureIDForKind(state, kind),
				"content":       map[string]any{"type": "text", "text": delta},
			}
		}
		// 回退：全文重发 + reset 换新 messageId
		if kind == thoughtTextKind {
			state.ThoughtText = text
		} else {
			state.AssistantText = text
		}
		return map[string]any{
			"sessionUpdate": updateKind,
			"messageId":     resetIDForKind(state, kind),
			"content":       map[string]any{"type": "text", "text": text},
		}
	}
	// 全新：直发全文 + ensure
	if kind == thoughtTextKind {
		state.ThoughtText = text
	} else {
		state.AssistantText = text
	}
	return map[string]any{
		"sessionUpdate": updateKind,
		"messageId":     ensureIDForKind(state, kind),
		"content":       map[string]any{"type": "text", "text": text},
	}
}

// cpMap 一层浅拷贝（e2a 包 copyMapAny 同构，acp 包内私有）。
func cpMap(m map[string]any) map[string]any {
	out := make(map[string]any, len(m))
	for k, v := range m {
		out[k] = v
	}
	return out
}

// ensureIDForKind 槽位对应的 ensure messageID。
func ensureIDForKind(state *AcpSessionUpdateState, kind buildIncrementalTextUpdateKind) string {
	if kind == thoughtTextKind {
		return ensureThoughtMessageID(state)
	}
	return ensureAssistantMessageID(state)
}

// resetIDForKind 槽位对应的 reset messageID。
func resetIDForKind(state *AcpSessionUpdateState, kind buildIncrementalTextUpdateKind) string {
	if kind == thoughtTextKind {
		return resetThoughtMessageID(state)
	}
	return resetAssistantMessageID(state)
}

	if v, ok := payload["is_processing"].(bool); ok {

	}
	return "processing"
}

// Python: handle_jsonrpc_request (L241-333)。
// 返回 true = 已处理；false = 非 JSON-RPC 请求（交回调用方）。
func (b *AcpGatewayBridge) HandleJsonrpcRequest(ctx context.Context, ws *websocket.Conn, data map[string]any) bool {
	if !IsJsonrpcRequest(data) {
		return false
	}
	rpcID := data["id"]
	method := strings.TrimSpace(strVal(data["method"]))
	params, _ := data["params"].(map[string]any)
	if params == nil {
		params = map[string]any{}
	}

	if method == "initialize" {
		// Python L254-256：build_acp_initialize_result()（clientCapabilities 合并在
		// Python protocol.py 构造内完成；Go BuildAcpInitializeResult 已含合并逻辑则直用，
		// 写码时核对 protocol_test.go——若未合并，此处 params["clientCapabilities"] 合入）
		result := acp.BuildAcpInitializeResult()
		if caps, ok := params["clientCapabilities"].(map[string]any); ok && len(caps) > 0 {
			result["clientCapabilities"] = cpMap(caps)
		}
		b.sendRawJsonrpcResult(ws, rpcID, result)
		return true
	}
	if method == "session/new" {
		sessionID := strings.TrimSpace(strVal(params["sessionId"]))
		if sessionID == "" {
			sessionID = "acp_" + randHexStr(12)
		}
		b.bindSession(ws, sessionID)
		b.sendRawJsonrpcResult(ws, rpcID, acp.BuildAcpSessionNewResult(sessionID))
		return true
	}
	if method == "session/prompt" {
		sessionID := strings.TrimSpace(strVal(params["sessionId"]))
		if sessionID == "" {
			b.sendRawJsonrpcError(ws, rpcID, -32602, "sessionId is required")
			return true
		}
		text := ExtractPromptText(params)
		if text == "" {
			b.sendRawJsonrpcError(ws, rpcID, -32602, "prompt is required")
			return true
		}
		requestID := "acp_" + randHexStr(12)
		b.bindSession(ws, sessionID)
		b.mu.Lock()
		b.requestCtxByReqID[requestID] = &AcpGatewayRequestCtx{
			JSONRPCID:     rpcID,
			SessionID:     sessionID,
			UserMessageID: strings.TrimSpace(strVal(params["messageId"])),
			ClientWS:      ws,
		}
		b.activePromptBySession[sessionID] = requestID
		b.mu.Unlock()
		mParams := cpMap(params)
		mParams["content"] = text
		mParams["query"] = text
		mParams["session_id"] = sessionID
		paramsJSON, _ := json.Marshal(mParams)
		msg := &schema.Message{
			ID:        requestID,
			Type:      schema.MessageTypeReq,
			ChannelID: b.channelID,
			SessionID: sessionID,
			Params:    paramsJSON,
			Timestamp: schema.NowTimestamp(),
			OK:        true,
			ReqMethod: schema.ReqMethodChatSend,
			Mode:      schema.ModeAgentPlan,
			Metadata:  map[string]any{"acp": map[string]any{"jsonrpc_id": rpcID, "method": method}},
		}
		b.dispatchMessage(ctx, msg)
		return true
	}
	if method == "session/cancel" {
		sessionID := strings.TrimSpace(strVal(params["sessionId"]))
		if sessionID != "" {
			b.mu.Lock()
			requestID := b.activePromptBySession[sessionID]
			delete(b.activePromptBySession, sessionID)
			var pctx *AcpGatewayRequestCtx
			if requestID != "" {
				pctx = b.requestCtxByReqID[requestID]
				delete(b.requestCtxByReqID, requestID)
			}
			b.mu.Unlock()
			if pctx != nil {
				b.sendRawJsonrpcResult(ws, pctx.JSONRPCID,
					acp.BuildAcpPromptResult("cancelled", pctx.UserMessageID))
			}
		}
		b.sendRawJsonrpcResult(ws, rpcID, nil)
		return true
	}
	if method == "session/list" {
		b.mu.Lock()
		sids := make([]string, 0, len(b.knownSessions))
		for sid := range b.knownSessions {
			sids = append(sids, sid)
		}
		sort.Strings(sids)
		b.mu.Unlock()
		b.sendRawJsonrpcResult(ws, rpcID, acp.BuildAcpSessionListResult(sids))
		return true
	}
	if method == "session/load" {
		b.sendRawJsonrpcError(ws, rpcID, -32601,
			"Method not supported by agent capabilities: session/load")
		return true
	}
	b.sendRawJsonrpcError(ws, rpcID, -32601, "Method not found: "+method)
	return true
}

// IsJsonrpcRequest 判定 JSON-RPC 请求。
// Python: is_jsonrpc_request (L462-468)：dict + jsonrpc=="2.0" + method 是 str。
func IsJsonrpcRequest(data map[string]any) bool {
	if data == nil {
		return false
	}
	if v, _ := data["jsonrpc"].(string); v != "2.0" {
		return false
	}
	_, ok := data["method"].(string)
	return ok
}

// ExtractPromptText 提取 prompt 文本。
// Python: extract_prompt_text (L470-486)：prompt list[text] 拼接（有 text 项时），
// 否则 text/content/query 依序取非空白串。
func ExtractPromptText(params map[string]any) string {
	if prompt, ok := params["prompt"].([]any); ok {
		texts := make([]string, 0, len(prompt))
		for _, item := range prompt {
			if m, ok := item.(map[string]any); ok && m["type"] == "text" {
				if text, ok := m["text"].(string); ok && text != "" {
					texts = append(texts, text)
				}
			}
		}
		if len(texts) > 0 {
			return strings.Join(texts, "\n")
		}
	}
	for _, key := range []string{"text", "content", "query"} {
		if v, ok := params[key].(string); ok && strings.TrimSpace(v) != "" {
			return strings.TrimSpace(v)
		}
	}
	return ""
}

// Python: inbound_intercept (L147-198)。
// 返回 true = 已拦截（响应帧或软忽略帧）；false = 非响应帧，交回路由。
func (b *AcpGatewayBridge) InboundIntercept(ws *websocket.Conn, data map[string]any) bool {
	if data == nil {
		return false
	}
	if v, _ := data["jsonrpc"].(string); strings.TrimSpace(v) != "2.0" {
		return false
	}
	idStr := strings.TrimSpace(strVal(data["id"]))
	_, hasResult := data["result"]
	_, hasError := data["error"]
	if idStr == "" || (!hasResult && !hasError) {
		return false
	}

	jsonrpcID := idStr
	b.mu.Lock()
	pending, ok := b.pendingRPCSessionByID[jsonrpcID]
	if ok {
		delete(b.pendingRPCSessionByID, jsonrpcID)
	}
	b.mu.Unlock()
	sessionID := ""
	if ok {
		sessionID = strings.TrimSpace(pending.sessionID)
	}

	if jsonrpcID == "" || sessionID == "" {
		logger.Info(logComponent).
			Str("jsonrpc_id", jsonrpcID).
			Str("session_id", sessionID).
			Msg("[ACP] ignoring unknown/late client rpc response")

			"type": "res",
			"id":   jsonrpcID,
			"ok":   true,
			"payload": map[string]any{
				"accepted": false,
				"ignored":  true,
				"reason":   "unknown_or_late_response",
			},
		}))
		return true
	}

	msg := e2a.BuildACPToolResponseMessage(jsonrpcID, cpMap(data), sessionID, b.channelID)

// OutboundIntercept 出站拦截器：acp.output_request 原样转发 IDE。
// Python: outbound_intercept (L199-228)。
// Go 侧拦截器收 {"raw": msg} 包裹帧（gateway_server.go Send() 既有约定）。
// 返回 true = 已拦截；false = 非本拦截器关注的消息。
func (b *AcpGatewayBridge) OutboundIntercept(ws *websocket.Conn, data map[string]any) bool {
	msg, _ := data["raw"].(*schema.Message)
	if msg == nil {
		return false
	}
	if msg.Type != schema.MessageTypeEvent || msg.Payload == nil {
		return false
	}
	if strVal(msg.Payload["event_type"]) != "acp.output_request" {
		return false
	}
	rawJsonrpc, ok := msg.Payload["jsonrpc"].(map[string]any)
	if !ok {
		logger.Warn(logComponent).
			Msg("[ACP] outbound_intercept: invalid jsonrpc payload (not dict), dropping frame")
		return true
	}
	jsonrpcID := strings.TrimSpace(strVal(rawJsonrpc["id"]))
	sessionID := strings.TrimSpace(msg.SessionID)
	if jsonrpcID == "" || sessionID == "" {
		logger.Warn(logComponent).
			Str("jsonrpc_id", jsonrpcID).
			Str("session_id", sessionID).
			Msg("[ACP] outbound_intercept: missing jsonrpc_id or session_id, forwarding without pending registration")
	} else {
		b.mu.Lock()
		b.pendingRPCSessionByID[jsonrpcID] = pendingRPCEntry{
			sessionID: sessionID,
			ws:        ws,
			createdAt: nowFloat(),
		}
		b.mu.Unlock()
		b.registerPendingRPC(sessionID)
	}
	b.sweepStalePending()
	writeAcpFrame(ws, mustJSON(rawJsonrpc))
	return true
}

// Cleanup 断连清理：按 ws 清 pending RPC 表。
// Python: cleanup (L229-240)。
func (b *AcpGatewayBridge) Cleanup(ws *websocket.Conn) {
	b.mu.Lock()
	stale := make([]string, 0)
	for jsonrpcID, entry := range b.pendingRPCSessionByID {
		if entry.ws == ws {
			stale = append(stale, jsonrpcID)
		}
	}
	for _, jsonrpcID := range stale {
		entry := b.pendingRPCSessionByID[jsonrpcID]
		delete(b.pendingRPCSessionByID, jsonrpcID)
		b.decreasePendingRPCCount(entry.sessionID)
	}
	b.mu.Unlock()
}

	return false

// markStreamActivityLocked 流活动重置 idle（调用方持 b.mu）。
// Python: _mark_stream_activity (L106-109)。
func markStreamActivityLocked(pctx *AcpGatewayRequestCtx) {
	pctx.SawProcessingIdle = false
	cancelIdleFinalizeTimerLocked(pctx)
}

// cancelIdleFinalizeTimerLocked 取消 idle finalize timer 后置 nil（调用方持 b.mu）。
// Python: _cancel_idle_finalize_task (L99-104)。
// Go 侧取消点统一在持锁段内执行（SendMessage/清理路径），timer.Stop 后置 nil。
func cancelIdleFinalizeTimerLocked(pctx *AcpGatewayRequestCtx) {
	if pctx.idleFinalizeTimer != nil {
		pctx.idleFinalizeTimer.Stop()
		pctx.idleFinalizeTimer = nil
	}
}

// shouldWaitForFinalTextBeforeEndTurn 判定是否等迟到 final text。
// Python: _should_wait_for_final_text_before_end_turn (L111-118)：
// sawChatFinal=true → false；assistant_text 或 assistant_message_id 非空 → true。
func shouldWaitForFinalTextBeforeEndTurn(pctx *AcpGatewayRequestCtx) bool {
	if pctx.SawChatFinal {
		return false
	}
	return pctx.State.AssistantText != "" || pctx.State.AssistantMessageID != ""
}

// registerPendingRPC 会话 pending 计数 +1 并取消该会话全部 idle finalize timer。
// Python: _register_pending_client_rpc (L503-516)。
func (b *AcpGatewayBridge) registerPendingRPC(sessionID string) {
	sid := strings.TrimSpace(sessionID)
	if sid == "" {
		return
	}
	b.mu.Lock()
	b.pendingRPCCountBySession[sid]++
	entries := make([]*AcpGatewayRequestCtx, 0)
	for _, c := range b.requestCtxByReqID {
		if c.SessionID == sid {
			entries = append(entries, c)
		}
	}
	b.mu.Unlock()
	for _, c := range entries {
		b.mu.Lock()
		cancelIdleFinalizeTimerLocked(c)
		b.mu.Unlock()
	}
}

// resolvePendingRPC 会话 pending 计数 -1 + maybe finalize。
// Python: _resolve_pending_client_rpc (L517-523)。
func (b *AcpGatewayBridge) resolvePendingRPC(sessionID string) {
	sid := strings.TrimSpace(sessionID)
	if sid == "" {
		return
	}
	b.decreasePendingRPCCount(sid)
	b.maybeFinalizeDeferredPrompts(sid)
}

// decreasePendingRPCCount 计数递减（<=1 删键）。
// Python: _decrease_pending_client_rpc_count (L524-533)。
func (b *AcpGatewayBridge) decreasePendingRPCCount(sessionID string) {
	sid := strings.TrimSpace(sessionID)
	if sid == "" {
		return
	}
	b.mu.Lock()
	current := b.pendingRPCCountBySession[sid]
	if current <= 1 {
		delete(b.pendingRPCCountBySession, sid)
	} else {
		b.pendingRPCCountBySession[sid] = current - 1
	}
	b.mu.Unlock()
}

// sessionHasPendingRPCs 会话 pending 判定。
// Python: _session_has_pending_client_rpcs (L534-538)。
func (b *AcpGatewayBridge) sessionHasPendingRPCs(sessionID string) bool {
	sid := strings.TrimSpace(sessionID)
	if sid == "" {
		return false
	}
	b.mu.Lock()
	defer b.mu.Unlock()
	return b.pendingRPCCountBySession[sid] > 0
}

// maybeFinalizeDeferredPrompts pending 清零后收尾挂起的 prompt。
// Python: _maybe_finalize_deferred_prompts (L540-575)。
func (b *AcpGatewayBridge) maybeFinalizeDeferredPrompts(sessionID string) {
	sid := strings.TrimSpace(sessionID)
	if sid == "" || b.sessionHasPendingRPCs(sid) {
		return
	}
	b.mu.Lock()
	matched := make([]*AcpGatewayRequestCtx, 0)
	for requestID, c := range b.requestCtxByReqID {
		if c.SessionID == sid && strings.TrimSpace(c.PendingStopReason) != "" {
			c.matchedRequestID = requestID
			matched = append(matched, c)
		}
	}
	b.mu.Unlock()

	for _, c := range matched {
		b.mu.Lock()
		stopReason := strings.TrimSpace(c.PendingStopReason)
		if stopReason == "" {
			b.mu.Unlock()
			continue
		}
		c.PendingStopReason = ""
		ws := c.ClientWS
		b.mu.Unlock()
		if ws == nil {
			continue
		}
		if shouldWaitForFinalTextBeforeEndTurn(c) {
			b.scheduleIdleFinalize(c.matchedRequestID, c, ws)
			continue
		}
		b.mu.Lock()
		cancelIdleFinalizeTimerLocked(c)
		b.mu.Unlock()
		b.sendRawJsonrpcResult(ws, c.JSONRPCID,
			acp.BuildAcpPromptResult(stopReason, c.UserMessageID))
		b.clearRequestContext(c.matchedRequestID, sid)
	}
}

// clearRequestContext 清理请求上下文。
// Python: _clear_request_context (L577-583)。
func (b *AcpGatewayBridge) clearRequestContext(requestID, sessionID string) {
	b.mu.Lock()
	if pctx, ok := b.requestCtxByReqID[requestID]; ok {
		cancelIdleFinalizeTimerLocked(pctx)
		delete(b.requestCtxByReqID, requestID)
	}
	if b.activePromptBySession[sessionID] == requestID {
		delete(b.activePromptBySession, sessionID)
	}
	b.mu.Unlock()
}

// scheduleIdleFinalize 调度 idle finalize timer。
// Python: _schedule_idle_finalize (L585-592)。
func (b *AcpGatewayBridge) scheduleIdleFinalize(requestID string, pctx *AcpGatewayRequestCtx, ws *websocket.Conn) {
	b.mu.Lock()
	cancelIdleFinalizeTimerLocked(pctx)
	secs := b.getIdleFinalizeSeconds()
	timer := time.AfterFunc(time.Duration(secs*float64(time.Second)), func() {
		b.idleFinalizeAfterTimeout(requestID, timer, ws)
	})
	pctx.idleFinalizeTimer = timer
	b.mu.Unlock()
}

// idleFinalizeAfterTimeout timer 超时回调：引用比对防 superseded。
// Python: _idle_finalize_after_timeout (L594-615)——asyncio.current_task 比对
// 的 Go 等价：回调闭包捕获 timer 自身引用，与 ctx.idleFinalizeTimer 比对。
func (b *AcpGatewayBridge) idleFinalizeAfterTimeout(requestID string, timer *time.Timer, ws *websocket.Conn) {
	b.mu.Lock()
	pctx, ok := b.requestCtxByReqID[requestID]
	if !ok || pctx.idleFinalizeTimer != timer {
		b.mu.Unlock()
		return
	}
	if b.pendingRPCCountBySession[pctx.SessionID] > 0 {
		pctx.PendingStopReason = "end_turn"
		pctx.idleFinalizeTimer = nil
		b.mu.Unlock()
		return
	}
	b.mu.Unlock()
	b.sendRawJsonrpcResult(ws, pctx.JSONRPCID,
		acp.BuildAcpPromptResult("end_turn", pctx.UserMessageID))
	b.clearRequestContext(requestID, pctx.SessionID)

	now := nowFloat()
	b.mu.Lock()
	stale := make([][2]string, 0)
	for jsonrpcID, entry := range b.pendingRPCSessionByID {
		if now-entry.createdAt > acpPendingRPCTimeoutSeconds {
			stale = append(stale, [2]string{jsonrpcID, strings.TrimSpace(entry.sessionID)})
			delete(b.pendingRPCSessionByID, jsonrpcID)
		}

	for _, pair := range stale {
		logger.Info(logComponent).
			Str("jsonrpc_id", pair[0]).
			Msg("[ACP] pending RPC entry expired")
		b.resolvePendingRPC(pair[1])
	}
}

// session_id + params.session_id；aliased 时 metadata 记原 external。
func (mh *MessageHandler) prepareAgentDispatchMessage(_ context.Context, msg *schema.Message) *schema.Message {
	if msg.ChannelID != acpChannelIDConst {
		return msg
	}
	if msg.ReqMethod == schema.ReqMethodInitialize || msg.ReqMethod == schema.ReqMethodSessionCreate {
		return msg
	}

	internalSessionID, aliased := mh.resolveAcpInternalSessionID(msg.SessionID)
	if internalSessionID == "" {
		return msg
	}

	params := map[string]any{}
	if len(msg.Params) > 0 {
		_ = json.Unmarshal(msg.Params, &params)
	}
	params["session_id"] = internalSessionID

	metadata := map[string]any{}
	for k, v := range msg.Metadata {
		metadata[k] = v
	}
	if aliased {
		if _, exists := metadata[acpOriginalSessionIDKey]; !exists {
			metadata[acpOriginalSessionIDKey] = msg.SessionID
		}

	out.SessionID = internalSessionID
	if data, err := json.Marshal(params); err == nil {
		out.Params = data
	}
	out.Metadata = metadata
	return &out

func (mh *MessageHandler) ensureAcpAgentSession(ctx context.Context, sessionID string) (string, error) {
	requestID := fmt.Sprintf("acp-session-create-%x-%s", time.Now().UnixMilli(), randHex6())
	env := e2a.E2AFromAgentFields(requestID,
		e2a.WithFieldChannelID(acpChannelIDConst),

		e2a.WithFieldIsStream(false),
		e2a.WithFieldTimestamp(float64(time.Now().UnixNano())/1e9),
	)
	resp, err := mh.agentClient.SendRequest(ctx, env)
	if err != nil {
		return "", fmt.Errorf("acp session.create failed: %w", err)
	}
	if !resp.OK {
		payload := resp.Payload
		if payload == nil {
			payload = map[string]any{}
		}
		return "", fmt.Errorf("acp session.create failed: %v", payload["error"])

	payload := resp.Payload

		resolved = sessionID
	}

	if resolved == "" {
		return "", fmt.Errorf("acp session.create returned empty session_id")

// Python: start (L700-742)。
func (ch *AcpChannel) Start(ctx context.Context) error {
	ch.runMu.Lock()
	if ch.running {
		ch.runMu.Unlock()
		return nil
	}
	ch.running = true
	ch.runMu.Unlock()
	defer func() {
		ch.runMu.Lock()
		ch.running = false
		ch.runMu.Unlock()
	}()

	if ch.onMessageCb == nil && ch.gatewayURL != "" {
		if err := ch.ensureGatewayConnection(); err != nil {
			return err
		}
	}

	reader := bufio.NewReader(ch.stdin)
	stdinEOF := false
	var stdinEOFSince float64
	for {
		ch.runMu.Lock()
		running := ch.running
		ch.runMu.Unlock()
		if !running {
			break
		}
		if stdinEOF {
			ch.mu.Lock()
			hasPending := len(ch.requestCtx) > 0 || len(ch.pendingRPCSessionByID) > 0
			ch.mu.Unlock()
			if !hasPending {
				if nowFloat()-stdinEOFSince >= ch.stdinEOFGraceSecs {
					break
				}
			}
			select {
			case <-ctx.Done():
				return ctx.Err()
			case <-time.After(50 * time.Millisecond):
			}
			continue
		}
		// Python readline 阻塞；ctx 取消时无法中断阻塞读——接受此限制
		//（测试用 fakeStdin 即时 EOF；生产 IDE 关 stdin 即退出）
		line, err := reader.ReadString('\n')
		if err != nil {
			stdinEOF = true
			stdinEOFSince = nowFloat()
			continue
		}
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		if err := ch.handleRawLine(ctx, line); err != nil {
			logger.Error(logger.ComponentGateway).
				Err(err).
				Msg("[ACP] stdio inbound failed")
			ch.writeE2AResponse(&e2a.E2AResponse{
				ResponseID:   "acp-err-" + randHex(8),
				Status:       e2a.E2AResponseStatusFailed,
				ResponseKind: e2a.E2AResponseKindACPJSONRPCError,
				Timestamp:    e2a.UTCNowISO(),
				Provenance:   *ch.provenance("stdio_error"),
				Body:         map[string]any{"code": -32603, "message": err.Error()},
			})
		}
	}
	return nil
}

// Stop 停止：清 ctx + pending + 关网关连接。
// Python: stop (L744-749)。
func (ch *AcpChannel) Stop() {
	ch.runMu.Lock()
	ch.running = false
	ch.runMu.Unlock()

	ch.mu.Lock()
	ids := make([]string, 0, len(ch.requestCtx))
	for id := range ch.requestCtx {
		ids = append(ids, id)
	}
	ch.mu.Unlock()
	for _, id := range ids {
		ch.clearRequestContext(id)
	}

	ch.mu.Lock()
	ch.pendingRPCSessionByID = make(map[string]acpPendingEntry)
	ch.mu.Unlock()

	ch.closeGatewayConnection()
}

// Send 出站消息（双模式分派）。
// Python: send (L768-785)。
func (ch *AcpChannel) Send(ctx context.Context, msg *schema.Message) {
	ch.mu.Lock()
	ctxObj := ch.requestCtx[msg.ID]
	ch.mu.Unlock()
	if ctxObj == nil {
		logger.Debug(logger.ComponentGateway).
			Str("id", msg.ID).
			Msg("[ACP] skip outbound without request context")
		return
	}

	if ctxObj.ResponseMode == "jsonrpc" {
		isFinal := ch.sendJsonrpcMessage(ctx, msg, ctxObj)
		if isFinal {
			ch.clearRequestContext(msg.ID)
		}
		return
	}

	response := ch.messageToE2aResponse(msg, ctxObj)
	if response == nil {
		return
	}
	ch.writeE2AResponse(response)
	if response.IsFinal {
		ch.clearRequestContext(msg.ID)
	}
}

func (ch *AcpChannel) dispatchJsonrpcMethod(ctx context.Context, rpcID any, method string, params map[string]any) error {

		ch.writeJsonrpcResult(rpcID, e2aacp.BuildAcpInitializeResult())
		ch.notifyAgentInitialize(ctx, params)
		return nil
	case "session/new":
		return ch.handleJsonrpcSessionNew(rpcID, params)
	case "session/prompt":
		return ch.handleJsonrpcSessionPrompt(ctx, rpcID, params)
	case "session/cancel":
		return ch.handleJsonrpcSessionCancel(ctx, rpcID, params)
	case "session/list":
		ch.mu.Lock()
		sids := make([]string, 0, len(ch.knownSessions))
		for sid := range ch.knownSessions {
			sids = append(sids, sid)
		}
		ch.mu.Unlock()
		sort.Strings(sids)
		ch.writeJsonrpcResult(rpcID, e2aacp.BuildAcpSessionListResult(sids))
		return nil
	case "session/load":
		ch.writeJsonrpcError(rpcID, -32601,
			"Method not supported by agent capabilities: session/load")
		return nil
	default:
		ch.writeJsonrpcError(rpcID, -32601, "Method not found: "+method)
		return nil
	}
}

// Python: _envelope_to_message (L840-878)。
func (ch *AcpChannel) envelopeToMessage(env *e2a.E2AEnvelope) (*schema.Message, error) {
	method := strings.TrimSpace(env.Method)
	params := cpStrAnyMap(env.Params)
	sessionID := env.SessionID
	if sessionID == "" {
		if v, ok := params["session_id"].(string); ok && v != "" {

		}

		sessionID = ch._defaultSessionID
	}

		text := chExtractPromptText(params)
		if _, exists := params["content"]; !exists {
			params["content"] = text
		}
		if _, exists := params["query"]; !exists {
			params["query"] = text
		}
		reqMethod = schema.ReqMethodChatSend
	}
	if reqMethod == "" {
		return nil, acpValueError(fmt.Sprintf("unsupported ACP/E2A method: %s", orStr(method, "<empty>")))

// ensureGatewayConnection 建立网关 WS 连接（12 次指数退避 0.15s×2ⁿ 封顶 2s）。
// Python: _ensure_gateway_connection (L1526-1571)。
func (ch *AcpChannel) ensureGatewayConnection() error {
	ch.gwConnMu.Lock()
	if ch.gwConn != nil {
		ch.gwConnMu.Unlock()
		return nil
	}
	ch.gwConnMu.Unlock()

	var lastErr error
	for attempt := 1; attempt <= acpGatewayConnectMaxAttempts; attempt++ {
		conn, _, err := websocket.DefaultDialer.Dial(ch.gatewayURL, nil)
		if err == nil {
			ch.gwConnMu.Lock()
			ch.gwConn = conn
			ch.gwConnMu.Unlock()
			done := make(chan struct{})
			ch.gwConnMu.Lock()
			ch.readerLoopDone = done
			ch.gwConnMu.Unlock()
			go ch.gatewayReaderLoop(conn, done)
			if attempt > 1 {
				logger.Info(logger.ComponentGateway).
					Int("attempts", attempt).
					Str("gateway_url", ch.gatewayURL).
					Msg("[ACP] gateway connected after retries")
			}
			return nil
		}
		lastErr = err
		if attempt >= acpGatewayConnectMaxAttempts {
			break
		}
		delay := acpGatewayConnectBaseDelaySec * float64(int(1)<<(attempt-1))
		if delay > acpGatewayConnectMaxDelaySec {
			delay = acpGatewayConnectMaxDelaySec
		}
		logger.Warn(logger.ComponentGateway).
			Int("attempt", attempt).
			Int("max_attempts", acpGatewayConnectMaxAttempts).
			Str("error", err.Error()).
			Float64("retry_in_seconds", delay).
			Msg("[ACP] gateway connect attempt failed")
		time.Sleep(time.Duration(delay * float64(time.Second)))
	}
	return fmt.Errorf("ACP gateway connect failed after %d attempts: %s (last error: %v)",
		acpGatewayConnectMaxAttempts, ch.gatewayURL, lastErr)
}

// closeGatewayConnection Python L1573-1589。
func (ch *AcpChannel) closeGatewayConnection() {
	ch.gwConnMu.Lock()
	conn := ch.gwConn
	ch.gwConn = nil
	done := ch.readerLoopDone
	ch.readerLoopDone = nil
	ch.gwConnMu.Unlock()
	if done != nil {
		select {
		case <-done:
		case <-time.After(time.Second):
		}
	}
	if conn != nil {
		if err := conn.Close(); err != nil {
			logger.Debug(logger.ComponentGateway).
				Err(err).
				Msg("[ACP] gateway websocket close ignored")
		}
	}
}

func (ch *AcpChannel) sendToGateway(_ context.Context, msg *schema.Message) error {
	if err := ch.ensureGatewayConnection(); err != nil {
		return err
	}
	params := map[string]any{}
	if len(msg.Params) > 0 {
		_ = json.Unmarshal(msg.Params, &params)
	}
	if msg.SessionID != "" {
		if _, exists := params["session_id"]; !exists {
			params["session_id"] = msg.SessionID
		}
	}
	if msg.Mode != "" {
		if _, exists := params["mode"]; !exists {
			params["mode"] = msg.Mode.ToRuntimeMode()
		}
	}
	reqMethod := string(msg.ReqMethod)
	if reqMethod == "" {
		return fmt.Errorf("gateway forward requires req_method")
	}
	frame := map[string]any{
		"type":   "req",
		"id":     msg.ID,
		"method": reqMethod,
		"params": params,
	}

	conn := ch.gwConn
	ch.gwConnMu.Unlock()

// gatewayReaderLoop 网关读循环。
// Python: _gateway_reader_loop (L1611-1630)。
func (ch *AcpChannel) gatewayReaderLoop(conn *websocket.Conn, done chan struct{}) {
	defer close(done)
	for {
		_, raw, err := conn.ReadMessage()
		if err != nil {
			// 连接关闭/错误：静默退出；gwConn 清理由 closeGatewayConnection 或
			// finally 语义承担（读错误时对齐 Python finally 自清）
			ch.gwConnMu.Lock()
			if ch.gwConn == conn {
				ch.gwConn = nil
			}
			ch.gwConnMu.Unlock()
			logger.Debug(logger.ComponentGateway).
				Err(err).
				Msg("[ACP] gateway reader stopped")
			return
		}
		var data map[string]any
		if err := json.Unmarshal(raw, &data); err != nil {
			logger.Debug(logger.ComponentGateway).
				Str("raw", string(raw)).
				Msg("[ACP] skip invalid gateway frame")
			continue
		}
		ch.handleGatewayFrame(data)
	}
}

		ch.handleGatewayJsonrpcRequest(data)

	ch.writeLine(mustJSON(data))
}

// gateway_url or agent_server_url or GATEWAY_URL or ws://{GATEWAY_HOST}:{GATEWAY_PORT}/acp。
// env 名字面保留（第三方 env 不改名）。
func resolveAcpGatewayURL(gatewayURL, agentServerURL string, getenv func(string) string) string {
	if gatewayURL != "" {
		return gatewayURL
	}
	if agentServerURL != "" {
		return agentServerURL
	}
	if getenv != nil {
		if v := getenv("GATEWAY_URL"); v != "" {
			return v
		}
		host := getenv("GATEWAY_HOST")
		if host == "" {
			host = "127.0.0.1"
		}
		port := getenv("GATEWAY_PORT")
		if port == "" {
			port = "19001"
		}
		return fmt.Sprintf("ws://%s:%s/acp", host, port)
	}
	return "ws://127.0.0.1:19001/acp"
}

	defer cancel()

	// stdout 重定向 stderr（Python sys.stdout = sys.stderr L1776）——须在

	agentServerURL, _ := cmd.Flags().GetString("agent-server-url")

	if err != nil {
		return err
	}

	var gs *gateway.AppGatewayServer

		// 模式一：进程内自举（复用 runAppCmd 服务器段，D3）

		}
		gatewayURL = resolveAcpGatewayURL("", agentServerURL, os.Getenv)
	} else {
		// 模式二：远端桥进程（Python jiuwenswarm-acp 1:1）
		gatewayURL = resolveAcpGatewayURL(gatewayURL, agentServerURL, os.Getenv)
	}

		Str("gateway_url", gatewayURL).
		Msg("[ACP] started: AcpChannel(stdio) -> Gateway -> AgentServer")

	channel := acp.NewAcpChannel(acp.LoadAcpChannelConfig(func() map[string]any {
		raw, rerr := cfg.Raw()
		if rerr != nil {
			return nil
		}
