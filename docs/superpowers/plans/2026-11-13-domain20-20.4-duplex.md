//go:build windows

package processcli

import (
	"os"
	"syscall"
	"unsafe"
)

// windowsPipeSource 每次 bounded read 前用 PeekNamedPipe 探测同步打开的管道。
// Python: duplex_input.py:56-91 _WindowsPipe
type windowsPipeSource struct {
	fd int
}

// readAvailable 探测可用字节；EOF 错误码返回空切片；其余报错。Python :79-91
func (s *windowsPipeSource) readAvailable(size int) ([]byte, error) {
	var available uint32
	handle := os.NewFile(uintptr(s.fd), "").Fd()
	kernel32 := syscall.MustLoadDLL("kernel32.dll")
	peek := kernel32.MustFindProc("PeekNamedPipe")
	r1, _, errNo := peek.Call(
		handle, 0, 0, 0,
		uintptr(unsafe.Pointer(&available)), 0,
	)
	if r1 == 0 {
		// Python _PIPE_EOF_ERRORS = (109, 232, 233)
		if errNo == syscall.ERROR_BROKEN_PIPE || errNo == 232 || errNo == 233 {
			return []byte{}, nil
		}
		return nil, &DuplexInputError{Message: "cannot inspect the input pipe"}
	}
	if available == 0 {
		return nil, nil
	}
	if int(available) < size {
		size = int(available)
	}
	buf := make([]byte, size)
	n, err := syscall.Read(syscall.Handle(handle), buf)
	if err != nil {
		return nil, &DuplexInputError{Message: "duplex input could not be read"}
	}
	return buf[:n], nil
}

// executeDuplex --run-jsonl：行读取器读首条 run 记录 + 真 DuplexController。
// Python: machine_entry.py:122-146 _execute_duplex
func executeDuplex(writer *OneShotWriter) *protocol.OneShotRunResult {
	reader, err := NewDuplexLineReader(nil, MaxDuplexLineBytes, duplexPollInterval)
	if err != nil {
		return failure(writer, "INVALID_INPUT", err.Error(), exitInvalidInput)
	}
	defer func() { _ = reader.Close() }()
	// Python: :130 line = await reader.read_line()
	line, err := reader.ReadLine()
	if err != nil {
		return failure(writer, "INVALID_INPUT", err.Error(), exitInvalidInput)
	}
	if line == nil {
		return failure(writer, "INVALID_INPUT", "A run record is required before control input.", exitInvalidInput)
	}
	// Python: :133 read_run_input("-", stdin=io.BytesIO(line))
	input, inputErr := readRunInputFromString(string(line))
	if inputErr != nil {
		return failureFromInputError(writer, asInputError(inputErr), exitInvalidInput)
	}
	if input.RequestID != nil && strings.TrimSpace(*input.RequestID) != "" {
		writer.RequestID = strings.TrimSpace(*input.RequestID)
	}
	prepared, err := PrepareWorkspace(input)
	if err != nil {
		return failureFromInputError(writer, asInputError(err), exitInvalidInput)
	}
	// Python: :138-143 DuplexController(reader, writer)（attended，无 unattended 参数）
	control := NewDuplexController(reader, writer, false)
	return runWithSignalsWithControl(prepared, writer, control)
}
