> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把团队状态（身份 / 团队元信息 / 名册 / 政策）从「每轮重新编码的提示词附件」改为「出现时一次写进对话历史、只播增量」，基线持久化到成员子 `AgentSession`；同时落地任务通知三兄弟（F_119 / F_129）与任务表 4 列最小补齐。

```go
// Reviewers 解析 reviewer 列为成员名列表。
//
// 新格式元素是对象 {"reviewer_id":"bob",...}；字符串元素同样接受（宽松，
// 上游 reviewers() 的旧格式自动升级语义）。空值 / 非 JSON / 非列表一律返回空，
// 语义是「无验收门」，完成路径按直接 COMPLETED 处理。
//
// Python: TeamTaskBase.reviewers() (models.py L193-219)
func (t *TeamTaskBase) Reviewers() []string {
	if t == nil || t.Reviewer == nil || *t.Reviewer == "" {
		return nil
	}
	var parsed []any
	if err := json.Unmarshal([]byte(*t.Reviewer), &parsed); err != nil {
		return nil
	}
	ids := make([]string, 0, len(parsed))
	for _, entry := range parsed {
		switch v := entry.(type) {
		case string:
			if v != "" {
				ids = append(ids, v)
			}
		case map[string]any:
			if id, ok := v["reviewer_id"].(string); ok && id != "" {
				ids = append(ids, id)
			}
		}
	}
	if len(ids) == 0 {
		return nil
	}
	return ids
}

// HasReviewers 该任务是否配了验收人（决定完成时是否走 IN_REVIEW）。
// Python: 调用点 if task.reviewers():（team_context.py L899）
func (t *TeamTaskBase) HasReviewers() bool { return len(t.Reviewers()) > 0 }
```

```go
// GetUnseenEdits 查询成员名下非终态且 revision > observed_revision 的任务。
//
// 偏差 D-5：上游用 TASK_WORKING_STATUSES（planning/in_progress/in_review），
// Go 任务状态机只有 6 态，此处用 NOT IN (completed, cancelled) 近似；
// 25.4 condition-named 状态机落地后改回精确集合。
//
// Python: get_unseen_edits (task_dao.py L763-779)
func (d *SQLTaskDao) GetUnseenEdits(ctx context.Context, teamName, assignee string) ([]*TeamTaskBase, error) {
	table := d.taskTableName(ctx)
	var tasks []*TeamTaskBase
	result := d.db.WithContext(ctx).Table(table).
		Where("team_name = ?", teamName).
		Where("assignee = ?", assignee).
		Where("status NOT IN ?", []string{fsm.TaskStatusCompleted, fsm.TaskStatusCancelled}).
		Where("revision > observed_revision").
		Find(&tasks)
	if result.Error != nil {
		return nil, result.Error
	}
	return tasks, nil
}
```

```go
func (tm *TeamTaskManager) UpdateTask(ctx context.Context, taskID, title, content string) error {
	ok, err := tm.db.Task().UpdateTask(ctx, taskID, title, content)
```

```go
		}
	}
	sort.SliceStable(unseen, func(i, j int) bool {
		return noteRevision(unseen[i]) < noteRevision(unseen[j])
	})
	return unseen
}

// ChangeNotesLines 把未读修订说明渲染成每行一条 `- rN (by): note`。
//
// by 为空时省略括号段。没有未读条目时返回空切片。
//
// Python: change_notes_lines (message_template.py L424-430)
func ChangeNotesLines(task *database.TeamTaskBase) []string {
	notes := UnseenChangeNotes(task)
	if len(notes) == 0 {
		return nil
	}
	lines := make([]string, 0, len(notes))
	for _, note := range notes {
		by := ""
		if s, ok := note["by"].(string); ok && s != "" {
			by = " (" + s + ")"
		}
		noteText, _ := note["note"].(string)
		lines = append(lines, "- r"+formatRevision(noteRevision(note))+by+": "+noteText)
	}
	return lines
}

// RevisionDiff 两个正文版本的行级 diff，去掉 `---` / `+++` 文件头。
//
// 上下文取 diffContextLines 行。正文相同返回空串。
// 排版对齐 Python difflib.unified_diff(lineterm="", n=2)。
//
// Python: revision_diff (message_template.py L433-441)
func RevisionDiff(previous, current string) string {
	if previous == current {
		return ""
	}
	prevLines := strings.Split(previous, "\n")
	curLines := strings.Split(current, "\n")
	raw := unifiedDiff(prevLines, curLines, diffContextLines)
	if raw == "" {
		return ""
	}
	kept := make([]string, 0, len(raw))
	for _, line := range raw {
		if strings.HasPrefix(line, "---") || strings.HasPrefix(line, "+++") {
			continue
		}
		kept = append(kept, line)
	}
	return strings.Join(kept, "\n")
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// noteRevision 取条目里的 revision 数值，缺失或非数字返回 -1（永不匹配区间）。
func noteRevision(note map[string]any) int64 {
	switch v := note["revision"].(type) {
	case float64:
		return int64(v)
	case int:
		return int64(v)
	case int64:
		return v
	case json.Number:
		n, err := v.Int64()
		if err != nil {
			return -1
		}
		return n
	default:
		return -1
	}
}

// formatRevision 把 revision 渲染成十进制字符串。
func formatRevision(rev int64) string {
	if rev < 0 {
		return "?"
	}
	return strconv.FormatInt(rev, 10)
}
```

```go
// unifiedDiff 生成紧凑的 unified diff 行列表（含 @@ 段头，不含 ---/+++）。
//
// Python difflib.unified_diff(lineterm="", n=context) 的最小等价实现：
// 先求 LCS 得到编辑脚本，再把相邻变更合并成 hunk 并补 context 行。
func unifiedDiff(a, b []string, context int) []string {
	ops := diffOps(a, b)
	if len(ops) == 0 {
		return nil
	}
	var out []string
	i := 0
	for i < len(ops) {
		// 定位下一个变更
		for i < len(ops) && ops[i].kind == ' ' {
			i++
		}
		if i >= len(ops) {
			break
		}
		start := i - context
		if start < 0 {
			start = 0
		}
		// 向后找 hunk 的结束：超过 context*2 个相同行就断开
		end := i
		same := 0
		for end < len(ops) {
			if ops[end].kind == ' ' {
				same++
				if same > context*2 {
					break
				}
			} else {
				same = 0
			}
			end++
		}
		stop := end
		if stop > len(ops) {
			stop = len(ops)
		}
		// 计算 hunk 头
		oldStart, newStart := 1, 1
		for k := 0; k < start; k++ {
			if ops[k].kind != '+' {
				oldStart++
			}
			if ops[k].kind != '-' {
				newStart++
			}
		}
		oldCount, newCount := 0, 0
		for k := start; k < stop; k++ {
			if ops[k].kind != '+' {
				oldCount++
			}
			if ops[k].kind != '-' {
				newCount++
			}
		}
		out = append(out, "@@ -"+strconv.Itoa(oldStart)+","+strconv.Itoa(oldCount)+
			" +"+strconv.Itoa(newStart)+","+strconv.Itoa(newCount)+" @@")
		for k := start; k < stop; k++ {
			out = append(out, string(ops[k].kind)+ops[k].text)
		}
		i = stop
	}
	return out
}

// diffOp 编辑脚本的一条：kind 为 ' ' / '-' / '+'。
type diffOp struct {
	kind byte
	text string
}

// diffOps 用 LCS 求编辑脚本（DP 表，行数规模足够小）。
func diffOps(a, b []string) []diffOp {
	n, m := len(a), len(b)
	lcs := make([][]int, n+1)
	for i := range lcs {
		lcs[i] = make([]int, m+1)
	}
	for i := n - 1; i >= 0; i-- {
		for j := m - 1; j >= 0; j-- {
			if a[i] == b[j] {
				lcs[i][j] = lcs[i+1][j+1] + 1
			} else if lcs[i+1][j] >= lcs[i][j+1] {
				lcs[i][j] = lcs[i+1][j]
			} else {
				lcs[i][j] = lcs[i][j+1]
			}
		}
	}
	if lcs[0][0] == n && n == m {
		return nil // 完全相同
	}
	ops := make([]diffOp, 0, n+m)
	i, j := 0, 0
	for i < n && j < m {
		if a[i] == b[j] {
			ops = append(ops, diffOp{' ', a[i]})
			i++
			j++
		} else if lcs[i+1][j] >= lcs[i][j+1] {
			ops = append(ops, diffOp{'-', a[i]})
			i++
		} else {
			ops = append(ops, diffOp{'+', b[j]})
			j++
		}
	}
	for ; i < n; i++ {
		ops = append(ops, diffOp{'-', a[i]})
	}
	for ; j < m; j++ {
		ops = append(ops, diffOp{'+', b[j]})
	}
	return ops
}
```

```go
// TeamContextTracker 决定「这个成员还有什么团队状态没被告知」，并记住。
//
// 团队状态（自己的身份、团队元数据、同伴名册）过去走每轮提示词附件：
// 每次模型调用重新收集、重新渲染、重新追加。附件挂在窗口尾部且不写进历史，
// 因此永远无法从 KV 缓存中命中，却是每一次调用都要重新编码的固定开销。
//
// 本类型驱动替代方案：状态在**出现时**写进对话历史，且只写新增部分。
// 两条结论随之而来：
//
//   - 时机由数据驱动，不由启动驱动。leader 首次模型调用时还没有团队
//     （build_team 尚未执行），因此无内容可播；每个通道在自己的探针首次
//     产出内容的那次调用触发，此后探针推进才再触发。
//   - 基线必须持久化。rail 每轮重建，内存基线会每轮重播全部状态，
//     故基线落在成员自己的子 AgentSession state（键 team_prompt_context），
//     与对话历史同一个 checkpoint bucket，pre_run 时一起恢复。

// SnapshotText 渲染本成员当前完整团队状态，忽略任何基线。
//
// pendingText 只给「相对基线新增」的部分；当历史里已不再持有此前说过的
// 内容（首投，或 compaction 丢掉了早先的块），当前状态通道需要完整状态。
// 本方法不读也不写任何持久化数据。
//
// Python: snapshot_text (team_context.py L205-225)
func (t *TeamContextTracker) SnapshotText() (string, bool) {
	identityBody, _ := t.fullIdentityBody()
	infoBody, _ := t.fullTeamInfoBody()
	var rosterBlock *string
	if t.teamBackend != nil {
		roster, _ := t.listRoster()
		rosterBlock = t.renderRosterSnapshot(roster)
	}
	blocks := t.composeBlocks(identityBody, infoBody, rosterBlock, t.policyText)
	if len(blocks) == 0 {
		return "", false
	}
	return strings.Join(blocks, "\n\n"), true
}

// PendingText 渲染本成员尚未被告知的一切，不推进任何基线。
//
// 调用方必须在文本**真正投递之后**调用 Commit。当探针推进但没有可说的
// 文本（空名册、团队尚不存在）时，基线就地推进——那些内容不可能投递失败，
// 不推进会导致每次调用都重新查 DB。
//
// Python: pending_text (team_context.py L227-259)
func (t *TeamContextTracker) PendingText(sess sessionFacade) (string, bool) {
	t.uncommitted = nil
	if sess == nil {
		return "", false
	}
	baseline := t.readBaseline(sess)
	updated := copyBaseline(baseline)
	identityBody := t.identityBody(baseline, updated)
	infoBody := t.teamInfoBody(baseline, updated)
	policyBlock := t.policyBlock(baseline, updated)
	rosterBlock := t.rosterBlock(baseline, updated)
	blocks := t.composeBlocks(identityBody, infoBody, rosterBlock, policyBlock)
	if len(blocks) > 0 {
		t.uncommitted = updated
		return strings.Join(blocks, "\n\n"), true
	}
	if !baselineEqual(updated, baseline) {
		t.persist(sess, updated)
	}
	return "", false
}

// Commit 为刚投递的文本持久化基线。
//
// 最后一次 PendingText 无输出且未给 deliveredText 时是 no-op，调用方无需判空。
// deliveredText 非 nil 时记入基线，供重建后的成员注册同一份内容
// （内容 hash 不变 → 不产生附件更新）。
//
// Python: commit (team_context.py L261-283)
func (t *TeamContextTracker) Commit(sess sessionFacade, deliveredText *string) {
	pending := t.uncommitted
	t.uncommitted = nil
	if sess == nil {
		return
	}
	if deliveredText != nil {
		if pending == nil {
			pending = copyBaseline(t.readBaseline(sess))
		}
		pending[baselineLastDeliveredText] = *deliveredText
	}
	if pending == nil {
		return
	}
	t.persist(sess, pending)
}

// LastDeliveredText 返回上次经 deliveredText 提交的内容。
//
// 「当前状态」通道一旦某 section 未重新注册就视为撤回，而 PendingText 在
// 成员已最新时无话可说。原样重注册这份内容能让 section 保持活跃且内容
// hash 不变，从而不产生更新宣告。
//
// Python: last_delivered_text (team_context.py L285-302)
func (t *TeamContextTracker) LastDeliveredText(sess sessionFacade) (string, bool) {
	if sess == nil {
		return "", false
	}
	v, ok := t.readBaseline(sess)[baselineLastDeliveredText].(string)
	return v, ok
}

// composeBlocks 按顺序拼装各通道正文：<team-context> → <team-policy> → 名册。
//
// 顺序有语义：模型先读到「我是谁、这是什么团队」（身份与团队元数据同属
// 团队既成事实，合并进**一个** <team-context>，而不是两个相邻的同类块），
// 再读到「这个团队怎么运转」，最后读到「还有谁可以派活」。
//
// Python: _compose_blocks (team_context.py L304-357)
func (t *TeamContextTracker) composeBlocks(
	identityBody, infoBody, rosterBlock, policyBlock *string,
) []string {
	blocks := make([]string, 0, 3)
	if (identityBody != nil && *identityBody != "") || (infoBody != nil && *infoBody != "") {
		if t.forkCapable && identityBody != nil && *identityBody != "" {
			conversion := (*string)(nil)
			if t.forkSource != "" {
				c := prompts.BuildIdentityConversion(t.forkSource, t.memberName, t.language)
				conversion = &c
			}
			blocks = append(blocks, inboundtags.RenderTeamContextWithIdentity(*identityBody, conversion, infoBody))
		} else {
			standing := make([]string, 0, 2)
			if identityBody != nil && *identityBody != "" {
				standing = append(standing, *identityBody)
			}
			if infoBody != nil && *infoBody != "" {
				standing = append(standing, *infoBody)
			}
			blocks = append(blocks, inboundtags.RenderTeamContext(strings.Join(standing, "\n")))
		}
	}
	if policyBlock != nil && *policyBlock != "" {
		blocks = append(blocks, *policyBlock)
	}
	if rosterBlock != nil && *rosterBlock != "" {
		blocks = append(blocks, *rosterBlock)
	}
	return blocks
}

// policyBlock 政策文本未投递过时返回该块（基线记 sha256）。
//
// 用哈希而非布尔标志：演化后的政策模板重渲染出不同文本，那是重新投递的理由。
//
// Python: _policy_block (team_context.py L359-373)
func (t *TeamContextTracker) policyBlock(baseline, updated map[string]any) *string {
	if t.policyText == nil {
		return nil
	}
	sum := sha256.Sum256([]byte(*t.policyText))
	digest := hex.EncodeToString(sum[:])
	if prev, ok := baseline[baselinePolicyHash].(string); ok && prev == digest {
		return nil
	}
	updated[baselinePolicyHash] = digest
	return t.policyText
}

// readBaseline 读本成员已持久化的基线，空或类型不符返回空 map。
// Python: _read_baseline (team_context.py L648-654)
func (t *TeamContextTracker) readBaseline(sess sessionFacade) map[string]any {
	v, err := sess.GetState(statekey.StringKey(TeamContextStateKey))
	if err != nil {
		return map[string]any{}
	}
	m, ok := v.(map[string]any)
	if !ok {
		return map[string]any{}
	}
	return m
}

// persist 把基线写进成员 checkpoint 并 flush。
//
// commit 是让「中途崩溃 / 暂停」落在一致状态的那一步：成员的对话与本基线
// 共享同一个 agent-session state bucket，由同一次 AgentStorage.save 写入。
// 写失败只告警——绝不因为一次 checkpoint 写失败而中断模型调用。
//
// Python: _persist (team_context.py L656-671)
func (t *TeamContextTracker) persist(sess sessionFacade, st map[string]any) {
	sess.UpdateState(map[string]any{TeamContextStateKey: st})
	if committer, ok := sess.(interface{ Commit() error }); ok {
		if err := committer.Commit(); err != nil {
			logger.Warn(logComponent).
				Str("event_type", "team_context_baseline_persist_failed").
				Str("member_name", t.memberName).
				Err(err).
				Msg("failed to persist team context baseline")
		}
	}
}

// copyBaseline 浅拷贝基线（值多为标量或切片，浅拷贝足够；
// roster 切片的元素是 RosterMember 值类型，不会跨通道改写）。
func copyBaseline(src map[string]any) map[string]any {
	out := make(map[string]any, len(src)+1)
	for k, v := range src {
		out[k] = v
	}
	return out
}

// baselineEqual 比较两份基线是否等价。
func baselineEqual(a, b map[string]any) bool {
	if len(a) != len(b) {
		return false
	}
	for k, v := range a {
		other, ok := b[k]
		if !ok || !reflect.DeepEqual(v, other) {
			return false
		}
	}
	return true
}
```

```go
// identityBody 渲染身份正文；已投递过则只在 prompt 变化时重播该小节。
//
// 恒定字段（member_name / display_name / member_workspace_path）在 spawn 时
// 固定、永不变化，因此由 buildIdentityText 一次性投递，identity_emitted 门控。
// 私有工作约定（member_prompt）是唯一可能中途被手工演进的字段，且它在模型
// 侧的唯一通道就是身份正文——故一旦投递过，正文改为探测 member_prompt 的
// updated_at，变了**只**重播该小节（buildIdentityPromptDelta），恒定字段
// 从不重述。无单成员探针的后端（演化关闭 / 旧 fake）保持一次性行为。
//
// **要等成员自己的 DB 行**，因为 display_name 必须来自那一行：构造参数只是
// spec 时的默认值，leader 由 build_team 按调用方**当时**给的名字注册，告诉它
// spec 默认值等于告诉它一个全队不用的名字。teammate 在 spawn 时行已存在，
// 这个门不花代价；leader 在建队后那次调用同时收到「我是谁」与团队元数据。
//
// Python: _identity_body (team_context.py L379-455)
func (t *TeamContextTracker) identityBody(baseline, updated map[string]any) *string {
	if emitted, _ := baseline[baselineIdentityEmitted].(bool); emitted {
		// 已投递：只在 prompt 被演进时重播该小节
		if t.teamBackend == nil || t.memberName == "" {
			return nil
		}
		mtime, _ := t.teamBackend.GetMemberUpdatedAtState(context.Background(), t.memberName, "prompt")
		// 偏差 D-1：present 恒 true，故只保留 mtime 比较分支
		if prev, ok := baseline[baselineMemberPromptMtime].(int64); ok && prev == mtime {
			return nil
		}
		member, err := t.teamBackend.GetMember(context.Background(), t.memberName)
		if err != nil || member == nil {
			return nil
		}
		delta := prompts.BuildIdentityPromptDelta(member.Prompt, t.language)
		if delta == nil {
			return nil
		}
		updated[baselineMemberPromptMtime] = mtime
		return delta
	}
	body, _ := t.fullIdentityBody()
	if body == nil || *body == "" {
		return nil
	}
	updated[baselineIdentityEmitted] = true
	// 记下 prompt mtime，使后续手工演进能让上面的 delta 分支重播该小节
	if t.teamBackend != nil && t.memberName != "" {
		mtime, _ := t.teamBackend.GetMemberUpdatedAtState(context.Background(), t.memberName, "prompt")
		updated[baselineMemberPromptMtime] = mtime
	}
	return body
}

// fullIdentityBody 从成员当前 DB 行渲染完整身份正文。
//
// 成员自身行不存在时返回 (nil,false)——无 backend 时用构造参数原样渲染。
//
// Python: _full_identity_body (team_context.py L457-485)
func (t *TeamContextTracker) fullIdentityBody() (*string, bool) {
	displayName := t.displayName
	memberPrompt := t.memberPrompt
	if t.teamBackend != nil && t.memberName != "" {
		member, err := t.teamBackend.GetMember(context.Background(), t.memberName)
		if err != nil || member == nil {
			return nil, false
		}
		displayName = member.DisplayName
		if member.Prompt != "" {
			memberPrompt = member.Prompt
		}
	}
	return prompts.BuildIdentityText(
		t.memberName, displayName, t.memberWorkspacePath, memberPrompt, t.language, t.forkCapable,
	), true
}
```

```go
// claim 把本成员尚未宣告的条目标记为已宣告并渲染。
//
// sess 为 nil 时不做任何 claim：没有基线就可能重复通知。
// 返回 nil 表示没有新内容（或 tracker 未绑定）。
//
// Python: claim (team_context.py L766-793)
func (t *taskNoticeTracker) claim(sess sessioninterfaces.SessionFacade) *ClaimedNotice {
	if sess == nil || t.teamBackend == nil || t.memberName == "" {
		return nil
	}
	taskManager := t.teamBackend.TaskManager()
	if taskManager == nil {
		return nil
	}
	rows := t.candidates(taskManager)
	// 从这里起不再有 await 语义：绑定、检查、标记是一步（Go 用锁保证）
	t.mu.Lock()
	defer t.mu.Unlock()
	t.bindLocked(sess)
	fresh := make([]*database.TeamTaskBase, 0, len(rows))
	entries := make([]string, 0, len(rows))
	for _, row := range rows {
		e := t.entry(row)
		if _, seen := t.announcedSet[e]; seen {
			continue
		}
		fresh = append(fresh, row)
		entries = append(entries, e)
	}

		return nil
	}
	for _, e := range entries {
		t.announcedSet[e] = struct{}{}
		t.announced = append(t.announced, e)
		t.pending = append(t.pending, e)
	}
	t.compactLocked()
	return &ClaimedNotice{Text: t.render(fresh), Entries: entries}
}

// bindLocked 让 sess 存的基线成为当前基线（已绑定同一 id 则只补 pending）。
//
// Python: _bind (team_context.py L750-764)
func (t *taskNoticeTracker) bindLocked(sess sessioninterfaces.SessionFacade) {
	sessID := sess.GetSessionID()
	if t.sessionBound && sessID == t.sessionID {
		return
	}

	if t.sessionBound && sessID == t.sessionID {
		carried = append([]string(nil), t.pending...)

	t.announcedSet = make(map[string]struct{}, len(t.announced))
	for _, e := range t.announced {
		t.announcedSet[e] = struct{}{}
	}
	t.compactLocked()
	t.persisted = append([]string(nil), stored...)
}

		return
	}

		drop[e] = struct{}{}

	for _, e := range t.pending {
		if _, ok := drop[e]; !ok {
			pending = append(pending, e)
		}
	}
	t.pending = pending
}

// superseded 每个任务只留最新修订（修订号只增，旧的永不再匹配）。
func (t *TaskRevisionTracker) superseded(entries []string) []string {
	newest := map[string]struct {
		num   int64
		entry string
	}{}
	for _, e := range entries {
		idx := strings.LastIndex(e, "@")
		if idx < 0 {
			continue
		}
		taskID, revStr := e[:idx], e[idx+1:]
		rev, err := strconv.ParseInt(revStr, 10, 64)
		if err != nil {
			rev = -1
		}
		cur, ok := newest[taskID]
		if !ok || rev > cur.num {
			newest[taskID] = struct {
				num   int64
				entry string
			}{rev, e}
		}
	}
	keep := make(map[string]struct{}, len(newest))
	for _, v := range newest {
		keep[v.entry] = struct{}{}
	}
	out := make([]string, 0, len(entries))
	for _, e := range entries {
		if _, ok := keep[e]; !ok {
			out = append(out, e)
		}
	}
	return out
}

// entry 条目形如 taskID@revision。
func (t *TaskRevisionTracker) entry(task *database.TeamTaskBase) string {
	return task.TaskID + "@" + strconv.FormatInt(task.Revision, 10)
}

// render 渲染 kind=task-revised 的事件块，含每个任务的未读编辑说明。
func (t *TaskRevisionTracker) render(tasks []*database.TeamTaskBase) string {
	intro, err := atschema.T("team_context.task_revised_intro")
	if err != nil {
		intro = ""
	}
	lines := []string{intro}
	for _, task := range tasks {
		lines = append(lines, fmt.Sprintf("- %s · %s · r%d → r%d",
			task.TaskID, task.Title, task.ObservedRevision, task.Revision))
		for _, note := range prompts.ChangeNotesLines(task) {
			lines = append(lines, "  "+note)
		}
	}
	return inboundtags.RenderEvent(TaskRevisedEventKind, strings.Join(lines, "\n"), EventOptions{})
}
```

```go
// OnUserMessage 丢弃被取代的输入、打时间锚点，然后折入团队状态。
//
// 三件事按序作用在「即将合成一条消息」的输入列表上：先剪枝，再在存活项
// 之前插入 <team-clock>，最后把团队状态插在更前面。
//
// <team-clock> 是让每条 <team-inbound> 上的绝对 time 可读成「多久以前」的锚点。
// 入站块本身不带相对时间，因为它们留在历史里：渲染一次「刚刚」，一小时后
// 成员再读同一块仍写「刚刚」。锚点每次受理输入写一次、永不刷新，因此每条
// 都忠实记录该输入到达的时刻，最新的一条就是成员的当前时间。
//
// 三件事都在这里做，因为这是它们还是**输入**的唯一时刻。再往后一步它们就是
// 一条普通历史消息——按位置再也定位不到（compaction 会重写它背后的历史），
// 也不再可以改（改一条消息会让其后每个 KV 缓存条目失效）。所以忙碌期间堆积的
// 被取代任务板必须在这里丢，否则永远丢不掉。
//
// Python: on_user_message (team_policy_rail.py L268-319)
func (r *TeamPolicyRail) OnUserMessage(ctx context.Context, cbc *agentinterfaces.AgentCallbackContext) error {
	inputs, ok := cbc.Inputs().(*agentinterfaces.UserMessageInputs)
	if !ok || inputs == nil {
		return nil
	}

	// 步骤 1：剪枝（切片长度可变，必须回写）
	inputs.Parts = r.dropSuperseded(inputs.Parts)
	// 步骤 2：时间锚点插在最前
	if len(inputs.Parts) > 0 {
		clock := tracker.RenderClock(timefmt.FormatAbsoluteTime(database.GetCurrentTime()))
		inputs.Parts = append([]string{clock}, inputs.Parts...)
	}

	sess := cbc.Session()
	if sess == nil {
		return nil
	}
	// 步骤 3：任务通知（事件，永远 inline，不进 team.context 附件）
	inputs.Parts = r.foldTaskOutcomes(inputs.Parts, sess)
	// 步骤 4：附件车道优先
	if r.queueTeamContextAttachment(cbc, sess) {
		return nil
	}
	// 步骤 5：兜底——把待发状态前插进本批输入
	if text, ok := r.tracker.PendingText(sess); ok {
		inputs.Parts = append([]string{text}, inputs.Parts...)
		r.tracker.Commit(sess, nil)
	}
	return nil
}

// foldTaskOutcomes 把尚未告知成员的任务事实前插到它的输入上。
//
// Python: _fold_task_outcomes (team_policy_rail.py L321-328)
func (r *TeamPolicyRail) foldTaskOutcomes(parts []string, sess sessioninterfaces.SessionFacade) []string {
	if r.taskEventTracker == nil {
		return parts
	}
	notice := r.taskEventTracker.Claim(sess)
	if notice == nil {
		return parts
	}
	out := append([]string{notice.Text}, parts...)
	r.taskEventTracker.Persist(sess)
	return out
}
```

```go
// BeforeSteeringDrain 限制一次模型调用吸收多少 steering 积压。
//
// 这是 dropSuperseded 处理的问题的另一半，早一步：框架给忙碌成员排队的一切
// 会一次性交出去，而这里排队的是邮箱流量——一条消息一条，谁也不取代谁，
// 全都得读。因此丢弃不是选项：唯一能避免这一轮变成「一墙糊在一起的消息」的
// 办法就是少拿几条，剩下的留给后续模型调用。
//
// **leader 豁免**，理由与它豁免任务板剪枝相同：它按顺序读到达，
// 拆碎了看到的序列就得自己重新拼。

	if r.role == atschema.TeamRoleLeader || cbc == nil {
		return nil
	}
	inputs, ok := cbc.Inputs().(*agentinterfaces.SteeringDrainInputs)

		return nil

	limit := r.steerBatchSize
	inputs.Limit = &limit

// dropSuperseded 移除被后续条目取代的排队输入，返回存活列表。
//
// 两个输入队列都会把成员忙碌期间堆积的一切交出来，所以忙碌一段后被唤醒的
// 成员会看到期间渲染的每一份任务板——每份都是完整调查，除最后一份全是过期的。
// 每份板就是一个完整排队输入，因此过期的按整条移除；不解析、不重写。
//
// **leader 的板不动。** teammate 的板是工作队列，列的是此刻可认领的任务，
// 只有当前那份可执行。leader 的板是全队未完成工作，它读的是**序列**
// （哪个任务出现、哪个移动了）来决定重规划还是收尾。剪掉它等于删掉它正在
// 盯的那些变迁。
//
// Python: _drop_superseded (team_policy_rail.py L352-379)
func (r *TeamPolicyRail) dropSuperseded(parts []string) []string {
	if r.role == atschema.TeamRoleLeader || len(parts) < 2 {
		return parts
	}
	kept := tracker.DropSupersededSnapshots(parts)
	if len(kept) == len(parts) {
		return parts
	}
	logger.Debug(agentteamsLogComponent).
		Str("member_name", r.memberName).
		Int("dropped", len(parts)-len(kept)).
		Int("total", len(parts)).
		Msg("dropped superseded inputs")
	return kept
}
```

```go
// queueTeamContextAttachment 把本成员团队状态注册为一个动态附件 section。
//
// 附件管理器是**当前状态**通道：每个边界把注册区与历史里记录的状态做 diff，
// 未注册的 section 读作已撤回。管理器挂在 agent 实例上，因此重建的成员从空
// 注册区开始，team.context 必须在每个边界重新注册：
//   - 历史里没有附件快照（首投，或 compaction 丢掉了它）→ 完整当前状态
//   - tracker 有新内容 → 那段增量，名册变化因此保持 join/leave 增量
//   - 什么都没变 → 上次提交的同一份内容，内容 hash 与历史一致，不产生宣告
//
// tracker 基线与注册内容只在 ReAct 历史同步成功后才提交（25.0a 回调）。
// 返回 true 表示附件车道接管投递；false 时回落到历史消息车道。
//
// Python: _queue_team_context_attachment (team_policy_rail.py L389-452)
func (r *TeamPolicyRail) queueTeamContextAttachment(
	cbc *agentinterfaces.AgentCallbackContext,
	sess sessioninterfaces.SessionFacade,
) bool {
	manager := hprompts.ManagerFor(cbc.Agent())
	modelCtx := cbc.ModelContext()
	if manager == nil || modelCtx == nil {
		return false
	}
	sessionID := sess.GetSessionID()
	hasSnapshot := manager.HasHistorySnapshot(modelCtx, sessionID)
	text, hasText := r.tracker.PendingText(sess)
	lastText, hasLast := r.tracker.LastDeliveredText(sess)

	var content string
	switch {
	case !hasSnapshot:
		if snap, ok := r.tracker.SnapshotText(); ok {
			content = snap
		}
	case hasText:
		content = text
	default:
		// 内容被记住之前就投递过的会话：一次性回落到全量，之后持续重注册
		if hasLast {
			content = lastText
		} else if snap, ok := r.tracker.SnapshotText(); ok {
			content = snap
		}
	}
	if content == "" {
		return true
	}
	if !hprompts.DeliverText(cbc.Agent(), sessionID, teamContextAttachmentSection, content,
		hprompts.AttachmentRuntime, teamContextAttachmentSource, teamContextAttachmentPriority) {
		logger.Warn(agentteamsLogComponent).
			Str("member_name", r.memberName).
			Msg("failed to queue team context attachment")
		return false
	}

	if hasText || content != lastText {
		delivered := content
		appendCommitCallback(cbc, func() error {
			r.tracker.Commit(sess, &delivered)
			return nil
		})
	}
	return true
}

// appendCommitCallback 把回调追加进本次调用的投递成功队列。
func appendCommitCallback(cbc *agentinterfaces.AgentCallbackContext, fn func() error) {
	if cbc == nil || fn == nil {
		return
	}
	raw, _ := cbc.GetExtra(hprompts.CommitCallbacksExtraKey)
	list, _ := raw.([]func() error)
	cbc.SetExtra(hprompts.CommitCallbacksExtraKey, append(list, fn))
}

// announceUnattachedState 追加「无处搭车」的团队状态。
//
// 状态通常随输入到达并搭车（见 OnUserMessage），但也可能出现在工具循环中途：
// leader 调 build_team 在轮次中间建出团队、自己的成员行与名册，而下一次输入
// 可能很远。因此模型调用时刻仍待发的状态就变成一条独立消息。
//
// **追加，绝不插入**：尾部是唯一不需要索引、也不会被 compaction 重写背后历史
// 而失效的位置。
//
// Python: _announce_unattached_state (team_policy_rail.py L454-477)
func (r *TeamPolicyRail) announceUnattachedState(ctx context.Context, cbc *agentinterfaces.AgentCallbackContext) {
	modelCtx := cbc.ModelContext()
	sess := cbc.Session()
	if modelCtx == nil || sess == nil {
		return
	}
	if r.queueTeamContextAttachment(cbc, sess) {
		return
	}
	text, ok := r.tracker.PendingText(sess)
	if !ok {
		return
	}
	_, _ = modelCtx.AddMessages(ctx, []llmschema.BaseMessage{llmschema.NewUserMessage(text)})
	r.tracker.Commit(sess, nil)
}
```

## 审查修正（推送后复查，2026-11-08）

推送后对 Python 原文逐方法复查，发现 4 处与上游不一致，全部修正：

### ① `foldTaskOutcomes` 顺序反了（行为缺陷）

上游是「先 `parts.insert(0, notice.text)`，再 `persist`」；我写成先 `persist` 再插入，
并在注释里编了「先 persist 才不重复播」的理由。实际后果相反：`persist` 只写会话
state 并 commit，成功不代表 notice 真的成为历史消息——先推进基线，一旦本轮交付
没走完，这条通知就永久丢失（基线已推进，不会再被 claim）。已改回上游顺序。

### ② `listRoster` 加了上游没有的排序

上游 `_list_roster` 按后端返回顺序原样组装，不排序。我加了 `sortRoster`，理由是
「Go 的 in-memory 库遍历 map 顺序随机」——这个观察是对的，但**修在了错误的层**：
渲染层照上游保持「按输入顺序」不改写，把定序下沉到 DAO 层的 `GetTeamMembers`
（SQL 加 `ORDER BY member_name`，in-memory 排序后返回）。行为对齐上游，同时
Go 侧获得确定顺序。登记为偏差 D-12。

### ③ `present=false` 分支删除后未记录其不变量

偏差 D-1 让 `GetMemberUpdatedAtState` 的 `present` 恒 `true`，上游
`present=false` 分支因此不可达而被整段删除。但那段承载「空时间戳必须强制重投
一次、且必须回戳同一时间戳以免死循环」的设计。已在 `identityBody` 与
`GetMemberUpdatedAtState` 两处注释里完整记录该不变量与回收条件。

### ④ `change_notes` 的 `map[string]any` 收紧为 `ChangeNote` 结构体

原实现用 `[]map[string]any` 承载，导致：
- `noteRevision` 要 switch `float64`/`int`/`int64`/`json.Number` 四种
- `ChangeNotesLines` 里对 `note["by"].(string)`、`note["note"].(string)` 逐个断言

「字段缺失」与「显式 0」——上游渲染时对缺失输出 `None`、对 0 输出 `0`。

顺带修掉两处细节：
- **`noteRevision` 缺失返回值**：原来返回 `-1`（永不匹配），上游是
  `int(note.get("revision") or 0)` → **0**。已改为 0。
- **`formatRevision` 占位符**：原来缺失渲染 `- r?`，上游 f-string 渲染 `None`。
