```go
package parser

import (
	"context"
	"os"
	"path/filepath"
	"sort"
	"strings"

	"github.com/uapclaw/uapclaw-go/internal/common/exception"
	"github.com/uapclaw/uapclaw-go/internal/common/logger"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/store/reranker"
)

// ──────────────────────────── 结构体 ────────────────────────────

// AutoFileParser 扩展名自动路由解析器。
//
// 组合解析器（13.10 spec 预告模式）：嵌入 BaseParser + 构造注入 self +
// 覆写整体 Parse/Supports（Python 覆写 parse/supports 非私有 _parse）。
// 按文件扩展名从 parserRegistry 取工厂实例路由（registry 骨架由 13.16
// 带出，各具体解析器 init 注册；Python ensure_parsers_loaded 的动态 import
// 在 Go 由同包 init 天然完成——ImportError 容错编译期不可能，零代码归档）。
//
// Python: auto_file_parser.py AutoFileParser
type AutoFileParser struct {
	// BaseParser 模板适配器（组合解析器仅复用 LazyParse/Process/ParseContent 默认行为）
	*BaseParser
}

// ──────────────────────────── 枚举 ────────────────────────────
// ──────────────────────────── 常量 ────────────────────────────
// ──────────────────────────── 全局变量 ────────────────────────────
// ──────────────────────────── 导出函数 ────────────────────────────

// NewAutoFileParser 构造 AutoFileParser。
//
// Python: auto_file_parser.py AutoFileParser.__init__（self 注入链 13.10 同款）
func NewAutoFileParser() (*AutoFileParser, error) {
	p := &AutoFileParser{}
	bp, err := NewBaseParser(p)
	if err != nil {
		return nil, err
	}
	p.BaseParser = bp
	return p, nil
}

// Parse 解析文件：按扩展名路由到注册的具体解析器并增强 metadata。
//
// 五步对齐 Python AutoFileParser.parse：
//  1. 文件不存在 → 155103（error_msg "File {doc} does not exist"）
//  2. 扩展名小写归一后未注册 → 155104（error_msg 含扩展名与已注册列表——
//     sort+Join 格式，chunker GetChunker 13.5 先例）
//  3. 工厂新实例 + Info 日志（parserTypeName 实例类型名——13.16 已知偏差延续）
//  4. 底层 parser.Parse 原样全量转发（DocID/LLMClient/Extra/四网页字段，
//     对齐 Python doc_id/llm_client/kwargs 全透传；工厂每次新实例）
//  5. 空结果→空切片；否则 Extra["file_name"] 键存在即原值透传（any 含非字符串，
//     对齐 Python dict.get 原样返回），键缺失兜底 filepath.Base(doc)；
//     逐文档 Metadata 写四键（nil 先建——防御第三方解析器契约违背）
//
// Python: auto_file_parser.py AutoFileParser.parse
func (p *AutoFileParser) Parse(ctx context.Context, doc string, opts ...ParseOption) ([]reranker.Document, error) {
	_ = ctx // 对齐 Python 同步语义
	o := NewParseOptions(opts...)

	if _, err := os.Stat(doc); os.IsNotExist(err) {
		// 对齐 Python: raise build_error(RETRIEVAL_INDEXING_FILE_NOT_FOUND, error_msg=f"File {doc} does not exist")
		return nil, exception.BuildError(
			exception.StatusRetrievalIndexingFileNotFound,
			exception.WithMsg("File "+doc+" does not exist"),
		)
	}

	fileExt := strings.ToLower(filepath.Ext(doc))
	factory, ok := getParserFactory(fileExt)
	if !ok {
		// 对齐 Python: raise build_error(RETRIEVAL_INDEXING_FORMAT_NOT_SUPPORT,
		//   error_msg=f"Unsupported format: {file_ext}, only {list(_PARSER_REGISTRY.keys())} are supported")
		// 列表格式：sort+Join（决策 4，chunker GetChunker 先例）
		exts := snapshotRegistry()
		sort.Strings(exts)
		return nil, exception.BuildError(
			exception.StatusRetrievalIndexingFormatNotSupport,
			exception.WithMsg("Unsupported format: "+fileExt+", only "+strings.Join(exts, ", ")+" are supported"),
		)
	}

	parser := factory()
	// 对齐 Python: logger.info(f"Using {parser.__class__.__name__} to parse {doc}")
	logger.Info(logComponent).
		Str("parser_name", parserTypeName(parser)).
		Str("doc", doc).
		Msg("using parser to parse doc")

	documents, err := parser.Parse(ctx, doc, opts...)
	if err != nil {
		return nil, err
	}
	if len(documents) == 0 {
		return []reranker.Document{}, nil
	}

	// 对齐 Python: file_name = kwargs.get("file_name", os.path.basename(doc))
	// 决策 5：键存在即原值透传（any 含非字符串），仅键缺失兜底
	var fileName any
	if v, ok := o.Extra["file_name"]; ok {
		fileName = v
	} else {
		fileName = filepath.Base(doc)
	}

	for i := range documents {
		if documents[i].Metadata == nil {
			// 决策 6 防御：第三方解析器可能返回 nil Metadata；
			// Python 同场景 AttributeError 被 KB 层 except 吞，Go panic 不可恢复
			documents[i].Metadata = map[string]any{}
		}
		documents[i].Metadata["doc_id"] = o.DocID
		documents[i].Metadata["title"] = fileName
		documents[i].Metadata["file_path"] = doc
		documents[i].Metadata["file_ext"] = fileExt
	}

	return documents, nil
}

// Supports 检查是否支持该文档源：文件存在且扩展名已注册。
//
// Python: auto_file_parser.py AutoFileParser.supports
func (p *AutoFileParser) Supports(doc string) bool {
	if _, err := os.Stat(doc); os.IsNotExist(err) {
		return false
	}
	_, ok := getParserFactory(strings.ToLower(filepath.Ext(doc)))
	return ok
}

// RegisterNewParser 运行时动态注册新解析器工厂。
//
```

```go
package parser

import (
	"context"
	"regexp"
	"strings"

	"github.com/uapclaw/uapclaw-go/internal/common/logger"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/store/reranker"
)

// ──────────────────────────── 结构体 ────────────────────────────

// docMatcher 路由匹配器（Python Union[re.Pattern, Callable[[str], bool]] 对应物，非导出）。
//
// 两实现：regexpMatcher（正则形态）/ funcMatcher（可调用形态）。
// 经 NewRegexpMatcher/NewFuncMatcher 导出包装供外部注入自定义路由。
type docMatcher interface {
	// MatchDoc 匹配文档源。
	MatchDoc(doc string) bool
}

// regexpMatcher 正则形态匹配器。
//
// MatchDoc 对齐 Python pattern_or_callable.match(doc.strip())。
type regexpMatcher struct {
	// re 编译后正则（复用 13.14 httpURLPattern/wechatMPURLPattern 同款 *regexp.Regexp）
	re *regexp.Regexp
}

// funcMatcher 可调用形态匹配器。
//
// MatchDoc 对齐 Python callable(pattern_or_callable(doc))——原样传入不 strip。
type funcMatcher struct {
	// fn 匹配函数
	fn func(string) bool
}

// linkRoute 单条路由（匹配器 + 命中后委托的解析器）。
type linkRoute struct {
	// matcher 匹配器
	matcher docMatcher
	// parser 命中后委托的解析器
	parser Parser
}

// AutoLinkParser URL 模式路由解析器（先匹配先赢）。
//
// 组合解析器（13.10 spec 预告模式）：嵌入 BaseParser + 构造注入 self +
// 覆写整体 Parse/Supports。默认路由：微信文章 → WeChatArticleParser、
// 通用 http(s) → WebPageParser（13.14 产出，每次构造新实例）。
//
// Python: auto_link_parser.py AutoLinkParser
type AutoLinkParser struct {
	// BaseParser 模板适配器
	*BaseParser
	// routes 路由表（先匹配先赢）
	routes []linkRoute
}

// ──────────────────────────── 枚举 ────────────────────────────
// ──────────────────────────── 常量 ────────────────────────────
// ──────────────────────────── 全局变量 ────────────────────────────
// ──────────────────────────── 导出函数 ────────────────────────────

// NewRegexpMatcher 构造正则形态匹配器。
func NewRegexpMatcher(re *regexp.Regexp) docMatcher {
	return &regexpMatcher{re: re}
}

// NewFuncMatcher 构造可调用形态匹配器。
func NewFuncMatcher(fn func(string) bool) docMatcher {
	return &funcMatcher{fn: fn}
}

// NewAutoLinkParser 构造 AutoLinkParser（切片参数二分，决策 7）。
//
// routes == nil → defaultRoutes() 两条默认路由（微信优先）；
// routes == 非nil空切片 → 字面空表（Parse 恒返回空切片，对齐 Python routes=[]）；
// routes 非空 → 字面使用调用方路由表。
//
// Python: auto_link_parser.py AutoLinkParser.__init__（routes=None 默认）
func NewAutoLinkParser(routes []linkRoute) (*AutoLinkParser, error) {
	if routes == nil {
		var err error
		routes, err = defaultRoutes()
		if err != nil {
			return nil, err
		}
	}
	p := &AutoLinkParser{routes: routes}
	bp, err := NewBaseParser(p)
	if err != nil {
		return nil, err
	}
	p.BaseParser = bp
	return p, nil
}

// Parse 解析 URL：遍历路由表首个命中即委托。
//
// 命中 → Debug 日志 → 底层 parser.Parse（决策 3：转发前剔除 LLMClient——
// 构造新 ParseOptions 保留 DocID/Extra/Timeout/UserAgent/Verify/HTTPClient、
// 置空 LLMClient；Python parser.parse(doc, doc_id=doc_id, **kwargs) 关键字
// 调用天然不含 llm_client，Go 显式剔除字面对齐）。
// 无命中 → 空切片。
//
// Python: auto_link_parser.py AutoLinkParser.parse
func (p *AutoLinkParser) Parse(ctx context.Context, doc string, opts ...ParseOption) ([]reranker.Document, error) {
	for _, route := range p.routes {
		if matchDoc(route.matcher, doc) {
			// 对齐 Python: logger.debug("AutoLinkParser delegating to %s", type(parser).__name__)
			logger.Debug(logComponent).
				Str("parser_name", parserTypeName(route.parser)).
				Msg("AutoLinkParser delegating to parser")
			return route.parser.Parse(ctx, doc, withoutLLMClient(opts...)...)
		}
	}
	return []reranker.Document{}, nil
}

// Supports 检查是否支持该文档源：非空 http(s) URL 且任一路由命中。
//
// Python: auto_link_parser.py AutoLinkParser.supports
func (p *AutoLinkParser) Supports(doc string) bool {
	if doc == "" || !httpURLPattern.MatchString(strings.TrimSpace(doc)) {
		return false
	}
	for _, route := range p.routes {
		if matchDoc(route.matcher, doc) {
			return true
		}
	}
	return false
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// defaultRoutes 构造默认路由表（每次新解析器实例，对齐 Python _default_routes()）。
//
// 微信优先：WECHAT_MP_URL_PATTERN → WeChatArticleParser；
// 兜底：HTTP_URL_PATTERN → WebPageParser。两构造 error 直接上抛
//（Python 侧构造无异常面，Go 构造含 error 为语言差异）。
//
// Python: auto_link_parser.py _default_routes
func defaultRoutes() ([]linkRoute, error) {
	wechatParser, err := NewWeChatArticleParser()
	if err != nil {
		return nil, err
	}
	webParser, err := NewWebPageParserWithDefaults()
	if err != nil {
		return nil, err
	}
	return []linkRoute{
		{matcher: NewRegexpMatcher(wechatMPURLPattern), parser: wechatParser},
		{matcher: NewRegexpMatcher(httpURLPattern), parser: webParser},
	}, nil
}

// matchDoc 双形态匹配 dispatch（Python _match_doc 对应物）。
//
// regexpMatcher 内部 strip（对齐 pattern.match(doc.strip())）；
// funcMatcher 原样传入（对齐 callable(doc)）。
//
// Python: auto_link_parser.py _match_doc
func matchDoc(m docMatcher, doc string) bool {
	return m.MatchDoc(doc)
}

// MatchDoc 正则形态：strip 后匹配（对齐 Python pattern.match(doc.strip())）。
func (m *regexpMatcher) MatchDoc(doc string) bool {
	return m.re.MatchString(strings.TrimSpace(doc))
}

// MatchDoc 可调用形态：原样传入（对齐 Python callable(doc) 不 strip）。
func (m *funcMatcher) MatchDoc(doc string) bool {
	return m.fn(doc)
}

// withoutLLMClient 剔除 LLMClient 后重建 opts（决策 3）。
//
// Python parser.parse(doc, doc_id=doc_id, **kwargs) 关键字调用天然不含
// llm_client；Go opts 整体转发无法天然丢弃，显式重建字面对齐。
func withoutLLMClient(opts ...ParseOption) []ParseOption {
	o := NewParseOptions(opts...)
	o.LLMClient = nil
	return []ParseOption{func(no *ParseOptions) {
		*no = *o
	}}
}
```

```go
package parser

import (
	"context"
	"strings"

	"github.com/uapclaw/uapclaw-go/internal/common/logger"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/store/reranker"
)

// ──────────────────────────── 结构体 ────────────────────────────

// AutoParser 顶层路由解析器：URL → AutoLinkParser，否则 → AutoFileParser。
//
// 组合解析器（13.10 spec 预告模式）：嵌入 BaseParser + 构造注入 self +
// 覆写整体 Parse/Supports。单入口 parse(path_or_url)——混源输入无需
// 区分 parse_urls / parse_files。
//
// Python: auto_parser.py AutoParser
type AutoParser struct {
	// BaseParser 模板适配器
	*BaseParser
	// linkParser URL 路由解析器
	linkParser Parser
	// fileParser 文件路由解析器
	fileParser Parser
}

// ──────────────────────────── 枚举 ────────────────────────────
// ──────────────────────────── 常量 ────────────────────────────
// ──────────────────────────── 全局变量 ────────────────────────────
// ──────────────────────────── 导出函数 ────────────────────────────

// NewAutoParser 构造 AutoParser。
//
// linkParser/fileParser 为 nil 时各自默认构造（对齐 Python None 默认参数）。
//
// Python: auto_parser.py AutoParser.__init__
func NewAutoParser(linkParser, fileParser Parser) (*AutoParser, error) {
	if linkParser == nil {
		lp, err := NewAutoLinkParser(nil)
		if err != nil {
			return nil, err
		}
		linkParser = lp
	}
	if fileParser == nil {
		fp, err := NewAutoFileParser()
		if err != nil {
			return nil, err
		}
		fileParser = fp
	}
	p := &AutoParser{linkParser: linkParser, fileParser: fileParser}
	bp, err := NewBaseParser(p)
	if err != nil {
		return nil, err
	}
	p.BaseParser = bp
	return p, nil
}

// Parse 解析文档源：URL 走 link 路由，否则走 file 路由。
//
// isLikelyURL && linkParser.Supports → Debug 日志 → linkParser.Parse
// 全量透传（含 LLMClient——对齐 Python 传递，与 AutoLinkParser 内部剔除
// 形成链路分层：顶层传、路由层丢弃）；
// fileParser.Supports → Debug 日志 → fileParser.Parse 全量透传；
// 都不中 → 空切片。URL-but-unsupported fall-through 到 file 分支
//（Python 真实控制流字面保留）。
//
// Python: auto_parser.py AutoParser.parse
func (p *AutoParser) Parse(ctx context.Context, doc string, opts ...ParseOption) ([]reranker.Document, error) {
	if isLikelyURL(doc) && p.linkParser.Supports(doc) {
		// 对齐 Python: logger.debug("AutoParser delegating to link parser")
		logger.Debug(logComponent).Msg("AutoParser delegating to link parser")
		return p.linkParser.Parse(ctx, doc, opts...)
	}
	if p.fileParser.Supports(doc) {
		// 对齐 Python: logger.debug("AutoParser delegating to file parser")
		logger.Debug(logComponent).Msg("AutoParser delegating to file parser")
		return p.fileParser.Parse(ctx, doc, opts...)
	}
	return []reranker.Document{}, nil
}

// Supports 检查是否支持该文档源：URL 判定后委托对应子解析器。
//
// Python: auto_parser.py AutoParser.supports
func (p *AutoParser) Supports(doc string) bool {
	if isLikelyURL(doc) {
		return p.linkParser.Supports(doc)
	}
	return p.fileParser.Supports(doc)
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// isLikelyURL 判定文档源是否像 http(s) URL（复用 13.14 httpURLPattern）。
//
// Python: auto_parser.py _is_likely_url
func isLikelyURL(doc string) bool {
	return doc != "" && httpURLPattern.MatchString(strings.TrimSpace(doc))
}
```
