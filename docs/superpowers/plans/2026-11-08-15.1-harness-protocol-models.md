package harness_protocol

import (
	"fmt"
	"math"
	"reflect"
)

// ──────────────────────────── 导出函数 ────────────────────────────

// FreezeJsonValue 校验并递归冻结一个 JSON 兼容值。
// 返回值不再与入参共享可变容器，调用方修改入参不影响结果。
// Python: models.py:30-48 freeze_json_value（5 个分支顺序一致）
func FreezeJsonValue(value any) (any, error) {
	// 分支 1：None（Go 的 nil）/ string / bool / 各类整数 原样返回
	// Python: models.py:33-34
	if value == nil {
		return nil, nil
	}
	switch v := value.(type) {
	case string, bool,
		int, int8, int16, int32, int64,
		uint, uint8, uint16, uint32, uint64:
		return v, nil
	case float64:
		// 分支 2：浮点必须有限
		// Python: models.py:35-38
		if math.IsNaN(v) || math.IsInf(v, 0) {
			return nil, fmt.Errorf("JSON numbers must be finite")
		}
		return v, nil
	case float32:
		f := float64(v)
		if math.IsNaN(f) || math.IsInf(f, 0) {
			return nil, fmt.Errorf("JSON numbers must be finite")
		}
		return v, nil
	}

	// 分支 3：切片/数组递归（list 与 tuple 归一到同一 Go 路径）
	// Python: models.py:39-40
	rv := reflect.ValueOf(value)
	if rv.Kind() == reflect.Slice || rv.Kind() == reflect.Array {
		out := make([]any, rv.Len())
		for i := 0; i < rv.Len(); i++ {
			item, err := FreezeJsonValue(rv.Index(i).Interface())
			if err != nil {
				return nil, err
			}
			out[i] = item
		}
		return out, nil
	}

	// 分支 4：映射递归，键必须是 string
	// Python: models.py:41-47
	if rv.Kind() == reflect.Map {
		if rv.Type().Key().Kind() != reflect.String {
			return nil, fmt.Errorf("JSON object keys must be strings")
		}
		out := make(map[string]any, rv.Len())
		iter := rv.MapRange()
		for iter.Next() {
			key := iter.Key().String()
			item, err := FreezeJsonValue(iter.Value().Interface())
			if err != nil {
				return nil, err
			}
			out[key] = item
		}
		return out, nil
	}

	// 分支 5：其余类型拒绝
	// Python: models.py:48
	return nil, fmt.Errorf("value of type %T is not JSON-compatible", value)
}

// FreezeJsonObject 校验、复制并递归冻结一个 JSON 对象。
// Python: models.py:51-57 freeze_json_object
func FreezeJsonObject(value any) (map[string]any, error) {
	frozen, err := FreezeJsonValue(value)
	if err != nil {
		return nil, err
	}
	if frozen == nil {
		// Python: models.py:56（isinstance(frozen, Mapping) 为 False → TypeError）
		return nil, fmt.Errorf("JSON object must be a mapping")
	}
	obj, ok := frozen.(map[string]any)
	if !ok {
		return nil, fmt.Errorf("JSON object must be a mapping")
	}
	return obj, nil
}

// JsonValueToBuiltin 把冻结后的协议 JSON 值转回标准 JSON 容器。
// Python: models.py:60-67 json_value_to_builtin
func JsonValueToBuiltin(value any) any {
	switch v := value.(type) {
	case []any:
		out := make([]any, len(v))
		for i, item := range v {
			out[i] = JsonValueToBuiltin(item)
		}
		return out
	case map[string]any:
		out := make(map[string]any, len(v))
		for key, item := range v {
			out[key] = JsonValueToBuiltin(item)
		}
		return out
	}
	return value
}

// NewTurnResult 创建轮次终态结果，10 步校验顺序与上游一致。
// Python: results.py:170-189
func NewTurnResult(status TurnStatus, opts ResultOptions) (TurnResult, error) {
	// 步骤 1：messages 拷贝为独立切片
	messages := make([]TurnMessage, len(opts.Messages))
	copy(messages, opts.Messages)
	// 步骤 2：冻结 final_output
	frozenFinal, err := FreezeJsonValue(opts.FinalOutput)
	if err != nil {
		return TurnResult{}, err
	}
	// 步骤 3：冻结 structured_output
	frozenStructured, err := FreezeJsonValue(opts.StructuredOutput)
	if err != nil {
		return TurnResult{}, err
	}
	// 步骤 4：冻结 provider_data
	frozenProviderData, err := FreezeJsonObject(opts.ProviderData)
	if err != nil {
		return TurnResult{}, err
	}
	// 步骤 5：failed 必须有 error
	if status == TurnStatusFailed && opts.Error == nil {
		return TurnResult{}, &HarnessProtocolError{Msg: "failed turn result requires error"}
	}
	// 步骤 6：非 failed 不得有 error
	if status != TurnStatusFailed && opts.Error != nil {
		return TurnResult{}, &HarnessProtocolError{Msg: "only failed turn result may contain error"}
	}
	// 步骤 7：interrupted 必须有 termination
	if status == TurnStatusInterrupted && opts.Termination == nil {
		return TurnResult{}, &HarnessProtocolError{Msg: "interrupted turn result requires termination"}
	}
	// 步骤 8：非 interrupted 不得有 termination
	if status != TurnStatusInterrupted && opts.Termination != nil {
		return TurnResult{}, &HarnessProtocolError{Msg: "only interrupted turn result may contain termination"}
	}
	// 步骤 9：时间戳有限且非负
	for _, ts := range []*float64{opts.StartedAt, opts.CompletedAt} {
		if ts != nil && (math.IsNaN(*ts) || math.IsInf(*ts, 0) || *ts < 0) {
			return TurnResult{}, &HarnessProtocolError{Msg: "turn timestamp must be a finite Unix timestamp"}
		}
	}
	// 步骤 10a：completed 不得早于 started
	if opts.StartedAt != nil && opts.CompletedAt != nil && *opts.CompletedAt < *opts.StartedAt {
		return TurnResult{}, &HarnessProtocolError{Msg: "turn completed_at must not precede started_at"}
	}
	// 步骤 10b：duration_ms 非负
	if opts.DurationMS != nil && *opts.DurationMS < 0 {
		return TurnResult{}, &HarnessProtocolError{Msg: "turn duration_ms must be non-negative monotonic elapsed time"}
	}
	return TurnResult{
		Status:           status,
		Messages:         messages,
		FinalOutput:      frozenFinal,
		StructuredOutput: frozenStructured,
		StopReason:       opts.StopReason,
		Termination:      opts.Termination,
		Error:            opts.Error,
		Usage:            opts.Usage,
		Cost:             opts.Cost,
		StartedAt:        opts.StartedAt,
		CompletedAt:      opts.CompletedAt,
		DurationMS:       opts.DurationMS,
		ProviderData:     frozenProviderData,
	}, nil
}

import "context"

// ──────────────────────────── 结构体 ────────────────────────────

// JsonObject provider 配置等 JSON 对象的 Go 形态。
// Python: models.py:27 JsonObject = Mapping[str, JsonValue]
type JsonObject = map[string]any

// HarnessProtocol 第三方 agent 必须满足的并发安全 harness 行为契约。
//
// Events 与 TurnEvents 是同一逻辑单消费者观测通道的两种视图；
// 实现必须自行串行化状态转换，调用方可从不同协程发起命令。
// Python: protocol.py:28-126 HarnessProtocol
type HarnessProtocol interface {
	// Card 返回静态实现身份与已声明能力
	Card() HarnessCard
	// State 返回当前高层生命周期相位
	State() HarnessState
	// ProviderSessionID 返回 provider 原生会话标识（尚不可用时为 nil）
	ProviderSessionID() *string
	// EventBufferConfig 返回观测使用的有界背压策略
	EventBufferConfig() EventBufferConfig
	// Start 校验宿主兼容性、开启一次周期并稳定在 idle
	Start(ctx context.Context, harnessContext HarnessContext) error
	// Stop 停止周期、关闭事件并稳定在 terminated（幂等）
	Stop(ctx context.Context) error
	// Events 返回周期级有序观测流；已有活跃迭代器时须返回 HarnessStateError
	Events() HarnessEventCursor
	// TurnEvents 返回单轮有限观测流；turnID 为 nil 表示下一轮未消费的轮次
	TurnEvents(turnID *string) HarnessEventCursor
	// Send 接受输入并返回其 message→turn 关联；不等待该轮结束
	Send(ctx context.Context, content HarnessInput, mode DeliveryMode) (SendReceipt, error)
	// Abort 按已声明能力中止在飞轮次
	Abort(ctx context.Context, mode AbortMode) error
	// Pause 在支持 pause_resume 时暂停在飞轮次
