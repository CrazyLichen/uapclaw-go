```go
// ──────────────────────────── 结构体 ────────────────────────────

// WireFrame CLI 线帧（CLI 2.181.0 实测，偏差 D-15.22-c）：
//
//	system/init        → 会话与能力协商
//	system/usage_update → _meta["codebuddy.ai/turnUsage"] 携带轮内用量
//	assistant          → message.content[] 为 Anthropic content blocks
//	result             → {subtype: success|error_max_turns|error_during_execution,
//	                      is_error, result, duration_ms, num_turns, usage}
//	user（input 方向）→ {"type":"user","message":{"role":"user","content":[{type:text,text}]}}
type WireFrame struct {
	Type      string          `json:"type"`
	Subtype   string          `json:"subtype,omitempty"`
	SessionID string          `json:"session_id,omitempty"`
	UUID      string          `json:"uuid,omitempty"`
	Message   json.RawMessage `json:"message,omitempty"` // assistant 帧 content blocks
	Result    string          `json:"result,omitempty"`
	IsError   bool            `json:"is_error,omitempty"`
	DurationMs *int64         `json:"duration_ms,omitempty"`
	NumTurns  *int            `json:"num_turns,omitempty"`
	Usage     json.RawMessage `json:"usage,omitempty"`
	Meta      map[string]any  `json:"_meta,omitempty"`
}

// Transport stdio 双工传输缝（测试注入点，对齐 dsh SDKFactory 手法）。
type Transport interface {
	// SendFrame 向 CLI stdin 写一帧（JSONL，自动补换行）
	SendFrame(frame map[string]any) error
	// ReadFrame 从 CLI stdout 读一帧；ctx 取消或轮超时返回错误
	ReadFrame(ctx context.Context) (WireFrame, error)
	// StderrTail 进程退出后可用的 stderr 尾巴（诊断用，≤2KB）
	StderrTail() string
	// Close 关闭传输（幂等）：杀进程、关管道
	Close() error
}

// TransportOptions 默认子进程实现的启动参数。
type TransportOptions struct {
	CliPath         string
	Args            []string            // model/resume/mcp/permission-mode 等已展开
	Cwd             string
	Env             map[string]string
	TurnTimeout     time.Duration
	InactivityTimeout time.Duration
}

// TransportFactory 传输缝工厂；nil 缝走默认子进程实现。
type TransportFactory func(options TransportOptions) (Transport, error)
```

默认实现（`defaultTransport`）：`exec.CommandContext` + `cmd.Start()`；`ReadFrame` 用 goroutine 逐行 Scanner + channel 转发，`select` 上加 `inactivityTicker`（无帧超时）与 `turnDeadline`；进程退出时收集 stderr 尾（对齐 15.13 externalcli 手法：`bytes.Buffer` 收 stderr + 退出后取尾部 2KB）。**不 import externalcli**（偏差 D-15.22-b 注：裁决 5，agent_teams 反向引用成环）。
