// Package observability 内 demand.go：
// 全局 trajectory span processor 单例面。
// 对齐 Python: openjiuwen/extensions/observability/demand.py:41/:52-73。
// 偏差 D-15.16-c2②：Python 懒构造分支构造
// agent_evolving.trajectory.processor.TrajectorySpanProcessor（Go 无该类型），
// 懒构造段 ⤴️ evolving trajectory 章节回填；本节 Get 未配置返回 nil。

var (
	trajectorySpanProcessorMu sync.Mutex
	trajectorySpanProcessor   sdktrace.SpanProcessor
)

// GetTrajectorySpanProcessor 返回进程级 trajectory span processor。
// Python: get_trajectory_span_processor (demand.py:52-73)——Go 未配置返回 nil，
// 懒构造分支 ⤴️evolving。
func GetTrajectorySpanProcessor() sdktrace.SpanProcessor {
	trajectorySpanProcessorMu.Lock()
	defer trajectorySpanProcessorMu.Unlock()
	return trajectorySpanProcessor
}

// SetTrajectorySpanProcessor 注入实例（观测初始化/测试/回填用）。
func SetTrajectorySpanProcessor(p sdktrace.SpanProcessor) {
	trajectorySpanProcessorMu.Lock()
	defer trajectorySpanProcessorMu.Unlock()
	trajectorySpanProcessor = p
}

// resetTrajectorySpanProcessorForTest 仅测试用。
func resetTrajectorySpanProcessorForTest() { SetTrajectorySpanProcessor(nil) }
