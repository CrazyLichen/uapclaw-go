```go
// Package requestsanitizer 提供 JSON provider 请求边界的非变更式归一化。
// Python: core/foundation/llm/utils/request_sanitizer.py
package requestsanitizer

import (
	"encoding/json"
	"unicode/utf16"
	"unicode/utf8"
)

// isSurrogateRune 判断是否 UTF-16 代理区字符。
func isSurrogateRune(r rune) bool { return r >= 0xD800 && r <= 0xDFFF }

// CleanUnicode 保留合法对，替换 DOM 切片产生的孤立 UTF-16 半体。
// Python: request_sanitizer.py:13 clean_unicode
func CleanUnicode(value any) any {
	switch v := value.(type) {
	case string:
		if !utf8.ValidString(v) || containsSurrogate(v) {
			return fixSurrogates(v)
		}
		return v
	case map[string]any:
		out := make(map[string]any, len(v))
		for k, item := range v {
			out[CleanUnicode(k).(string)] = CleanUnicode(item)
		}
		return out
	case []any:
		out := make([]any, len(v))
		for i, item := range v {
			out[i] = CleanUnicode(item)
		}
		return out
	default:
		return value
	}
}
```

`fixSurrogates`：按 rune 扫描，成对代理位用 `utf16.DecodeRune` 组合，孤立代理位替换为 `utf8.RuneError`（等价 Python `decode("utf-16-le","replace")` 的行为）。`containsSurrogate` 辅助。`JSONArguments`（:28）与 `SanitizeChatRequest`（:35，遍历 messages[].tool_calls[].function.arguments）按 Python 逐行转换。
