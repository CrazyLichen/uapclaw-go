```python
def gen_docx_golden(path):
    """跑 word_parser.py 逻辑产出黄金基准（复刻 _parse 不含 captioner——captioner 禁用态）。"""
    from docx import Document
    from docx.table import Table
    from docx.text.paragraph import Paragraph

    def table_to_markdown(table):
        rows = [[cell.text.strip().replace("|", "\\|") for cell in row.cells] for row in table.rows]
        if not rows:
            return ""
        lines = ["| " + " | ".join(rows[0]) + " |"]
        lines.append("| " + " | ".join("---" for _ in rows[0]) + " |")
        for row in rows[1:]:
            lines.append("| " + " | ".join(row) + " |")
        lines.append("")
        return "\n".join(lines)

    def paragraph_to_markdown(paragraph):
        text = paragraph.text.strip()
        if not text:
            return ""
        style_name = None
        try:
            if paragraph.style is not None:
                style_name = getattr(paragraph.style, "name", None)
        except Exception:
            style_name = None
        if style_name is None:
            return text
        style_name = str(style_name).strip()
        if style_name == "Title":
            return "# " + text
        if style_name.startswith("Heading "):
            try:
                level = int(style_name.split()[1])
                if 1 <= level <= 9:
                    return "#" * (level + 1) + " " + text
            except (IndexError, ValueError):
                pass
        return text

    doc = Document(path)
    content = []
    for block in doc.iter_inner_content():
        if isinstance(block, Paragraph):
            md = paragraph_to_markdown(block)
            if md:
                content.append(md)
        elif isinstance(block, Table):
            md = table_to_markdown(block)
            if md:
                content.append(md)
    result = "\n".join([line for line in content if line.strip()])
    return {"result": result if result else None}
```

```go
package parser

import (
	"archive/zip"
	"bytes"
	"encoding/xml"
	"fmt"
	"io"
	"path"
	"strings"
)

// ──────────────────────────── 结构体 ────────────────────────────

// docxModel DOCX 解析中间模型（stdlib 自研，非导出）。
//
// 持有 zip 解包后的部件字节与三张解析表，为 WordParser 提供四保真点数据：
//  1. 保序块遍历（parseBlocks）
//  2. 样式名（styleName + babelFishInternal2UI）
//  3. 合并单元格（rowCells 展开）
//  4. paragraph.text（paragraphText 拼装）
//
// 对齐 python-docx Document 的读侧子集（word_parser.py 实际消费面）。
type docxModel struct {
	// docBytes word/document.xml 原始字节
	docBytes []byte
	// styles styleId → w:name（styles.xml 的 w:style 表）
	styles map[string]string
	// rels rId → target（word/_rels/document.xml.rels 的关系表）
	rels map[string]string
	// files zip 全部部件（图片按 rels target 查字节）
	files map[string][]byte
	// docDir document.xml 所在目录（media 路径拼接基准，通常 "word"）
	docDir string
}

// ──────────────────────────── 枚举 ────────────────────────────
// ──────────────────────────── 常量 ────────────────────────────

const (
	// docxMainNS wml 主命名空间
	docxMainNS = "http://schemas.openxmlformats.org/wordprocessingml/2006/main"
	// docxRelNS r 命名空间（r:id/r:embed 属性）
	docxRelNS = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"
	// docxDrawingNS a 命名空间（a:blip）
	docxDrawingNS = "http://schemas.openxmlformats.org/drawingml/2006/main"
)

// ──────────────────────────── 全局变量 ────────────────────────────

// babelFishUI python-docx BabelFish 内部名 → UI 名转换表。
//
// Python: docx/styles/__init__.py BabelFish.style_aliases（11 对，逐字复制）
var babelFishUI = map[string]string{
	"caption":   "Caption",
	"footer":    "Footer",
	"header":    "Header",
	"heading 1": "Heading 1",
	"heading 2": "Heading 2",
	"heading 3": "Heading 3",
	"heading 4": "Heading 4",
	"heading 5": "Heading 5",
	"heading 6": "Heading 6",
	"heading 7": "Heading 7",
	"heading 8": "Heading 8",
	"heading 9": "Heading 9",
}

// ──────────────────────────── 导出函数 ────────────────────────────
// ──────────────────────────── 非导出函数 ────────────────────────────

// loadDocxModel 打开 DOCX zip 并解析三部件。
//
// 部件缺失容忍度对齐 python-docx：document.xml 缺失 → error（打开即失败，
// 外层软失败）；styles.xml/rels 缺失 → 空表（python-docx 容忍无样式文档）。
func loadDocxModel(filePath string) (*docxModel, error) {
	zr, err := zip.OpenReader(filePath)
	if err != nil {
		return nil, err
	}
	defer zr.Close()

	dm := &docxModel{
		styles: map[string]string{},
		rels:   map[string]string{},
		files:  map[string][]byte{},
	}

	// 全量解包到内存（fixture 级文档 KB 量级；python-docx 也是惰性但读侧等效）
	for _, f := range zr.File {
		rc, err := f.Open()
		if err != nil {
			return nil, fmt.Errorf("open zip entry %s: %w", f.Name, err)
		}
		data, err := io.ReadAll(rc)
		rc.Close()
		if err != nil {
			return nil, fmt.Errorf("read zip entry %s: %w", f.Name, err)
		}
		// zip 条目名统一正斜杠（Windows 产物容错，godocx ReadFromZip 同款处理）
		name := strings.ReplaceAll(f.Name, "\\", "/")
		dm.files[name] = data
	}

	docBytes, ok := dm.files["word/document.xml"]
	if !ok {
		return nil, fmt.Errorf("word/document.xml not found in docx")
	}
	dm.docBytes = docBytes
	dm.docDir = "word"

	// rels：word/_rels/document.xml.rels（缺失容忍）
	if relsBytes, ok := dm.files["word/_rels/document.xml.rels"]; ok {
		if err := dm.parseRels(relsBytes); err != nil {
			return nil, err
		}
	}

	// styles：rel target 解析（对齐 python-docx 经 document.xml.rels 找 styles part）
	if target, ok := dm.relsByType(docxRelTypeStyles); ok {
		stylePath := joinDocPath(dm.docDir, target)
		if styleBytes, ok := dm.files[stylePath]; ok {
			if err := dm.parseStyles(styleBytes); err != nil {
				return nil, err
			}
		}
	}
	return dm, nil

// parseRels 解析关系表。
//
// Python: docx.opc.package 惰性解析；Go 直接全量建表。
func (dm *docxModel) parseRels(data []byte) error {
	dec := xml.NewDecoder(bytes.NewReader(data))
	for {
		tok, err := dec.Token()
		if err == io.EOF {
			return nil
		}
		if err != nil {
			return err
		}
		if se, ok := tok.(xml.StartElement); ok && se.Name.Local == "Relationship" {
			id, target, relType := "", "", ""
			for _, a := range se.Attr {
				switch a.Name.Local {
				case "Id":
					id = a.Value
				case "Target":
					target = a.Value
				case "Type":
					relType = a.Value
				}
			}
			if id != "" && target != "" {
				// rId → target（hyperlink 外链与 styles/image 关系共用表）
				dm.rels[id] = target
				if relType == docxRelTypeStyles {
					dm.rels["__styles__"] = target
				}
			}
		}
	}
}

// parseStyles 解析 styles.xml 的 w:style → styleId/w:name 表。
func (dm *docxModel) parseStyles(data []byte) error {
	dec := xml.NewDecoder(bytes.NewReader(data))
	var inStyle bool
	var styleID, styleName string
	flush := func() {
		if inStyle && styleID != "" && styleName != "" {
			dm.styles[styleID] = styleName
		}
		inStyle, styleID, styleName = false, "", ""
	}
	for {
		tok, err := dec.Token()
		if err == io.EOF {
			flush()
			return nil
		}
		if err != nil {
			return err
		}
		switch elem := tok.(type) {
		case xml.StartElement:
			switch elem.Name.Local {
			case "style":
				flush()
				inStyle = true
				for _, a := range elem.Attr {
					if a.Name.Local == "styleId" {
						styleID = a.Value
					}
				}
			case "name":
				if inStyle {
					for _, a := range elem.Attr {
						if a.Name.Local == "val" {
							styleName = a.Value
						}
					}
				}
			}
		case xml.EndElement:
			if elem.Name.Local == "style" {
				flush()
			}
		}
	}
}

// styleName 返回 styleId 对应的原始 w:name（BabelFish 输入侧）。
func (dm *docxModel) styleName(styleID string) string {
	return dm.styles[styleID]
}

	if t, ok := dm.rels["__styles__"]; ok {

	}

}

	if dir == "" {

	}

}
```

```go
func (dm *docxModel) parseBlocks() []docxBlock {
	var blocks []docxBlock
	dec := xml.NewDecoder(bytes.NewReader(dm.docBytes))
	// 手写深度跟踪（StartElement/EndElement 配对计数）
	depth := 0
	const bodyDepth = 2 // w:document(1) > w:body(2)
	for {
		tok, err := dec.Token()
		if err == io.EOF {
			return blocks
		}
		if err != nil {
			return blocks
		}
		switch te := tok.(type) {
		case xml.StartElement:
			depth++
			if depth == bodyDepth+1 && te.Name.Local == "p" {
				blk := dm.parseParagraphBlock(dec, te)
				blocks = append(blocks, blk)
				// parseParagraphBlock 消费到 </w:p>，深度回到 bodyDepth
				depth = bodyDepth
			} else if depth == bodyDepth+1 && te.Name.Local == "tbl" {
				blk := dm.parseTableBlock(dec, te)
				blocks = append(blocks, blk)
				depth = bodyDepth
			}
		case xml.EndElement:
			depth--
		}
	}
}

// parseParagraphBlock 消费一个 w:p 元素到结束，产出段落块。
//
// 同遍历完成三件事：text 拼装（w:t/w:tab/w:br/hyperlink 子 runs）、
// styleID 提取（w:pStyle）、paraXML 重建（EncodeToken 重建原始字节）。
func (dm *docxModel) parseParagraphBlock(dec *xml.Decoder, start xml.StartElement) docxBlock {
	blk := docxBlock{kind: "para"}
	var buf bytes.Buffer
	enc := xml.NewEncoder(&buf)
	_ = enc.EncodeToken(start)
	// 段落深度（1 = w:p 自身）
	depth := 1
	inHyperlink := false
	for {
		tok, err := dec.Token()
		if err != nil {
			break
		}
		switch te := tok.(type) {
		case xml.StartElement:
			depth++
			_ = enc.EncodeToken(te)
			switch te.Name.Local {
			case "hyperlink":
				inHyperlink = true
			case "pStyle":
				for _, a := range te.Attr {
					if a.Name.Local == "val" {
						blk.styleID = a.Value
					}
				}
			case "t":
				// 文本节点：读 CharData
				if txt, err := dec.Token(); err == nil {
					if cd, ok := txt.(xml.CharData); ok {
						blk.text += string(cd)
					}
					// 紧随的 EndElement(t) 由通用 EndElement 分支处理
				}
			case "tab":
				blk.text += "\t"
			case "br":
				blk.text += "\n"
			}
		case xml.EndElement:
			depth--
			_ = enc.EncodeToken(te)
			if te.Name.Local == "hyperlink" {
				inHyperlink = false
			}
			if depth == 0 {
				_ = enc.Flush()
				blk.paraXML = buf.Bytes()
				// 样式名解析（BabelFish UI 名）
				blk.styleUI = dm.styleUI(blk.styleID)
				return blk
			}
		default:
			_ = enc.EncodeToken(tok)
		}
	}
	return blk
}

// styleUI styleId → w:name → BabelFish UI 名（空 styleID 返回空串）。
func (dm *docxModel) styleUI(styleID string) string {

	internal := dm.styleName(styleID)
	if internal == "" {
		return ""
	}
	if ui, ok := babelFishUI[internal]; ok {
		return ui
	}
	return internal
}
```

```go
// parseTableBlock 消费一个 w:tbl 元素到结束，产出网格展开的表格块。
//
// 对齐 python-docx Table.rows → row.cells 语义：
//  - gridSpan：cell 横向重复 N 次（同内容）
//  - vMerge restart：本格是纵向合并 master 起点
//  - vMerge 续格（无 val 或 val="continue"）：内容视为空，**回填所在列
//    上方最近 master 的已拼装文本**（python-docx CT_Tc 跨行合并语义：
//    续格 cell 与 master 指向同一 tc 元素，读出即 master 文本）
func (dm *docxModel) parseTableBlock(dec *xml.Decoder, start xml.StartElement) docxBlock {
	blk := docxBlock{kind: "table"}
	// 行级状态
	var rows [][]string
	var curRow []string
	// 列位置跟踪（gridSpan 展开后）
	curCol := 0
	// 每列最近的 vMerge master 文本（列号 → 文本）
	masterText := map[int]string{}

	depth := 1
	// cell 级状态
	var cellText strings.Builder
	var gridSpan int
	var vMergeVal string
	var hasVMerge bool
	inCell := false

	endCell := func() {
		if !inCell {
			return
		}
		text := cellText.String()
		span := gridSpan
		if span < 1 {
			span = 1
		}
		if hasVMerge && vMergeVal != "restart" {
			// 续格：回填本列最近 master 文本（无 master 时空串）
			text = masterText[curCol]
			span = 1 // 续格自身无 gridSpan 语义（Python row.cells 续格占 1 列）
		} else if hasVMerge {
			// master：span 覆盖的每列都记录（供下方续格按展开后列号回填——
			// python-docx 续格按网格列对齐，master 若带 gridSpan 其覆盖列都需登记）
			for i := 0; i < span; i++ {
				masterText[curCol+i] = text
			}
		}
		for i := 0; i < span; i++ {
			curRow = append(curRow, text)
		}
		curCol += span
		inCell = false
		cellText.Reset()
		gridSpan = 0
		vMergeVal = ""
		hasVMerge = false
	}

	for {
		tok, err := dec.Token()
		if err != nil {
			break
		}
		switch te := tok.(type) {
		case xml.StartElement:
			depth++
			switch te.Name.Local {
			case "tr":
				curRow = []string{}
				curCol = 0
			case "tc":
				endCell()
				inCell = true
				cellText.Reset()
				gridSpan = 0
				vMergeVal = ""
				hasVMerge = false
			case "tcPr":
				// 属性节点：等子元素 gridSpan/vMerge（通用 token 循环内处理）
			case "gridSpan":
				for _, a := range te.Attr {
					if a.Name.Local == "val" {
						if n, err := strconv.Atoi(a.Value); err == nil {
							gridSpan = n
						}
					}
				}
			case "vMerge":
				hasVMerge = true
				for _, a := range te.Attr {
					if a.Name.Local == "val" {
						vMergeVal = a.Value
					}
				}
			case "t":
				if inCell {
					if txt, err := dec.Token(); err == nil {
						if cd, ok := txt.(xml.CharData); ok {
							cellText.Write(cd)
						}
					}
				}
			case "tab":
				if inCell {
					cellText.WriteString("\t")
				}
			case "br":
				if inCell {
					cellText.WriteString("\n")
				}
			}
		case xml.EndElement:
			depth--
			switch te.Name.Local {
			case "tc":
				endCell()
			case "tr":
				if curRow != nil {
					rows = append(rows, curRow)
					curRow = nil
				}
			case "tbl":
				if depth == 0 {
					blk.rows = rows
					return blk
				}
			}
		}
	}
	blk.rows = rows
	return blk
}
```

**vMerge 判定语义**（主代码已内联）：
- `restart` = master 起点（正常文本，span 覆盖的每列记入 masterText）
- 空 val 或 `continue` = 续格（回填本列上方最近 master 文本；无 master 时空串）

```go
// extractImages 从段落 XML 提取图片并落盘。
//
// 对齐 Python _extract_images_from_paragraph 逐字语义：
//  1. 段落元素内 findall(".//a:blip")——Go 深度优先全树遍历找 blip 节点
//  2. r:embed 空 → continue
//  3. media 字节不存在 → Python related_parts.get 返回 None 跳过（**不炸**）——
//     只有解码失败才炸。此处对齐：查不到字节 continue（对齐 image_part 为 None 分支）
//  4. 解码失败 → error 向上传播（对齐 Image.open 抛异常炸整篇）
//  5. RGBA/P → RGB（Go：解码后统一转 *image.RGBA? 不——PIL convert("RGB")
//     是去 alpha；Go 用 draw.Draw 到 RGBA 画布再 png.Encode）
//  6. 落盘命名 images/{filename}__para_{n}__img_{i}.png
func (dm *docxModel) extractImages(paraXML []byte, filename string, paragraphNum int, outputDir string) ([]string, error) {
	var images []string
	imagesPerParagraph := 0

	// 深度优先遍历找所有 a:blip 节点（≈ findall(".//a:blip")）
	dec := xml.NewDecoder(bytes.NewReader(paraXML))
	var walkBlips func()
	_ = walkBlips

	// 简化：token 流扫描（blip 无嵌套 blip，token 扫描等效）
	var blipEmbeds []string
	for {
		tok, err := dec.Token()
		if err == io.EOF {
			break
		}
		if err != nil {
			return nil, err
		}
		if se, ok := tok.(xml.StartElement); ok && se.Name.Local == "blip" {
			for _, a := range se.Attr {
				if a.Name.Local == "embed" {
					blipEmbeds = append(blipEmbeds, a.Value)
				}
			}
		}
	}

	for _, rEmbed := range blipEmbeds {
		if rEmbed == "" {
			continue
		}
		data, ok := dm.mediaBytes(rEmbed)
		if !ok {
			// 对齐 Python image_part 为 None 跳过（不炸）
			continue
		}
		img, err := decodeImage(data)
		if err != nil {
			// 坏图炸整篇（对齐 Python Image.open 异常冒泡）
			return nil, fmt.Errorf("decode image %s: %w", rEmbed, err)
		}
		// RGBA/P → RGB：Go 统一转 NRGBA/RGBA 画布
		rgb := toRGBImage(img)
		if err := os.MkdirAll(outputDir, 0o755); err != nil {
			return nil, err
		}
		imagePath := filepath.Join(outputDir, fmt.Sprintf("%s__para_%d__img_%d.png", filename, paragraphNum, imagesPerParagraph))
		f, err := os.Create(imagePath)
		if err != nil {
			return nil, err
		}
		if err := png.Encode(f, rgb); err != nil {
			f.Close()
			return nil, err
		}
		f.Close()
		images = append(images, imagePath)
		imagesPerParagraph++
	}
	return images, nil
}

// decodeImage 按字节解码图片（png/jpeg/gif/webp 嗅探）。
//
// 对齐 PIL Image.open 格式自动识别。image.Decode 需要 注册解码器：
// png/jpeg/gif 在 stdlib image 包 import 时自注册；webp 需
// import golang.org/x/image/webp（副作用注册）。
func decodeImage(data []byte) (image.Image, error) {
	// image.Decode 自动按注册解码器嗅探（RegisterFormat）
	return image.Decode(bytes.NewReader(data))
}

// toRGBImage 转 RGBA 画布（≈ PIL RGBA/P convert("RGB")）。
func toRGBImage(img image.Image) *image.RGBA {
	if rgba, ok := img.(*image.RGBA); ok {
		return rgba
	}
	b := img.Bounds()
	out := image.NewRGBA(image.Rect(0, 0, b.Dx(), b.Dy()))
	draw.Draw(out, out.Bounds(), img, b.Min, draw.Src)
	return out
}
```

```go
// paragraphToMarkdown 对齐 Python _paragraph_to_markdown。
//
// Title → "#"、Heading N → N+1 个 "#"（1≤N≤9）；解析失败 Error 日志回落原文
// （对齐 except (IndexError, ValueError) 分支）。
func paragraphToMarkdown(text, styleName string) string {
	text = strings.TrimSpace(text)
	if text == "" {
		return ""
	}
	if styleName == "" {
		return text
	}
	styleName = strings.TrimSpace(styleName)
	if styleName == "Title" {
		return "# " + text
	}
	if strings.HasPrefix(styleName, "Heading ") {
		parts := strings.Fields(styleName)
		if len(parts) >= 2 {
			if level, err := strconv.Atoi(parts[1]); err == nil && level >= 1 && level <= 9 {
				return strings.Repeat("#", level+1) + " " + text
			}
		}
		// 对齐 Python: logger.error("Error while parsing docx paragraph with style %s: %r", ...)
		logger.Error(logComponent).
			Str("style_name", styleName).
			Msg("error while parsing docx paragraph style level")
	}
	return text
}

// tableToMarkdown 对齐 Python _table_to_markdown。
//
// 首行表头 + "---" 分隔 + cell strip + "\|" 转义 + 尾空行。
func tableToMarkdown(rows [][]string) string {
	if len(rows) == 0 {
		return ""
	}
	esc := func(c string) string {
		return strings.ReplaceAll(strings.TrimSpace(c), "|", "\\|")
	}
	var lines []string
	head := make([]string, len(rows[0]))
	for i, c := range rows[0] {
		head[i] = esc(c)
	}
	lines = append(lines, "| "+strings.Join(head, " | ")+" |")
	dashes := make([]string, len(rows[0]))
	for i := range dashes {
		dashes[i] = "---"
	}
	lines = append(lines, "| "+strings.Join(dashes, " | ")+" |")
	for _, row := range rows[1:] {
		cells := make([]string, len(row))
		for i, c := range row {
			cells[i] = esc(c)
		}
		lines = append(lines, "| "+strings.Join(cells, " | ")+" |")
	}
	lines = append(lines, "") // Markdown table need a one-line separation
	return strings.Join(lines, "\n")
}
```

```go
	pyDateTimeFormat = "2006-01-02 15:04:05"
	// pyMicroFormat 微秒格式（%.6f 对齐）

// pyFloatStr 还原 Python str(float) 记法。
//
// Python repr 规则（CPython float_repr）：
//   - abs ∈ [1e-4, 1e16) 用普通记法（d 格式最短往返），整数值补 ".0"
//   - 其余用 e 记法（Python 风格：1e+16 / 1e-05，指数不带前导零补齐，
//     两位以上指数照常）
//   - nan/inf/-inf 特殊字面
func pyFloatStr(f float64) string {
	if math.IsNaN(f) {
		return "nan"
	}
	if math.IsInf(f, 1) {
		return "inf"
	}
	if math.IsInf(f, -1) {
		return "-inf"
	}
	abs := math.Abs(f)
	// Python 17 位精度 d 格式 + 最短化：Go FormatFloat 'f' 与 'e' 选短者
	// Python 实际用 repr 算法（David Gay 最短表示），Go strconv 的 'g'/'f'
	// 均为最短往返算法，d 格式区域内等价。
	if abs >= 1e-4 && abs < 1e16 {
		s := strconv.FormatFloat(f, 'f', -1, 64)
		// 整数值补 .0（Python "2.0" 风格）
		if !strings.Contains(s, ".") {
			s += ".0"
		}
		return s
	}
	// e 记法：Go 'e' 与 Python 格式差异——Go "1e+16" 恰好同形；
	// 指数两位时 Python 也是 e+16/e-05 两/一位混用：
	// Python 指数最少两位（1e-05），Go e 记法指数也是两位（1e-05 同形）。
	e := strconv.FormatFloat(f, 'e', -1, 64)
	// Go e 记法形如 "1e+16"；Python 同 "1e+16"。负指数 "1e-05" 同形。
	return e
}

// cellStr 对齐 Python _cell_str：nil→""，str(value).strip()。
//
// 逐类型模拟 Python str()：
//   - int → 十进制
//   - float64 → pyFloatStr
//   - bool → "True"/"False"
//   - time.Time → Python str(datetime) 格式（微秒非 0 补 .%06d）
//   - string → 原文（strip）
func cellStr(v any) string {
	if v == nil {
		return ""
	}
	switch val := v.(type) {
	case string:
		return strings.TrimSpace(val)
	case int:
		return strconv.Itoa(val)
	case int64:
		return strconv.FormatInt(val, 10)
	case float64:
		return pyFloatStr(val)
	case bool:
		if val {
			return "True"
		}
		return "False"
	case time.Time:
		s := val.Format(pyDateTimeFormat)
		if val.Nanosecond() != 0 {
			s += fmt.Sprintf(".%06d", val.Nanosecond()/1000)
		}
		return s
	default:
		return strings.TrimSpace(fmt.Sprintf("%v", val))
	}
}

// excelSheet 单 sheet 数据（名称 + 类型化行）。
type excelSheet struct {
	// name sheet 名称
	name string
	// rows 类型化单元格行（openpyxl values_only 语义）
	rows [][]any
}

// excelSheetRows excelize adapter：读 xlsx 产 sheet 列表（名称 + [][]any）。
//
// 对齐 openpyxl load_workbook(read_only=True, data_only=True) +
// iter_rows(values_only=True)：
//   - 逐 sheet 顺序（excelize GetSheetList 同序）
//   - cell 值还原类型：数字（int/float64）/bool/string/nil
//   - 公式取缓存值（data_only=True）
func excelSheetRows(filePath string) ([]excelSheet, error) {
	f, err := excelize.OpenFile(filePath)
	if err != nil {
		return nil, err
	}
	defer f.Close()

	var sheets []excelSheet
	for _, sheet := range f.GetSheetList() {
		rows, err := f.GetRows(sheet)
		if err != nil {
			return nil, err
		}
		// GetRows 丢类型——逐 cell 按原始类型还原
		typed := make([][]any, len(rows))
		for ri, row := range rows {
			typedRow := make([]any, len(row))
			for ci := range row {
				axis, _ := excelize.CoordinatesToCellName(ci+1, ri+1)
				typedRow[ci] = excelCellValue(f, sheet, axis)
			}
			typed[ri] = typedRow
		}
		sheets = append(sheets, excelSheet{name: sheet, rows: typed})
	}
	return sheets, nil
}

// excelCellValue 还原单 cell 的 openpyxl 类型语义。
//
// excelize GetCellType 返回枚举：Bool/Number/String/InlineString/Error。
// Number cell：无小数点整数值 → int，否则 float64（对齐 openpyxl
// 按 xlsx 存储推断 int/float）。Bool → bool。空 cell → nil。
func excelCellValue(f *excelize.File, sheet, axis string) any {
	ct, err := f.GetCellType(sheet, axis)
	if err != nil {
		return nil
	}
	switch ct {
	case excelize.CellTypeBool:
		v, err := f.GetCellValue(sheet, axis)
		if err != nil {
			return nil
		}
		// excelize bool 显示 "TRUE"/"FALSE"
		return strings.EqualFold(v, "TRUE")
	case excelize.CellTypeNumber, excelize.CellTypeUnset:
		raw, err := f.GetCellValue(sheet, axis)
		if err != nil || raw == "" {
			return nil
		}
		if !strings.ContainsAny(raw, ".eE") {
			if n, err := strconv.ParseInt(raw, 10, 64); err == nil {
				return int(n)
			}
		}
		if fv, err := strconv.ParseFloat(raw, 64); err == nil {
			return fv
		}
		return raw
	default:
		// String/InlineString/Error → 原文
		v, err := f.GetCellValue(sheet, axis)
		if err != nil {
			return nil
		}
		if v == "" {
			return nil
		}
		return v
	}
}
```

```go
// 对齐 Python parse 整体覆写：
//   - 文件不存在 → 155103
//   - CSV/TSV 解码或语法失败 → 155104（cause 链）
//   - xlsx 解析失败 → 155104（cause 链）
//   - 成功 → Info 日志 + Document 列表

// include_header 经 opts.Extra["include_header"] 传入（对齐 Python kwargs.get）。

// Python: excel_parser.py ExcelParser.parse
func (p *ExcelParser) Parse(ctx context.Context, doc string, opts ...ParseOption) ([]reranker.Document, error) {
	o := NewParseOptions(opts...)

	if _, err := os.Stat(doc); os.IsNotExist(err) {
		// 对齐 Python: raise build_error(RETRIEVAL_INDEXING_FILE_NOT_FOUND, error_msg=f"File {doc} does not exist")
		return nil, exception.BuildError(
			exception.StatusRetrievalIndexingFileNotFound,
			exception.WithMsg("File "+doc+" does not exist"),
		)
	}

	ext := strings.ToLower(filepath.Ext(doc))
	baseID := o.DocID
	if baseID == "" {
		baseID = doc
	}
	if baseID == "" {
		baseID = "tabular"
	}

	includeHeader := true
	if v, ok := o.Extra["include_header"]; ok {
		if b, ok := v.(bool); ok {
			includeHeader = b
		}
	}

	var documents []reranker.Document

	if ext == ".csv" || ext == ".tsv" {
		delimiter := ","
		if ext == ".tsv" {
			delimiter = "\t"
		}
		rows, err := loadDelimitedRows(doc, delimiter)
		if err != nil {
			// 对齐 Python: logger.exception("Failed to parse %s %s: %s", ...)
			logger.Error(logComponent).
				Str("ext", strings.TrimPrefix(strings.ToUpper(ext), ".")).
				Str("doc", doc).
				Err(err).
				Msg("failed to parse delimited file")
			return nil, exception.BuildError(
				exception.StatusRetrievalIndexingFormatNotSupport,
				exception.WithMsg("Parse failed for "+doc+": "+err.Error()),
				exception.WithCause(err),
			)
		}
		sheetName := filepath.Base(doc)
		if sheetName == "" {
			sheetName = "default"
		}
		// rows [][]string → [][]any 包装（rowsToDocuments 消费 any）

		}

}

// ──────────────────────────── 非导出函数 ────────────────────────────

// rowsToDocuments 对齐 Python _rows_to_documents（黄金基准锁定语义）。
//
// 行文档：include_header 且表头非空 → "h: v" 无条件 append（值空也产出）；
// 空表头整格跳过；include_header=false 仅非空值 append；parts 空 continue。
// 列文档：空表头列回落 "Column N"；values 非空过滤；include_header=true
// → "Column name: X. Values: ..." 或 "(empty)"；false → join(values)。
func rowsToDocuments(rows [][]any, sheetName, baseID string, sheetIndex int, includeHeader bool) []reranker.Document {
	var docs []reranker.Document
	if len(rows) == 0 {
		return docs
	}

	headers := make([]string, len(rows[0]))
	for i, v := range rows[0] {
		headers[i] = cellStr(v)
	}

	dataRows := rows[1:]

	// 行文档（row_idx 从 2 起，对齐 Python enumerate(start=2)）
	for rowIdx, row := range dataRows {
		rowIdx += 2
		var parts []string
		n := len(headers)
		if len(row) < n {
			n = len(row) // zip 短边截断
		}
		for colIdx := 0; colIdx < n; colIdx++ {
			h := headers[colIdx]
			v := cellStr(row[colIdx])
			if includeHeader && h != "" {
				parts = append(parts, h+": "+v)
			} else if !includeHeader && v != "" {
				parts = append(parts, v)
			}
		}
		if len(parts) == 0 {
			continue
		}
		text := strings.Join(parts, ", ")
		docID := fmt.Sprintf("%s_s%d_r%d", baseID, sheetIndex, rowIdx)
		docs = append(docs, reranker.Document{
			ID:   docID,
			Text: text,
			Metadata: map[string]any{
				"sheet_name":  sheetName,
				"row_index":   rowIdx,
				"source_type": "row",
			},
		})
	}

	// 列文档
	for colIdx, colName := range headers {
		if colName == "" {
			colName = fmt.Sprintf("Column %d", colIdx+1)
		}
		var values []string
		for _, row := range dataRows {
			if colIdx < len(row) {
				if v := cellStr(row[colIdx]); v != "" {
					values = append(values, v)
				}
			}
		}
		var text string
		if includeHeader {
			if len(values) > 0 {
				text = fmt.Sprintf("Column name: %s. Values: %s", colName, strings.Join(values, ", "))
			} else {
				text = fmt.Sprintf("Column name: %s. Values: (empty)", colName)
			}
		} else {
			text = strings.Join(values, ", ")
		}
		docID := fmt.Sprintf("%s_s%d_c%d", baseID, sheetIndex, colIdx)
		docs = append(docs, reranker.Document{
			ID:   docID,
			Text: text,
			Metadata: map[string]any{
				"sheet_name":   sheetName,
				"column_name":  colName,
				"source_type":  "column",
			},
		})
	}

	return docs
}

// loadDelimitedRows 对齐 Python _load_delimited_rows。
//
// utf-8-sig：BOM 剥离 + 非法 UTF-8 字节 → error（对齐 Python
// UnicodeDecodeError → 155104 路径）。
func loadDelimitedRows(path, delimiter string) ([][]string, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	// utf-8-sig BOM 剥离
	data = bytes.TrimPrefix(data, []byte{0xEF, 0xBB, 0xBF})
	if !utf8.Valid(data) {
		return nil, fmt.Errorf("invalid UTF-8 bytes in %s", path)
	}
	r := csv.NewReader(bytes.NewReader(data))
	r.Comma = []rune(delimiter)[0]
	r.FieldsPerRecord = -1 // 行长度可变（对齐 Python csv.reader 默认宽松）
	return r.ReadAll()
}
```
