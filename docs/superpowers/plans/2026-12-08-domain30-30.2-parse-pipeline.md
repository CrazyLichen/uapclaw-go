```go
package kbadmin

import (
	"encoding/json"
)

// ──────────────────────────── 常量 ────────────────────────────

// 默认值常量（spec §1：上限无配置回退 50MiB；chunker 默认对齐 13.x 文本切分基线）
const (
	defaultMaxFileMB          = 50
	defaultChunkerType        = "text"
	defaultChunkerChunkSize   = 500
	defaultChunkerChunkOverlap = 50
)

// ──────────────────────────── 结构体 ────────────────────────────

// KBConfig kb_info.config_json 的解析结果（kbadmin 编排消费面）
type KBConfig struct {
	// MaxFileMB 单文件上限（MB），D-30-5-1 回填消费点
	MaxFileMB int `json:"max_file_mb"`
	// Chunker 切分配置
	Chunker ChunkerConfig `json:"chunker"`
	// Embedding 向量化配置，D-30-1-4 回填消费点
	Embedding EmbeddingEndpointConfig `json:"embedding"`
}

// ChunkerConfig 切分器配置
type ChunkerConfig struct {
	Type         string `json:"type"`
	ChunkSize    int    `json:"chunk_size"`
	ChunkOverlap int    `json:"chunk_overlap"`
}

// EmbeddingEndpointConfig embedding 服务端点配置
type EmbeddingEndpointConfig struct {
	Provider  string `json:"provider"`
	ModelName string `json:"model_name"`
	BaseURL   string `json:"base_url"`
	APIKey    string `json:"api_key"`
}

// ──────────────────────────── 导出函数 ────────────────────────────

// ParseKBConfig 解析 config_json；空/坏 JSON 回退默认（防御性：坏配置不让管理面挂死）
func ParseKBConfig(raw string) KBConfig {
	cfg := KBConfig{
		MaxFileMB: defaultMaxFileMB,
		Chunker:   ChunkerConfig{Type: defaultChunkerType, ChunkSize: defaultChunkerChunkSize, ChunkOverlap: defaultChunkerChunkOverlap},
	}
	if raw == "" {
		return cfg
	}
	var parsed KBConfig
	if err := json.Unmarshal([]byte(raw), &parsed); err != nil {
		return cfg
	}
	if parsed.MaxFileMB > 0 {
		cfg.MaxFileMB = parsed.MaxFileMB
	}
	if parsed.Chunker.Type != "" {
		cfg.Chunker.Type = parsed.Chunker.Type
	}
	if parsed.Chunker.ChunkSize > 0 {
		cfg.Chunker.ChunkSize = parsed.Chunker.ChunkSize
	}
	if parsed.Chunker.ChunkOverlap >= 0 && parsed.Chunker.ChunkOverlap != 0 || parsed.Chunker.ChunkOverlap > 0 {
		cfg.Chunker.ChunkOverlap = parsed.Chunker.ChunkOverlap
	}
	if parsed.Embedding.Provider != "" {
		cfg.Embedding = parsed.Embedding
	}
	return cfg
}
```

```go
package kbadmin

import (
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"os"
	"time"

	"github.com/google/uuid"
)

// ──────────────────────────── 全局变量 ────────────────────────────

// ErrDocDuplicate 同 KB 内 content_hash 冲突（spec §4 DOC_DUPLICATE）
var ErrDocDuplicate = errors.New("doc duplicate in kb")

// ──────────────────────────── 结构体 ────────────────────────────

// DocService 文档管理编排（30.2 真实现：解析任务/重解析/取消/列表）
type DocService struct {
	store   *Store
	uploads *UploadStore
	// worker 解析任务提交口（Domain 组装期接线；nil 时任务仅落库不入队——测试用）
	enqueue func(taskID string)
}

// ──────────────────────────── 导出函数 ────────────────────────────

// SetEnqueue 注入任务唤醒回调（worker 装配期调用）
func (s *DocService) SetEnqueue(fn func(taskID string)) { s.enqueue = fn }

// SubmitParseTask 建解析任务：hash 去重 → 建 kb_doc(pending) + kb_task(pending) → 唤醒 worker
func (s *DocService) SubmitParseTask(kbID, fileName, finalPath string, size int64, contentHash string) (*KBTask, *KBDoc, error) {
	// 去重判据：同 KB 内已存在同 hash 且非 failed/cancelled 的文档
	existing, err := s.store.Doc().ListByKB(kbID)
	if err != nil {
		return nil, nil, err
	}
	for _, d := range existing {
		if d.ContentHash == contentHash && d.ParseStatus != "failed" && d.ParseStatus != "cancelled" {
			return nil, nil, ErrDocDuplicate
		}
	}
	// contentHash 为空时现场计算
	if contentHash == "" {
		contentHash, err = fileSHA256(finalPath)
		if err != nil {
			return nil, nil, err
		}
	}
	now := time.Now().Unix()
	doc := &KBDoc{ID: uuid.NewString(), KBID: kbID, FileName: fileName,
		ParseStatus: "pending", ContentHash: contentHash,
		MetaJSON: mustJSONString(map[string]any{"size": size, "final_path": finalPath}),
		LastActivityAt: now, CreatedAt: now, UpdatedAt: now}
	if err := s.store.Doc().Create(doc); err != nil {
		return nil, nil, err
	}
	task := &KBTask{ID: uuid.NewString(), DocID: doc.ID, KBID: kbID,
		Status: "pending", CreatedAt: now}
	if err := s.store.Task().Create(task); err != nil {
		_ = s.store.Doc().Delete(doc.ID) // 回滚
		return nil, nil, err
	}
	if s.enqueue != nil {
		s.enqueue(task.ID)
	}
	return task, doc, nil
}

// fileSHA256 现场计算文件哈希
func fileSHA256(path string) (string, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(data)
	return hex.EncodeToString(sum[:]), nil
}

// mustJSONString map→JSON 字符串（失败返回 "{}"，仅元数据展示字段）
func mustJSONString(v any) string {
	data, err := json.Marshal(v)
	if err != nil {
		return "{}"
	}
	return string(data)
}
```

```go
// runPipeline 执行骨架：按序执行、每步 ctx 检查、错误即停
func runPipeline(ctx context.Context, job *ParseJob, steps []stageStep, env *pipelineEnv) error {
	for _, step := range steps {
		if err := ctx.Err(); err != nil {
			return err
		}
		env.sink.OnStageStart(job, step.name)
		if err := step.run(ctx, job, env); err != nil {
			env.sink.OnStageError(job, step.name, err)
			return err
		}
		env.sink.OnStageEnd(job, step.name)
	}
	return nil
}
```

```go
package kbadmin

import (
	"encoding/json"
	"time"

// TaskEvent 解析进度事件（kb.task.subscribe 推送载荷）
type TaskEvent struct {
	TaskID    string `json:"task_id"`
	DocID     string `json:"doc_id"`
	KBID      string `json:"kb_id"`
	Stage     string `json:"stage"`
	Phase     string `json:"phase"` // started / finished / error / status（终态）
	Status    string `json:"status,omitempty"`
	Error     string `json:"error,omitempty"`
	Timestamp int64  `json:"timestamp"`
}

// ProgressSink 观察者：stage 台账写入 + 事件发布 + 停滞扫描
type ProgressSink struct {
	store     *Store
	publisher func(TaskEvent)
}

// NewProgressSink 构造（publisher 为既有事件通道接缝；nil 允许=仅落台账）
func NewProgressSink(store *Store, publisher func(TaskEvent)) *ProgressSink {
	return &ProgressSink{store: store, publisher: publisher}
}

// emit 发布事件（publisher 为 nil 时跳过）
func (s *ProgressSink) emit(e TaskEvent) {
	e.Timestamp = time.Now().Unix()
	if s.publisher != nil {
		s.publisher(e)
	}
}

// OnStageStart 阶段开始：stages_json 追加 {stage, started_at} + last_activity_at 刷新
func (s *ProgressSink) OnStageStart(job *ParseJob, stage string) {
	s.touchStage(job, stage, func(st *stageRecord) { st.StartedAt = time.Now().Unix() })
	s.emit(TaskEvent{TaskID: job.TaskID, DocID: job.DocID, KBID: job.KBID, Stage: stage, Phase: "started"})
}

// OnStageEnd 阶段成功结束
func (s *ProgressSink) OnStageEnd(job *ParseJob, stage string) {
	s.touchStage(job, stage, func(st *stageRecord) { st.FinishedAt = time.Now().Unix() })
	s.emit(TaskEvent{TaskID: job.TaskID, DocID: job.DocID, KBID: job.KBID, Stage: stage, Phase: "finished"})
}

// OnStageError 阶段失败
func (s *ProgressSink) OnStageError(job *ParseJob, stage string, err error) {
	s.touchStage(job, stage, func(st *stageRecord) {
		st.FinishedAt = time.Now().Unix()
		st.Error = err.Error()
	})
	s.emit(TaskEvent{TaskID: job.TaskID, DocID: job.DocID, KBID: job.KBID, Stage: stage, Phase: "error", Error: err.Error()})
}

// EmitStatus 终态事件（worker 收尾调用）
func (s *ProgressSink) EmitStatus(job *ParseJob, status, errMsg string) {
	s.emit(TaskEvent{TaskID: job.TaskID, DocID: job.DocID, KBID: job.KBID, Phase: "status", Status: status, Error: errMsg})
}

// stageRecord stages_json 单阶段记录
type stageRecord struct {
	Stage      string `json:"stage"`
	StartedAt  int64  `json:"started_at"`
	FinishedAt int64  `json:"finished_at"`
	Error      string `json:"error,omitempty"`
}

// touchStage 读改写 kb_doc.stages_json + last_activity_at
func (s *ProgressSink) touchStage(job *ParseJob, stage string, mutate func(*stageRecord)) {
	doc, err := s.store.Doc().Get(job.DocID)
	if err != nil {
		return
	}
	var stages []stageRecord
	if doc.StagesJSON != "" {
		_ = json.Unmarshal([]byte(doc.StagesJSON), &stages)
	}
	target := &stageRecord{Stage: stage}
	for i := range stages {
		if stages[i].Stage == stage {
			target = &stages[i]
			break
		}
	}
	if target.Stage == "" {
		target.Stage = stage
		stages = append(stages, *target)
		target = &stages[len(stages)-1]
	}
	mutate(target)
	data, _ := json.Marshal(stages)
	doc.StagesJSON = string(data)
	doc.LastActivityAt = time.Now().Unix()
	doc.UpdatedAt = doc.LastActivityAt
	_ = s.store.Doc().Update(doc)
}

// ScanStalled 停滞扫描：last_activity_at 超阈值的 processing 文档标 suspected_stalled
// （仅标记+cancel 入口，不自动取消——对齐 WeKnora；worker 单 goroutine 周期调）
func (s *ProgressSink) ScanStalled(threshold time.Duration) {
	docs, err := s.store.Doc().ListByStatus("processing")
	if err != nil {
		return
	}
	cutoff := time.Now().Add(-threshold).Unix()
	for _, doc := range docs {
		if doc.LastActivityAt < cutoff && doc.StallState == "normal" {
			doc.StallState = "suspected_stalled"
			_ = s.store.Doc().Update(&doc)
		}
	}
}
```

```go
package kbadmin

import (
	"context"
	"sync"
	"time"

	"github.com/google/uuid"

	"github.com/uapclaw/uapclaw-go/internal/common/logger"
	"github.com/uapclaw/uapclaw-go/internal/openuapclaw/core/retrieval/embedding"
)

// ParseWorker 固定池解析执行器（spec §2 方案A）
type ParseWorker struct {
	store    *Store
	factory  *RetrievalKBFactory
	sink     *ProgressSink
	size     int
	wake     chan string // 任务 ID 唤醒通道（缓冲=池大小，防提交方阻塞）
	cancels  sync.Map    // taskID → context.CancelFunc
	stopOnce sync.Once
	stopCh   chan struct{}
	wg       sync.WaitGroup
}

// NewParseWorker 构造（size 并发度；publisher 接 progressSink 事件出口）
func NewParseWorker(store *Store, factory *RetrievalKBFactory, size int) *ParseWorker {
	return &ParseWorker{store: store, factory: factory, size: size,
		wake: make(chan string, size), stopCh: make(chan struct{})}
}

// SetSink 注入观察者（Domain 组装期；含 publisher）
func (w *ParseWorker) SetSink(sink *ProgressSink) { w.sink = sink }

// Enqueue 提交唤醒（非阻塞：缓冲满则丢弃——启动恢复期会重捞）
func (w *ParseWorker) Enqueue(taskID string) {
	select {
	case w.wake <- taskID:
	default:
	}
}

// Start 启动：孤儿恢复 + N 个常驻 worker
func (w *ParseWorker) Start() {
	w.recoverOrphans()
	for i := 0; i < w.size; i++ {
		w.wg.Add(1)
		go w.loop()
	}
}

// Stop 优雅停机（测试与进程退出用）
func (w *ParseWorker) Stop() {
	w.stopOnce.Do(func() { close(w.stopCh) })
	w.wg.Wait()
}

// recoverOrphans 启动轮询：processing → cancelled(server_restart)（只认确切信号，
// 不推断中途进度——videotask 结算纪律）；pending → 重新入队
func (w *ParseWorker) recoverOrphans() {
	orphans, err := w.store.Task().ListByStatus("processing")
	if err == nil {
		for _, t := range orphans {
			t.Status = "cancelled"
			t.CancelReason = "server_restart"
			t.FinishedAt = time.Now().Unix()
			if uerr := w.store.Task().Update(&t); uerr != nil {
				logger.Error(logComponent).Str("task_id", t.ID).Err(uerr).
					Msg("orphan task recovery failed")
				continue
			}
			w.syncDocStatus(t.DocID, "cancelled")
		}
	}
	pending, err := w.store.Task().ListByStatus("pending")
	if err == nil {
		for _, t := range pending {
			w.Enqueue(t.ID)
		}
	}
}

// loop worker 主循环
func (w *ParseWorker) loop() {
	defer w.wg.Done()
	for {
		select {
		case <-w.stopCh:
			return
		case taskID := <-w.wake:
			w.runOne(taskID)
		}
	}
}

// claimTask CAS 单认领：UPDATE kb_task SET status='processing', started_at=?
// WHERE id=? AND status='pending'——影响行数=1 才算认领成功
func (w *ParseWorker) claimTask(taskID string) bool {
	res := w.store.DB().Model(&KBTask{}).
		Where("id = ? AND status = ?", taskID, "pending").
		Updates(map[string]any{"status": "processing", "started_at": time.Now().Unix()})
	return res.Error == nil && res.RowsAffected == 1
}

// runOne 单任务执行：认领 → 装配 env → runPipeline → 终态落库
func (w *ParseWorker) runOne(taskID string) {
	if !w.claimTask(taskID) {
		return // 他 worker 已认领或任务已取消
	}
	task, err := w.store.Task().Get(taskID)
	if err != nil {
		return
	}
	doc, err := w.store.Doc().Get(task.DocID)
	if err != nil {
		w.finishTask(task, "failed", err.Error())
		return
	}
	kb, err := w.store.KB().Get(task.KBID)
	if err != nil {
		w.finishTask(task, "failed", err.Error())
		return
	}
	job := &ParseJob{TaskID: task.ID, DocID: doc.ID, KBID: task.KBID, KBType: kb.Type,
		Config: ParseKBConfig(kb.ConfigJSON)}
	var meta struct {
		FinalPath string `json:"final_path"`
	}
	_ = json.Unmarshal([]byte(doc.MetaJSON), &meta)
	job.FilePath = meta.FinalPath

	ctx, cancel := context.WithCancel(context.Background())
	w.cancels.Store(taskID, cancel)
	defer func() {
		cancel()
		w.cancels.Delete(taskID)
	}()

	steps, ok := pipelinePlans[job.KBType]
	if !ok {
		w.finishTask(task, "failed", "未知库类型: "+job.KBType)
		return
	}
	env := &pipelineEnv{store: w.store, kbFactory: w.factory,
		embedModel: w.buildEmbedModel(job.Config), sink: w.sink}
	if err := runPipeline(ctx, job, steps, env); err != nil {
		switch {
		case errors.Is(err, context.Canceled):
			w.finishTask(task, "cancelled", "")
			w.syncDocStatus(doc.ID, "cancelled")
		case errors.Is(err, ErrExtractingNotImplemented):
			w.finishTask(task, "failed", "extracting 阶段 30.4 实现")
			w.syncDocStatus(doc.ID, "failed")
		default:
			w.finishTask(task, "failed", err.Error())
			w.syncDocStatus(doc.ID, "failed")
			// 失败分级由 progressSink 阶段 error 记录承载（parse/index 归属 stage 名）
		}
		return
	}
	w.finishTask(task, "completed", "")
	w.syncDocStatus(doc.ID, "completed")
}

// finishTask 终态落库 + 状态事件
func (w *ParseWorker) finishTask(task *KBTask, status, errMsg string) {
	task.Status = status
	task.FinishedAt = time.Now().Unix()
	if err := w.store.Task().Update(task); err != nil {
		logger.Error(logComponent).Str("task_id", task.ID).Err(err).
			Msg("task finish update failed")
	}
	if w.sink != nil {
		w.sink.EmitStatus(&ParseJob{TaskID: task.ID, DocID: task.DocID, KBID: task.KBID}, status, errMsg)
	}
}

// syncDocStatus 文档状态跟随任务终态（retry_count 语义归 reparse）

	if err != nil {
		return
	}

		doc.ErrorMsg = doc.StagesJSON // 失败信息指向最新阶段明细
	}

}
```

```go
// Reparse 重解析：failed/cancelled/completed 文档建新任务行（旧行保留作历史）
func (s *DocService) Reparse(docID string) (*KBTask, error) {
	doc, err := s.store.Doc().Get(docID)
	if err != nil {
		return nil, ErrDocNotFound
	}
	if doc.ParseStatus == "pending" || doc.ParseStatus == "processing" {
		return nil, ErrDocParseActive
	}
	now := time.Now().Unix()
	doc.ParseStatus = "pending"
	doc.RetryCount++
	doc.ErrorMsg = ""
	doc.UpdatedAt = now
	if err := s.store.Doc().Update(doc); err != nil {
		return nil, err
	}
	task := &KBTask{ID: uuid.NewString(), DocID: doc.ID, KBID: doc.KBID, Status: "pending", CreatedAt: now}
	if err := s.store.Task().Create(task); err != nil {
		return nil, err
	}
	if s.enqueue != nil {
		s.enqueue(task.ID)
	}
	return task, nil
}

// CancelParse 取消解析：worker 在途走 ctx 取消；否则 DAO 直接标 cancelled
func (s *DocService) CancelParse(taskID string, cancelFn func(string) bool) error {
	task, err := s.store.Task().Get(taskID)
	if err != nil {
		return ErrTaskNotFound
	}
	if task.Status == "pending" || task.Status == "processing" {
		if cancelFn != nil && cancelFn(taskID) {
			return nil // worker 收尾落 cancelled 终态
		}
		task.Status = "cancelled"
		task.CancelReason = "user_cancel"
		task.FinishedAt = time.Now().Unix()
		if err := s.store.Task().Update(task); err != nil {
			return err
		}
		doc, derr := s.store.Doc().Get(task.DocID)
		if derr == nil {
			doc.ParseStatus = "cancelled"
			doc.UpdatedAt = time.Now().Unix()
			_ = s.store.Doc().Update(doc)
		}
	}
	return nil
}
```
