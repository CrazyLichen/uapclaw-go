**Goal:** 新增 `internal/swarm/server/kbadmin/` 域：五表 GORM AutoMigrate + DAO、KBService 真实 CRUD（SimpleKB 编排打通）、三 Service 骨架、kb.* 方法族 E2A 分发全族注册（未实现返 `KB_NOT_IMPLEMENTED`）、集成测试随批交付。

**Tech Stack:** Go + GORM（SQLite，`CGO_ENABLED=1`）+ 既有 `swarm/schema` ReqMethod / `core/retrieval/knowledgebase` / 集成双层审计门禁。

package kbadmin

import (
	"fmt"

	"gorm.io/driver/sqlite"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
)

// ──────────────────────────── 结构体 ────────────────────────────

// Store kbadmin 域 SQLite 存储（AutoMigrate 自建五表，agent_teams 惯例）
type Store struct {
	// db GORM 句柄
	db *gorm.DB
}

// ──────────────────────────── 导出函数 ────────────────────────────

// AllModels 返回五表模型切片（AutoMigrate 与集成测试共用）
func AllModels() []any {
	return []any{&KBInfo{}, &KBDoc{}, &KBChunkMeta{}, &KBChunkRevision{}, &KBTask{}}
}

// OpenStore 打开（或创建）kbadmin SQLite 存储并自动迁移五表
func OpenStore(dbPath string) (*Store, error) {
	db, err := gorm.Open(sqlite.Open(dbPath), &gorm.Config{
		Logger: logger.Default.LogMode(logger.Warn),
	})
	if err != nil {
		return nil, fmt.Errorf("打开 kbadmin 存储失败: %w", err)
	}
	if err := db.AutoMigrate(AllModels()...); err != nil {
		return nil, fmt.Errorf("kbadmin 五表 AutoMigrate 失败: %w", err)
	}
	return &Store{db: db}, nil
}

// DB 返回 GORM 句柄（DAO 使用）
func (s *Store) DB() *gorm.DB { return s.db }

// Close 关闭底层连接
func (s *Store) Close() error {
	sqlDB, err := s.db.DB()
	if err != nil {
		return err
	}
	return sqlDB.Close()
}
```

package kbadmin

import (
	"errors"

	"gorm.io/gorm"
)

// ──────────────────────────── 全局变量 ────────────────────────────

// ErrKBNotFound 知识库不存在
var ErrKBNotFound = errors.New("kb not found")

// ──────────────────────────── 结构体 ────────────────────────────

// KBDAO kb_info 表数据访问
type KBDAO struct {
	db *gorm.DB
}

// ──────────────────────────── 导出函数 ────────────────────────────

// NewKBDAO 构造 KBDAO
func NewKBDAO(db *gorm.DB) *KBDAO { return &KBDAO{db: db} }

// Create 新建知识库记录
func (d *KBDAO) Create(kb *KBInfo) error {
	return d.db.Create(kb).Error
}

// Get 按 ID 查询；不存在返回 ErrKBNotFound
func (d *KBDAO) Get(id string) (*KBInfo, error) {
	var kb KBInfo
	if err := d.db.First(&kb, "id = ?", id).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, ErrKBNotFound
		}
		return nil, err
	}
	return &kb, nil
}

// List 全量列表（置顶优先、更新时间倒序）
func (d *KBDAO) List() ([]KBInfo, error) {
	var list []KBInfo
	err := d.db.Order("pinned DESC, updated_at DESC").Find(&list).Error
	return list, err
}

// Update 更新知识库记录
func (d *KBDAO) Update(kb *KBInfo) error {
	return d.db.Save(kb).Error
}

// Delete 按 ID 删除
func (d *KBDAO) Delete(id string) error {
	return d.db.Delete(&KBInfo{}, "id = ?", id).Error
}
```

	CreateCollection(ctx context.Context, kbID, configJSON string) error
	DeleteCollection(ctx context.Context, kbID string) error
}

}

// Create 创建知识库：DAO 落行 + collection 创建（向量库失败则 DAO 回滚）
func (s *KBService) Create(ctx context.Context, id, name, typ, configJSON, folderPath string) (*KBInfo, error) {
	now := time.Now().Unix()
	if id == "" {
		id = uuid.NewString()
	}
	kb := &KBInfo{ID: id, Name: name, Type: typ, ConfigJSON: configJSON,
		FolderPath: folderPath, CreatedAt: now, UpdatedAt: now}
	if err := s.store.KB().Create(kb); err != nil {
		return nil, err
	}
	if err := s.kbFactory.CreateCollection(ctx, id, configJSON); err != nil {
		_ = s.store.KB().Delete(id) // 回滚
		return nil, err
	}
	return kb, nil
}

// Get 查询知识库（纯 DAO）
func (s *KBService) Get(id string) (*KBInfo, error) { return s.store.KB().Get(id) }

// List 列表（纯 DAO）
func (s *KBService) List() ([]KBInfo, error) { return s.store.KB().List() }

// Update 更新（纯 DAO）
func (s *KBService) Update(kb *KBInfo) (*KBInfo, error) {
	kb.UpdatedAt = time.Now().Unix()
	if err := s.store.KB().Update(kb); err != nil {
		return nil, err
	}
	return kb, nil
}

	if err := s.kbFactory.DeleteCollection(ctx, id); err != nil {
		return err
	}
	return s.store.KB().Delete(id)

	var err error
	switch request.ReqMethod {
	case schema.ReqMethodKBKBList:
		err = respondList(domain, resp)
	case schema.ReqMethodKBKBGet:
		err = respondKBGet(domain, request, resp)
	case schema.ReqMethodKBKBCreate:
		err = respondKBCreate(domain, request, resp)
	case schema.ReqMethodKBKBDelete:
		err = respondKBDelete(domain, request, resp)
	case schema.ReqMethodKBKBUpdate:
		err = respondKBUpdate(domain, request, resp)
	case schema.ReqMethodKBDocUpload,
		schema.ReqMethodKBDocImportURL, schema.ReqMethodKBDocImportManual,
		schema.ReqMethodKBDocList, schema.ReqMethodKBDocGet, schema.ReqMethodKBDocUpdate,
		schema.ReqMethodKBDocDelete, schema.ReqMethodKBDocBatchDelete,
		schema.ReqMethodKBDocReparse, schema.ReqMethodKBDocBatchReparse,
		schema.ReqMethodKBDocCancelParse, schema.ReqMethodKBDocRegenerateSummary,
		schema.ReqMethodKBDocPreview, schema.ReqMethodKBDocDownload,
		schema.ReqMethodKBChunkList, schema.ReqMethodKBChunkUpdate,
		schema.ReqMethodKBChunkRevisions, schema.ReqMethodKBChunkRevert,
		schema.ReqMethodKBChunkDelete, schema.ReqMethodKBChunkGenerateQuestions,
		schema.ReqMethodKBChunkListQuestions, schema.ReqMethodKBChunkerPreview,
		schema.ReqMethodKBSearchTest, schema.ReqMethodKBTaskProgress,
		schema.ReqMethodKBTaskSubscribe:
		resp.Code = ErrCodeKBNotImplemented
		resp.Message = "方法将在后续批次实现"
	case schema.ReqMethodKBKBPin, schema.ReqMethodKBKBMove,
		schema.ReqMethodKBKBSetConfig, schema.ReqMethodKBKBStatistics,
		schema.ReqMethodKBKBGenerateProfile:
		resp.Code = ErrCodeKBNotImplemented
		resp.Message = "方法将在后续批次实现"
	default:
		resp.Code = ErrCodeKBInternal
		resp.Message = "未知的 kb 方法"
	}
	_ = err
	return resp
}

// respondList 填充 kb 列表响应
func respondList(domain *Domain, resp *schema.AgentResponse) error {
	list, err := domain.KB.List()
	if err != nil {
		resp.Code = ErrCodeKBInternal
		resp.Message = err.Error()
		return nil
	}
	data, err := json.Marshal(list)
	if err != nil {
		return err
	}
	resp.Data = data
	return nil
}
