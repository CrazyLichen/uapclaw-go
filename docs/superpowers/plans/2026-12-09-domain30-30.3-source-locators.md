package common

// ──────────────────────────── 枚举 ────────────────────────────

// SourceLocator 类型常量（对齐 WeKnora types.SourceLocator 语义子集）
const (
	// LocatorTypeText pdf 文本锚（页码+行级 bbox）
	LocatorTypeText = "text"
	// LocatorTypeImage 图片锚（caption 块宿主：pdf=图片 placement bbox）
	LocatorTypeImage = "image"
	// LocatorTypeParagraph docx 段落号锚
	LocatorTypeParagraph = "paragraph"
	// LocatorTypeCharRange 字符区间锚（txt/md）
	LocatorTypeCharRange = "char_range"
)

// MetaKeySourcePageMap Document.Metadata 内部传递键：pdf 页-词映射 sidecar。
// 仅 parser→chunker 内部传递用；chunker 消费后必须从 chunk metadata 剔除，
// 坐标数据不进向量库（spec §4 拍板）。
const MetaKeySourcePageMap = "source_page_map"

// MetaKeySourceParagraphMap Document.Metadata 内部传递键：docx 段落映射 sidecar。
// 消费纪律同 MetaKeySourcePageMap。
const MetaKeySourceParagraphMap = "source_paragraph_map"

// MetaKeySourceCharRange Document.Metadata 标记键：值为 true 时 chunker
// 产 char_range 型 locator（txt/md 专用）。
const MetaKeySourceCharRange = "source_char_range"

// ──────────────────────────── 结构体 ────────────────────────────

// BBox 页面矩形 [x0, y0, x1, y1]，PDF 原生坐标（1/72 inch）
type BBox [4]float64

// PageWord 页内单词：文本 + 词框（gomupdf page.Words() 对应物）
type PageWord struct {
	// Text 单词文本
	Text string
	// BBox 词框
	BBox BBox
}

// PageImage 页内展示图片：placement bbox + 提取字节
type PageImage struct {
	// BBox 页面 placement 矩形
	BBox BBox
	// Data 原始编码字节（PNG/JPEG）
	Data []byte
}

// PageMap 单页映射（pdf sidecar 元素）
type PageMap struct {
	// PageNo 页号（1 起）
	PageNo int
	// PageText 该页拼装文本（与全文组装同语义）
	PageText string
	// Words 页内单词（含 bbox）
	Words []PageWord
	// Images 页内展示图片
	Images []PageImage
	// CaptionAnchors caption 锚：全文区间 + 宿主图（本页第 idx 张图）
	CaptionAnchors []CaptionAnchor
}

// CaptionAnchor caption 文本块锚（caption 非原文，锚到宿主图）
type CaptionAnchor struct {
	// Start/End 全文字符区间（半开）
	Start, End int
	// ImageIndex 宿主图在本页 Images 的下标
	ImageIndex int
}

// ParagraphMap docx 段落映射 sidecar 元素
type ParagraphMap struct {
	// Paragraph 段落号（word_parser 既有 paragraphNum 语义）
	Paragraph int
	// Text 段落拼装文本（含 caption 追加段）
	Text string
}

// SourceLocator 单条原文定位
type SourceLocator struct {
	// Type 定位类型（LocatorType* 常量）
	Type string `json:"type"`
	// PageNo pdf 页号（1 起），其余类型 0
	PageNo int `json:"page_no,omitempty"`
	// LineBBoxes pdf 锚：行级合并矩形
	LineBBoxes []BBox `json:"line_bboxes,omitempty"`
	// Paragraph docx 段落号
	Paragraph int `json:"paragraph,omitempty"`
	// CharStart/CharEnd txt/md 全文字符区间（半开）
	CharStart int `json:"char_start,omitempty"`
	CharEnd   int `json:"char_end,omitempty"`
	// Quote 引用原文（WeKnora 同款：前端页内定位句子）
	Quote string `json:"quote,omitempty"`
}
```

package parser

import (
	"fmt"

	"github.com/srijanmukherjee/gomupdf"

	"github.com/uapclaw/uapclaw-go/internal/openuapclaw/core/retrieval/common"
)

// pdfPageData 单页提取产物（gomupdf 词框 + 图片）
type pdfPageData struct {
	PageNo int
	Words  []common.PageWord
	Images []common.PageImage
}

// extractPDFPages 用 gomupdf 逐页提取词框与嵌入图片（placement + 字节）。
// 每个 Document 独占 MuPDF context，用后即关（gomupdf 并发纪律）。
func extractPDFPages(path string) ([]pdfPageData, error) {
	doc, err := gomupdf.Open(path)
	if err != nil {
		return nil, fmt.Errorf("open pdf: %w", err)
	}
	defer doc.Close() // 关闭失败仅资源回收语义，不覆盖主错误

	pages := make([]pdfPageData, 0, doc.NumPage())
	for i, page := range doc.Pages() {
		pd := pdfPageData{PageNo: i + 1}

		words, werr := page.Words() // 词框 = 字符 quad 并集（PDF points）
		if werr != nil {
			return nil, fmt.Errorf("page %d words: %w", i+1, werr)
		}
		for _, w := range words {
			pd.Words = append(pd.Words, common.PageWord{
				Text: w.Text,
				BBox: common.BBox{w.Rect.X0, w.Rect.Y0, w.Rect.X1, w.Rect.Y1},
			})
		}

		infos, ierr := page.GetImages() // placement bbox + 原始尺寸
		if ierr != nil {
			return nil, fmt.Errorf("page %d images: %w", i+1, ierr)
		}
		for idx := range infos {
			data, derr := page.ExtractImage(idx)
			if derr != nil {
				return nil, fmt.Errorf("page %d image %d: %w", i+1, idx, derr)
			}
			r := infos[idx].Rect // placement；字段名以 Task 0 冒烟为准，若 API 名有出入在此单点修正
			pd.Images = append(pd.Images, common.PageImage{
				BBox: common.BBox{r.X0, r.Y0, r.X1, r.Y1},
				Data: data,
			})
		}
		pages = append(pages, pd)
	}
	return pages, nil
}
```

⚠️ **API 对齐注意**：`page.Words()` 返回元素的字段名（`Text`/`Rect`）与 `GetImages()` 返回元素结构以 pkg.go.dev 实测为准；如与上述不符，**只在本文件单点修正**，门面接口（`extractPDFPages` 签名与 `pdfPageData`）不变。Task 0 冒烟时先写一个临时 main 打印实际类型确认。

组装语义逐条保持（偏差 D-30-3-1 已登记字符级差异）：
1. `extractPDFPages` 取词/图
2. 每页文本 = 词按读取序空格 join，页内按换行分组（词 y 差 > 行高阈值 2pt 分行，行内空格 join、行间 "\n"）
3. 非空页文本入全文；每页图片字节经既有归档（`CpImage` 语义：写 `images/` 目录）→ `CaptionImages` 生成 caption，非空 caption 紧随该页文本（`\n` 追加）
4. 全文过滤空行后 "\n".join；空串返回 ("", nil)（对齐旧软失败语义）
5. sidecar：每词在全文中的 [charStart,charEnd) 用**前向游标**定位（页文本按全文偏移切窗内 `strings.Index`）；caption 块记 `CaptionAnchor{Start,End,ImageIndex}`
6. sidecar 塞 `doc.Metadata[common.MetaKeySourcePageMap]`（`[]*common.PageMap` 序列化），由 BaseParser 装 Document 的既有路径携带

核心函数骨架（pdf_parser.go，替换全部 ledongthuc/pdfcpu 引用）：

```go
// buildPageMaps 组装页文本 + 全文 + 词区间 sidecar。
// 返回（全文, PageMaps）。caption 由调用方追加后再补 CaptionAnchor。
func buildPageMaps(pages []pdfPageData) (string, []*common.PageMap, error) {
	var b strings.Builder
	maps := make([]*common.PageMap, 0, len(pages))
	for _, pd := range pages {
		pageText := assemblePageText(pd.Words) // 行分组 join，见下
		maps = append(maps, &common.PageMap{
			PageNo: pd.PageNo, PageText: pageText,
			Words: pd.Words, Images: pd.Images,
		})
		if pageText != "" {
			if b.Len() > 0 {
				b.WriteString("\n")
			}
			b.WriteString(pageText)
		}
	}
	full := b.String()
	// 前向游标定位每词全文区间
	cursor := 0
	for _, pm := range maps {
		for wi := range pm.Words {
			w := &pm.Words[wi]
			rel := strings.Index(full[cursor:], w.Text)
			if rel < 0 { // 词跨行被 join 改写（连字等）：跳过该词定位
				continue
			}
			// 全文 offset 由页起始 offset 折算（页间含 "\n"）
			wStart := pageStartOffset(full, maps, pm) + relInPage(pm, w)
			_ = wStart // 落地时以页起始 offset 表实现，保持 O(n) 单遍
			cursor = wStart + len(w.Text)
		}
	}
	return full, maps, nil
}
```

```go
package chunker

// 本文件为 30.3 source_locators 的 chunk 级映射（WeKnora internal/sourceloc
// 同构：区间索引 → chunk locators；偏差 D-30-3-2/3 已登记）。

// mapChunkLocators 计算 chunk 全文区间 [start,end) 的定位。
// 优先级：pdf PageMap > docx ParagraphMap > char_range 标记 > nil。
func mapChunkLocators(docText string, start, end int,
	pageMaps []*common.PageMap, paraMaps []common.ParagraphMap) []common.SourceLocator
```

实现要点（每条对应一个测试）：
1. **词区间求交**：词 [ws,we) 与 chunk 区间相交 → 收集（词区间来自 PageMap，缺省时按 `strings.Index` 页内前向定位补算）
2. **行级合并**：同页词按 bbox y0 相差 ≤ 2pt 分组，组内矩形并集 `[minX0,minY0,maxX1,maxY1]`；组序按 y 升序
3. **Quote**：`docText[start:end]` 原文截取（WeKnora 同款）
4. **caption 宿主锚**：CaptionAnchor 区间与 chunk 区间相交 → `image` 型 locator（LineBBoxes=宿主图 bbox），与 text 锚并存
5. **paragraph**：段落文本与 chunk 文本互含或相交 → `paragraph` 型
6. **char_range**：`MetaKeySourceCharRange` 标记 → `{Type: char_range, CharStart: start, CharEnd: end}`

`text_chunker.go` 的 `ChunkDocuments` 挂接（**在既有对齐步骤之后追加，不改动原步骤序**）：

```go
		// 30.3 source_locators：chunk 区间前向定位 + sidecar 映射 + sidecar 键剥离
		texts, err := c.inner.ChunkText(docText) // ← 既有步骤不动
		...
		cursor := 0
		for i, text := range texts {
			// 前向游标定位 chunk 在 docText 的区间（重叠重叠段取首次出现）
			rel := strings.Index(docText[cursor:], text)
			var start, end int
			if rel >= 0 {
				start, end = cursor+rel, cursor+rel+len(text)
				cursor = start + 1 // 重叠 chunk 前进 1 防同位重复命中
			}
			// sidecar 取自 doc.Metadata（构造方塞入），映射后从 chunk metadata 剔除
			locs := mapChunkLocators(docText, start, end, pageMaps, paraMaps)
			if charRange { // doc.Metadata 标记
				locs = append(locs, common.SourceLocator{Type: common.LocatorTypeCharRange, CharStart: start, CharEnd: end})
			}
			uid := uuid.NewString()
			meta := mergeChunkMetadata(doc.Metadata, i, len(texts), uid)
			stripSidecarKeys(meta) // delete MetaKeySourcePageMap/ParagraphMap/CharRange
			chunks = append(chunks, common.TextChunk{ID: uid, Text: text, DocID: doc.ID, Metadata: meta, SourceLocators: locs})
		}
```

⚠️ `chunkUnit=token` 路径 chunk 文本可能被 tokenizer 规整（与 docText 非精确子串）：`strings.Index` 落空时该 chunk locators 留空（不猜、不近似），Warn 日志 `chunk_locator_miss`。kbadmin 链为 char 路径不受影响。
