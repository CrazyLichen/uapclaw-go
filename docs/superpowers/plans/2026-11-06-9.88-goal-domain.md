```go
// parseGoalTimestamp 解析必须带时区的时间戳。
// Python: _parse_timestamp (schema.py L56)
func parseGoalTimestamp(value any, fieldName string) (time.Time, error) {
	text, ok := value.(string)
	if !ok || text == "" {
		return time.Time{}, fmt.Errorf("invalid persisted GoalRecord %s", fieldName)
	}
	for _, layout := range []string{time.RFC3339Nano, goalTimestampLayoutSpace} {
		parsed, err := time.Parse(layout, text)
		if err == nil {
			return parsed, nil
		}
	}
	return time.Time{}, fmt.Errorf("invalid persisted GoalRecord %s", fieldName)
}

// pyInt 复刻 Python int()：接受整数、浮点（向零截断）与数字字符串。
//
// Python 侧值来自内存 int 或 checkpoint JSON 还原的 float64；
// 只认 int 会在会话恢复后翻译失败。
func pyInt(value any) (int, error) {
	switch v := value.(type) {
	case int:
		return v, nil
	case int8:
		return int(v), nil
	case int16:
		return int(v), nil
	case int32:
		return int(v), nil
	case int64:
		return int(v), nil
	case uint:
		return int(v), nil
	case uint8:
		return int(v), nil
	case uint16:
		return int(v), nil
	case uint32:
		return int(v), nil
	case uint64:
		return int(v), nil
	case float32:
		return int(v), nil
	case float64:
		return int(v), nil
	case string:
		trimmed := strings.TrimSpace(v)
		parsed, err := strconv.ParseInt(trimmed, 10, 64)
		if err != nil {
			asFloat, floatErr := strconv.ParseFloat(trimmed, 64)
			if floatErr != nil {
				return 0, err
			}
			return int(asFloat), nil
		}
		return int(parsed), nil
	default:
		return 0, fmt.Errorf("cannot convert %T to int", value)
	}
}

// optionalPositiveInt 解析可选的正整数持久化字段。
// Python: _optional_positive_int (schema.py L33)
func optionalPositiveInt(value any, fieldName string) (*int, error) {
	if value == nil {
		return nil, nil
	}
	parsed, err := pyInt(value)
	if err != nil {
		return nil, fmt.Errorf("invalid persisted GoalRecord %s", fieldName)
	}
	if parsed <= 0 {
		return nil, fmt.Errorf("invalid persisted GoalRecord %s", fieldName)
	}
	return &parsed, nil
}

// pyTruthy 复刻 Python 真值判定，用于持久化与 assessor JSON 的可选字段。
func pyTruthy(value any) bool {
	switch v := value.(type) {
	case nil:
		return false
	case bool:
		return v
	case string:
		return v != ""
	case int:
		return v != 0
	case int64:
		return v != 0
	case float64:
		return v != 0
	case []any:
		return len(v) > 0
	case map[string]any:
		return len(v) > 0
	default:
		return true
	}
}

// pyStr 复刻 str(value) 的可观测部分（上游只对字符串做有意义的渲染）。
func pyStr(value any) string {
	if text, ok := value.(string); ok {
		return text
	}
	if value == nil {
		return "None"
	}
	return fmt.Sprint(value)
}

// pyOptionalStr 复刻 `data.get(x) or None`：falsy → nil。
func pyOptionalStr(value any) *string {
	if !pyTruthy(value) {
		return nil
	}
	text := pyStr(value)
	return &text
}

// optionalStringValue 把可选字符串转成可直接放进 map/JSON 的值。
func optionalStringValue(value *string) any {
	if value == nil {
		return nil
	}
	return *value
}

// derefStr 取可选字符串的值，nil 视为空串。
func derefStr(value *string) string {
	if value == nil {
		return ""
	}
	return *value
}
```

```go
// SettleActiveTime 结算当前计时段的秒数。
// Python: GoalRecord.settle_active_time (schema.py L195)
func (r *GoalRecord) SettleActiveTime(keepActive bool) {
	now := utcNow()
	nowISO := now.Format(goalTimestampLayout)
	if r.ActiveStartedAt != nil {
		startedAt, err := parseGoalTimestamp(*r.ActiveStartedAt, "active_started_at")
		if err != nil {
			// 上游会抛 ValueError；进入内存的记录已过 FromDict 校验，
			// 这里按"零耗时"兜底并记录，避免一次脏时间戳打断整轮。
			startedAt = now
		}
		elapsed := int(now.Sub(startedAt).Seconds())
		if elapsed < 0 {
			elapsed = 0
		}
		r.TimeUsedSeconds += elapsed
		if keepActive {
			r.ActiveStartedAt = &nowISO
		} else {
			r.ActiveStartedAt = nil
		}
	} else if keepActive && r.Status == GoalStatusActive {
		r.ActiveStartedAt = &nowISO
	}
	r.UpdatedAt = nowISO
}

// FromGoalRecordDict 严格还原持久化记录；任何字段非法都返回错误（由 store 清空）。
// Python: GoalRecord.from_dict (schema.py L229)
func FromGoalRecordDict(data map[string]any) (*GoalRecord, error) {
	objective, objectiveOK := nonEmptyString(data["objective"])
	sessionID, sessionOK := nonEmptyString(data["session_id"])
	goalID, goalOK := nonEmptyString(data["goal_id"])
	if !objectiveOK || !sessionOK || !goalOK {
		return nil, errors.New("invalid persisted GoalRecord identity")
	}

	statusRaw := any(string(GoalStatusActive))
	if raw, exists := data["status"]; exists {
		statusRaw = raw
	}
	status := GoalStatus(pyStr(statusRaw))
	switch status {
	case GoalStatusActive, GoalStatusPaused, GoalStatusCompleted, GoalStatusBlocked:
	default:
		return nil, errors.New("invalid persisted GoalRecord status")
	}

	revision := 0
	if raw, exists := data["revision"]; exists {
		parsed, err := pyInt(raw)
		if err != nil {
```

```go
// 本文件对应 Python openjiuwen/harness/goal/store.py，
// 用 Session state 作为 GoalRecord 的唯一持久化真源。
//
// 禁止 sidecar JSON / SQLite / history 作为替代状态源（上游约束）。
package goal

import (
	"context"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/session/state"
	"github.com/uapclaw/uapclaw-go/internal/common/logger"
)

// ──────────────────────────── 结构体 ────────────────────────────

// GoalSession Goal 状态持久化所需的会话能力。
// Python: store.py 直接持有 Session（getattr 探测可选 commit）；Go 收窄为 4 方法接口。
type GoalSession interface {
	// GetSessionID 返回会话标识
	GetSessionID() string
	// GetState 读取状态
	GetState(key state.StateKey) (any, error)
	// UpdateState 写入状态
	UpdateState(data map[string]any)
	// Commit 显式提交到检查点
	Commit(ctx context.Context) error
}

// SessionGoalStore 会话级目标持久化。
// Python: SessionGoalStore (store.py L25)
type SessionGoalStore struct {
	// session 绑定的会话
	session GoalSession
}

// ──────────────────────────── 常量 ────────────────────────────

// SessionGoalRecordKey 目标记录在 Session state 中的键。
// Python: SESSION_GOAL_RECORD_KEY (store.py L22)
const SessionGoalRecordKey = "harness.goal.record"

// ──────────────────────────── 导出函数 ────────────────────────────

// NewSessionGoalStore 绑定会话创建存储。
// Python: SessionGoalStore(session) (store.py L28)
func NewSessionGoalStore(session GoalSession) *SessionGoalStore {
	return &SessionGoalStore{session: session}
}

// SessionID 返回绑定会话标识。
// Python: SessionGoalStore.session_id (store.py L31)
func (s *SessionGoalStore) SessionID() string {
	if s.session == nil {
		return ""
	}
	return s.session.GetSessionID()
}

// Load 读取当前目标记录；类型不符或还原失败时清空并返回 nil。
// Python: SessionGoalStore.load (store.py L35)
func (s *SessionGoalStore) Load() *GoalRecord {
	if s.session == nil {
		return nil
	}
	raw, err := s.session.GetState(state.StringKey(SessionGoalRecordKey))
	if err != nil {
		logger.Warn(logger.ComponentAgentCore).
			Err(err).
			Str("event_type", "GOAL_STORE_GET_STATE_ERROR").
			Str("session_id", s.SessionID()).
			Msg("failed to read goal state")
		return nil
	}
	if raw == nil {
		return nil
	}
	data, ok := raw.(map[string]any)
	if !ok {
		logger.Warn(logger.ComponentAgentCore).
			Str("event_type", "GOAL_STORE_INVALID_TYPE").
			Str("session_id", s.SessionID()).
			Str("state_type", pyTypeName(raw)).
			Msg("invalid goal state type in session, clearing")
		s.Clear()
		return nil
	}
	record, err := FromGoalRecordDict(data)
	if err != nil {
		logger.Warn(logger.ComponentAgentCore).
			Err(err).
			Str("event_type", "GOAL_STORE_DESERIALIZE_ERROR").
			Str("session_id", s.SessionID()).
			Msg("failed to deserialize goal state, clearing")
		s.Clear()
		return nil
	}
	return record
}

// Save 持久化目标记录（dict 形态、零值按删键处理）。
// Python: SessionGoalStore.save (store.py L59)
func (s *SessionGoalStore) Save(record *GoalRecord) {
	if s.session == nil || record == nil {
		return
	}
	s.session.UpdateState(map[string]any{SessionGoalRecordKey: record.ToDict()})
}

// Clear 从会话状态中移除目标记录。
// Python: SessionGoalStore.clear (store.py L63)
func (s *SessionGoalStore) Clear() {
	if s.session == nil {
		return
	}
	s.session.UpdateState(map[string]any{SessionGoalRecordKey: nil})
}

// Commit 显式落盘。
// Python: SessionGoalStore.commit (store.py L67)
func (s *SessionGoalStore) Commit(ctx context.Context) error {
	if s.session == nil {
		return nil
	}
	return s.session.Commit(ctx)
}

// pyTypeName 返回 Python 风格的轻量类型名，仅用于日志。
func pyTypeName(value any) string {
	switch value.(type) {
	case nil:
		return "NoneType"
	case string:
		return "str"
	case bool:
		return "bool"
	case int, int64:
		return "int"
	case float64:
		return "float"
	case []any:
		return "list"
	case map[string]any:
		return "dict"
	default:
		return "object"
	}
}
```

```go
// Python: _JSON_BLOCK_PATTERN (evaluation.py L33)，re.DOTALL + 非贪婪
var jsonBlockPattern = regexp.MustCompile("(?s)```(?:json)?\\s*\\n?(.*?)\\n?\\s*```")

// ──────────────────────────── 非导出函数 ────────────────────────────

// parseAssessmentJSON 解析 assessor 原始输出，失败时尝试围栏块。
// Python: _parse_assessment_json (evaluation.py L39)
func parseAssessmentJSON(text string) *GoalAssessment {
	trimmed := strings.TrimSpace(text)

	var data any
	if err := json.Unmarshal([]byte(trimmed), &data); err != nil {
		if match := jsonBlockPattern.FindStringSubmatch(trimmed); match != nil {
			if blockErr := json.Unmarshal([]byte(strings.TrimSpace(match[1])), &data); blockErr != nil {
				data = nil
			}
		}
	}

	payload, ok := data.(map[string]any)
	if !ok {
		return nil
	}

	statusText, ok := payload["status"].(string)
	if !ok || statusText == "" {
		return nil
	}
	if _, exists := payload["evidence"]; !exists || !pyTruthy(payload["evidence"]) {
		return nil
	}

	status := GoalAssessmentStatus(statusText)
	switch status {
	case GoalAssessmentStatusContinue,
		GoalAssessmentStatusComplete,
		GoalAssessmentStatusBlocked:
	default:
		return nil
	}

	return &GoalAssessment{
		Status:          status,
		Evidence:        pyStr(payload["evidence"]),
		RemainingWork:   pyOptionalStr(payload["remaining_work"]),
		NextInstruction: pyOptionalStr(payload["next_instruction"]),
	}
}

// fallbackContinue 返回保守的 continue 结论。
// Python: _FALLBACK_CONTINUE (evaluation.py L26)，Go 每次新建（无模块级可变共享）
func fallbackContinue() GoalAssessment {
	remaining := "Continue making verifiable progress toward the objective."
	instruction := "Continue advancing the objective with verifiable evidence."
	return GoalAssessment{
		Status:          GoalAssessmentStatusContinue,
		Evidence:        "assessment_error: no reliable assessment available, treating as continue.",
		RemainingWork:   &remaining,
		NextInstruction: &instruction,
	}
}

// effectiveLimit 复刻 Python 的 `config.x or record.x`（0 视为未设置）。
func effectiveLimit(configValue, recordValue *int) *int {
	if configValue != nil && *configValue != 0 {
		return configValue
	}
	if recordValue != nil && *recordValue != 0 {
		return recordValue
	}
	return nil
}
```

```go
// ShouldSpotCheck 判定 CONTINUE 是否要抽检复核。
// Python: GoalEvaluator.should_spot_check (evaluation.py L264)
func (e *GoalEvaluator) ShouldSpotCheck(record *GoalRecord) bool {
	if record == nil || e.config.VerificationInterval == nil {
		return false
	}
	interval := *e.config.VerificationInterval
	if interval <= 0 {
		return false
	}
	return record.AttemptCount > 0 && record.AttemptCount%interval == 0

// Assess 按配置策略产出评估结论。
//
// transcriptResponse 为 nil 表示监理未被调用；空串表示调用了但返回空。
// Python: GoalEvaluator.assess (evaluation.py L88)
func (e *GoalEvaluator) Assess(
	record *GoalRecord,
	agentReport *GoalAssessment,
	transcriptResponse *string,
) GoalAssessment {
	var assessment GoalAssessment
	switch e.config.Strategy {
	case GoalStopStrategyAgentReport:
		assessment = assessAgentReport(agentReport)
	case GoalStopStrategyTranscript:
		assessment = assessTranscript(transcriptResponse)
	default:
		assessment = e.assessHybrid(record, agentReport, transcriptResponse)
	}
	return e.applyHardLimits(record, assessment)
}
```

```go
// assessAgentReport 直接信任主模型自报。
// Python: GoalEvaluator._assess_agent_report (evaluation.py L123)
func assessAgentReport(agentReport *GoalAssessment) GoalAssessment {

		return GoalAssessment{

			Evidence:        "no_report: no submit_goal_report was accepted for this attempt.",
			RemainingWork:   &remaining,

// assessTranscript 解析独立监理模型的 JSON 输出。
// Python: GoalEvaluator._assess_transcript (evaluation.py L138)
func assessTranscript(transcriptResponse *string) GoalAssessment {
	if transcriptResponse == nil {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_TRANSCRIPT_MISSING").
			Msg("transcript strategy: no response, fallback continue")
		return fallbackContinue()
	}
	parsed := parseAssessmentJSON(*transcriptResponse)
	if parsed == nil {
		logger.Warn(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_TRANSCRIPT_PARSE_FAILED").
			Msg("transcript strategy: failed to parse response, fallback continue")
		return fallbackContinue()
	}
	return *parsed
}

// assessHybrid 默认策略：信任 continue，终态必须复核。
// Python: GoalEvaluator._assess_hybrid (evaluation.py L155)
func (e *GoalEvaluator) assessHybrid(
	record *GoalRecord,
	agentReport *GoalAssessment,
	transcriptResponse *string,
) GoalAssessment {
	if agentReport == nil {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_HYBRID_NO_REPORT").
			Msg("hybrid: no agent report, using transcript")
		return assessTranscript(transcriptResponse)
	}

	normalized := normalizeReport(agentReport)

	if normalized.Status == GoalAssessmentStatusContinue {
		if e.ShouldSpotCheck(record) {
			logger.Info(logger.ComponentAgentCore).
				Str("event_type", "GOAL_ASSESS_HYBRID_SPOT_CHECK").
				Int("attempt_count", record.AttemptCount).
				Msg("hybrid: spot-check triggered")
			verified := verifyWithTranscript(normalized, transcriptResponse, true)
			return verified
		}
		return normalized
	}

	if transcriptResponse == nil {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_HYBRID_TRANSCRIPT_NOT_INVOKED").
			Str("agent_status", string(normalized.Status)).
			Msg("hybrid: transcript not invoked, falling back to continue")
	} else {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_HYBRID_VERIFYING").
			Str("agent_status", string(normalized.Status)).
			Msg("hybrid: verifying via transcript")
	}
	return verifyWithTranscript(normalized, transcriptResponse, false)
}

// verifyWithTranscript 用监理结论复核自报；冲突时监理赢。
// Python: GoalEvaluator._verify_with_transcript (evaluation.py L210)
func verifyWithTranscript(
	agentAssessment GoalAssessment,
	transcriptResponse *string,
	trustOnFailure bool,
) GoalAssessment {
	if transcriptResponse == nil {
		if trustOnFailure {
			return agentAssessment
		}
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_TRANSCRIPT_UNAVAILABLE").
			Msg("transcript unavailable, fallback continue")
		evidence := fmt.Sprintf(
			"transcript_unavailable: agent reported %s but transcript assessor was not invoked for verification.",
			agentAssessment.Status,
		)
		return GoalAssessment{
			Status:          GoalAssessmentStatusContinue,
			Evidence:        evidence,
			RemainingWork:   agentAssessment.RemainingWork,
			NextInstruction: agentAssessment.NextInstruction,
		}
	}

	transcriptResult := parseAssessmentJSON(*transcriptResponse)
	if transcriptResult == nil {
		if trustOnFailure {
			return agentAssessment
		}
		logger.Warn(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_TRANSCRIPT_PARSE_FAILED").
			Msg("transcript parse failed, fallback continue")
		evidence := fmt.Sprintf(
			"assessment_error: agent reported %s but transcript assessor response could not be parsed.",
			agentAssessment.Status,
		)
		return GoalAssessment{
			Status:          GoalAssessmentStatusContinue,
			Evidence:        evidence,
			RemainingWork:   agentAssessment.RemainingWork,
			NextInstruction: agentAssessment.NextInstruction,
		}
	}

	if transcriptResult.Status != agentAssessment.Status {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_CONFLICT").
			Str("agent_status", string(agentAssessment.Status)).
			Str("transcript_status", string(transcriptResult.Status)).
			Msg("conflict between agent report and transcript, using transcript")
	}
	return *transcriptResult
}

// normalizeReport 归一化自报状态；非法值降级为 continue。
// Python: GoalEvaluator._normalize_report (evaluation.py L271)
func normalizeReport(report *GoalAssessment) GoalAssessment {
	switch report.Status {
	case GoalAssessmentStatusContinue,
		GoalAssessmentStatusComplete,
		GoalAssessmentStatusBlocked:
		return *report
	default:
		logger.Warn(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_INVALID_REPORT_STATUS").
			Str("report_status", string(report.Status)).
			Msg("invalid report status, normalizing to continue")
		return GoalAssessment{
			Status:          GoalAssessmentStatusContinue,
			Evidence:        fmt.Sprintf("normalized_invalid_status: original was %s", report.Status),
			RemainingWork:   report.RemainingWork,
			NextInstruction: report.NextInstruction,
		}
	}
}

// applyHardLimits 施加硬上限；只有 CONTINUE 会被改判。
// Python: GoalEvaluator._apply_hard_limits (evaluation.py L289)
func (e *GoalEvaluator) applyHardLimits(record *GoalRecord, assessment GoalAssessment) GoalAssessment {
	if assessment.Status != GoalAssessmentStatusContinue {
		return assessment
	}
	effectiveMax := effectiveLimit(e.config.MaxAttempts, record.MaxAttempts)
	effectiveBudget := effectiveLimit(e.config.TokenBudget, record.TokenBudget)
	if effectiveMax != nil && record.AttemptCount >= *effectiveMax {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_MAX_ATTEMPTS_EXHAUSTED").
			Int("attempt_count", record.AttemptCount).
			Int("max_attempts", *effectiveMax).
			Msg("max_attempts exhausted")
		return GoalAssessment{
			Status:          GoalAssessmentStatusBlocked,
			Evidence:        fmt.Sprintf("max_attempts_exhausted: %d/%d", record.AttemptCount, *effectiveMax),
			RemainingWork:   assessment.RemainingWork,
			NextInstruction: assessment.NextInstruction,
		}
	}
	if effectiveBudget != nil && record.TokenUsage.TotalTokens >= *effectiveBudget {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "GOAL_ASSESS_TOKEN_BUDGET_EXHAUSTED").
			Int("total_tokens", record.TokenUsage.TotalTokens).
			Int("token_budget", *effectiveBudget).
			Msg("token budget exhausted")
		return GoalAssessment{
			Status: GoalAssessmentStatusBlocked,
			Evidence: fmt.Sprintf(
				"token_budget_exhausted: %d/%d", record.TokenUsage.TotalTokens, *effectiveBudget,
			),
			RemainingWork:   assessment.RemainingWork,
			NextInstruction: assessment.NextInstruction,
		}
	}
	return assessment
}
```

```go
// BuildGoalTaskQuery 构造一轮 goal 尝试的用户 query。
// Python: build_goal_task_query (goal.py L171)
func BuildGoalTaskQuery(view GoalTaskView, language string) string {
	previous := formatPreviousAssessment(view.Previous, language)
	instruction := BuildGoalCurrentInstruction(view, language)
	budget := formatBudgetNotice(view, language)

	replacer := strings.NewReplacer(
		"{objective}", view.Objective,
		"{previous_assessment}", previous,
		"{current_instruction}", instruction,
		"{budget_notice}", budget,
	)
	return replacer.Replace(goalTaskTemplate(language))
}

// BuildGoalCurrentInstruction 返回本轮权威指令。
// Python: build_goal_current_instruction (goal.py L199)

	if view.Previous != nil && view.Previous.NextInstruction != "" {
		return view.Previous.NextInstruction

// formatPreviousAssessment 渲染 previous_assessment 段落。
// Python: _format_previous_assessment (goal.py L138)
func formatPreviousAssessment(previous *GoalTaskAssessment, language string) string {
	if previous == nil {
		if language == "en" {
			return goalTaskNoPreviousEN
		}
		return goalTaskNoPreviousCN
	}
	parts := []string{fmt.Sprintf("status: %s", previous.Status)}
	if previous.Evidence != "" {
		parts = append(parts, fmt.Sprintf("evidence: %s", previous.Evidence))
	}
	if previous.RemainingWork != "" {
		parts = append(parts, fmt.Sprintf("remaining_work: %s", previous.RemainingWork))
	}
	return strings.Join(parts, "\n")
}

// formatBudgetNotice 渲染 budget_notice 段落。
// Python: _format_budget_notice (goal.py L152)
func formatBudgetNotice(view GoalTaskView, language string) string {
	notices := make([]string, 0, 2)
	if view.MaxAttempts != nil {
		remaining := *view.MaxAttempts - view.AttemptCount
		if remaining < 0 {
			remaining = 0
		}
		if language == "en" {
			notices = append(notices,
				fmt.Sprintf("Remaining attempts: %d/%d", remaining, *view.MaxAttempts))
		} else {
			notices = append(notices,
				fmt.Sprintf("剩余尝试次数：%d/%d", remaining, *view.MaxAttempts))
		}
	}
	if view.TokenBudget != nil {
		if language == "en" {
			notices = append(notices,
				fmt.Sprintf("Token budget: %d/%d", view.TotalTokens, *view.TokenBudget))
		} else {
			notices = append(notices,
				fmt.Sprintf("Token 预算：%d/%d", view.TotalTokens, *view.TokenBudget))
		}
	}
	if len(notices) == 0 {
		if language == "en" {
			return goalTaskNoBudgetEN
		}
		return goalTaskNoBudgetCN
	}
	return strings.Join(notices, "\n")
}
```

```go
import (
```

```go
// ──────────────────────────── 结构体 ────────────────────────────

// GoalStore GoalManager 所需的持久化面。
// Python: GoalStore Protocol (manager.py L27)
type GoalStore interface {
	// SessionID 返回绑定会话标识
	SessionID() string
	// Load 读取当前记录（无则 nil）
	Load() *GoalRecord
	// Save 持久化记录
	Save(record *GoalRecord)
	// Clear 移除记录

	// Commit 显式落盘

// Coordinator 交互监督器所需的 Goal 能力面，由 *GoalManager 实现。
//
// Python: DeepAgent.goal_manager 是具体类 GoalManager；Go 用窄接口避免
// hschema → goal 的反向依赖（规则 10.2：这是 DeepAgent 依赖他人的窄接口）。
type Coordinator interface {
	// HasActiveRecord 当前会话是否存在 ACTIVE 记录
	HasActiveRecord() bool
	// EnsureActiveGoalWorkLocked 在 ACTIVE 目标缺少待办时补排（调用方持控制锁）
	EnsureActiveGoalWorkLocked() bool
	// BeginAttempt 开始一次尝试；返回 nil 表示该代际不应开始
	BeginAttempt(ctx context.Context, goalID string, revision int) (*GoalRecord, error)
}

// GoalManagerConfig GoalManager 构造参数。
// Python: GoalManager(*, store, event_manager, control_lock, has_output_stream,
//
//	cancel_active_round, emit_event, notify_work, language) (manager.py L54)
type GoalManagerConfig struct {
	// Store 持久化面
	Store GoalStore
	// EventManager 双队列调度器（无内部锁）
	EventManager *task_loop.EventManager
	// ControlLock 与 DeepAgent 共用的交互控制锁
	ControlLock *sync.Mutex
	// HasOutputStream 是否有输出消费者
	HasOutputStream func() bool
	// CancelActiveRound 取消在飞轮回调（调用方持控制锁，不得再取锁）
	CancelActiveRound func(ctx context.Context, kind hschema.RoundWorkKind, goalID, reason string)
	// EmitEvent 发出交互事件
	EmitEvent func(event hschema.InteractionEvent)
	// NotifyWork 唤醒监督器
	NotifyWork func()
	// Language 提示词语言
	Language string
}

// GoalManager 会话级 Goal 能力与唯一写者。
//
// ⚠️ 所有状态写操作共用 DeepAgent 的交互控制锁（Python 同）。
// 带 Locked 后缀的方法要求调用方已持锁，方法内部绝不再取锁。
// Python: GoalManager (manager.py L47)
type GoalManager struct {

	store GoalStore
	// eventManager 双队列调度器
	eventManager *task_loop.EventManager

	controlMu *sync.Mutex
	// hasOutputStreamFn 输出消费者探测
	hasOutputStreamFn func() bool
	// cancelActiveRound 取消在飞轮
	cancelActiveRound func(ctx context.Context, kind hschema.RoundWorkKind, goalID, reason string)
	// emitEvent 事件出口
	emitEvent func(event hschema.InteractionEvent)
	// notifyWork 唤醒监督器
	notifyWork func()
	// language 提示词语言
	language string
}

// Peek 返回深拷贝记录，不等待控制锁。
// Python: GoalManager.peek (manager.py L89)
```

```go
// Set 创建目标记录；首次创建或（确认后）覆盖既有目标。
//
// 注意：三处参数校验在取锁**之前**执行，与上游一致（非法参数不排队等锁）。
// Python: GoalManager.set (manager.py L104)
func (m *GoalManager) Set(ctx context.Context, objective string, opts SetOptions) (*GoalRecord, error) {
	normalized := strings.TrimSpace(objective)
	if normalized == "" {
		return nil, newGoalOperationError(
			"set", "invalid_objective", "goal objective must not be empty", nil,
		)
	}
	if opts.TokenBudget != nil && *opts.TokenBudget <= 0 {
		return nil, newGoalOperationError(
			"set", "invalid_objective", "token_budget must be positive", nil,
		)
	}
	if opts.MaxAttempts != nil && *opts.MaxAttempts <= 0 {
		return nil, newGoalOperationError(
			"set", "invalid_objective", "max_attempts must be positive", nil,
		)
	}

	m.lock()
	defer m.unlock()

	existing := m.store.Load()
	if existing != nil && !opts.OverwriteConfirmed {
		return nil, newGoalOperationError(
			"set", "already_exists", "a goal already exists for this session", existing,
		)
	}
	if existing != nil && m.eventManager != nil {
		m.eventManager.DiscardGoalWork(existing.SessionID, existing.GoalID)
	}

	record := CreateGoalRecord(m.store.SessionID(), normalized, opts.MaxAttempts, opts.TokenBudget)
	m.store.Save(record)
	if err := m.commitStoreLocked(ctx); err != nil {
		return nil, err
	}

	// 既有输出流仍是唯一消费者：先排替代轮再中止旧轮，流会自然续到新目标。
	if m.hasOutputStream() {
		m.ensureGoalWorkLocked(record)
		m.emitGoalUpdatedLocked(record)
	}

	if existing != nil {
		m.cancelRound(ctx, hschema.RoundWorkKindGoal, existing.GoalID, "goal_overwrite")
	}

	return record.Clone(), nil
}
```

```go
func (m *GoalManager) ensureGoalWorkLocked(record *GoalRecord) bool {
	if record == nil || record.Status != GoalStatusActive || !m.hasOutputStream() {
		return false
	}
	if m.eventManager == nil {
		return false
	}

		map[string]any{"query": query},
		record.GoalID,
		record.Revision,

	}

}
```

```go
// Pause 暂停目标；不提升 revision，让在飞轮自然收尾并提交评估。
// Python: GoalManager.pause (manager.py L173)
func (m *GoalManager) Pause(ctx context.Context) (*GoalRecord, error) {
	m.lock()
	defer m.unlock()

	record := m.store.Load()
	if record == nil {
		return nil, nil
	}
	if record.Status == GoalStatusActive {
		// 不提升 revision：在飞轮（含评估落盘）必须仍然有效。
		if m.eventManager != nil {
			m.eventManager.DiscardGoalWork(record.SessionID, record.GoalID)
		}
		// 先结算，保证只显示 time_used 的宿主在暂停时不会跳到 0。
		record.SettleActiveTime(false)
		// 在飞轮仍在跑：为重开时钟以便后端继续计时。
		if m.hasInFlightGoalAttempt(record) {
			record.StartTiming()
		}
		record.Status = GoalStatusPaused
		record.Touch(false)
		m.store.Save(record)
		if err := m.commitStoreLocked(ctx); err != nil {
			return nil, err
		}
		if m.hasOutputStream() {
			m.emitGoalUpdatedLocked(record)
		}
	}
	return record.Clone(), nil
}

// Resume 恢复 PAUSED/BLOCKED 目标；空闲时提升 revision 并补排一轮。
// Python: GoalManager.resume (manager.py L203)
func (m *GoalManager) Resume(ctx context.Context) (*GoalRecord, error) {
	m.lock()
	defer m.unlock()

	record := m.store.Load()
	if record == nil {
		return nil, nil
	}
	if record.Status == GoalStatusPaused || record.Status == GoalStatusBlocked {
		inFlight := m.hasInFlightGoalAttempt(record)
		// 暂停期若尝试仍在跑，先把这段时间折入 time_used。
		record.SettleActiveTime(false)
		record.Status = GoalStatusActive
		record.StartTiming()
		// 空闲/阻塞恢复提升代际；同一尝试仍在飞则不提升、也不重复排队。
		record.Touch(!inFlight)
		m.store.Save(record)
		if err := m.commitStoreLocked(ctx); err != nil {
			return nil, err
		}
		if m.hasOutputStream() {
			if !inFlight {
				m.ensureGoalWorkLocked(record)
			}
			m.emitGoalUpdatedLocked(record)
		}
	}
	return record.Clone(), nil
}
```

```go
// Python: GoalManager.clear (manager.py L230)
func (m *GoalManager) Clear(ctx context.Context) (*GoalRecord, error) {
	m.lock()
	defer m.unlock()

	record := m.store.Load()
	if record == nil {

	}
	return record.Clone(), nil
```

```go
// AccumulateUsage 累加一次 token 用量，并结算已流逝的工时段。
//
// 注意：上游此处只 save 不 commit（manager.py L289-301），Go 照抄。
// Python: GoalManager.accumulate_usage (manager.py L280)
func (m *GoalManager) AccumulateUsage(
	goalID string,
	revision int,
	inputTokens int,
	outputTokens int,
	cachedInputTokens int,
) error {
	m.lock()
	defer m.unlock()

	record := m.store.Load()
	if !matchesInFlight(record, goalID, revision) {
		return nil
	}
	// PAUSED 的在飞轮仍持有 active_started_at，同样要结算。
	if record.ActiveStartedAt != nil &&
		(record.Status == GoalStatusActive || record.Status == GoalStatusPaused) {
		record.SettleActiveTime(true)
	}
	record.TokenUsage.Accumulate(inputTokens, outputTokens, cachedInputTokens)
	record.Touch(false)
	m.store.Save(record)
	return nil
}

// ApplyAssessment 在当前代际仍有效时提交一次评估。
//
// 接受 ACTIVE 与 PAUSED：暂停只停"续跑"，允许在飞轮把评估写完。
// Python: GoalManager.apply_assessment (manager.py L303)
func (m *GoalManager) ApplyAssessment(
	ctx context.Context,
	goalID string,
	revision int,
	assessment GoalAssessment,
) (*GoalRecord, error) {
	m.lock()
	defer m.unlock()

	record := m.store.Load()
	if !matchesInFlight(record, goalID, revision) {
		return nil, nil
	}
	record.LastAssessment = &assessment
	switch assessment.Status {
	case GoalAssessmentStatusComplete:
		record.SettleActiveTime(false)
		record.Status = GoalStatusCompleted
		record.LastStopReason = strPtr("completed")
	case GoalAssessmentStatusBlocked:
		record.SettleActiveTime(false)
		record.Status = GoalStatusBlocked
		record.LastStopReason = strPtr("blocked")
	default:
		if record.Status == GoalStatusActive {
			record.SettleActiveTime(true)
		} else if record.Status == GoalStatusPaused {
			// PAUSED 下收尾：关时钟，恢复时另开新段，空闲暂停不计费。
			record.SettleActiveTime(false)
		}
	}
	record.Touch(false)
	m.store.Save(record)
	if err := m.commitStoreLocked(ctx); err != nil {
		return nil, err
	}

	if record.Status == GoalStatusActive && m.hasOutputStream() {
		m.ensureGoalWorkLocked(record)
	}
	if m.hasOutputStream() {
		m.emitGoalUpdatedLocked(record)
	}
	return record.Clone(), nil
}
```
