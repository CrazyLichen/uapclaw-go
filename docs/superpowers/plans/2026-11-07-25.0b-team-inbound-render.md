> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把上游 `inbound_render.py` / `prompts/messages.py` / `timefmt.py` / `external/format.py` 的六类团队 XML 标签渲染搬到 Go，并补齐 `TeamRole` 四取值、10 个 i18n 键、`ReplyHintFor` 与 `inbound_tags` 静态 Section。

package agent_teams

import (
	"time"

	"github.com/uapclaw/uapclaw-go/internal/agent_teams/schema"

// ──────────────────────────── 结构体 ────────────────────────────

// ──────────────────────────── 枚举 ────────────────────────────

// ──────────────────────────── 常量 ────────────────────────────

const (
	// justNowSeconds 「刚刚」的时间桶上限（秒）
	justNowSeconds = 10
	// minuteSeconds 一分钟（秒）
	minuteSeconds = 60
	// hourSeconds 一小时（秒）
	hourSeconds = 60 * 60
	// daySeconds 一天（秒）
	daySeconds = 24 * 60 * 60
)

// ──────────────────────────── 全局变量 ────────────────────────────

// ──────────────────────────── 导出函数 ────────────────────────────

// FormatAbsoluteTime 把毫秒 epoch 渲染为绝对本地时间，不带相对差。
//
// 用于写进 agent 对话历史、之后会被重新读取的文本。相对差只在渲染那一刻为真：
// 一条消息以「刚刚」送达，一小时后成员重新推理时仍读作「刚刚」。
// 这类文本只携带绝对时间，读者用每次输入附带的当前时间锚点
// （inbound_render.RenderClock）自行计算经过时长。
//
// Python: format_absolute_time(timestamp_ms) (timefmt.py L102-122)
func FormatAbsoluteTime(timestampMs int64) string {
	return formatAbsolute(time.Local, timestampMs)
}

// FormatTimeContext 把时间戳渲染为「绝对本地时间 (相对差)」。
//
// timestampMs 为 nil 时（源字段未设置，例如从未流转过的任务）返回 time.unknown 文案。
// nowMs 永远由调用方传入，保持本函数纯函数可测。
//
// Python: format_time_context(timestamp_ms, now_ms) (timefmt.py L82-99)
func FormatTimeContext(timestampMs *int64, nowMs int64) string {
	return formatTimeContextAt(time.Local, timestampMs, nowMs)
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// relativeKeyAndValue 选出相对时间的 i18n key 与数值。
//
// 纯数值逻辑，不带语言耦合：调用方用 schema.T 渲染返回的 key。
// 负 delta（时间戳在未来，即时钟偏斜）与 10 秒以内都归入「刚刚」，
// 让模型永远看不到负数或噪声计数。
//
// Python: _relative_key_and_value(delta_ms) (timefmt.py L32-58)
func relativeKeyAndValue(deltaMs int64) (string, *int64) {
	if deltaMs < 0 {
		return "time.just_now", nil
	}
	seconds := deltaMs / 1000
	if seconds < justNowSeconds {
		return "time.just_now", nil
	}
	if seconds < minuteSeconds {
		return "time.seconds_ago", &seconds
	}
	if seconds < hourSeconds {
		v := seconds / minuteSeconds
		return "time.minutes_ago", &v
	}
	if seconds < daySeconds {
		v := seconds / hourSeconds
		return "time.hours_ago", &v
	}
	v := seconds / daySeconds
	return "time.days_ago", &v
}

// formatAbsolute 把 epoch 渲染为带数字时区偏移的本地墙钟时间。
//
// 用传入的 loc 解析（对应上游 astimezone() 取运行时本地时区），并把偏移写成
// +08:00 形式，让跨机器的读者仍能无歧义地对齐两个绝对时间。
// loc 可注入是为了测试固定时区，避免改全局 time.Local 污染其他测试。
//
// Python: _format_absolute(timestamp_ms) (timefmt.py L61-79)
func formatAbsolute(loc *time.Location, timestampMs int64) string {
	t := time.UnixMilli(timestampMs).In(loc)
	return t.Format("2006-01-02 15:04:05 -07:00")
}

// formatTimeContextAt 可注入时区的 format_time_context 实现（供测试固定时区）。
//
// Python 侧直接依赖 astimezone() 无可注入点，此处仅为测试可确定性而拆出。
func formatTimeContextAt(loc *time.Location, timestampMs *int64, nowMs int64) string {
	if timestampMs == nil {
		msg, err := schema.T("time.unknown")
		if err != nil {
			return "time.unknown"
		}
		return msg
	}
	key, value := relativeKeyAndValue(nowMs - *timestampMs)
	var relative string
	if value == nil {
		rendered, err := schema.T(key)
		if err != nil {
			return formatAbsolute(loc, *timestampMs)
		}
		relative = rendered
	} else {
		rendered, err := schema.T(key, map[string]any{"value": *value})
		if err != nil {
			return formatAbsolute(loc, *timestampMs)
		}
		relative = rendered
	}
	return formatAbsolute(loc, *timestampMs) + " (" + relative + ")"
}

⚠️ **注意 `-07:00` 布局**：Go 的 `-07:00` 与 Python `%z` 拼冒号后的结果一致（`+0800` → `+08:00`，`+0000` → `+00:00`）。**不要**用 `Z07:00`，那会在 UTC 下输出 `Z` 而非 `+00:00`。

package agent_teams

import (
	"html"
	"strings"
)

// 本文件提供成员入站消息与框架事件的 XML 渲染。
//
// 团队成员的 harness 输入混了两类东西：另一个成员或用户发来的**原始消息**，
// 以及运行时包裹在它周围的**框架元数据与指令**（message id、时间、回复提示、
// 事件通知）。早期的 i18n 模板把两者拼成一整条扁平字符串，模型分不清「谁说的」
// 与「框架加的」。本文件改用语义化 XML，给模型一条清晰边界：
//
//   - <team-inbound> 原样包裹**原始消息**，sender / message_id / type / time 作为属性。
//   - <team-clock> 每次准入输入附带一个当前时间锚点。
//   - <team-event> 包裹**框架事件**，kind 属性标明类型。
//   - <team-context> 包裹**团队既成事实**（自己的身份、团队元数据）。
//   - <team-policy> 包裹外部 CLI 成员 system prompt 里的**团队准则**。
//   - <team-note> 携带框架附加的提示或约束，**嵌套在所修饰块的内部**作为最后一个
//     子元素：某条提示在说哪件事，是树上的事实，不需要模型按前后位置去猜。
//
// 这些都是纯结构函数：调用方传入动态数据与已本地化的文本片段（由 schema.T 解析），
// 拿回一段 XML 字符串。文案不进这一层，双语契约留在 i18n（唯一真相），
// 标签结构留在这里。type / kind / for 的属性值是稳定的英文契约 token，
// **永不本地化**，因为 inbound-tags 系统提示词段按名字引用它们。
//
// Python: openjiuwen/agent_teams/inbound_render.py (417 行)

// ──────────────────────────── 结构体 ────────────────────────────

// InboundOptions RenderInbound 的可选参数。
//
// Python 侧全用 keyword-only 参数，Go 无此能力，故聚成结构体。
type InboundOptions struct {
	// Sender 发送者的 member_name
	Sender string
	// MessageID 消息 id，让模型能引用 / 标记已读
	MessageID string
	// MsgType 稳定契约 token：InboundTypeDirect 或 InboundTypeBroadcast（不本地化）
	MsgType string
	// TimeInfo 绝对发送时间（由 FormatAbsoluteTime 渲染）。不带相对差：
	// 本块写进历史后会被再次读取。
	TimeInfo string
	// ForController 为真时添加 for="controller"（HITT 头像视为给其控制者的通知）
	ForController bool
	// NoteKind 可选 <team-note> 的 kind（如 "reply-hint" / "hitt-silence"）
	NoteKind string
	// NoteText 可选 <team-note> 正文；NoteKind 与 NoteText 都非空才渲染
	NoteText string
}

// EventOptions RenderEvent 的可选参数。
type EventOptions struct {
	// TaskID 可选任务 id。用指针区分「未设置」与「显式空串」——
	// 上游是 `if task_id is not None`，空串仍会渲染 task_id=""。
	TaskID *string
	// ForController 为真时添加 for="controller"（HITT）
	ForController bool
	// NoteKind 可选 <team-note> kind，嵌套在事件内
	NoteKind string
	// NoteText 可选 <team-note> 正文
	NoteText string
}

// ──────────────────────────── 枚举 ────────────────────────────

// ──────────────────────────── 常量 ────────────────────────────

const (
	// InboundTypeDirect <team-inbound> type 属性的直接消息契约 token
	InboundTypeDirect = "direct"
	// InboundTypeBroadcast <team-inbound> type 属性的广播消息契约 token
	InboundTypeBroadcast = "broadcast"
	// ControllerSender 人类成员自己控制者的 from 取值。
	// 不是成员名：控制者没有名册条目、没有 agent 进程，头像永远无法通过
	// send_message 找到它。
	ControllerSender = "controller"
)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// snapshotEventKinds 每次出现都是某块状态的**完整快照**的事件 kind 集合，
	// 因此最新一次就足以表达此前所有的信息。
	//
	// 只有具备该性质的事件才能进这里。增量（roster-change）或带主体的
	// 一次性事件（stale-claim，携带 task_id）一旦丢掉较早的一条就会丢信息。
	//
	// 偏差 D-3：上游是 frozenset，Go 用 map 无迭代顺序；当前只有 1 个元素
	// 所以无歧义。若增到 ≥2 个元素，须改为有序结构以保证与上游一致。
	snapshotEventKinds = map[string]struct{}{
		"task-board": {},
	}

	// eventOpenTagFormat 某 kind 的 <team-event> 开标签。RenderEvent 把 kind
	// 放在最前，因此可用它从首字符识别一个已渲染的块。
	eventOpenTagFormat = `<team-event kind="%s"`
)

// ──────────────────────────── 导出函数 ────────────────────────────

// RenderInbound 把一条成员/用户入站消息渲染为 <team-inbound> XML。
//
// content 原样渲染在元素内（XML 转义，绝不改写）。
// 有 note 时 <team-note> 作为最后一个子元素嵌套在内部。
//
// Python: render_inbound (inbound_render.py L115-157)
func RenderInbound(content string, opts InboundOptions) string {
	attrs := []string{
		`from="` + escAttr(opts.Sender) + `"`,
		`message_id="` + escAttr(opts.MessageID) + `"`,
		`type="` + escAttr(opts.MsgType) + `"`,

// RenderTeamContextWithIdentity 渲染带嵌套 <identity> 子块的 <team-context>。
//
// RenderTeamContext 的 fork 感知变体：成员自己的身份被包进 <identity> 元素
// （可选 <identity-conversion> 子元素标注 fork 来源），让模型能把**当前**身份块
// 与通过 fork 继承来的更早身份块区分开。仅在 fork 开启时使用——非 fork 团队的
// 纯 RenderTeamContext 保持字节一致。
//
// identityConversion / infoBody 为 nil 表示不存在（上游用 truthiness 判定，
// 空串等价于 None）。
//
// Python: render_team_context_with_identity (inbound_render.py L266-300)
func RenderTeamContextWithIdentity(identityBody string, identityConversion, infoBody *string) string {
	identityParts := []string{escText(identityBody)}
	if identityConversion != nil && *identityConversion != "" {
		identityParts = append(identityParts, renderBlock("identity-conversion", nil, *identityConversion, ""))
	}
	identity := "<identity>\n" + strings.Join(identityParts, "\n") + "\n</identity>"
	inner := []string{identity}
	if infoBody != nil && *infoBody != "" {
		inner = append(inner, escText(*infoBody))
	}
	return "<team-context>\n" + strings.Join(inner, "\n") + "\n</team-context>"
}

// DropSupersededSnapshots 返回去掉「被更晚快照取代」的输入后的 parts。
//
// 成员忙碌时回来后面对期间积压的一切。任务看板可能在里面出现好几次，每次都是
// 相隔数秒的完整survey——但只有最后一次描述成员即将据以行动的看板，更早的都是
// 噪音，离变成永久历史只差一步。
//
// 条目整条丢弃，绝不编辑：每条是一个输入，且只丢 snapshotEventKinds 里的 kind。
// 其余一切（包括名册增量、单任务催促）都携带别的条目不会重复的信息。
//
// Python: drop_superseded_snapshots (inbound_render.py L372-399)
func DropSupersededSnapshots(parts []string) []string {
	newest := make(map[string]int, len(parts))
	for i, part := range parts {
		if kind := snapshotKindOf(part); kind != "" {
			newest[kind] = i
		}
	}
	out := make([]string, 0, len(parts))
	for i, part := range parts {
		// 非快照条目 kind 为空，map 查不到 → 取自身下标 → 自然保留，无需单独分支。
		keep, ok := newest[snapshotKindOf(part)]
		if !ok || keep == i {
			out = append(out, part)
		}
	}
	return out
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// escText 转义元素正文：只转 & < >，不转引号。
// 与 Python html.escape(text, quote=False) 逐字节一致。
//
// Python: _esc_text (inbound_render.py L78-80)
func escText(text string) string {
	r := strings.NewReplacer("&", "&amp;", "<", "&lt;", ">", "&gt;")
	return r.Replace(text)
}

// escAttr 转义属性值（转义引号）。
//
// Python: _esc_attr (inbound_render.py L83-85) —— html.escape(quote=True)，
// 单引号输出 &#x27;、双引号输出 &quot;。
// 本实现用 html.EscapeString：单引号输出 &#39;、双引号输出 &#34;。
// XML 解析后等值，属已知偏差 D-1（spec §7）。
func escAttr(value string) string {
	return html.EscapeString(value)
}

// renderNote 渲染嵌套的 <team-note> 子元素，缺失时返回空串。
//
// 自带尾换行，让外层块不需要对「有没有 note」做分支。
//
// Python: _render_note (inbound_render.py L88-96)
func renderNote(noteKind, noteText string) string {
	if noteKind == "" || noteText == "" {
		return ""
	}
	return `<team-note kind="` + escAttr(noteKind) + `">` + "\n" + escText(noteText) + "\n</team-note>\n"
}

// renderBlock 把一个块元素渲染在其 body 与可选嵌套 note 之外。
//
// tag 元素名；attrs 已转义的 name="value" 片段（按契约顺序）；body 元素正文
// （在此转义）；note 已渲染的 <team-note> 子元素或空串。
//
// Python: _render_block (inbound_render.py L99-112)
func renderBlock(tag string, attrs []string, body, note string) string {
	openTag := tag
	if len(attrs) > 0 {
		openTag = tag + " " + strings.Join(attrs, " ")
	}
	return "<" + openTag + ">\n" + escText(body) + "\n" + note + "</" + tag + ">"
}

// snapshotKindOf 返回 text 所属的快照 kind，不是则返回空串。
//
// 只有「整条输入就是一个快照事件」才符合条件：这些事件每次 deliver_input
// 投递一个，所以整条条目就是这个块，丢掉它不带走走任何别的东西。
// <team-note> 不算「别的东西」——它嵌套在块内部、只修饰那个块，因此随它所挂的
// 快照一起过期、一起被取代。
//
// Python: snapshot_kind_of (inbound_render.py L350-369)
func snapshotKindOf(text string) string {
	stripped := strings.TrimSpace(text)
	if !strings.HasSuffix(stripped, "</team-event>") {
		return ""
	}
	for kind := range snapshotEventKinds {
		if strings.HasPrefix(stripped, sprintfEventOpenTag(kind)) {
			return kind
		}
	}
	return ""
}

// sprintfEventOpenTag 用 eventOpenTagFormat 生成某 kind 的开标签。
// 单独抽出来避免为非格式化场景引入 fmt 的依赖面。
func sprintfEventOpenTag(kind string) string {
	// 等价 fmt.Sprintf(eventOpenTagFormat, kind)；kind 只做字符串拼接，
	// 不含需要格式化的字符。
	return strings.Replace(eventOpenTagFormat, "%s", kind, 1)
}

// DiffRoster 以 member_name 为键 diff 两份同伴名册。
//
// 用 orderedmap 对齐 Python dict 的两条语义：迭代保首个插入位置、重复键后写覆盖。
// 若用 map[string]RosterMember 迭代顺序会被随机化，字节级对拍必失败。
//
// Python: diff_roster (messages.py L164-189)
func DiffRoster(old, new []RosterMember) RosterDelta {
	oldByName := orderedmap.New[string, RosterMember]()
	for _, m := range old {
		oldByName.Set(m.MemberName, m)
	}
	newByName := orderedmap.New[string, RosterMember]()
	for _, m := range new {
		newByName.Set(m.MemberName, m)
	}

	var joined, left, changed []RosterMember
	for pair := newByName.Oldest(); pair != nil; pair = pair.Next() {
		if _, ok := oldByName.Get(pair.Key); !ok {
			joined = append(joined, pair.Value)
		}
	}
	for pair := oldByName.Oldest(); pair != nil; pair = pair.Next() {
		if _, ok := newByName.Get(pair.Key); !ok {
			left = append(left, pair.Value)
		}
	}
	for pair := newByName.Oldest(); pair != nil; pair = pair.Next() {
		prev, ok := oldByName.Get(pair.Key)
		if !ok {
			continue
		}
		if memberFieldChanged(prev, pair.Value) {
			changed = append(changed, pair.Value)
		}
	}
	return RosterDelta{Joined: joined, Left: left, Changed: changed}
}

// FormatMemberLine 渲染一行名册。
//
// markHumans 为真时给人类成员（human_agent 或 passive_human）追加 [human]。
// prefix 是放在字段前的方括号标记，名册增量正文用它说明该成员是加入/退出/更新。
//
// Python: format_member_line (messages.py L197-237)
func FormatMemberLine(member RosterMember, markHumans bool, prefix string) string {
	displayName := member.DisplayName
	if displayName == "" {
		displayName = "unknown"
	}
	head := "- "
	if prefix != "" {
		head = "- [" + prefix + "] "
	}
	line := head + "member_name=" + member.MemberName + " display_name=" + displayName
	if member.ModelGroup != "" {
		line += " model_group=" + member.ModelGroup
	}
	if member.ModelsUsed != "" {
		line += " models_used=" + member.ModelsUsed
	}
	// 两种人类形态都是真人——同伴必须知道要相应放慢沟通节奏
	// （且工具可能代表他们行动）。
	if markHumans &&
		(member.Role == atschema.TeamRoleHumanAgent || member.Role == atschema.TeamRolePassiveHuman) {
		line += " [human]"
	}
	if member.Desc != "" {
		line += " :: " + member.Desc
	}
	return line
}

// BuildIdentityText 渲染成员自己的身份正文，无可用字段时返回 nil。
//
// 携带这个成员特有的一切：它被两个名字知晓，以及它的私有工作约定（公开 desc
// 的成员私有对应物，从不进入任何同伴的名册）。这些都在 spawn 时固定、之后不变，
// 所以本正文只投递一次。
//
// displayName 在这里，是因为同伴的名册用**两个**名字列出成员：没有自己的标签，
// 成员就分不清名册里哪一行是自己，也无法像团队其他人那样称呼自己。
//
// memberWorkspacePath 是成员自己的产物目录，逐成员且恒定，因此与名字同处一个正文。
//
// fork 处理（仅 enable_fork=True 的团队）：forkCapable 为真时前置一段恒定的能力
// 声明，让成员知道自己的身份可能被 fork 转换通知重新指派。转换通知本身由
// BuildIdentityConversion 渲染，作为嵌套 <identity-conversion> 元素投递。
//
// Python: build_identity_text (messages.py L247-315)
func BuildIdentityText(memberName, displayName, memberWorkspacePath, memberPrompt, language string, forkCapable bool) *string {
	privatePrompt := strings.TrimSpace(memberPrompt)
	label := strings.TrimSpace(displayName)
	workspace := strings.TrimSpace(memberWorkspacePath)
	if memberName == "" && label == "" && workspace == "" && privatePrompt == "" {
		return nil
	}
	lbl := messageLabelsFor(language)
	var lines []string
	if forkCapable {
		lines = append(lines, lbl["identity_capability"], "")
	}
	lines = append(lines, lbl["identity_heading"], "")
	if memberName != "" {
		lines = append(lines, lbl["member_name_line"]+": "+memberName)
	}
	if label != "" {
		lines = append(lines, lbl["display_name_line"]+": "+label)
	}
	if workspace != "" {
		purpose := parenthesized(lbl["member_workspace_purpose"], language)
		lines = append(lines, lbl["member_workspace_line"]+": `"+workspace+"`"+purpose)
	}
	if privatePrompt != "" {
		lines = append(lines, "", lbl["private_prompt_heading"], "", privatePrompt)
	}
	out := strings.Join(lines, "\n") + "\n"
	return &out
}

// BuildIdentityPromptDelta 只渲染身份正文里的私有工作约定小节。
//
// 恒定的身份字段由 BuildIdentityText 一次性投递、从不变化，所以每次编辑后
// 重新公告它们是噪音。但私有工作约定（member_prompt.md）**可以**在会话中途
// 手工演进，而它在模型侧的唯一通道就是身份正文——那是一次性的。
// 本 delta 只渲染该小节，让身份正文能在不重述恒定字段的前提下重新公告演进后的约定。
//
// 空白 memberPrompt 返回 nil（无可重新公告的内容）。
//
// Python: build_identity_prompt_delta (messages.py L318-347)
func BuildIdentityPromptDelta(memberPrompt, language string) *string {
	privatePrompt := strings.TrimSpace(memberPrompt)
	if privatePrompt == "" {
		return nil
	}
	lbl := messageLabelsFor(language)
	out := "\n" + lbl["private_prompt_heading"] + "\n\n" + privatePrompt + "\n"
	return &out
}

// BuildIdentityConversion 渲染 fork 转换通知。
//

// BuildTeamInfoText 渲染团队元数据正文，无可用字段时返回 nil。

// teamInfo 取 team_name / display_name / desc 三键（TeamBackend.GetTeamInfo 的形状）。

//
// Python: build_team_info_text (messages.py L379-428)
func BuildTeamInfoText(teamInfo map[string]string, teamWorkspacePath, teamOutputsDir, language string) *string {
	lbl := messageLabelsFor(language)
	teamName := teamInfo["team_name"]
	displayName := teamInfo["display_name"]
	desc := teamInfo["desc"]
	path := strings.TrimSpace(teamWorkspacePath)
	outputs := strings.TrimSpace(teamOutputsDir)
	if teamName == "" && displayName == "" && desc == "" && path == "" && outputs == "" {
		return nil
	}

	lines := []string{lbl["info_heading"], ""}
	if teamName != "" {
		lines = append(lines, "- "+lbl["team_name_label"]+": "+teamName)
	}
	if displayName != "" {
		lines = append(lines, "- "+lbl["display_name_label"]+": "+displayName)
	}
	if desc != "" {
		lines = append(lines, "- "+lbl["team_desc"]+": "+desc)
	}
	if path != "" {
		lines = append(lines, "- "+lbl["team_workspace"]+": `"+path+"`")
		lines = append(lines, "  - "+lbl["team_workspace_purpose"])
		lines = append(lines, "  - "+lbl["team_workspace_config_note"])
	}
	if outputs != "" {
		lines = append(lines, "- "+lbl["team_outputs_dir"]+": `"+outputs+"`")
		lines = append(lines, "  - "+lbl["team_outputs_dir_purpose"])
	}
	out := strings.Join(lines, "\n") + "\n"
	return &out

//
// 在成员第一次有同伴需要被告知时投递一次；此后每次变化都用增量公告

// markHumans 为真时给人类成员标 [human]。调用方按「查看者角色 +

//
// Python: build_roster_snapshot_text (messages.py L431-458)
func BuildRosterSnapshotText(members []RosterMember, markHumans bool, language string) *string {
	if len(members) == 0 {
		return nil
	}
	lbl := messageLabelsFor(language)
	rows := make([]string, 0, len(members))
	for _, m := range members {
		rows = append(rows, FormatMemberLine(m, markHumans, ""))
	}
	out := lbl["members_heading"] + "\n\n" + strings.Join(rows, "\n") + "\n"
	return &out
}

// BuildRosterDeltaText 渲染只列出变动项的名册变更正文，无变更时返回 nil。
//
// Python: build_roster_delta_text (messages.py L461-487)
func BuildRosterDeltaText(delta RosterDelta, markHumans bool, language string) *string {
	if delta.IsEmpty() {
		return nil
	}
	lbl := messageLabelsFor(language)
	rows := make([]string, 0, len(delta.Joined)+len(delta.Left)+len(delta.Changed))
	for _, m := range delta.Joined {
		rows = append(rows, FormatMemberLine(m, markHumans, lbl["roster_joined"]))
	}
	for _, m := range delta.Left {
		rows = append(rows, FormatMemberLine(m, markHumans, lbl["roster_left"]))
	}
	for _, m := range delta.Changed {
		rows = append(rows, FormatMemberLine(m, markHumans, lbl["roster_updated"]))
	}
	out := lbl["roster_change_heading"] + "\n\n" + strings.Join(rows, "\n") + "\n"
	return &out
}

// messageLabelsFor 返回某语言的标签表，缺失时回退 cn。
//
// Python: labels_for (messages.py L133-135)
func messageLabelsFor(language string) map[string]string {
	if tbl, ok := messageLabels[language]; ok {
		return tbl
	}
	return messageLabels["cn"]
}

// parenthesized 用该语言实际使用的括号包裹一个尾随说明。

// Python: _parenthesized (messages.py L240-244) —— cn 用全角，en 用半角且带前导空格。
func parenthesized(text, language string) string {
	if language == "cn" {
		return "（" + text + "）"
	}
	return " (" + text + ")"
}

func memberFieldChanged(prev, cur RosterMember) bool {
	if cur.DisplayName != prev.DisplayName {
		return true
	}

		return true
	}
	if cur.Role != prev.Role {
		return true
	}

}

package external

import (
	"fmt"

	agentteams "github.com/uapclaw/uapclaw-go/internal/agent_teams"
	atschema "github.com/uapclaw/uapclaw-go/internal/agent_teams/schema"
)

// 本文件是进程外成员（外部 CLI 驱动）的入站消息渲染。
//
// 与进程内 MessageHandler 的渲染语义一致：原始消息原文进 <team-inbound>
// （sender / message_id / type / 绝对发送时间作为属性），框架提示进独立的
// <team-note>。human-agent 头像得到 for="controller" 加 hitt-silence note；
// 其余成员得到 reply-hint note。
//
// Python: openjiuwen/agent_teams/external/format.py (213 行) 的 render_message /
// render_messages。render_task_line / render_task_board 待 25.4 回填。

// ──────────────────────────── 结构体 ────────────────────────────

// MessageLike 渲染用的团队消息行结构视图。
//
// 用接口而非具体类型，对齐上游的 Protocol 结构化视图：调用方传数据库行或
// 测试假实现都可。
//
// Python: _MessageLike (format.py L31-38)
type MessageLike interface {
	// MessageID 消息 id
	MessageID() string
	// FromMemberName 发送者成员名
	FromMemberName() string
	// Content 消息正文
	Content() string
	// Broadcast 是否广播
	Broadcast() bool
	// TimestampMs 毫秒 UTC epoch 发送时间
	TimestampMs() int64
}

// RenderMessageOptions RenderMessage 的可选参数。
type RenderMessageOptions struct {
	// IsHumanAgent 接收方是否为 human-agent 头像
	IsHumanAgent bool
	// Body 框架模板消息的投递时正文，由调用方渲染。
	// 传值同时会**去掉** reply-hint：框架指令用工具调用答复，而不是回复。
	// nil 表示普通消息——渲染它自己的 content 并保留提示。
	Body *string
}

// RenderMessagesOptions RenderMessages 的可选参数。
type RenderMessagesOptions struct {
	// IsHumanAgent 接收方是否为 human-agent 头像
	IsHumanAgent bool
	// Bodies 批次中框架模板消息的投递时正文，按 message_id 索引。
	// 普通消息不在映射里，渲染自己的 content。
	Bodies map[string]string
}

// ──────────────────────────── 枚举 ────────────────────────────

// ──────────────────────────── 常量 ────────────────────────────

// ──────────────────────────── 全局变量 ────────────────────────────

// ──────────────────────────── 导出函数 ────────────────────────────

// RenderMessage 把一条入站消息渲染为 <team-inbound> XML。
//
// 对应进程内 MessageHandler 的渲染：消息正文原文进 <team-inbound>，
// 框架提示进独立的 <team-note>。
//
// Python: render_message (format.py L52-108)
func RenderMessage(message MessageLike, opts RenderMessageOptions) (string, error) {
	msgType := agentteams.InboundTypeDirect
	if message.Broadcast() {
		msgType = agentteams.InboundTypeBroadcast
	}
	timeInfo := agentteams.FormatAbsoluteTime(message.TimestampMs())

	if opts.IsHumanAgent {
		silence, err := atschema.T("hitt.silence_note")
		if err != nil {
			return "", fmt.Errorf("解析 hitt.silence_note 失败: %w", err)
		}
		content := message.Content()
		if opts.Body != nil {
			content = *opts.Body
		}
		return agentteams.RenderInbound(content, agentteams.InboundOptions{
			Sender:        message.FromMemberName(),
			MessageID:     message.MessageID(),
			MsgType:       msgType,
			TimeInfo:      timeInfo,
			ForController: true,
			NoteKind:      "hitt-silence",
			NoteText:      silence,
		}), nil
	}

	if opts.Body != nil {
		return agentteams.RenderInbound(*opts.Body, agentteams.InboundOptions{
			Sender:    message.FromMemberName(),
			MessageID: message.MessageID(),
			MsgType:   msgType,
			TimeInfo:  timeInfo,
		}), nil
	}

	hint, err := atschema.ReplyHintFor(message.FromMemberName())
	if err != nil {
		return "", fmt.Errorf("解析 reply_hint 失败: %w", err)
	}
	return agentteams.RenderInbound(message.Content(), agentteams.InboundOptions{
		Sender:    message.FromMemberName(),
		MessageID: message.MessageID(),
		MsgType:   msgType,
		TimeInfo:  timeInfo,
		NoteKind:  "reply-hint",
		NoteText:  hint,
	}), nil
}

// RenderMessages 按给定顺序（最旧在前）渲染一批入站消息。
//
// bodies 是批次中框架模板消息的投递时正文，按 message_id 索引。
// 普通消息不在映射里，渲染自己的 content。
//
// Python: render_messages (format.py L111-128)
func RenderMessages(messages []MessageLike, opts RenderMessagesOptions) (string, error) {
	rendered := make([]string, 0, len(messages))
	for _, m := range messages {
		one := RenderMessageOptions{IsHumanAgent: opts.IsHumanAgent}
		if opts.Bodies != nil {
			if body, ok := opts.Bodies[m.MessageID()]; ok {
				b := body
				one.Body = &b
			}
		}
		text, err := RenderMessage(m, one)
		if err != nil {
			return "", err
		}
		rendered = append(rendered, text)
	}
	return stringsJoin(rendered, "\n\n"), nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// stringsJoin 避免为一次 Join 引入 strings 导入面（本包其余部分不需要 strings）。
func stringsJoin(parts []string, sep string) string {
	if len(parts) == 0 {
		return ""
	}
	out := parts[0]
	for _, p := range parts[1:] {
		out += sep + p
	}
	return out
}
