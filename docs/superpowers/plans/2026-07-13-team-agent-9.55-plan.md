```go
func (a *TeamAgent) onTeammateCreated(ctx context.Context, teammateID string) error {
	runtimeCtx, err := a.spawnManager.BuildContextFromDB(ctx, teammateID)

	return a.SpawnTeammate(ctx, runtimeCtx, prompt, a.SessionID(ctx), nil)
```

```go
// registerTeamCompletionCallbacks 注册团队完成回调到 TeamCompletionHandler。
// Python: TeamAgent._register_team_completion_callbacks()
func (a *TeamAgent) registerTeamCompletionCallbacks() {
	harness := a.configurator.Harness()
	dispatcher := a.coordination.Dispatcher()
	if harness == nil || dispatcher == nil { return }
	if dispatcher.TeamCompletion == nil { return }

	evolutionRailType := reflect.TypeOf(&evolution.TeamSkillEvolutionRail{})
	createRailType := reflect.TypeOf(&evolution.TeamSkillCreateRail{})

	for _, railType := range []reflect.Type{evolutionRailType, createRailType} {
		for _, rail := range harness.FindRails(railType) {
			notify := a.extractNotifyTeamCompleted(rail)
			if notify != nil {
				dispatcher.TeamCompletion.RegisterCompletionCallback(notify)
			}
		}
	}
}

// extractNotifyTeamCompleted 从 Rail 中提取 notify_team_completed 回调。
func (a *TeamAgent) extractNotifyTeamCompleted(rail any) func(ctx context.Context) error {
	switch r := rail.(type) {
	case *evolution.TeamSkillEvolutionRail:
		return func(ctx context.Context) error {
			_, err := r.NotifyTeamCompleted(ctx)
			return err
		}
	case *evolution.TeamSkillCreateRail:
		return func(ctx context.Context) error {
			r.NotifyTeamCompleted(nil)
			return nil
		}
	default:
		return nil
	}
}
```

```go
func (m *SpawnManager) BuildContextFromDB(ctx context.Context, memberName string) (atschema.TeamRuntimeContext, error) {

		TeamSpec:      runtimeCtx.TeamSpec,
		MessagerConfig: m.configurator.BuildMemberMessagerConfig(teammate.MemberName),
```

```go
// Start 启动协调子系统，编排完整初始化链。
// 对齐 Python: CoordinationKernel.start(session)
func (k *CoordinationKernel) Start(ctx context.Context, session any) {
	if k.eventBus == nil { return }
	memberName := k.host.MemberName()
	if memberName == "" { memberName = "?" }
	logger.Info(logComponent).Str("member_name", memberName).Msg("coordination starting")

	// 步骤 1: setMemberId
	k.host.SetMemberID(memberName)

	// 步骤 2: DB 初始化
	backendAccessor := k.host.TeamBackendAccessor()
	if backendAccessor != nil && backendAccessor.DB() != nil {
		_ = backendAccessor.DB().Initialize(ctx)
	}

	// 步骤 3: Session bind/release
	sessMgr := k.host.SessionManager()
	if session != nil && sessMgr != nil {
		_, _ = sessMgr.BindSession(ctx, session)
	} else if sessMgr != nil {
		_ = sessMgr.ReleaseSession(ctx)
	}

	// 步骤 4: [Leader] 检查全 SHUTDOWN → clean_team 或 recover
	if k.host.Role() == schema.TeamRoleLeader && backendAccessor != nil {
		existing, _ := backendAccessor.DB().Team().GetTeam(ctx, backendAccessor.TeamName())
		if existing != nil {
			members, _ := backendAccessor.ListMembers(ctx)
			if members != nil && allMembersShutdown(members) {
				logger.Warn(logComponent).Str("team_name", backendAccessor.TeamName()).
					Msg("team found with all teammates in SHUTDOWN — finalizing prior incomplete cleanup")
				_ = backendAccessor.CleanTeam(ctx)
			} else {
				_, _ = k.host.RecoverTeam(ctx)
			}
		}
	}

	// 步骤 5: Workspace 初始化
	wsMgr := k.host.WorkspaceManager()
	infra := k.host.(types.InfraAccessor)
	if wsMgr != nil && infra != nil {
		specAccessor := k.host.(types.BlueprintAccessor)
		var remoteURL string
		if specAccessor != nil && specAccessor.Spec() != nil {
			remoteURL = extractWorkspaceRemoteURL(specAccessor.Spec())
		}
		_ = wsMgr.Initialize(ctx, remoteURL)
		infra.SetWorkspaceInitialized()
	}

	// 步骤 6: Memory toolkit init
	memMgr := k.host.MemoryManager()
	harnessAccessor := k.host.HarnessAccessor()
	if memMgr != nil && harnessAccessor != nil {
		success, _ := memMgr.InitToolkit(ctx)
		if success {
			harnessAccessor.RegisterMemberTools(memMgr)
			if memMgr.ExtractionModel() == nil {
				memMgr.SetExtractionModel(harnessAccessor.Model())
			}
			_ = harnessAccessor.InjectMemberMemory(ctx, memMgr)
		}
	}

	// 步骤 7: 更新状态
	_ = k.host.UpdateStatus(ctx, schema.MemberStatusReady)

	// 步骤 8: EventBus 启动
	if k.dispatcher == nil {
		logger.Error(logComponent).Msg("CoordinationKernel.start() requires setup()")
		return
	}
	if !k.eventBus.IsRunning() {
		k.eventBus.Start(ctx, func(ctx context.Context, event types.CoordinationEvent) {
			k.dispatcher.Dispatch(ctx, event)
		})
	}

	// 步骤 9: Subscribe transport
	_ = k.host.SubscribeTransport(ctx)

	// 步骤 10: TeamCompletion rearmed
	if k.dispatcher != nil && k.dispatcher.TeamCompletion != nil {
		k.dispatcher.TeamCompletion.Rearm()
	}

	k.lifecycleState = kernelStateRunning
}
```

```go
// allMembersShutdown 检查所有成员是否都是 SHUTDOWN 状态。
func allMembersShutdown(members []any) bool {
	for _, m := range members {
		// 通过类型断言检查 status 字段
		if statusExtractor, ok := m.(interface{ Status() string }); ok {
			if statusExtractor.Status() != string(schema.MemberStatusShutdown) {
				return false
			}
		}
	}
	return true
}
```
