package harness_protocol

// 本文件对应 Python openjiuwen/harness_protocol/hooks.py（163 行）：
// harness 执行边界上宿主策略扩展的控制面契约。
// 本文件只定义契约与构造期校验，不含任何调用点 —— 上游 harness_providers/
// 内亦无 hooks 调用（HostCapability.HOOKS 全仓 0 命中），首个消费方预计为
// 15.5 provider 基座或 15.7 claudecode 的 permission prompt 场景。
// 注意：events 的 HookObservedEvent 只是钩子调用的观测镜像，不得用于决策。

// ──────────────────────────── 结构体 ────────────────────────────

// BeforePromptContext 输入到达外部 agent 之前提供的上下文。
// Python: hooks.py:30-38 BeforePromptContext
type BeforePromptContext struct {
	// AgentName agent 名
	AgentName string
	// ProviderSessionID provider 会话（nil = 未提供）
	ProviderSessionID *string
	// TurnID 所属轮次（nil = 未提供）
	TurnID *string
	// Prompt 待送入模型的输入
	Prompt string
}

// BeforePromptResult 钩子返回的提示词改写或停轮决策。
// Python: hooks.py:40-48 BeforePromptResult
type BeforePromptResult struct {
	// Prompt 改写后的提示词
	Prompt string
	// ContinueExecution 是否继续执行（Python 默认 True）
	ContinueExecution bool
	// Reason 停轮原因（nil = 未提供）
	Reason *string
	// AdditionalContext 追加上下文（nil = 未提供）
	AdditionalContext *string
}

// BeforePromptResultOptions BeforePromptResult 的可选字段。
// Python: hooks.py:45-47 dataclass 默认值
type BeforePromptResultOptions struct {
	// Reason 停轮原因
	Reason *string
	// AdditionalContext 追加上下文
	AdditionalContext *string
}

// ──────────────────────────── 枚举 ────────────────────────────

// ToolDecisionKind 工具执行前返回的决策种类。
// Python: hooks.py:20-28 ToolDecisionKind（5 项）
type ToolDecisionKind string

const (
	// ToolDecisionKindAllow 允许执行
	ToolDecisionKindAllow ToolDecisionKind = "allow"
	// ToolDecisionKindDeny 拒绝执行
	ToolDecisionKindDeny ToolDecisionKind = "deny"
	// ToolDecisionKindRewrite 改写参数后允许执行
	ToolDecisionKindRewrite ToolDecisionKind = "rewrite"
	// ToolDecisionKindAsk 转交宿主交互审批
	ToolDecisionKindAsk ToolDecisionKind = "ask"
	// ToolDecisionKindProviderPolicy 交给 provider 原生权限策略
	ToolDecisionKindProviderPolicy ToolDecisionKind = "provider_policy"
)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewBeforePromptContext 创建提示词前置上下文（无校验）。
// Python: hooks.py:30-38 无 __post_init__
func NewBeforePromptContext(
	agentName string,
	providerSessionID *string,
	turnID *string,
	prompt string,
) BeforePromptContext {
	return BeforePromptContext{
		AgentName:         agentName,
		ProviderSessionID: providerSessionID,
		TurnID:            turnID,
		Prompt:            prompt,
	}
}

// NewBeforePromptResult 创建提示词前置结果。
// Python: hooks.py:40-48 无 __post_init__；continue_execution 默认 True，
// Go Options 零值 bool 为 false，故在构造器内填默认值 true。
func NewBeforePromptResult(prompt string, opts BeforePromptResultOptions) (BeforePromptResult, error) {
	return BeforePromptResult{
		Prompt:            prompt,
		ContinueExecution: true, // Python 默认值
		Reason:            opts.Reason,
		AdditionalContext: opts.AdditionalContext,
	}, nil
}

// NewBeforeToolContext 创建工具前置上下文。
// Python: hooks.py:61-62 __post_init__（冻结 arguments）
func NewBeforeToolContext(
	agentName string,
	providerSessionID *string,
	turnID *string,
	callID string,
	toolName string,
	arguments JsonObject,
) (BeforeToolContext, error) {
	// 步骤 1：冻结 arguments（Python freeze_json_object(None) 抛 TypeError，
	// 故 nil 在此被拒绝 —— 与上游一致）
	frozenArguments, err := FreezeJsonObject(arguments)
	if err != nil {
		return BeforeToolContext{}, err
	}
	return BeforeToolContext{
		AgentName:         agentName,
		ProviderSessionID: providerSessionID,
		TurnID:            turnID,
		CallID:            callID,
		ToolName:          toolName,
		Arguments:         frozenArguments,
	}, nil
}

// NewToolDecision 创建工具决策。
// Python: hooks.py:74-80 ToolDecision.__post_init__（3 步，顺序不可调）
// Q9 决策 A：校验跨越「形参 decision」与「Options.UpdatedArguments」边界，
// 仍按 Python __post_init__ 顺序内联。
func NewToolDecision(decision ToolDecisionKind, opts ToolDecisionOptions) (ToolDecision, error) {
	updated := opts.UpdatedArguments
	// 步骤 1：updated_arguments 非 nil 则冻结
	if updated != nil {
		frozen, err := FreezeJsonObject(updated)
		if err != nil {
			return ToolDecision{}, err
		}
		updated = frozen
	}
	// 步骤 2：REWRITE 必须携带 updated_arguments
	if decision == ToolDecisionKindRewrite && updated == nil {
		return ToolDecision{}, &HarnessProtocolError{Msg: "rewrite tool decision requires updated_arguments"}
	}
	// 步骤 3：非 REWRITE 不得携带 updated_arguments
	if decision != ToolDecisionKindRewrite && updated != nil {
		return ToolDecision{}, &HarnessProtocolError{Msg: "updated_arguments is only valid for a rewrite tool decision"}
	}
	return ToolDecision{
		Decision:          decision,
		Reason:            opts.Reason,
		UpdatedArguments:  updated,
		AdditionalContext: opts.AdditionalContext,
	}, nil
}

// NewAfterToolContext 创建工具后置上下文。
// Python: hooks.py:96-98 __post_init__（冻结 arguments 与 result）

// validateRequestCommon 五个请求类型共用的构造期校验。
// Python: interactions.py:22-31 _validate_request_common（3 步，顺序不可调）
//
// Q7 决策 A 的核心：provider_session_id / turn_id 用 *string，
// nil = 未提供（Python None，放行），"" = 显式空串（拒绝）。
func validateRequestCommon(
	requestID string,
	providerSessionID *string,
	turnID *string,
	deadlineAt *float64,
) error {
	// 步骤 1：request_id 非空
	if requestID == "" {
		return &HarnessProtocolError{Msg: "interaction request_id must not be empty"}
	}
	// 步骤 2：provider_session_id / turn_id 若给了值则不得为空串
	if providerSessionID != nil && *providerSessionID == "" {
		return &HarnessProtocolError{Msg: "interaction provider_session_id must not be empty"}
	}
	if turnID != nil && *turnID == "" {
		return &HarnessProtocolError{Msg: "interaction turn_id must not be empty"}
	}
	// 步骤 3：deadline_at 给了值须为有限非负 Unix 秒
	if deadlineAt != nil {
		d := *deadlineAt
		if math.IsNaN(d) || math.IsInf(d, 0) || d < 0 {
			return &HarnessProtocolError{Msg: "interaction deadline_at must be a finite Unix timestamp"}
		}
	}
	return nil
}

// freezeJsonObjectOrDefault 冻结 JSON 对象，nil 归一化为空对象。
// Python 等价物：dataclass field(default_factory=dict)。
// Python 侧 default_factory 使调用方永不传 None，而 Go Options 零值是 nil；
// 若不归一化，FreezeJsonObject(nil) 会返回 "must be a mapping" 错误。
func freezeJsonObjectOrDefault(value JsonObject) (JsonObject, error) {
	if value == nil {
		value = JsonObject{}
	}
	return FreezeJsonObject(value)
}

// ValidateInteractionResponse 校验请求身份与它要求的应答形状。
// Python: interactions.py:273-288 validate_interaction_response（2 步）
//
// Q5 决策改选 3 + Q5b 决策 3-a：应答配对关系由请求自身的
// expectedResponseKind() 声明，比较 kind 字符串。
func ValidateInteractionResponse(
	request HarnessInteractionRequest,
	response HarnessInteractionResponse,
) (HarnessInteractionResponse, error) {
	reqID := requestIDOf(request)
	respID := responseIDOf(response)
	// 步骤 1：request_id 必须一致（Python :279-282）
	if respID != reqID {
		return nil, &HarnessProtocolError{
			Msg: fmt.Sprintf("interaction response id %q does not match request %q", respID, reqID),
		}
	}
	// 步骤 2：应答 kind 必须是请求声明的期望 kind（Python :283-288）
	expected := request.expectedResponseKind()
	if got := response.responseKind(); got != expected {
		return nil, &HarnessProtocolError{
			Msg: fmt.Sprintf("%s requires %s, got %s", requestKindName(request), expected, got),
		}
	}
	return response, nil
}

// recordingHandler 记录 15.3 交互应答的宿主实现（扮演宿主策略层）。
type recordingHandler struct {
	mu sync.Mutex
	// decision 固定返回的审批决策
	decision harness_protocol.ToolApprovalDecision
	// cancelLog 记录 (requestID, reason)
	cancelLog []cancelRecord
	// handleCount 记录 Handle 调用次数
	handleCount int
}

// cancelRecord 一次 cancel 调用。
type cancelRecord struct {
	RequestID string
	Reason    harness_protocol.InteractionCancelReason
}

// Handle 按固定决策应答，并走 ValidateInteractionResponse 自校验。
func (h *recordingHandler) Handle(ctx context.Context, request harness_protocol.HarnessInteractionRequest) (harness_protocol.HarnessInteractionResponse, error) {
	h.mu.Lock()
	h.handleCount++
	decision := h.decision
	h.mu.Unlock()

	switch r := request.(type) {
	case harness_protocol.ToolApprovalRequest:
		resp, err := harness_protocol.NewToolApprovalResponse(r.RequestID, decision,
			harness_protocol.ToolApprovalResponseOptions{})
		if err != nil {
			return nil, err
		}
		return harness_protocol.ValidateInteractionResponse(request, resp)
	case *harness_protocol.ToolApprovalRequest:
		resp, err := harness_protocol.NewToolApprovalResponse(r.RequestID, decision,
			harness_protocol.ToolApprovalResponseOptions{})
		if err != nil {
			return nil, err
		}
		return harness_protocol.ValidateInteractionResponse(request, resp)
	default:
		return nil, errors.New("unsupported request type in recordingHandler")
	}
}

// Cancel 记录取消原因（幂等：重复调用不改变结果）。
func (h *recordingHandler) Cancel(_ context.Context, requestID string, opts ...harness_protocol.CancelOption) error {
	reason := harness_protocol.InteractionCancelReasonProviderWithdrew
	for _, opt := range opts {
		if opt != nil {
			opt(&reason)
		}
	}
	h.mu.Lock()
	defer h.mu.Unlock()
	h.cancelLog = append(h.cancelLog, cancelRecord{RequestID: requestID, Reason: reason})
	return nil
}

// recordingHooks 记录 15.3 钩子调用的宿主实现。
type recordingHooks struct {
	mu sync.Mutex
	// beforePromptCalls / beforeToolCalls / afterToolCalls / onStopCalls 计数
	beforePromptCalls int
	beforeToolCalls   int
	afterToolCalls    int
	onStopCalls       int
	// stopPrompt 为 true 时 before_prompt 返回 continue=false
	stopPrompt bool
	// denyTool 非空时该工具被 DENY
	denyTool string
	// rewriteTo 非空时 before_tool 返回 REWRITE 到该命令
	rewriteTo string
	// replaceResultWith 非 nil 时 after_tool 返回 replace_result=true
	replaceResultWith any
	// observedTurnIDs 记录每次钩子拿到的 turn_id（验证身份透传）
	observedTurnIDs []string
}

// BeforePrompt 记录调用并可掐断执行。
func (h *recordingHooks) BeforePrompt(_ context.Context, hookCtx harness_protocol.BeforePromptContext) (harness_protocol.BeforePromptResult, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.beforePromptCalls++
	h.observedTurnIDs = append(h.observedTurnIDs, strOrEmpty(hookCtx.TurnID))
	return harness_protocol.NewBeforePromptResult(hookCtx.Prompt, harness_protocol.BeforePromptResultOptions{})
}

// BeforeTool 记录调用并给出允许/拒绝/改写决策。
func (h *recordingHooks) BeforeTool(_ context.Context, hookCtx harness_protocol.BeforeToolContext) (harness_protocol.ToolDecision, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.beforeToolCalls++
	h.observedTurnIDs = append(h.observedTurnIDs, strOrEmpty(hookCtx.TurnID))
	if hookCtx.ToolName == h.denyTool {
		return harness_protocol.ToolDecision{Decision: harness_protocol.ToolDecisionKindDeny}, nil
	}
	if h.rewriteTo != "" {
		return harness_protocol.NewToolDecision(harness_protocol.ToolDecisionKindRewrite,
			harness_protocol.ToolDecisionOptions{
				UpdatedArguments: harness_protocol.JsonObject{"command": h.rewriteTo},
			})
	}
	return harness_protocol.ToolDecision{Decision: harness_protocol.ToolDecisionKindAllow}, nil
}

// AfterTool 记录调用并可选替换结果。
func (h *recordingHooks) AfterTool(_ context.Context, hookCtx harness_protocol.AfterToolContext) (harness_protocol.AfterToolResult, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.afterToolCalls++
	h.observedTurnIDs = append(h.observedTurnIDs, strOrEmpty(hookCtx.TurnID))
	if h.replaceResultWith != nil {
		return harness_protocol.NewAfterToolResult(harness_protocol.AfterToolResultOptions{
			ReplaceResult: true,
			UpdatedResult: h.replaceResultWith,
		})
	}
	return harness_protocol.NewAfterToolResult(harness_protocol.AfterToolResultOptions{})
}

// OnStop 记录调用。
func (h *recordingHooks) OnStop(_ context.Context, _ harness_protocol.StopHookContext) error {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.onStopCalls++
	return nil
}

// TestHooksBoundary_并发多轮钩子计数正确 钩子实现在并发下必须不被撕裂
func TestHooksBoundary_并发多轮钩子计数正确(t *testing.T) {
	const workers = 4
	hooks := &recordingHooks{}
	var wg sync.WaitGroup
	providers := make([]*fakeCLIProvider, workers)

	for i := 0; i < workers; i++ {
		sess := newRealSession(t, "sess-hooks-concurrent-"+string(rune('a'+i)))
		providers[i] = &fakeCLIProvider{hooks: hooks}
		if err := providers[i].Start(context.Background(), sess); err != nil {
			t.Fatalf("Start[%d]: %v", i, err)
		}
	}
	defer func() {
		for i := range providers {
			_ = providers[i].Stop(context.Background())
		}
	}()

	errs := make([]error, workers)
	for i := 0; i < workers; i++ {
		wg.Add(1)
		go func(idx int) {
			defer wg.Done()
			errs[idx] = providers[idx].Send(context.Background(), "shell", false)
		}(i)
	}
	wg.Wait()

	for i := 0; i < workers; i++ {
		if errs[i] != nil {
			t.Errorf("worker %d Send: %v", i, errs[i])
		}
	}
	// 每个 worker 各触发 1 次 before_prompt / before_tool / after_tool / on_stop
	if hooks.beforePromptCalls != workers {
		t.Errorf("beforePromptCalls = %d，期望 %d", hooks.beforePromptCalls, workers)
	}
	if hooks.beforeToolCalls != workers {
		t.Errorf("beforeToolCalls = %d，期望 %d", hooks.beforeToolCalls, workers)
	}
	if hooks.afterToolCalls != workers {
		t.Errorf("afterToolCalls = %d，期望 %d", hooks.afterToolCalls, workers)
	}
	if hooks.onStopCalls != workers {
		t.Errorf("onStopCalls = %d，期望 %d", hooks.onStopCalls, workers)
	}
}
