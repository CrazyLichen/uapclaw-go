```go
// TodoModifyToolFactory todo_modify 工具工厂。
// swarm 层（todo_compat）通过此工厂替换默认的 update 分派逻辑，
// 等价 Python install_todo_modify_compat_patch() 的模块级 monkey-patch。
// Python 无此类型（机制差异显式化），Go 用函数注入替代继承覆盖。
type TodoModifyToolFactory func(todoTool TodoTool, language, agentID string) tool.Tool

// NewTodoModifyToolWithUpdateFn 创建带自定义 update 分派的 todo_modify 工具。
// updateFn 为 nil 时走原生 updateTodos（与 NewTodoModifyTool 行为一致）。
func NewTodoModifyToolWithUpdateFn(todoTool TodoTool, language, agentID string, updateFn func(todos []hschema.TodoItem, updates []map[string]any) ([]hschema.TodoItem, string, error)) tool.Tool {
	card, _ := tools.BuildToolCard("todo_modify", "TodoModifyTool", language, nil, agentID)

	fn := func(ctx context.Context, input TodoModifyInput, opts ...tool.ToolOption) (map[string]any, error) {
		result, err := func() (map[string]any, error) {
			sessionID, err := extractSessionID(opts)
			if err != nil {
				return nil, err
			}
			lock := todoTool.lockManager.Operation(sessionID)
			lock.Lock()
			defer lock.Unlock()

			todos, err := todoTool.LoadTodos(ctx, sessionID)
			if err != nil {
				return nil, err
			}

			var updatedTodos []hschema.TodoItem
			var msg string
			switch input.Action {
			case "update":
				if updateFn != nil {
					updatedTodos, msg, err = updateFn(todos, input.Todos)
				} else {
					updatedTodos, msg, err = updateTodos(todos, input.Todos)
				}
			case "delete":
				updatedTodos, msg, err = deleteTodos(todos, input.IDs)
			case "cancel":
				updatedTodos, msg, err = cancelTodos(todos, input.IDs)
			case "append":
				updatedTodos, msg, err = appendTodos(todos, input.Todos)
			case "insert_after":
				if input.TodoData == nil {
					return nil, exception.BuildError(
						exception.StatusToolTodosValidationInvalid,
						exception.WithParam("reason", "无效的插入操作输入: 'todo_data' 必须为包含 'target_id' 和 'items' 的对象"),
					)
				}
				for _, item := range input.TodoData.Items {
					if err := validateSingleTodoItem(item); err != nil {
						return nil, err
					}
				}
				updatedTodos, msg, err = insertAfterTodos(todos, input.TodoData.TargetID, input.TodoData.Items)
			case "insert_before":
				if input.TodoData == nil {
					return nil, exception.BuildError(
						exception.StatusToolTodosValidationInvalid,
						exception.WithParam("reason", "无效的插入操作输入: 'todo_data' 必须为包含 'target_id' 和 'items' 的对象"),
					)
				}
				for _, item := range input.TodoData.Items {
					if err := validateSingleTodoItem(item); err != nil {
						return nil, err
					}
				}
				updatedTodos, msg, err = insertBeforeTodos(todos, input.TodoData.TargetID, input.TodoData.Items)
			default:
				return nil, exception.BuildError(
					exception.StatusToolTodosValidationInvalid,
					exception.WithParam("reason", fmt.Sprintf("无效操作: %s", input.Action)),
				)
			}

			if err != nil {
				return nil, err
			}
			if err := todoTool.SaveTodos(ctx, sessionID, updatedTodos); err != nil {
				return nil, err
			}
			logger.Info(logComponent).
				Str("session_id", sessionID).
				Str("action", input.Action).
				Int("task_count", len(updatedTodos)).
				Msg("TodoModifyTool modified todo successfully")
			return map[string]any{"message": msg}, nil
		}()
		if err != nil {
			logger.Error(logComponent).Err(err).
				Str("event_type", "TOOL_CALL_ERROR").
				Str("tool_name", "todo_modify").
				Msg("Todo modify tool invocation failed")
			return nil, exception.BuildError(exception.StatusToolTodosInvokeFailed,
				exception.WithParam("reason", err.Error()),
			)
		}
		return result, nil
	}

	invokeFn, _ := tool.NewTool(fn, tool.WithToolCard(card), tool.WithToolInputParams(card.InputParams))
	return invokeFn
}

// CreateTodosToolWithModifyFactory 创建全部待办事项工具集，todo_modify 项由工厂构造。
// modifyFactory 为 nil 时等价 CreateTodosTool（modify 项走原生 NewTodoModifyTool）。
func CreateTodosToolWithModifyFactory(workspace string, fs sys_operation.FsOperation, language, agentID string, modifyFactory TodoModifyToolFactory) ([]tool.Tool, TodoTool) {
	lockManager := NewTodoLockManager()
	todoTool := newTodoTool(workspace, fs, lockManager)
	var modifyTool tool.Tool
	if modifyFactory != nil {
		modifyTool = modifyFactory(todoTool, language, agentID)
	} else {
		modifyTool = NewTodoModifyTool(todoTool, language, agentID)
	}
	toolsList := []tool.Tool{
		NewTodoCreateTool(todoTool, language, agentID),
		NewTodoListTool(todoTool, language, agentID),
		NewTodoGetTool(todoTool, language, agentID),
		modifyTool,
	}
	return toolsList, todoTool
}
```

**重要 Go 差异处理（写码时注意）：**
- `unquotedWindowsPathPattern` 的 lookbehind `(?<![\w/])`：Go RE2 **不支持 lookbehind**！替代实现：用 `regexp.MustCompile(`[A-Za-z]:\\[^\s|&;]+`)` findAllStringSubmatchIndex 后手动检查前一个字符是否 `\w` 或 `/`，是则跳过该匹配起点（在此函数注释中标注 RE2 差异）
- `powershellCommandArgPattern` 的 `(?P<script>...)` 命名组：Go 用 `SubexpNames()[1]` 索引取值（RE2 支持 `(?P<name>...)`）
- shlex.split(posix=False)：segmentBaseCommand 自实现时无需完整 shlex，Python tokens[0] 等价实现——按空白切分取第一段再 stripMatchingQuotes
- 危险正则 `\bdel\s+/[a-z]*[fsq][a-z]*\b` 的 `\b` RE2 支持，无需改

```go
func GetMcpTools() []tool.Tool {

// Python: _has_paid_search_api_key
func hasPaidSearchAPIKey() bool {
	for _, key := range []string{"BOCHA_API_KEY", "PERPLEXITY_API_KEY", "SERPER_API_KEY", "JINA_API_KEY"} {
		if os.Getenv(key) != "" {
			return true
		}
	}
	return false
}

// isFreeSearchEnabled 检查免费搜索引擎开关
// Python: _is_free_search_enabled
func isFreeSearchEnabled() bool {
	return envFlag("FREE_SEARCH_DDG_ENABLED", false) || envFlag("FREE_SEARCH_BING_ENABLED", false)
}

// envFlag 环境变量布尔解析（1/true/yes/on/enabled）
// Python: _env_flag (mcp_toolkits.py L26-30)
func envFlag(name string, defaultVal bool) bool {
	raw := strings.ToLower(strings.TrimSpace(os.Getenv(name)))
	if raw == "" {
		return defaultVal
	}
	return raw == "1" || raw == "true" || raw == "yes" || raw == "on" || raw == "enabled"
}
```

```go
// syncToolGroup 统一处理一组工具的热更新：启用时注册，禁用时移除。
// Python: _sync_tool_group (interface_deep.py:1318-1350)
// 返回 (updated_tools, updated_registered)。
func (d *DeepAdapter) syncToolGroup(
	currentTools []tool.Tool, registered bool, enabled bool,
	createFn func() []tool.Tool, warnLabel string,
) ([]tool.Tool, bool) {
	if !enabled {
		if registered {
			d.removeRegisteredTools(currentTools)
			namesToRemove := make(map[string]bool, len(currentTools))
			for _, t := range currentTools {
				namesToRemove[t.Card().Name] = true
			}
			d.toolCards = d.pruneToolCards(d.toolCards, namesToRemove)
		}
		return nil, false
	}
	if !registered {
		newTools := d.safeCreateTools(createFn, warnLabel)
		if newTools == nil {
			return nil, false
		}
		rm := runner.GetResourceMgr()
		am := d.getAbilityManager()
		for _, t := range newTools {
			if rm != nil {
				if err := rm.AddTool(t); err != nil {
					logger.Warn(logComponent).Err(err).Str("warn_label", warnLabel).Msg("syncToolGroup: add tool to ResourceMgr failed")
				}
			}
			d.appendToolCard([]*tool.ToolCard{t.Card()})
			if am != nil {
				am.Add(t.Card())
			}
		}
		return newTools, len(newTools) > 0
	}
	return currentTools, registered
}

// safeCreateTools 执行 createFn 并 recover 异常（对齐 Python try/except Exception warn）。
func (d *DeepAdapter) safeCreateTools(createFn func() []tool.Tool, warnLabel string) (tools []tool.Tool) {
	defer func() {
		if r := recover(); r != nil {
			logger.Warn(logComponent).Any("panic", r).Str("warn_label", warnLabel).Msg("syncToolGroup reload failed")
			tools = nil
		}
	}()
	return createFn()
}
```

package tools

import (

```go
	"strings"
```

```go
// ──────────────────────────── 导出函数 ────────────────────────────

// NewCompatibleTodoModifyTool 构造兼容版 todo_modify 工具。
// Python: CompatibleTodoModifyTool（继承 SDK TodoModifyTool 覆盖 _update_todos）
// Go 机制差异：不继承，用 NewTodoModifyToolWithUpdateFn 注入 compatibleUpdateTodos。
// 卡名保持 todo_modify（对齐 Python 子类不换名）。
func NewCompatibleTodoModifyTool(todoTool todo.TodoTool, language, agentID string) tool.Tool {
	return todo.NewTodoModifyToolWithUpdateFn(todoTool, language, agentID, compatibleUpdateTodos)
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// compatibleUpdateTodos 兼容模式批量更新。
// Python: CompatibleTodoModifyTool._update_todos (todo_compat.py L25-85)
// 语义：status ∈ {deleted, delete} → 收集删除；"canceled" → 归一 "cancelled"；
// 其余字段就地更新；最终过滤 deletedIDs → validateSingleInProgress；
// 消息 "Successfully updated N task(s)" + "Successfully deleted M task(s) (IDs: ...)"
// 或 "No task changes applied"。
// 注意：Python 版在 _update_todos 内 save_todos；Go 版外层 NewTodoModifyToolWithUpdateFn
// 统一 SaveTodos（分派结构差异，语义等价——spec 偏差 #2 同型）。
func compatibleUpdateTodos(todos []hschema.TodoItem, updates []map[string]any) ([]hschema.TodoItem, string, error) {
	if updates == nil {
		updates = []map[string]any{}
	}
	todoMap := make(map[string]*hschema.TodoItem, len(todos))
	for i := range todos {
		todoMap[todos[i].ID] = &todos[i]
	}
	deletedIDs := make(map[string]bool)
	updatedCount := 0

	for _, todoData := range updates {
		todoID := utils.StrValFromMap(todoData, "id")
		if todoID == "" {
			return nil, "", exception.BuildError(
				exception.StatusToolTodosValidationInvalid,
				exception.WithParam("reason", "Batch update failed: Missing required field: 'id'"),
			)
		}
		if _, ok := todoMap[todoID]; !ok {
			return nil, "", exception.BuildError(
				exception.StatusToolTodosValidationInvalid,
				exception.WithParam("reason", fmt.Sprintf("Batch update failed: Task with ID '%s' not found", todoID)),
			)
		}

		statusValue := utils.StrValFromMap(todoData, "status")
		if statusValue == "deleted" || statusValue == "delete" {
			deletedIDs[todoID] = true
			continue
		}

		currentTodo := todoMap[todoID]
		if _, exists := todoData["content"]; exists {
			currentTodo.Content = utils.StrValFromMap(todoData, "content")
		}
		if _, exists := todoData["activeForm"]; exists {
			currentTodo.ActiveForm = utils.StrValFromMap(todoData, "activeForm")
		}
		if _, exists := todoData["description"]; exists {
			currentTodo.Description = utils.StrValFromMap(todoData, "description")
		}
		if _, exists := todoData["status"]; exists {
			if statusValue == "canceled" {
				statusValue = "cancelled" // 归一（对齐 Python "canceled" → "cancelled"）
			}
			parsed, err := hschema.ParseTodoStatus(statusValue)
			if err != nil {
				return nil, "", exception.BuildError(
					exception.StatusToolTodosValidationInvalid,
					exception.WithParam("reason", fmt.Sprintf("无效的状态 '%s'（任务 '%s'）", statusValue, todoID)),
				)
			}
			currentTodo.Status = parsed
		}
		if _, exists := todoData["selected_model_id"]; exists {
			currentTodo.SelectedModelID = utils.StrValFromMap(todoData, "selected_model_id")
		}
		updatedCount++
	}

	updatedTodos := make([]hschema.TodoItem, 0, len(todos))
	for _, item := range todos {
		if !deletedIDs[item.ID] {
			updatedTodos = append(updatedTodos, item)
		}
	}
	if err := validateTodoSingleInProgress(updatedTodos); err != nil {
		return nil, "", err
	}

	var parts []string
	if updatedCount > 0 {
		parts = append(parts, fmt.Sprintf("Successfully updated %d task(s)", updatedCount))
	}
	if len(deletedIDs) > 0 {
		sorted := make([]string, 0, len(deletedIDs))
		for id := range deletedIDs {
			sorted = append(sorted, id)
		}
		sort.Strings(sorted) // Python sorted(deleted_ids)
		parts = append(parts, fmt.Sprintf("Successfully deleted %d task(s) (IDs: %s)", len(deletedIDs), strings.Join(sorted, ", ")))
	}
	if len(parts) == 0 {
		return updatedTodos, "No task changes applied", nil
	}
	return updatedTodos, strings.Join(parts, "; "), nil
}

// validateTodoSingleInProgress 单一进行中校验。
// agentcore todo 包 validateSingleInProgress 为非导出无法跨包调用，本地等价实现，
// 错误文案与 agentcore 版逐字一致（todo.go:981 已核实）。
func validateTodoSingleInProgress(todos []hschema.TodoItem) error {
	inProgressCount := 0
	for _, item := range todos {
		if item.Status == hschema.TodoStatusInProgress {
			inProgressCount++
		}
	}
	if inProgressCount > 1 {
		return exception.BuildError(
			exception.StatusToolTodosValidationInvalid,
			exception.WithParam("reason", "超过一个任务被标记为 'in_progress'（仅允许一个）"),
		)
	}
	return nil
}
```

```go
// cancelPendingTodos 将未完成的 todo 项标记为 cancelled。
// Python: _cancel_pending_todos (interface_deep.py:4351-4410)
// 返回前端格式 todo 列表（附加到 interrupt_result 通知前端刷新），无 todo 或失败返回 nil。
//
// Python 先从 ability_manager.get("todo_modify") + resource_mgr.get_tool() 取已注册实例，
// 失败才 fallback 构造。Go 的 tool.Tool 接口无法还原 TodoTool 基类（LoadTodos/SaveTodos
// 在基类上），因此统一走 fallback 构造——语义等价：todos 文件路径相同（同一 workspace 的
// todo 节点），读写同一持久层（偏差 #2 说明栏）。
func (d *DeepAdapter) cancelPendingTodos(ctx context.Context, sessionID string) []map[string]any {
	if d.instance == nil {
		return nil
	}

	defer func() {
		if r := recover(); r != nil {
			logger.Warn(logComponent).Any("panic", r).Str("session_id", sessionID).Msg("cancel pending todos failed")
		}
	}()

	// fallback 构造：从 DeepConfig 的 workspace todo 节点构造 TodoTool
	if d.instance.DeepConfig() == nil || d.instance.DeepConfig().Workspace == nil {
		return nil
	}
	todoPath := d.instance.DeepConfig().Workspace.GetNodePath(hworkspace.WorkspaceNodeTODO)
	if todoPath == nil {
		return nil
	}
	var fs sysop.FsOperation
	if d.sysOperation != nil {
		fs = d.sysOperation.Fs()
	}
	_, todoBase := todo.CreateTodosTool(*todoPath, fs, d.resolveRuntimeLanguage(), "")
	modifyTool := &todoBase

	todos, err := modifyTool.LoadTodos(ctx, sessionID)
	if err != nil || len(todos) == 0 {
		return nil
	}

	// 收集非 completed/cancelled 的 ID
	var idsToCancel []string
	for _, item := range todos {
		if item.Status != hschema.TodoStatusCompleted && item.Status != hschema.TodoStatusCancelled {
			idsToCancel = append(idsToCancel, item.ID)
		}
	}

	if len(idsToCancel) > 0 {
		// Python: modify_tool._cancel_todos(ids, todos)（SDK 版内含 save）
		// Go: cancelTodos 不含 save，外层显式两步（spec 偏差 #2）
		updated, _, cErr := todo.CancelTodos(todos, idsToCancel)
		if cErr != nil {
			logger.Warn(logComponent).Err(cErr).Str("session_id", sessionID).Msg("cancel pending todos failed")
			return nil
		}
		if sErr := modifyTool.SaveTodos(ctx, sessionID, updated); sErr != nil {
			logger.Warn(logComponent).Err(sErr).Str("session_id", sessionID).Msg("cancel pending todos save failed")
			return nil
		}
		logger.Info(logComponent).Str("session_id", sessionID).Int("cancelled_count", len(idsToCancel)).Msg("cancelled pending todos marked as cancelled")
	}

	// 重新加载并返回前端格式
	updatedTodos, err := modifyTool.LoadTodos(ctx, sessionID)
	if err != nil || len(updatedTodos) == 0 {
		return nil
	}
	// formatTodosForFrontend 是 commrails 包非导出函数；
	// 本地等价实现 formatTodosForFrontendLocal（映射逻辑逐字对齐 stream_event_emit.go:234）
	return formatTodosForFrontendLocal(updatedTodos), nil
}

// formatTodosForFrontendLocal 前端 Todo 格式化（本地等价实现，
// 对齐 commrails.formatTodosForFrontend —— 该函数非导出无法跨包引用）。
func formatTodosForFrontendLocal(todosData []hschema.TodoItem) []map[string]any {
	statusMapping := map[hschema.TodoStatus]string{
		hschema.TodoStatusPending:    "pending",
		hschema.TodoStatusInProgress: "in_progress",
		hschema.TodoStatusCompleted:  "completed",
	}
	result := make([]map[string]any, 0, len(todosData))
	for _, item := range todosData {
		if item.Status == hschema.TodoStatusCancelled {
			continue
		}
		statusStr, ok := statusMapping[item.Status]
		if !ok {
			statusStr = item.Status.String()
		}
		result = append(result, map[string]any{
			"id":         item.ID,
			"content":    item.Content,
			"activeForm": item.ActiveForm,
			"status":     statusStr,
		})
	}
	return result
}
```
