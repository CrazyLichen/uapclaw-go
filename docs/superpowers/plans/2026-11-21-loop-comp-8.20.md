func NewBaseCondition(
	invokeImpl func(ctx context.Context, inputs map[string]any, sess session.BaseSession) (bool, map[string]any, error),
	inputSchema any,
) BaseCondition {
	if invokeImpl == nil {
		panic("invokeImpl 不能为 nil（对齐 Python 抽象方法 invoke 必须实现）")
	}
	b := BaseCondition{inputSchema: inputSchema}
	b.AtomicNode = graph.NewAtomicNode(func(ctx context.Context, kwargs map[string]any) (any, error) {
		sess, _ := kwargs["session"].(session.BaseSession)
		inputs := map[string]any{}
		if b.inputSchema != nil {
			ws, ok := sess.State().(state.WorkflowState)
			if !ok {
				return nil, exception.BuildError(exception.StatusGraphStateCommitError,
					exception.WithParam("reason", "session is not support commit state"))
			}
			inputs, _ = ws.GetInputs(b.inputSchema.(state.StateKey)).(map[string]any)
		}
		// Python condition.py:21-23：result = self.invoke(...)；元组分支
		// isinstance(result, tuple) → set_outputs(result[1]) → result = result[0]

			}
		}

	})

// 鸭子怪癖逐字保留：invoke 按 inputs.get(key, []) 取数组，而 inputs 恒按

// updates 写 state 返回 (True, io_updates)）。
func (c *ArrayCondition) Invoke(ctx context.Context, inputs map[string]any, sess session.BaseSession) (bool, error) {
	flag, _, err := c.invokeImpl(ctx, inputs, sess)
	return flag, err
}

// TraceInfo 返回 "ArrayCondition"（Python 默认 trace_info 返回 ""，D-20-x：基类缺省）。
func (c *ArrayCondition) TraceInfo(_ session.BaseSession) any { return "" }

func (c *ArrayCondition) invokeImpl(_ context.Context, inputs map[string]any, sess session.BaseSession) (bool, map[string]any, error) {
	// Python: current_idx = session.state().get(INDEX)
	cur := getConditionIndex(sess) // 包级助手：sess.State().Get(constants.IndexKey)，nil→-1 语义见实现
	minLength := DefaultMaxLoopNumber
	updates := map[string]any{}
	for key := range c.arrays {
		arr, _ := inputs[key].([]any) // Python: inputs.get(key, [])，非数组 → 空
		if len(arr) < minLength {
			minLength = len(arr)
		}
		if cur >= minLength {
			return false, nil, nil // Python: return False（裸 bool）
		}
		updates[key] = arr[cur]
	}
	// Python: session.state().update(updates); io_updates = updates.copy()
	if ws, ok := sess.State().(state.WorkflowState); ok {
		if err := ws.Update(updates); err != nil {
			return false, nil, err
		}
	}
	return true, updates, nil // Python: return True, io_updates（tuple）
}

// ArrayConditionInSession 会话版数组条件（loop_comp.py:177 消费；array.py:29-60）。
type ArrayConditionInSession struct {
	BaseCondition
	arrays    map[string]any
	minLength int
}

// NewArrayConditionInSession 构造（array.py:30-33：super().__init__() 无 schema；
// _check_arrays 构造期算 min_length，None/非 list|tuple 即 ValueError）。
func NewArrayConditionInSession(arrays map[string]any) (*ArrayConditionInSession, error) {
	minLength, err := checkArrays(arrays) // _check_arrays 一比一
	if err != nil {
		return nil, err
	}
	c := &ArrayConditionInSession{arrays: arrays, minLength: minLength}
	c.BaseCondition = NewBaseCondition(c.invokeImpl, nil)
	return c, nil
}

// Invoke/TraceInfo 同 ArrayCondition 形状（trace_info 默认 ""）。

func (c *ArrayConditionInSession) invokeImpl(_ context.Context, _ map[string]any, sess session.BaseSession) (bool, map[string]any, error) {
	cur := getConditionIndex(sess)
	if cur >= c.minLength {
		return false, nil, nil // array.py:34-35 裸 False
	}
	updates := map[string]any{}
	for key, info := range c.arrays {
		arr, ok := info.([]any)
		if !ok {
			return false, nil, fmt.Errorf("Expected list/tuple for '%s' in loop_array, got %T", key, info)
		}
		if cur >= len(arr) {
			// Python except (TypeError, IndexError, KeyError) → ValueError("Array loop error in '%s': <error_details>")
			return false, nil, fmt.Errorf("Array loop error in '%s': <error_details>", key)
		}
		updates[key] = arr[cur]
	}
	if ws, ok := sess.State().(state.WorkflowState); ok {
		if err := ws.Update(updates); err != nil {
			return false, nil, err
		}
	}
	return true, updates, nil
}

// checkArrays 构造期校验（array.py:52-59 一比一）。
func checkArrays(arrays map[string]any) (int, error) {
	minLength := DefaultMaxLoopNumber
	for key, info := range arrays {
		if info == nil {
			return 0, fmt.Errorf("Value for key '%s' in loop_array cannot be None", key)
		}
		if _, ok := info.([]any); !ok {
			return 0, fmt.Errorf("Expected list/tuple for '%s' in loop_array, got %T", key, info)
		}
		if len(info.([]any)) < minLength {
			minLength = len(info.([]any))
		}
	}
	return minLength, nil
}

		stage, _ := kwargs["loop_stage"].(string)

		case StageStartRound:

func (c *OutputCallback) generateOutput(sess session.BaseSession, results []any, root []string, outputFormat any) any {
	if m, ok := outputFormat.(map[string]any); ok {

		for key, value := range m {
			path := append(root[:len(root):len(root)], key)
			output[key] = c.generateOutput(sess, results, path, value)
		}
		return output
	}
	if s, ok := outputFormat.(string); ok && sessionutils.IsRefPath(s) {
		refStr := sessionutils.ExtractOriginKey(s)
		path := strings.Split(refStr, sessionutils.NestedPathSplit)
		if path[0] == sess.NodeID() {
			if len(results) == 0 {
				return nil
			}
			data := results[len(results)-1]
			for _, key := range root {
				data = data.(map[string]any)[key]
			}
			return data
		}
	}
	output := []any{}
	for _, result := range results {
		data := result
		for _, key := range root {
			data = data.(map[string]any)[key]
		}
		output = append(output, data)
	}
	return output
}
