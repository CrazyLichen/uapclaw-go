```go
package chunker

import (
	"context"
	"strconv"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/store/reranker"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/retrieval/common"
	"github.com/uapclaw/uapclaw-go/internal/common/exception"
)

// ──────────────────────────── 结构体 ────────────────────────────

// Chunker 文本分块抽象接口。
//
// 三方法对齐 Python Chunker：chunk_text（子类必覆写）、
// chunk_documents（基类默认实现，子类可选覆写）、process（Processor 契约入口）。
//
// 知识库入库时调用 ChunkDocuments；ChunkText 是子类唯一必须实现的分块算法。
//
// Python: openjiuwen/core/retrieval/indexing/processor/chunker/base.py (Chunker)
type Chunker interface {
	// ChunkText 对单段文本分块，返回分块文本列表。空文本返回空切片。
	ChunkText(text string) ([]string, error)
	// ChunkDocuments 对文档列表分块，产出带溯源元数据的 TextChunk。
	ChunkDocuments(ctx context.Context, docs []reranker.Document) ([]common.TextChunk, error)
	// Process 处理文档（Processor 契约入口），等价于 ChunkDocuments。
	Process(ctx context.Context, docs []reranker.Document) ([]common.TextChunk, error)
}

// BaseChunker 分块器默认适配器。
//
// 对齐 Python Chunker 基类的可复用部分：chunk_size/chunk_overlap 构造校验与
// length_function。具体分块器嵌入本结构体，仅覆写 ChunkText，自动获得
// ChunkDocuments/Process 默认实现。
//
// Python: openjiuwen/core/retrieval/indexing/processor/chunker/base.py (Chunker)
type BaseChunker struct {
	// ChunkSize 分块大小，必须大于 0
	ChunkSize int
	// ChunkOverlap 相邻块重叠，取值 [0, ChunkSize)
	ChunkOverlap int
	// lengthFunction 长度计算函数，nil 时默认按字符计数（utf8.RuneCountInString）
	lengthFunction func(string) int
	// self 完整 Chunker 引用（子类构造时经 SetSelf 注入），
	// 使默认 ChunkDocuments 能路由到子类覆写的 ChunkText
	self Chunker
}

// ──────────────────────────── 枚举 ────────────────────────────
// ──────────────────────────── 常量 ────────────────────────────
// ──────────────────────────── 全局变量 ────────────────────────────
// ──────────────────────────── 导出函数 ────────────────────────────

// NewBaseChunker 构造 BaseChunker 并执行参数校验。
//
// 校验规则（一比一对齐 Python Chunker.__init__）：
//   - chunkSize <= 0 → 155100 RETRIEVAL_INDEXING_CHUNK_SIZE_INVALID
//   - chunkOverlap < 0 → 155101 RETRIEVAL_INDEXING_CHUNK_OVERLAP_INVALID
//   - chunkOverlap >= chunkSize → 155101 RETRIEVAL_INDEXING_CHUNK_OVERLAP_INVALID
//
// lengthFn 可选，默认按字符计数。
func NewBaseChunker(chunkSize, chunkOverlap int, lengthFn ...func(string) int) (*BaseChunker, error) {
	if chunkSize <= 0 {
		return nil, exception.BuildError(
			exception.StatusRetrievalIndexingChunkSizeInvalid,
			exception.WithParam("error_msg", "chunk_size must be greater than 0, current value: "+strconv.Itoa(chunkSize)),
		)
	}
	if chunkOverlap < 0 {
		return nil, exception.BuildError(
			exception.StatusRetrievalIndexingChunkOverlapInvalid,
			exception.WithParam("error_msg", "chunk_overlap must be greater than or equal to 0, current value: "+strconv.Itoa(chunkOverlap)),
		)
	}
	if chunkOverlap >= chunkSize {
		return nil, exception.BuildError(
			exception.StatusRetrievalIndexingChunkOverlapInvalid,
			exception.WithParam("error_msg", "chunk_overlap must be less than chunk_size"),
		)
	}
	bc := &BaseChunker{
		ChunkSize:    chunkSize,
		ChunkOverlap: chunkOverlap,
	}
	if len(lengthFn) > 0 && lengthFn[0] != nil {
		bc.lengthFunction = lengthFn[0]
	}
	return bc, nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────
```

注意：import 需补 `"strconv"`。`lengthFunction` 默认值问题：Python 是 `length_function or len`（构造期固化）；Go 保持字段 nil、`Length()` 方法运行期 fallback，语义等价且测试断言 `assert.Nil`。

```go
// chunk_documents（基类默认实现，子类可选覆写）、process（Processor 契约入口）。

//
// Python: openjiuwen/core/retrieval/indexing/processor/chunker/base.py (Chunker)
type Chunker interface {
	// ChunkText 对单段文本分块，返回分块文本列表。空文本返回空切片。
	ChunkText(text string) ([]string, error)
	// ChunkDocuments 对文档列表分块，产出带溯源元数据的 TextChunk。
	ChunkDocuments(ctx context.Context, docs []reranker.Document) ([]common.TextChunk, error)
```

在 BaseChunker 上追加方法（导出函数区块）：

```go
// ChunkDocuments 分块文档列表（默认实现，可被嵌入方继承）。
//
// 对每个文档调用 ChunkText，为每块生成 uuid 并合并溯源元数据：
// {**doc.Metadata, "chunk_index": i, "total_chunks": len(texts), "chunk_id": uid}。
// 任何文档分块失败即中断返回。
//
// Python: base.py Chunker.chunk_documents
func (b *BaseChunker) ChunkDocuments(ctx context.Context, docs []reranker.Document) ([]common.TextChunk, error) {
	_ = ctx // 对齐 Python 同步签名，当前无 ctx 消费
	chunks := make([]common.TextChunk, 0)
	for _, doc := range docs {
		texts, err := b.ChunkTextFor(doc.Text)
		if err != nil {
			return nil, err
		}
		for i, text := range texts {
			uid := uuid.NewString()
			meta := make(map[string]any, len(doc.Metadata)+3)
			for k, v := range doc.Metadata {
				meta[k] = v
			}
			meta["chunk_index"] = i
			meta["total_chunks"] = len(texts)
			meta["chunk_id"] = uid
			chunks = append(chunks, common.TextChunk{
				ID:       uid,
				Text:     text,
				DocID:    doc.ID,
				Metadata: meta,
			})
		}
	}
	return chunks, nil
}
```

**关键设计说明——chunkTextOf 路由**：嵌入方覆写 `ChunkText(text)`；BaseChunker.ChunkDocuments 需要调用"嵌入方的覆写版本"，但 Go 方法无法从嵌入结构体调用外部覆写方法。采用 **self 注入模式**：BaseChunker 增加非导出字段 `self Chunker`，子类构造函数调用 `SetSelf` 注入自身引用。ChunkDocuments 内经 `chunkTextOf` 路由到 `self.ChunkText`。

在 base.go 的 BaseChunker 结构体追加字段：

```go
	// self 完整 Chunker 引用（子类构造时经 SetSelf 注入），
	// 使默认 ChunkDocuments 能路由到子类覆写的 ChunkText
	self Chunker
```

```go
package chunker

import (
	"fmt"
	"sort"
	"strings"
	"sync"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/store/reranker"
)

// ──────────────────────────── 结构体 ────────────────────────────

// ChunkerOptions 分块器工厂参数，对齐 Python get_chunker(**kwargs)。
type ChunkerOptions struct {
	// ChunkSize 分块大小（默认 512）
	ChunkSize int
	// ChunkOverlap 相邻块重叠（默认 50）
	ChunkOverlap int
	// InnerChunker hybrid 工厂的 inner 分块器（对齐 inner_chunker）
	InnerChunker Chunker
	// NoSplitWhen hybrid 单块判定谓词（对齐 no_split_when）
	NoSplitWhen func(*reranker.Document) bool
	// Extra 其他透传参数（对齐 **kwargs 剩余项）
	Extra map[string]any
}

// ChunkerOption 工厂可选参数函数。
type ChunkerOption func(*ChunkerOptions)

// ChunkerFactory 分块器工厂签名。返回 error 承载构造校验失败
//（对齐 Python 工厂内构造抛 ValueError）。
//
// 错误风格说明：Python registry 系错误（ValueError/KeyError/TypeError）
// 是原生异常非 StatusCode，Go 侧按项目先例（retrieval/embedding/utils.go 等）
// 统一用 fmt.Errorf。
type ChunkerFactory func(opts ...ChunkerOption) (Chunker, error)

// ──────────────────────────── 枚举 ────────────────────────────
// ──────────────────────────── 常量 ────────────────────────────

const (
	// defaultChunkSize 默认分块大小（对齐 Python CharChunker 默认值）
	defaultChunkSize = 512
	// defaultChunkOverlap 默认分块重叠
	defaultChunkOverlap = 50
)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// chunkerRegistry 名称 → 工厂。并发访问由 registryMu 保护
	//（Python 无并发问题，Go 包级 map 需要互斥）
	chunkerRegistry = map[string]ChunkerFactory{}
	// registryMu registry 互斥锁
	registryMu sync.Mutex
)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewChunkerOptions 构建带默认值的选项并应用可变参数。
func NewChunkerOptions(opts ...ChunkerOption) *ChunkerOptions {
	o := &ChunkerOptions{
		ChunkSize:    defaultChunkSize,
		ChunkOverlap: defaultChunkOverlap,
	}
	for _, opt := range opts {
		opt(o)
	}
	return o
}

// WithChunkSize 设置分块大小。
func WithChunkSize(size int) ChunkerOption {
	return func(o *ChunkerOptions) { o.ChunkSize = size }
}

// WithChunkOverlap 设置分块重叠。
func WithChunkOverlap(overlap int) ChunkerOption {
	return func(o *ChunkerOptions) { o.ChunkOverlap = overlap }
}

// WithInnerChunker 设置 hybrid 工厂的 inner 分块器。
func WithInnerChunker(c Chunker) ChunkerOption {
	return func(o *ChunkerOptions) { o.InnerChunker = c }
}

// WithNoSplitWhen 设置 hybrid 单块判定谓词。
func WithNoSplitWhen(fn func(*reranker.Document) bool) ChunkerOption {
	return func(o *ChunkerOptions) { o.NoSplitWhen = fn }
}

// WithExtra 设置透传参数（对齐 **kwargs）。
func WithExtra(key string, value any) ChunkerOption {
	return func(o *ChunkerOptions) {
		if o.Extra == nil {
			o.Extra = make(map[string]any)
		}
		o.Extra[key] = value
	}
}

// RegisterChunker 按名注册分块器工厂。
//
// 对齐 Python register_chunker：空名/重名（overwrite=false）报错。
func RegisterChunker(name string, factory ChunkerFactory, overwrite bool) error {
	if strings.TrimSpace(name) == "" {
		return fmt.Errorf("chunker name must be a non-empty string")
	}
	registryMu.Lock()
	defer registryMu.Unlock()
	if _, exists := chunkerRegistry[name]; exists && !overwrite {
		return fmt.Errorf("Chunker '%s' is already registered. Use overwrite to replace it", name)
	}
	chunkerRegistry[name] = factory
	return nil
}

// GetChunker 按名获取分块器实例。
//
// 对齐 Python get_chunker：未知名报错且错误信息含已注册名列表；
// 工厂返回 nil 视为类型错误（对齐 Python TypeError 断言）。
func GetChunker(name string, opts ...ChunkerOption) (Chunker, error) {
	registryMu.Lock()
	factory, exists := chunkerRegistry[name]
	if !exists {
		names := make([]string, 0, len(chunkerRegistry))
		for k := range chunkerRegistry {
			names = append(names, k)
		}
		registryMu.Unlock()
		sort.Strings(names)
		return nil, fmt.Errorf("Unknown chunker: %s. Registered: %s", name, strings.Join(names, ", "))
	}
	registryMu.Unlock()

	chunker, err := factory(opts...)
	if err != nil {
		return nil, err
	}
	if chunker == nil {
		return nil, fmt.Errorf("Chunker entry '%s' must return a Chunker instance, got nil", name)
	}
	return chunker, nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// init 内置注册。对齐 Python __init__.py 尾部的 register_chunker 调用。
func init() {
	// "char" 注册随 13.6 CharChunker 落地 ⤴️
	_ = RegisterChunker("hybrid", hybridFactory, false)
}

// hybridFactory hybrid 分块器工厂。
//
// 对齐 Python _hybrid_factory：
//   - inner 缺省：由 chunk_size/chunk_overlap 构造默认 CharChunker
//     （13.6 前占位报错 ⤴️），多余参数透传给 CharChunker（前向兼容）
//   - inner 已给：多余 kwargs 报错（TypeError 语义）；inner 必须实现 Chunker
//   - HybridChunker 实体在 13.8 回填 ⤴️，当前 inner 分支为占位
func hybridFactory(opts ...ChunkerOption) (Chunker, error) {
	o := NewChunkerOptions(opts...)
	if o.InnerChunker == nil {
		// ⤴️ 13.6：替换为 CharChunker 构造（多余 Extra 透传）
		return nil, fmt.Errorf("default inner chunker (char) not implemented until 13.6")
	}
	if len(o.Extra) > 0 {
		keys := make([]string, 0, len(o.Extra))
		for k := range o.Extra {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		return nil, fmt.Errorf("Unknown kwargs for 'hybrid' chunker: %s", strings.Join(keys, ", "))
	}
	// ⤴️ 13.8：替换为 HybridChunker(inner, noSplitWhen) 构造
	_ = o.NoSplitWhen
	return nil, fmt.Errorf("HybridChunker not implemented until 13.8")
}
```
