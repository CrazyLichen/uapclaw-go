# 11.26 A2A 协议通道 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 uapclaw Gateway 对外暴露 A2A 1.0 入站通道（JSON-RPC + SSE + Agent Card），把外部 A2A 调用转成内部 `chat.send` 请求并回流结果。

**Architecture:** 协议栈交给官方 Go SDK `github.com/a2aproject/a2a-go/v2`（`a2asrv` 提供 JSON-RPC 分派、任务状态机、SSE、TaskStore、Agent Card 路由），我们只写「网关桥」：`A2AChannel`（实现 `channel_manager.BaseChannel`，管理 HTTP 生命周期与 pending 表）+ `a2aExecutor`（`a2asrv.AgentExecutor`，把 `MessageHandler` 回包事件序列化成 A2A 事件）+ 6 个纯映射函数（1:1 移植 Python `a2a_connect.py`）。

**Tech Stack:** Go 1.26 / `github.com/a2aproject/a2a-go/v2 v2.7.0`（Apache-2.0）/ `net/http` / 既有 `swarm/schema`、`swarm/gateway/channel_manager`、`common/logger`

**设计文档:** `docs/superpowers/specs/2026-10-31-a2a-channel-11.26-design.md`（含 4 项拍板与 D1-D12 偏差归档，实施时逐条核对）

package a2a

import (
	"encoding/json"
	"fmt"
	"strings"

	sdka2a "github.com/a2aproject/a2a-go/v2/a2a"

	"github.com/uapclaw/uapclaw-go/internal/swarm/schema"
)

// ──────────────────────────── 常量 ────────────────────────────

const (
	// agentRequestFailedText 错误兜底文案
	agentRequestFailedText = "agent request failed"
)

// ──────────────────────────── 导出函数 ────────────────────────────

// IsCompletionSentinelText 判断文本是否为"完成哨兵"（应被过滤，不发给 A2A 客户端）。
//
// 去掉全部空白后小写比较，等价 Python "".join(text.split()).lower()。
// Python: a2a_connect.py is_completion_sentinel_text (L425-428)
func IsCompletionSentinelText(text string) bool {
	compact := strings.ToLower(strings.Join(strings.Fields(text), ""))
	return compact == "{'is_complete':true}" || compact == `{"is_complete":true}`
}

// IsTerminalMessage 判断内部消息是否为一次 A2A 任务的终态消息。
//
// Python: a2a_connect.py is_terminal_message (L430-444)
func IsTerminalMessage(msg *schema.Message) bool {
	if msg == nil {
		return false
	}
	if msg.Type == schema.MessageTypeRes {
		return true
	}
	if msg.Type != schema.MessageTypeEvent {
		return false
	}
	if msg.EventType == schema.EventTypeChatError || msg.EventType == schema.EventTypeChatInterruptResult {
		return true
	}
	if v, ok := msg.Payload["is_complete"].(bool); ok && v {
		return true
	}
	return false
}

// MessageToText 把内部消息压成纯文本（A2A fallback 用）。
//
// Python: a2a_connect.py message_to_text (L348-357)
func MessageToText(msg *schema.Message) string {
	if msg == nil {
		return ""
	}
	payload := msg.Payload
	if msg.Type == schema.MessageTypeEvent && msg.EventType == schema.EventTypeChatError {
		if v := firstNonEmptyStr(payload["error"], payload["content"]); v != "" {
			return v
		}
		return agentRequestFailedText
	}
	if content, ok := payload["content"]; ok {
		return strVal(content)
	}
	if len(payload) > 0 {
		// 偏差 D5：Python str(dict) 为单引号 repr，Go 用 JSON（确定性、可读）
		if b, err := json.Marshal(payload); err == nil {
			return string(b)
		}
	}
	return ""
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// strVal 把任意值转字符串；nil 返回空串。
func strVal(v any) string {
	switch t := v.(type) {
	case nil:
		return ""
	case string:
		return t
	default:
		return fmt.Sprint(t)
	}
}

// firstNonEmptyStr 返回第一个非空字符串化值。
func firstNonEmptyStr(vals ...any) string {
	for _, v := range vals {
		if s := strVal(v); s != "" {
			return s
		}
	}
	return ""
}

// joinTextParts 把 A2A 消息的文本段按换行拼接（对应 Python context.get_user_input() 兜底）。
func joinTextParts(m *sdka2a.Message) string {
	if m == nil {
		return ""
	}
	segs := make([]string, 0, len(m.Parts))
	for _, p := range m.Parts {
		if p == nil {
			continue
		}
		if t := strings.TrimSpace(p.Text()); t != "" {
			segs = append(segs, t)
		}
	}
	return strings.TrimSpace(strings.Join(segs, "\n"))
}

// MapA2APartsToParams 把 A2A 消息的 parts 映射为内部 query 与 files 参数。
//
// 文本段按 "\n" 合并为 query；含 url/data/raw 的 Part 转成 files 条目，
// 键与 web 通道保持兼容（name/filename、media_type/type、url/uri、encoding=base64）。
// Python: a2a_connect.py map_a2a_parts_to_params (L457-500)
func MapA2APartsToParams(m *sdka2a.Message) (string, []map[string]any) {
	if m == nil {
		return "", nil
	}

	textSegments := make([]string, 0, len(m.Parts))
	var files []map[string]any

	for idx, part := range m.Parts {
		if part == nil {
			continue
		}
		if t := strings.TrimSpace(part.Text()); t != "" {
			textSegments = append(textSegments, t)
		}

		filename := strings.TrimSpace(part.Filename)
		mediaType := strings.TrimSpace(part.MediaType)
		url := strings.TrimSpace(string(part.URL()))
		data := part.Data()
		raw := part.Raw()

		if url == "" && data == nil && len(raw) == 0 {
			continue
		}

		name := filename
		if name == "" {
			name = fmt.Sprintf("a2a_part_%d", idx)
		}
		entry := map[string]any{
			// web 通道兼容键
			"name":     name,
			"filename": name,
		}
		if mediaType != "" {
			entry["media_type"] = mediaType
			// 通用消费方检查 type
			entry["type"] = mediaType
		}
		if url != "" {
			entry["url"] = url
			entry["uri"] = url
		}
		if data != nil {
			entry["data"] = strVal(data)
			entry["encoding"] = "base64"
		}
		if len(raw) > 0 {
			// 偏差 D4：Python 直接 str(bytes) 会 TypeError，Go 走 base64
			entry["raw"] = base64.StdEncoding.EncodeToString(raw)
		}
		files = append(files, entry)
	}

	query := strings.TrimSpace(strings.Join(textSegments, "\n"))
	return query, files
}

// MessageToA2AParts 把内部消息 payload 映射为 A2A 响应 parts。
//
// CHAT_ERROR 直接转错误文本；content/result 转文本（过滤哨兵）；
// 工具事件文本化为 [tool_call]/[tool_result]；payload.files 转 url/data/raw 三类 Part。
// fallbackToText 为真且无任何 part 时回退 MessageToText。
// Python: a2a_connect.py message_to_a2a_parts (L359-423)
func MessageToA2AParts(msg *schema.Message, fallbackToText bool) []*sdka2a.Part {
	if msg == nil {
		return nil
	}
	payload := msg.Payload
	if payload == nil {
		payload = map[string]any{}
	}

	var parts []*sdka2a.Part

	// 错误信息保持可读
	if msg.Type == schema.MessageTypeEvent && msg.EventType == schema.EventTypeChatError {
		text := firstNonEmptyStr(payload["error"], payload["content"])
		if text == "" {
			text = agentRequestFailedText
		}
		return []*sdka2a.Part{sdka2a.NewTextPart(text)}
	}

	switch content := payload["content"].(type) {
	case string:
		if t := strings.TrimSpace(content); t != "" && !IsCompletionSentinelText(t) {
			parts = append(parts, sdka2a.NewTextPart(t))
		}
	case nil, map[string]any, []any:
		// 缺失 / 结构化内容不转文本（对齐 Python 跳过 dict/list）
	default:
		parts = append(parts, sdka2a.NewTextPart(strVal(content)))
	}

	if result, ok := payload["result"].(string); ok {
		if t := strings.TrimSpace(result); t != "" && !IsCompletionSentinelText(t) {
			parts = append(parts, sdka2a.NewTextPart(t))
		}
	}

	// 流式模式暴露工具进度
	if msg.EventType == schema.EventTypeChatToolCall {
		if toolCall, ok := payload["tool_call"].(map[string]any); ok && toolCall != nil {
			name := strings.TrimSpace(strVal(toolCall["name"]))
			if name == "" {
				name = "tool"
			}
			parts = append(parts, sdka2a.NewTextPart("[tool_call] "+name))
		}
	}
	if msg.EventType == schema.EventTypeChatToolResult {
		toolName := strings.TrimSpace(strVal(payload["tool_name"]))
		if toolResult, ok := payload["result"].(string); ok && strings.TrimSpace(toolResult) != "" {
			label := "[tool_result] "
			if toolName != "" {
				label = "[tool_result:" + toolName + "] "
			}
			parts = append(parts, sdka2a.NewTextPart(label+strings.TrimSpace(toolResult)))
		}
	}

	parts = append(parts, filesToParts(payload["files"])...)

	if len(parts) > 0 {
		return parts
	}
	if fallbackToText {
		return []*sdka2a.Part{sdka2a.NewTextPart(MessageToText(msg))}
	}
	return nil
}

// BuildRequestParams 构造内部 chat.send 的 params。
//
// Python: a2a_connect.py _build_request_params (L446-455)
func BuildRequestParams(query string, files []map[string]any) map[string]any {
	params := map[string]any{"query": query}
	if len(files) > 0 {
		params["files"] = files
	}
	return params
}

// filesToParts 把 payload.files 转成 A2A 文件 Part（url/data/raw 三类）。
//
// Python: a2a_connect.py L396-417
func filesToParts(raw any) []*sdka2a.Part {
	list, ok := raw.([]any)
	if !ok {
		return nil
	}
	var parts []*sdka2a.Part
	for idx, item := range list {
		fi, ok := item.(map[string]any)
		if !ok {
			continue
		}
		name := strings.TrimSpace(firstNonEmptyStr(fi["filename"], fi["name"]))
		if name == "" {
			name = fmt.Sprintf("file_%d", idx)
		}
		mediaType := strings.TrimSpace(firstNonEmptyStr(fi["media_type"], fi["type"]))
		url := strings.TrimSpace(firstNonEmptyStr(fi["url"], fi["uri"]))
		data := strings.TrimSpace(strVal(fi["data"]))
		rawText := strings.TrimSpace(strVal(fi["raw"]))

		if url != "" {
			parts = append(parts, &sdka2a.Part{Content: sdka2a.URL(url), Filename: name, MediaType: mediaType})
		}
		if data != "" {
			parts = append(parts, &sdka2a.Part{Content: sdka2a.Data{Value: data}, Filename: name, MediaType: mediaType})
		}
		if rawText != "" {
			rawBytes, err := base64.StdEncoding.DecodeString(rawText)
			if err != nil {
				rawBytes = []byte(rawText)
			}
			parts = append(parts, &sdka2a.Part{Content: sdka2a.Raw(rawBytes), Filename: name, MediaType: mediaType})
		}
	}
	return parts
}

// pendingRequest 在途 A2A 请求的应答队列。
//
// Python: a2a_connect.py _PendingA2ARequest (L38-40)
type pendingRequest struct {
	// ch 内部消息回包队列（有界，满则丢弃告警，偏差 D11）
	ch chan *schema.Message
}

// A2AChannel A2A 入站通道。
//
// 协议栈交给 a2asrv，本类型只做网关桥：pending 配对、内部 Message 构造、HTTP 生命周期。
// Python: a2a_connect.py A2AChannel (L179-500)
type A2AChannel struct {
	// cfg 通道配置
	cfg A2AChannelConfig
	// mu 保护 pending / onMessageCb / srv
	mu sync.RWMutex
	// pending requestID → 应答队列
	pending map[string]*pendingRequest
	// onMessageCb 入站回调（由 ChannelManager.RegisterChannel 注入）
	onMessageCb func(*schema.Message) bool
	// running 运行标志
	running atomic.Bool
	// srv HTTP 服务器（Start 后非 nil）
	srv *http.Server
}

// OnMessage 注册入站回调。
//
// Python: a2a_connect.py on_message (L194-195)
func (ch *A2AChannel) OnMessage(callback func(*schema.Message) bool) {
	ch.mu.Lock()
	defer ch.mu.Unlock()

// Send 把内部回包投递给对应的在途请求；未命中静默丢弃。
//
// Python: a2a_connect.py send (L305-309)
func (ch *A2AChannel) Send(_ context.Context, msg *schema.Message) error {
	if msg == nil {
		return nil
	}
	ch.mu.RLock()
	pending := ch.pending[msg.ID]
	ch.mu.RUnlock()
	if pending == nil {
		return nil
	}
	select {
	case pending.ch <- msg:
	default:
		// 偏差 D11：有界队列满则丢弃并告警，避免阻塞出站派发
		logger.Warn(logComponent).
			Str("msg_id", msg.ID).
			Msg("A2A pending queue full, dropping message")
	}
	return nil
}

// DispatchA2ARequest 构造内部 chat.send 请求并交给入站回调，返回应答队列。
//
// Python: a2a_connect.py dispatch_a2a_request (L311-343)
func (ch *A2AChannel) DispatchA2ARequest(
	_ context.Context,
	requestID, sessionID, query string,
	files []map[string]any,
	metadata map[string]any,
) (*pendingRequest, error) {
	ch.mu.Lock()
	cb := ch.onMessageCb
	pending := &pendingRequest{ch: make(chan *schema.Message, pendingBufferSize)}
	ch.pending[requestID] = pending
	ch.mu.Unlock()

	if cb == nil {
		ch.ClearPendingRequest(requestID)
		return nil, fmt.Errorf("A2AChannel on_message callback is not set")
	}

	rawParams, err := json.Marshal(BuildRequestParams(query, files))
	if err != nil {
		ch.ClearPendingRequest(requestID)
		return nil, fmt.Errorf("marshal request params: %w", err)
	}

	msg := &schema.Message{
		ID:        requestID,
		Type:      schema.MessageTypeReq,
		ChannelID: ch.ChannelID(),
		SessionID: sessionID,
		Params:    rawParams,
		Timestamp: float64(time.Now().UnixNano()) / 1e9,
		OK:        true,
		ReqMethod: schema.ReqMethodChatSend,
		IsStream:  true,
		Metadata:  metadata,
	}
	cb(msg)
	return pending, nil
}

// ClearPendingRequest 移除在途请求（不关闭队列，避免 Send 与关闭竞态）。
//
// Python: a2a_connect.py clear_pending_request (L345-346)
func (ch *A2AChannel) ClearPendingRequest(requestID string) {
	ch.mu.Lock()
	defer ch.mu.Unlock()
	delete(ch.pending, requestID)
}

// Stop 停止通道：关闭 HTTP 服务并向所有在途请求投递唤醒消息。
//
// Python: a2a_connect.py stop (L276-303)
func (ch *A2AChannel) Stop(ctx context.Context) error {
	ch.running.Store(false)

	ch.mu.Lock()
	srv := ch.srv
	ch.mu.Unlock()

	var shutdownErr error
	if srv != nil {
		shutdownErr = srv.Shutdown(ctx)
	}

	ch.mu.Lock()
	for id, pending := range ch.pending {
		wakeup := &schema.Message{
			ID:        "a2a_shutdown",
			Type:      schema.MessageTypeEvent,
			ChannelID: ch.ChannelID(),
			Timestamp: float64(time.Now().UnixNano()) / 1e9,
			OK:        false,
			Payload:   map[string]any{"error": shutdownWakeupError, "is_complete": true},
			EventType: schema.EventTypeChatError,
		}
		select {
		case pending.ch <- wakeup:
		default:
			logger.Warn(logComponent).Str("request_id", id).Msg("failed to deliver A2A shutdown wakeup")
		}
	}
	ch.pending = make(map[string]*pendingRequest)
	ch.mu.Unlock()

	logger.Info(logComponent).Msg("A2A channel stopped")
	return shutdownErr
}

// C 返回应答队列的只读通道（executor 与测试消费用）。
func (p *pendingRequest) C() <-chan *schema.Message { return p.ch }

// collectEvents 收集序列产出的事件类型标签
func collectEvents(seq func(func(sdka2a.Event, error) bool)) []string {
	var kinds []string
	seq(func(evt sdka2a.Event, err error) bool {
		if err != nil {
			kinds = append(kinds, "error")
			return false
		}
		switch v := evt.(type) {
		case *sdka2a.Task:
			kinds = append(kinds, "task:"+string(v.Status.State))
		case *sdka2a.TaskStatusUpdateEvent:
			kinds = append(kinds, "status:"+string(v.Status.State))
		case *sdka2a.TaskArtifactUpdateEvent:
			label := "artifact"
			if v.LastChunk {
				label += ":last"
			}
			if v.Append {
				label += ":append"
			}
			kinds = append(kinds, label)
		default:
			kinds = append(kinds, "other")
		}
		return true
	})
	return kinds
}

// ──────────────────────────── 结构体 ────────────────────────────

// a2aExecutor A2A 执行器：把内部消息事件流翻译成 A2A 事件序列。
//
// 协议侧（任务建档、状态机、SSE）由 a2asrv 负责，本类型只做桥接。
// Python: a2a_connect.py _A2AAgentExecutor (L43-176)
type a2aExecutor struct {
	// ch 所属通道
	ch *A2AChannel
}

// ──────────────────────────── 导出函数 ────────────────────────────

// executorFor 构造通道对应的 A2A 执行器。
func executorFor(ch *A2AChannel) a2asrv.AgentExecutor { return &a2aExecutor{ch: ch} }

// Execute 执行一次任务：产出 A2A 事件序列。
//
// 序列：Task(submitted) → StatusUpdate(WORKING) → Artifact... → StatusUpdate(终态)。
// 首事件必须是 Task（偏差 D1，a2asrv 强制）。
// Python: _A2AAgentExecutor.execute (L49-162)
func (e *a2aExecutor) Execute(ctx context.Context, ec *a2asrv.ExecutorContext) iter.Seq2[sdka2a.Event, error] {
	return func(yield func(sdka2a.Event, error) bool) {
		requestID := string(ec.TaskID)
		defer e.ch.ClearPendingRequest(requestID)

		if !yield(sdka2a.NewSubmittedTask(ec, ec.Message), nil) {
			return
		}

		query, files := MapA2APartsToParams(ec.Message)
		if query == "" {
			query = joinTextParts(ec.Message)
		}
		if query == "" {
			yield(sdka2a.NewStatusUpdateEvent(ec, sdka2a.TaskStateFailed, nil), nil)
			return
		}

		if !yield(sdka2a.NewStatusUpdateEvent(ec, sdka2a.TaskStateWorking, nil), nil) {
			return
		}

		pending, err := e.ch.DispatchA2ARequest(ctx, requestID, ec.ContextID, query, files, ec.Metadata)
		if err != nil {
			logger.Error(logComponent).Err(err).Str("request_id", requestID).Msg("A2A dispatch failed")
			yield(sdka2a.NewStatusUpdateEvent(ec, sdka2a.TaskStateFailed, nil), nil)
			return
		}

		artifactStarted := false
		for {
			var msg *schema.Message
			select {
			case <-ctx.Done():
				return
			case msg = <-pending.C():
			}
			if msg == nil {
				return
			}

			parts := MessageToA2AParts(msg, false)
			terminal := IsTerminalMessage(msg)
			if terminal && len(parts) == 0 && msg.EventType == schema.EventTypeChatError {
				// 错误响应保持可读
				parts = MessageToA2AParts(msg, true)
			}
			parts = filterSentinelParts(parts)

			if len(parts) > 0 {
				artifact := &sdka2a.Artifact{
					ID:       sdka2a.ArtifactID(requestID + "_response"),
					Name:     "response",
					Parts:    sdka2a.ContentParts(parts),
					Metadata: msg.Metadata,
				}
				if !yield(&sdka2a.TaskArtifactUpdateEvent{
					ContextID: ec.ContextID,
					TaskID:    ec.TaskID,
					Artifact:  artifact,
					Append:    artifactStarted,
					LastChunk: terminal,
				}, nil) {
					return
				}
				artifactStarted = true
			}

			if terminal {
				yield(sdka2a.NewStatusUpdateEvent(ec, terminalState(msg.EventType), nil), nil)
				return
			}
		}
	}
}

// Cancel 响应客户端取消请求。
//
// Python: _A2AAgentExecutor.cancel (L164-176)
func (e *a2aExecutor) Cancel(_ context.Context, ec *a2asrv.ExecutorContext) iter.Seq2[sdka2a.Event, error] {
	return func(yield func(sdka2a.Event, error) bool) {
		defer e.ch.ClearPendingRequest(string(ec.TaskID))
		// 任务尚未建档时先补 Task（a2asrv 要求首事件为 Task/Message）
		if ec.StoredTask == nil && ec.Message != nil {
			if !yield(sdka2a.NewSubmittedTask(ec, ec.Message), nil) {
				return
			}
		}
		yield(sdka2a.NewStatusUpdateEvent(ec, sdka2a.TaskStateCanceled, nil), nil)
	}
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// terminalState 把内部终态事件映射为 A2A 任务状态。
//
// Python: _A2AAgentExecutor.execute (L136-149)
func terminalState(eventType schema.EventType) sdka2a.TaskState {
	switch eventType {
	case schema.EventTypeChatError:
		return sdka2a.TaskStateFailed
	case schema.EventTypeChatInterruptResult:
		return sdka2a.TaskStateCanceled
	default:
		return sdka2a.TaskStateCompleted
	}
}

// filterSentinelParts 过滤完成哨兵文本 Part。
//
// Python: _A2AAgentExecutor.execute (L113-119)
func filterSentinelParts(parts []*sdka2a.Part) []*sdka2a.Part {
	if len(parts) == 0 {
		return nil
	}
	filtered := make([]*sdka2a.Part, 0, len(parts))
	for _, p := range parts {
		if p == nil {
			continue
		}
		if IsCompletionSentinelText(strings.TrimSpace(p.Text())) {
			continue
		}
		filtered = append(filtered, p)
	}
	return filtered
}

// Start 启动 A2A HTTP 服务（未启用则直接返回）。
//
// 对齐 Python start (L197-274)：绑定失败或服务提前退出都向调用方返错。
// ctx 不参与（生命周期由 Stop 控制），形参保持接口一致。
func (ch *A2AChannel) Start(_ context.Context) error {
	if ch.running.Load() {
		return nil
	}
	if !ch.cfg.Enabled {
		logger.Info(logComponent).Msg("A2A channel disabled by config")
		return nil
	}

	srv := &http.Server{
		Handler:           ch.newMux(),
		ReadHeaderTimeout: 10 * time.Second,
	}
	addr := fmt.Sprintf("%s:%d", ch.cfg.Host, ch.cfg.Port)
	ln, err := net.Listen("tcp", addr)
	if err != nil {
		return fmt.Errorf("listen %s: %w", addr, err)
	}

	ch.mu.Lock()
	ch.srv = srv
	ch.mu.Unlock()

	errCh := make(chan error, 1)
	go func() { errCh <- srv.Serve(ln) }()

	// 存活检查窗口，对齐 Python await asyncio.sleep(0.2) + done() 判定
	select {
	case serveErr := <-errCh:
		if serveErr != nil && !errors.Is(serveErr, http.ErrServerClosed) {
			ch.mu.Lock()
			ch.srv = nil
			ch.mu.Unlock()
			return serveErr
		}
	case <-time.After(startProbeWindow):
	}

	ch.running.Store(true)
	logger.Info(logComponent).
		Str("url", fmt.Sprintf("http://%s%s", addr, ch.cfg.RPCPath)).
		Msg("A2A channel started")
	return nil
}

// newMux 组装三条路由：JSON-RPC / Agent Card / Extended Agent Card。
func (ch *A2AChannel) newMux() *http.ServeMux {
	card := BuildAgentCard(&ch.cfg)

	mux := http.NewServeMux()
	mux.Handle(ch.cfg.RPCPath, a2asrv.NewJSONRPCHandler(requestHandler))
	mux.Handle(ch.cfg.CardPath, a2asrv.NewStaticAgentCardHandler(card))

			a2asrv.ExtendedAgentCardProducerFn(func(context.Context, *sdka2a.GetExtendedAgentCardRequest) (*sdka2a.AgentCard, error) {
				return card, nil
			}),
		))
	}
	return mux
}
