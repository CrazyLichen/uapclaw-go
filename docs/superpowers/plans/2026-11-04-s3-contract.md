// IsPromptAttachmentUser 判断消息是否为附件投递产生的伪用户消息。
//
// Python: agent_evolving/trajectory/messages.py:104-119 `_is_prompt_attachment_user`
// 判定顺序：metadata 标记优先；其次内容含 "<system-reminder>"（兼容丢失 metadata 的形态）。
func IsPromptAttachmentUser(msg llmschema.BaseMessage) bool {
	if msg == nil {
		return false
	}
	if meta := msg.GetMetadata(); meta != nil && meta[MetadataKeyAttachmentHistory] == true {
		return true
	}
	return strings.Contains(msg.GetContent().Text(), "<system-reminder>")
}

// IsPromptAttachmentHistory 判断是否为附件历史消息（marker + mode 双重判定）。
//
// 用于观测层区分"真 system prompt"与"附件投影后的 system"。
func IsPromptAttachmentHistory(msg llmschema.BaseMessage) bool {
	if msg == nil {
		return false
	}
	meta := msg.GetMetadata()
	if meta == nil || meta[MetadataKeyAttachmentHistory] != true {
		return false
	}
	mode, _ := meta[MetadataKeyMode].(string)
	return mode == ModeSnapshot || mode == ModeDelta
}
```

package trajectory

import (
	llmschema "github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/llm/schema"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/harness/prompts"

// ──────────────────────────── 导出函数 ────────────────────────────

// TrimPromptToLastUser 只保留最后一条真实用户消息起的片段。
//
// Python: agent_evolving/trajectory/messages.py `_trim_prompt_to_last_user`
// 附件消息（IsPromptAttachmentUser）不算真实用户消息，不得作为切片起点。
func TrimPromptToLastUser(prompt []llmschema.BaseMessage) []llmschema.BaseMessage {
	for i := len(prompt) - 1; i >= 0; i-- {
		msg := prompt[i]
		if msg == nil || msg.GetRole() != llmschema.RoleTypeUser {
			continue
		}
		if prompts.IsPromptAttachmentUser(msg) {
			continue
		}
		out := make([]llmschema.BaseMessage, len(prompt)-i)
		copy(out, prompt[i:])
		return out
	}
	return prompt
}
```

// attachmentMarkOf 返回附件消息的观测标注；非附件消息返回空。
//
// 上游在新版 semconv 下区分 gen_ai.system_instructions 与 chat history
// （callback_handler.py:1718-1733 / 1915-1930）；本项目使用 OpenLLMetry 的
// gen_ai.prompt.<i>.* 扁平属性，等价做法是在消息条目上加 kind/mode 标注。
func attachmentMarkOf(msg any) (kind string, mode string) {
	base, ok := msg.(llmschema.BaseMessage)
	if !ok || !prompts.IsPromptAttachmentHistory(base) {
		return "", ""
	}
	raw, _ := base.GetMetadata()[prompts.MetadataKeyMode].(string)
	return "prompt_attachment_history", raw
}
