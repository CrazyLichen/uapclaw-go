```go
// IntelliRouterDeployment IntelliRouter 背后的单个物理部署。
// 镜像 intelli_router.Deployment：单端点服务单模型，自带凭证与限流预算。
// 与 ModelPoolEntry（分配器逐成员下发）不同：deployments 永不单独分配，
// 整份列表交给每个成员的客户端，由路由器逐请求挑选、失败换台。
// Python: IntelliRouterDeployment (pool.py:241)
type IntelliRouterDeployment struct {
	// ModelName 部署服务的模型名
	ModelName string `json:"model_name"`
	// APIKey 部署 API Key
	APIKey string `json:"api_key"`
	// APIBase Provider 根 URL —— 不要带 /v1 后缀！
	// 这是唯一与本仓其余 api_base 约定不同的字段：ModelClientConfig.api_base
	// 指向 OpenAI 兼容 API base（通常以 /v1 结尾），而 IntelliRouter 的各
	// provider 适配器自行拼路径（OpenAI 适配器拼 f"{api_base}/v1/chat/completions"）。
	// 传了 /v1 结尾的 base 会拼出 /v1/v1/chat/completions，且故障形态不是 404
	// 而是路由器报 ResponseNotRead（错误路径未先消费流就读 body）。
	// 正确示例：https://api.deepseek.com
	APIBase string `json:"api_base"`
	// ID 稳定部署标识，出现在路由器统计与日志中
	ID string `json:"id,omitempty"`
	// ModelID 编译快照中的业务模型标识
	ModelID string `json:"model_id,omitempty"`
	// Provider intelli_router 解释的上游 provider 名（必填）
	Provider string `json:"provider"`
	// EndpointProfile 端点档案名（可选）
	EndpointProfile string `json:"endpoint_profile,omitempty"`
	// CustomHeaders 自定义请求头
	CustomHeaders map[string]string `json:"custom_headers,omitempty"`
	// FallbackTag 模型组成员点名（tag-filtered）用的标签
	FallbackTag string `json:"fallback_tag,omitempty"`
	// ModelDescription 模型描述
	ModelDescription string `json:"model_description,omitempty"`
	// RequestDefaults 模型层编译出的部署级请求默认值
	RequestDefaults map[string]any `json:"request_defaults,omitempty"`
	// TPM 每分钟 token 预算（限流感知策略用）
	TPM *int `json:"tpm,omitempty"`
	// RPM 每分钟请求数预算
	RPM *int `json:"rpm,omitempty"`
	// Tags 部署标签列表（ToDeploymentDict 时 tags[0] 兜底 FallbackTag）
	Tags []string `json:"tags,omitempty"`
	// Timeout 部署级超时；未设回退路由器级
	Timeout *float64 `json:"timeout,omitempty"`
	// VerifySSL 部署级 TLS 校验；未设回退路由器级（仅显式 true/false 才下发）
	VerifySSL *bool `json:"verify_ssl,omitempty"`
}

// ToDeploymentDict 渲染 IntelliRouterModelClient 消费的 wire dict。
// 可选字段省略而非发 null，让客户端层应用自己的回退——尤其 verify_ssl，
// 只有键缺失时才回退路由器级值。
// Python: to_deployment_dict (pool.py:297)
func (d *IntelliRouterDeployment) ToDeploymentDict() map[string]any {
	payload := map[string]any{
		"model_name": d.ModelName,
		"api_key":    d.APIKey,
		"api_base":   d.APIBase,
		"provider":   d.Provider,
	}
	if len(d.Tags) > 0 {
		payload["fallback_tag"] = d.Tags[0]
	}
	optional := map[string]any{}
	putIf := func(k string, v any) { if v != nil && !isZeroAny(v) { optional[k] = v } }
	putIf("id", strOrNil(d.ID))
	// ...（model_id/endpoint_profile/custom_headers/fallback_tag/model_description/
	//     request_defaults/tpm/rpm/timeout/verify_ssl 逐字段同构，指针字段 nil 即省略，
	//     map/string 字段空即省略）
	payload = mergePayload(payload, optional)
	return payload
}
```

实现注：`isZeroAny`/`strOrNil`/`mergePayload` 若包内已有等价助手则复用，没有则在本文件非导出函数区新增；**严禁**新开 `xxx_helpers.go` 文件（规则 9）。

```go
// entrySignature 条目全量配置的规范签名（剔除 model_id 与 metadata.client.client_id）。
// 两个签名相同的条目描述同一逻辑端点+同一 auth/请求旋钮——foundation 客户端缓存
// 可安全复用。client_id 是每次 dump 的新 uuid，计入签名会让暂停团队每次请求被
// 拆建重盘（上游注释明确记载的教训）。签名用 JSON 序列化 + 键排序复刻
// Python json.dumps(sort_keys=True)，不依赖 struct 字段序。
// Python: _entry_signature (pool.py:544)
func entrySignature(entry ModelPoolEntry) string {
	payload := map[string]any{}
	raw, _ := json.Marshal(entry)
	_ = json.Unmarshal(raw, &payload)
	delete(payload, "model_id")
	if client, ok := payload["metadata"].(map[string]any)["client"].(map[string]any); ok {
		delete(client, "client_id")
	}
	sorted, _ := marshalSortedJSON(payload)
	return sorted
}

// InheritPoolIds 把 current_pool 的 model_id 按 bit-exact 签名带入 new_pool。
// Python: inherit_pool_ids (pool.py:566)
func InheritPoolIds(currentPool, newPool []ModelPoolEntry) []ModelPoolEntry {
	oldBySig := map[string][]string{} // 签名 → model_id 队列（按池序）
	for _, e := range currentPool {
		sig := entrySignature(e)
		oldBySig[sig] = append(oldBySig[sig], e.ModelID)
	}
	result := make([]ModelPoolEntry, 0, len(newPool))
	for _, e := range newPool {
		sig := entrySignature(e)
		if ids := oldBySig[sig]; len(ids) > 0 {
			e.ModelID = ids[0]
			oldBySig[sig] = ids[1:]
		}
		result = append(result, e)
	}
	return result
}
```

`marshalSortedJSON`：map → 递归键排序输出（或直接 `json.Marshal` Go map——Go encoding/json 对 map 已按键排序输出，**可直接用 `json.Marshal`**，在注释注明这一点）。
