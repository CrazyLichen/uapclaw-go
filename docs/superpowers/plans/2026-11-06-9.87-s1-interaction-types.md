// 本文件对应 Python openjiuwen/harness/schema/interaction.py，
// 提供会话级交互监督器使用的领域对象。
//
// 这些对象不包含任何 JiuwenSwarm 协议类型：一个会话要么收到完整的用户
// inputs，要么收到"继续执行当前持续目标"的指令；只有拿到唯一输出租约的
// 调用方才会得到输出消费者。
package schema

import (
	"github.com/uapclaw/uapclaw-go/internal/common/utils"
)

// ──────────────────────────── 结构体 ────────────────────────────

// InteractionPhase 会话级交互循环的内部生命周期相位。
// Python: InteractionPhase (interaction.py L28)
type InteractionPhase string

// InputDispatchMode 轮次活跃期间的用户输入路由策略。
// Python: InputDispatchMode (interaction.py L36)
type InputDispatchMode string

// RoundWorkKind work item 类型。
// Python: RoundWorkItem.kind (Literal["user", "goal"])
type RoundWorkKind string

// SendInputRequest 宿主投递输入的请求。
//
// inputs["query"] 是唯一的用户文本来源；整个 mapping 原样保留，
// 使既有 DeepAgent 调用元数据（会话 id、可信目录等）随 work item 传递。
// Python: SendInputRequest (interaction.py L43)
type SendInputRequest struct {
	// RequestID 请求标识
	RequestID string
	// Inputs 完整调用载荷（含 query）
	Inputs map[string]any
	// Mode 分发模式（仅对用户文本生效；InteractiveInput 恢复请求忽略）
	Mode InputDispatchMode
}

// RoundContext 单个 work item 的执行上下文。
// Python: RoundWorkItem.context (Dict[str, object])
//
// ⚠️ 必须用 NewUserWorkItem / NewGoalWorkItem 构造：ResetLoop 的默认值
// 由工厂按 Python 语义（默认 True）显式赋值，零值 false 不代表"未设置"。
type RoundContext struct {
	// GoalID 目标标识（仅 kind=goal 有值）
	GoalID string
	// Revision 目标修订号（上游默认 0，0 是合法值而非"未设置"）
	Revision int
	// SessionID 会话标识（仅 kind=goal 有值）
	SessionID string
	// ResetLoop 执行前是否重置任务循环（迭代计数 + 停止条件状态）
	ResetLoop bool
}

// RoundWorkItem 交互监督器消费的一个工作单元。
// Python: RoundWorkItem (interaction.py L59)
type RoundWorkItem struct {
	// Kind 类型（user / goal）
	Kind RoundWorkKind
	// RequestID 宿主请求标识（goal 轮为空）
	RequestID string
	// Inputs 原样传给 invoke 的载荷
	Inputs map[string]any
	// Context 执行上下文
	Context RoundContext
	// IsFollowUp 是否为同一轮的续跑输入
	IsFollowUp bool
}

// ActiveInteractionRound 当前由监督器执行的 work item 记账。
// Python: ActiveInteractionRound (interaction.py L119)
type ActiveInteractionRound struct {
	// Work 当前 work item
	Work *RoundWorkItem
	// TaskID 轮次任务标识
	TaskID string
}

// ──────────────────────────── 常量 ────────────────────────────

const (
	// InteractionPhaseIdle 空闲（无输出消费者或无待办）
	InteractionPhaseIdle InteractionPhase = "idle"
	// InteractionPhaseRunning 有轮次在执行
	InteractionPhaseRunning InteractionPhase = "running"
	// InteractionPhaseTerminated 已终止（不可逆）
	InteractionPhaseTerminated InteractionPhase = "terminated"
)

const (
	// InputDispatchModeFollowUp 作为后续输入排队
	InputDispatchModeFollowUp InputDispatchMode = "follow_up"
	// InputDispatchModeSteer 直接塞入当前活跃轮的转向队列
	InputDispatchModeSteer InputDispatchMode = "steer"
)

const (
	// RoundWorkKindUser 宿主投递的用户轮
	RoundWorkKindUser RoundWorkKind = "user"
	// RoundWorkKindGoal 由 openjiuwen 全权持有的目标轮
	RoundWorkKindGoal RoundWorkKind = "goal"
)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewUserWorkItem 由宿主输入构造隔离的 user work item。
// Python: RoundWorkItem.user (interaction.py L69)
func NewUserWorkItem(requestID string, inputs map[string]any, isFollowUp bool, resetLoop bool) *RoundWorkItem {
	return &RoundWorkItem{
		Kind:       RoundWorkKindUser,
		RequestID:  requestID,
		Inputs:     utils.DeepCopyMap(inputs),
		Context:    RoundContext{ResetLoop: resetLoop},
		IsFollowUp: isFollowUp,
	}
}

// NewGoalWorkItem 构造由 openjiuwen 全权持有的 goal work item。
// Python: RoundWorkItem.goal (interaction.py L87)
func NewGoalWorkItem(inputs map[string]any, goalID string, revision int, sessionID string) *RoundWorkItem {
	return &RoundWorkItem{
		Kind:      RoundWorkKindGoal,
		RequestID: "",
		Inputs:    utils.DeepCopyMap(inputs),
		Context: RoundContext{
			GoalID:    goalID,
			Revision:  revision,
			SessionID: sessionID,
			ResetLoop: true,
		},
	}
}

// Query 返回任务循环使用的查询，刻意只从 Inputs 派生。
// Python: RoundWorkItem.query (interaction.py L109)
func (w *RoundWorkItem) Query() any {
	if w == nil || w.Inputs == nil {
		return nil
	}
	return w.Inputs["query"]
}

// ResetLoop 返回执行前是否重置任务循环。
// Python: RoundWorkItem.reset_loop (interaction.py L114)
func (w *RoundWorkItem) ResetLoop() bool {
	if w == nil {
		return true
	}
	return w.Context.ResetLoop
}

// RunKind 返回运行类型。
// Python: ActiveInteractionRound.run_kind (L127)
func (r *ActiveInteractionRound) RunKind() RoundWorkKind { return r.Work.Kind }

// RunContext 返回运行上下文。
// Python: ActiveInteractionRound.run_context (L131)
func (r *ActiveInteractionRound) RunContext() RoundContext { return r.Work.Context }

// GoalID 返回目标标识（仅 goal 轮有值）。
// Python: ActiveInteractionRound.goal_id (L135)
func (r *ActiveInteractionRound) GoalID() string { return r.Work.Context.GoalID }

// Revision 返回目标修订号（仅 goal 轮有值）。
// Python: ActiveInteractionRound.revision (L139)
func (r *ActiveInteractionRound) Revision() int { return r.Work.Context.Revision }
```

	if s.closed.Load() {
		return nil, false

	item, ok := s.driver.NextOutput(ctx, s.lease)

		return nil, false

	return func(yield func(stream.Schema) bool) {
		for {
			item, ok := s.Next(ctx)
			if !ok {
				return
			}
			if !yield(item) {
				return
			}
		}
	}
}

// Python: OutputLeaseManager.attach (interaction.py L259)
func (m *OutputLeaseManager) Attach() *OutputLease {
	m.mu.Lock()
	defer m.mu.Unlock()
	if m.closed || (m.lease != nil && !m.lease.IsClosed()) {
		return nil
	}
	lease := NewOutputLease(uuid.NewString())
	m.lease = lease
	return lease
}

func (m *OutputLeaseManager) Detach(token string, discardBuffer bool) bool {
	m.mu.Lock()
	defer m.mu.Unlock()
	lease := m.lease
	if lease == nil || lease.Token != token {
		return false
	}
	m.lease = nil
	lease.Close()
	if discardBuffer {
		m.drainQueueLocked()
	}
	return true
}

	m.mu.Lock()
	defer m.mu.Unlock()
	m.closed = true
	if m.lease != nil {
		m.lease.Close()
		m.lease = nil
	}
	m.drainQueueLocked()
}

// Python: OutputLeaseManager.emit (interaction.py L287)
func (m *OutputLeaseManager) Emit(item stream.Schema, expectedToken string) {
	m.mu.Lock()
	defer m.mu.Unlock()

	lease := m.lease
	if m.closed || lease == nil || lease.IsClosed() || lease.Finishing.Load() {
		return
	}
	if expectedToken != "" && lease.Token != expectedToken {
		return
	}

	select {
	case m.queue <- leaseFrame{schema: item}:
	default:
		m.dropped.Add(1)
		m.logDropThrottled(lease.Token)
	}
}

// NextItem 取一条输出；租约作废、上下文取消或读到结束哨兵时返回 false。
// Python: OutputLeaseManager.next_item (interaction.py L297)
func (m *OutputLeaseManager) NextItem(ctx context.Context, lease *OutputLease) (stream.Schema, bool) {
	if lease == nil {
		return nil, false
	}

	m.mu.Lock()
	current := m.lease
	if current != lease || lease.IsClosed() {
		m.mu.Unlock()
		return nil, false
	}
	m.mu.Unlock()

	for {
		select {
		case <-lease.closed:
			return nil, false
		case <-ctx.Done():
			return nil, false
		case frame := <-m.queue:
			if !frame.end {
				return frame.schema, true
			}
			m.mu.Lock()
			if m.lease == lease {
				m.lease = nil
				lease.Close()
			}
			m.mu.Unlock()
			return nil, false
		}
	}
}

// FinishCurrent 在已转发分片排空后结束当前流。
// Python: OutputLeaseManager.finish_current (interaction.py L278)
func (m *OutputLeaseManager) FinishCurrent() {
	m.mu.Lock()
	defer m.mu.Unlock()

	lease := m.lease
	if lease == nil || lease.IsClosed() || lease.Finishing.Load() {
		return
	}
	lease.Finishing.Store(true)
	m.enqueueEndLocked()
}

func (m *OutputLeaseManager) DroppedCount() int64 { return m.dropped.Load() }

		Str("lease_token", token).

package task_loop

import (
	"errors"
	"fmt"

	hschema "github.com/uapclaw/uapclaw-go/internal/agentcore/harness/schema"
)

// ──────────────────────────── 结构体 ────────────────────────────

// EventManager 把 user 与 goal work 放进同一个调度域。
//
// 监督器是唯一的弹出消费者；两个独立队列实现 user 优先，而不暴露两套宿主流 API。
//
// ⚠️ 并发约定：本类型**不含内部锁**。所有方法要求调用方持有 DeepAgent 的
// interaction control lock（spec K3 / D4）。Python 靠 asyncio 单线程满足此约定，
// Go 由 S2 的 supervisor 在 NextWork/MarkStarted/PushUser/MarkFinished 四处加锁。
// Python: EventManager (event_manager.py L15)
type EventManager struct {
	// userQueue 用户轮队列（高优先级）
	userQueue []*hschema.RoundWorkItem
	// goalQueue 目标轮队列（低优先级）
	goalQueue []*hschema.RoundWorkItem
	// dequeued 已弹出但尚未开始的 work
	dequeued *hschema.RoundWorkItem
	// active 正在执行的 work
	active *hschema.RoundWorkItem
}

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// ErrWorkKindMismatch work item 类型与目标队列不匹配
	// Python: ValueError("push_user requires a user RoundWorkItem") (L30)
	ErrWorkKindMismatch = errors.New("work item kind mismatch")
)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewEventManager 创建调度管理器。
// Python: EventManager() (event_manager.py L22)
func NewEventManager() *EventManager {
	return &EventManager{}
}

// PushUser 入队 user work；kind 不匹配时返回 ErrWorkKindMismatch。
// Python: EventManager.push_user (event_manager.py L28)
func (m *EventManager) PushUser(work *hschema.RoundWorkItem) error {
	if work.Kind != hschema.RoundWorkKindUser {
		return fmt.Errorf("%w: push_user requires a user RoundWorkItem", ErrWorkKindMismatch)
	}
	m.userQueue = append(m.userQueue, work)
	return nil
}

// NextWork 弹出下一个待执行 work：user 队列优先，其次 goal 队列。
// Python: EventManager.next_work (event_manager.py L50)
func (m *EventManager) NextWork() *hschema.RoundWorkItem {
	if len(m.userQueue) > 0 {
		m.dequeued = m.userQueue[0]
		m.userQueue = m.userQueue[1:]
		return m.dequeued
	}
	if len(m.goalQueue) > 0 {
		m.dequeued = m.goalQueue[0]
		m.goalQueue = m.goalQueue[1:]
		return m.dequeued
	}
	return nil
}

// MarkStarted 标记 work 开始执行：清 dequeued、置 active。
// Python: EventManager.mark_started (event_manager.py L59)
func (m *EventManager) MarkStarted(work *hschema.RoundWorkItem) {
	if m.dequeued == work {
		m.dequeued = nil
	}
	m.active = work
}

// MarkFinished 标记 work 结束：清 active。
// Python: EventManager.mark_finished (event_manager.py L64)
func (m *EventManager) MarkFinished(work *hschema.RoundWorkItem) {
	if m.active == work {
		m.active = nil
	}
}

// HasPendingWork 返回是否有待办（不含正在执行）。
// Python: EventManager.has_pending_work (event_manager.py L68)
func (m *EventManager) HasPendingWork() bool {
	return len(m.userQueue) > 0 || len(m.goalQueue) > 0
}

// HasPendingUserWork 返回是否有待办 user work。
// Python: EventManager.has_pending_user_work (event_manager.py L71)
func (m *EventManager) HasPendingUserWork() bool {
	return len(m.userQueue) > 0
}

// ActiveWork 返回正在执行的 work（无则 nil）。
// Python: EventManager.active_work (event_manager.py L141)
func (m *EventManager) ActiveWork() *hschema.RoundWorkItem { return m.active }

			work.Kind == hschema.RoundWorkKindGoal &&
			work.Context.GoalID == goalID &&
			work.Context.Revision == revision
	}
	for _, work := range m.goalQueue {
		if matches(work) {
			return true
		}
	}
	return matches(m.dequeued) || matches(m.active)
}

// HasRunningGoal 返回指定目标是否已有 dequeued/active 的尝试。
//
// 忽略 revision 与 pending 队列：pause/resume 用它识别"仍在收尾的尝试"，
// 即使记录 revision 已被提升，且被丢弃的排队延续不计入。
// Python: EventManager.has_running_goal (event_manager.py L89)
func (m *EventManager) HasRunningGoal(goalID string) bool {
	matches := func(work *hschema.RoundWorkItem) bool {
		return work != nil &&
			work.Kind == hschema.RoundWorkKindGoal &&
			work.Context.GoalID == goalID
	}
	return matches(m.dequeued) || matches(m.active)
}

// DiscardGoalWork 丢弃被清除/替换记录所属的待办 goal work。
//
// goalID 为空串表示丢弃该会话的全部目标待办。
// Python: EventManager.discard_goal_work (event_manager.py L106)
func (m *EventManager) DiscardGoalWork(sessionID string, goalID string) int {
	before := len(m.goalQueue)
	kept := make([]*hschema.RoundWorkItem, 0, before)
	for _, work := range m.goalQueue {
		sameSession := work.Context.SessionID == sessionID
		sameGoal := goalID == "" || work.Context.GoalID == goalID
		if sameSession && sameGoal {
			continue
		}
		kept = append(kept, work)
	}
	m.goalQueue = kept
	discarded := before - len(m.goalQueue)
	if discarded > 0 {
		logger.Info(logger.ComponentAgentCore).
			Str("event_type", "INTERACTION_GOAL_DISCARDED").
			Int("count", discarded).
			Str("session_id", sessionID).
			Str("goal_id", goalID).
			Msg("discarded pending goal work items")
	}
	return discarded
}

// DiscardAllWork 在输出租约断开时丢弃两个队列的待办。
//
// 持久化的 GoalRecord 刻意保留：之后一次 GOAL 发送会在前端重新 attach 后
// 重建下一个 work item。
// Python: EventManager.discard_all_work (event_manager.py L132)
func (m *EventManager) DiscardAllWork() {
	m.userQueue = nil
	m.goalQueue = nil
}
```
