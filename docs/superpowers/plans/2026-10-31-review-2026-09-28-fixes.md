- [ ] **Step 2: InMemory 实现**

```go
// CompareAndDelete 原子性地删除 key，仅当当前值等于 expected 时成功。
func (s *InMemoryKVStore) CompareAndDelete(_ context.Context, key string, expected []byte) (bool, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	e, ok := s.store[key]
	if !ok {
		return false, nil
	}
	now := time.Now().Unix()
	if e.expiryTs != 0 && now > e.expiryTs {
		// 已过期视为不存在
		delete(s.store, key)
		return false, nil
	}
	if !bytes.Equal(e.value, expected) {
		return false, nil
	}
	delete(s.store, key)
	return true, nil
}
```

```go
// compareAndDeleteScript 原子比较删除脚本：值相等才删除。
const compareAndDeleteScript = `
if redis.call('get', KEYS[1]) == ARGV[1] then
  return redis.call('del', KEYS[1])
end
return 0
`

// CompareAndDelete 原子性地删除 key，仅当当前值等于 expected 时成功。
func (s *RedisStore) CompareAndDelete(ctx context.Context, key string, expected []byte) (bool, error) {
	res, err := s.client.Eval(ctx, compareAndDeleteScript, []string{key}, string(expected)).Int()
	if err != nil {
		return false, err
	}
	return res == 1, nil
}
```

- [ ] **Step 4: File 实现**

```go
// CompareAndDelete 原子性地删除 key，仅当解包后的值等于 expected 时成功。
// 使用 bbolt 写事务保证 read-compare-delete 三者原子。
func (s *FileKVStore) CompareAndDelete(_ context.Context, key string, expected []byte) (bool, error) {
	matched := false
	err := s.db.Update(func(tx *bolt.Tx) error {
		b := tx.Bucket([]byte(s.bucketName))
		raw := b.Get([]byte(key))
		if raw == nil {
			return nil
		}
		var entry fileEntry
		if err := json.Unmarshal(raw, &entry); err != nil {
			return err
		}
		value, err := base64.StdEncoding.DecodeString(entry.Value)
		if err != nil {
			return err
		}
		if !bytes.Equal(value, expected) {
			return nil
		}
		matched = true
		return b.Delete([]byte(key))
	})
	return matched, err
}
```

- [ ] **Step 5: DbBased 实现**

```go
// CompareAndDelete 原子性地删除 key，仅当当前值等于 expected 时成功。
// 在单个事务内完成读取、解包、比较与删除。
func (s *DbBasedKVStore) CompareAndDelete(ctx context.Context, key string, expected []byte) (bool, error) {
	s.ensureTable()
	matched := false
	err := s.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
		var row KVStoreRow
		err := tx.Where("key = ?", key).First(&row).Error
		if err == gorm.ErrRecordNotFound {
			return nil
		}
		if err != nil {
			return err
		}
		stored := row.Value
		if decoded, ok := decodeExclusiveValue(row.Value); ok {
			stored = decoded
		}
		if !bytes.Equal(stored, expected) {
			return nil
		}
		matched = true
		return tx.Where("key = ?", key).Delete(&KVStoreRow{}).Error
	})
	if err != nil {
		return false, err
	}
	return matched, nil
}
```

- [ ] **Step 2: 改造 Release 与 Acquire**

```go
// Acquire 获取分布式锁。自旋调用 ExclusiveSet 直到成功或 ctx 被取消。
//
// 一个 DistributedLock 实例只代表一个逻辑持有者，不可跨 goroutine 共享：
// 实例内 lockValue 会在每次尝试时重写，共享会让持有者的锁无法被正确释放。
func (l *DistributedLock) Acquire(ctx context.Context) error {
	if l.lockValue != "" {
		return fmt.Errorf("distributed lock already acquired by this instance")
	}
	for {
		l.lockValue = uuid.New().String()
		ok, err := l.store.ExclusiveSet(ctx, l.lockKey, []byte(l.lockValue), l.ttl)
		if err != nil {
			l.lockValue = ""
			return err
		}
		if ok {
			return nil
		}
		select {
		case <-ctx.Done():
			l.lockValue = ""
			return ctx.Err()
		case <-time.After(l.retryDelay):
		}
	}
}

// Release 释放分布式锁。通过 CompareAndDelete 原子校验并删除，防止误删他人持有的锁。
//
// 返回 nil 不代表锁一定已释放：key 已过期或被他人持有时会跳过删除并记录 Warn，
// 与 Python 的 try/except 静默处理对齐。
func (l *DistributedLock) Release(ctx context.Context) error {
	released, err := l.store.CompareAndDelete(ctx, l.lockKey, []byte(l.lockValue))
	l.lockValue = ""
	if err != nil {
		logger.Error(logComponent).Err(err).
			Str("lock_key", l.lockKey).
			Str("lock_value", l.lockValue).
			Str("event_type", logEventMemoryStore).
			Msg("failed to release lock")
		return nil
	}
	if !released {
		logger.Warn(logComponent).
			Str("lock_key", l.lockKey).
			Str("event_type", logEventMemoryStore).
			Msg("lock not released: expired or owned by another")
	}
	return nil
}
```

> 注意：`lock_value` 必须在调用 `CompareAndDelete` **之前**取快照，因为它会被清空：
