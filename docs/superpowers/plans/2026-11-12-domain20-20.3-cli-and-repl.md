// ResolvedMode 一次请求解析出的完整运行模式视图。
// Python: gateway/common/mode_matrix.py:196-221 ResolvedMode（10 字段）
type ResolvedMode struct {
	// ManagerMode AgentManager / adapter 选型用的一级模式
	ManagerMode string
	// SubMode 子模式（normal / plan / team，可空）
	SubMode string
	// CanonicalMode 写回 params["mode"] 的规范值
	CanonicalMode string
	// WorkMode 归一化后的 work / code；历史请求为空
	WorkMode string
	// IsPlan 本次请求是否要求处于 plan
	IsPlan bool
	// IsTeam 本次请求是否集群模式
	IsTeam bool
	// IsCodeProfile 是否使用 CodeAdapter / code 团队 profile
	IsCodeProfile bool
	// Profile 本次请求使用的 profile（normal / code）
	Profile string
	// NormalMode 退出 plan 后应回到的 canonical 模式
	NormalMode string
	// FromWebComposition 是否由 Web 的 mode + work_mode 组合而来
	FromWebComposition bool
}

// DeprecateMode 旧 canonical 静默映射到新 canonical；非旧串原样返回。
// Python: gateway/common/mode_matrix.py:341-358 deprecate_mode
//
// 铁律 3：空/空白串原样返回，不做 canonicalize_mode_text 的空串回落，
// 避免把空值误映射成 agent.work.normal。归一化由调用方在上游完成。
func DeprecateMode(mode string) string {
	if strings.TrimSpace(mode) == "" {
		return mode
	}
	text := CanonicalizeModeText(mode)
	if mapped, ok := deprecatedModes[text]; ok {
		return mapped
	}
	return text
}

// NormalizeWorkMode 把任意来源的 work_mode 归一成 work / code；非法时返回空串。
// Python: gateway/common/mode_matrix.py:237-242 normalize_work_mode
func NormalizeWorkMode(raw any) string {
	s, ok := raw.(string)
	if !ok {
		return ""
	}
	switch strings.ToLower(strings.TrimSpace(s)) {
	case "code":
		return "code"
	case "work":
		return "work"
	default:
		return ""
	}
}

// ReadRequestWorkMode 读取请求中显式携带的 work_mode；缺失或非法时返回空串。
// Python: gateway/common/mode_matrix.py:245-253 read_request_work_mode
//
// 只有 Web 会携带该字段，因此它同时充当「这是 Web 组合模式请求」的判据。
func ReadRequestWorkMode(params map[string]any) string {
	if params == nil {
		return ""
	}
	return NormalizeWorkMode(params["work_mode"])
}

// BaseModeWithoutPlan canonical plan 模式退出后应回到的普通模式。
// Python: gateway/common/mode_matrix.py:330-333 base_mode_without_plan
func BaseModeWithoutPlan(canonical string) string {
	text := CanonicalizeModeText(canonical)
	if strings.HasSuffix(text, ".plan") {
		return strings.TrimSuffix(text, ".plan") + ".normal"
	}
	return text
}

// ResolveRequestMode 解析一次请求的运行模式。
// Python: gateway/common/mode_matrix.py:413-489 resolve_request_mode
//
// 优先走 Web 组合分支；只有「存在合法 work_mode」且「mode 是 Web 支持的基础模式」
// 同时成立才生效。其余取值都交给 legacyResolver。
//
// legacyResolver 对应上游 legacy_resolver 参数（resolve_agent_request_mode），
// 返回 (managerMode, subMode, canonicalMode)。
func ResolveRequestMode(
	params map[string]any,
	legacyResolver func(mode, workMode string) (managerMode, subMode, canonicalMode string),
	workMode string,
) ResolvedMode {
	var rawMode any
	if params != nil {
		rawMode = params["mode"]
	}
	modeText := CanonicalizeModeText(modeTextValue(rawMode))

	work := NormalizeWorkMode(workMode)
	if work == "" {
		work = ReadRequestWorkMode(params)
	}

	// Python: :438-461 web 组合分支
	if work != "" && IsWebComposableMode(modeText) {
		if m, s, c, ok := ComposeWebMode(modeText, work); ok {
			return ResolvedMode{
				ManagerMode:        m,
				SubMode:            s,
				CanonicalMode:      c,
				WorkMode:           work,
				IsPlan:             IsPlanMode(c),
				IsTeam:             IsTeamMode(c),
				IsCodeProfile:      work == "code",
				Profile:            profileOf(work),
				NormalMode:         BaseModeWithoutPlan(c),
				FromWebComposition: true,
			}
		}
	}

	// Python: :463-489 legacy 分支
	m, s, c := legacyResolver(modeText, work)
	return ResolvedMode{
		ManagerMode:        m,
		SubMode:            s,
		CanonicalMode:      c,
		WorkMode:           work,
		IsPlan:             IsPlanMode(c),
		IsTeam:             IsTeamMode(c),
		IsCodeProfile:      m == "code",
		Profile:            profileOf(m),
		NormalMode:         BaseModeWithoutPlan(c),
		FromWebComposition: false,
	}
}

// profileOf web 分支按 work_mode 判、legacy 分支按 manager_mode 判。
// Python: mode_matrix.py:457（work_mode == "code"）与 :484（manager_mode == "code"）
func profileOf(flag string) string {
	if flag == "code" {
		return "code"
	}
	return "normal"

const (
	// endpointProfileOverridesKey config.yaml 顶层键名
	endpointProfileOverridesKey = "endpoint_profile_overrides"

// ──────────────────────────── 导出函数 ────────────────────────────

// GetEndpointProfileOverrides 返回用户配置的 api_base host → endpoint_profile 映射。
// Python: gateway/common/config.py:308-327 get_endpoint_profile_overrides
//
// 读 config.yaml 顶层 endpoint_profile_overrides 键。用于自建网关（如
// DashScope 风格端点）——其思考控制方言（enable_thinking / thinking_budget）
// 无法从域名推断，用户在配置里声明后，保存/校验/运行各路径自动补
// endpoint_profile 透传给 core。
//
// host 归一为小写；host 或 profile 为空的条目丢弃。源码不内置任何 host。
func GetEndpointProfileOverrides() map[string]string {
	cfg := readPermissionsConfigFile()
	raw, _ := cfg[endpointProfileOverridesKey].(map[string]any)
	overrides := make(map[string]string, len(raw))
	for host, profile := range raw {
		hostKey := strings.ToLower(strings.TrimSpace(toString(host)))
		profileValue := strings.TrimSpace(toString(profile))
		if hostKey != "" && profileValue != "" {
			overrides[hostKey] = profileValue
		}
	}
	return overrides
}

// apiBaseHost 提取 api_base 的主机名（小写）；缺 scheme 时补 https。
// Python: reasoning_config.py:20-24 _parse_api_base
func apiBaseHost(apiBase string) string {
	value := strings.TrimSpace(apiBase)
	if value == "" {
		return ""
	}
	if !strings.Contains(value, "://") {
		value = "https://" + value
	}
	parsed, err := url.Parse(value)
	if err != nil {
		return ""
	}
	return strings.ToLower(parsed.Hostname())
}

func cloneSampling(rule map[string]float64) map[string]float64 {
	out := make(map[string]float64, len(rule))
	for k, v := range rule {
		out[k] = v
	}
	return out
}

// ──────────────────────────── 导出函数 ────────────────────────────

// InjectReasoningParams 把 reasoning_level 翻译成 core 的 reasoning 配置。
// Python: gateway/common/reasoning_injector.py:71-88 inject_reasoning_params
//
// 三步（顺序即语义）：
//  1. level = NormalizeReasoningLevel(model_config_dict["reasoning_level"])
//  2. override = ResolveSamplingOverride(api_base || base_url, model_name || model)
//     override 非空则覆盖 runtime_model_config 的 temperature/top_p
//  3. level 为空 → 原样返回（不注入 reasoning）
//     level 非空 → runtime_model_config["reasoning"] = ReasoningConfigForLevel(level)
//
// reasoning_level 是内部提示，无论是否命中都从结果里 pop 掉。
func InjectReasoningParams(modelClientConfig, modelConfigObj map[string]any) map[string]any {
	modelConfigDict := copyStringAnyMap(modelConfigObj)
	level := NormalizeReasoningLevel(modelConfigDict["reasoning_level"])

	runtimeModelConfig := copyStringAnyMap(modelConfigDict)
	delete(runtimeModelConfig, "reasoning_level")

	override := ResolveSamplingOverride(
		firstNonEmptyString(modelClientConfig["api_base"], modelClientConfig["base_url"]),
		firstNonEmptyString(modelClientConfig["model_name"], modelClientConfig["model"]),
	)
	for key, value := range override {
		runtimeModelConfig[key] = value
	}
	if level == "" {
		return runtimeModelConfig
	}
	runtimeModelConfig["reasoning"] = ReasoningConfigForLevel(level)
	return runtimeModelConfig
}

// BuildReasoningModelRequestKwargs 共享的模型构造 kwargs 计算入口。
// Python: gateway/common/reasoning_injector.py:145-165
//
// 被 BuildModelFromEntry / 配置校验 / 图像模态探测预热等多处共用。
func BuildReasoningModelRequestKwargs(
	modelClientConfig, modelConfigObj map[string]any, modelName string,
) map[string]any {
	effectiveModelName := resolveModelName(modelName, modelConfigObj)

	reasoningClientConfig := copyStringAnyMap(modelClientConfig)
	if effectiveModelName != "" {
		reasoningClientConfig["model_name"] = effectiveModelName
	}
	runtimeModelConfig := InjectReasoningParams(reasoningClientConfig, modelConfigObj)

	return buildModelRequestKwargs(effectiveModelName, runtimeModelConfig)
}

// buildModelRequestKwargs 归一 model_config_obj 为 core 的 kwargs 形状。
// Python: reasoning_injector.py:91-138 _build_model_request_kwargs
func buildModelRequestKwargs(modelName string, modelConfigObj map[string]any) map[string]any {
	requestKwargs := copyStringAnyMap(modelConfigObj)

	// Python: :97-105 agentos 老配置用 max_tokens 表示上下文窗口
	if source, _ := requestKwargs["_source"].(string); source == "agentos" {
		legacy, hasLegacy := requestKwargs["max_tokens"]
		if _, hasNew := requestKwargs["context_window"]; !hasNew && hasLegacy {
			requestKwargs["context_window"] = legacy
		}
		delete(requestKwargs, "max_tokens")
	}

	// Python: :106-114 清理内部键，不得进 core 的 ModelRequestConfig
	delete(requestKwargs, "model")
	delete(requestKwargs, "model_name")
	delete(requestKwargs, "reasoning_level")
	delete(requestKwargs, "_source")

	// Python: :123-137 context_window 归一
	//
	// Go 侧无 core_has_context_window_field() 探测（偏差 E-5）：Go 的
	// ModelRequestConfig.ContextWindow 是正式字段，永远走「保留」分支。
	if raw, ok := requestKwargs["context_window"]; ok {
		if normalized, valid := config.ParsePositiveTokens(raw); valid {
			requestKwargs["context_window"] = normalized
		} else {
			delete(requestKwargs, "context_window")
		}
	}

	requestKwargs["model"] = modelName
	return requestKwargs
}

// resolveModelName 解析生效的模型名：显式参数优先，其次 mco 里的 model/model_name。
// Python: reasoning_injector.py:55-61 _resolve_model_name
func resolveModelName(modelName string, modelConfigObj map[string]any) string {
	if trimmed := strings.TrimSpace(modelName); trimmed != "" {
		return trimmed
	}
	return strings.TrimSpace(firstNonEmptyString(
		modelConfigObj["model"], modelConfigObj["model_name"],
	))
}

// copyStringAnyMap 浅拷贝 map（nil 安全）。
func copyStringAnyMap(source map[string]any) map[string]any {
	out := make(map[string]any, len(source))
	for k, v := range source {
		out[k] = v
	}
	return out
}

// firstNonEmptyString 返回第一个非空字符串（非字符串视为空）。
func firstNonEmptyString(values ...any) string {
	for _, value := range values {
		if s, ok := value.(string); ok && strings.TrimSpace(s) != "" {
			return s
		}
	}
	return ""
}
