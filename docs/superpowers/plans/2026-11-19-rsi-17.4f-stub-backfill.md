```go
package harnessrsi

import (

	"errors"

// ──────────────────────────── 常量 ────────────────────────────

// ──────────────────────────── 全局变量 ────────────────────────────

// ErrTaskCancelled 协作式取消哨兵错误（对齐 Python asyncio.CancelledError 的桩期形态）。
var ErrTaskCancelled = errors.New("task cancelled")

// ──────────────────────────── 导出函数 ────────────────────────────

// RaiseIfCancelled 协作式取消检查入口：ctx 已取消/超时即返回 ErrTaskCancelled。
//
// 对齐 Python single_harness/iterative.py:91 _raise_if_cancelled（模块级函数，
// asyncio.current_task().cancelling() > 0 时抛 CancelledError）。Go 侧无 asyncio，
// 协作式取消等价物为 context.Context。
//
// ⤴️ 17.4f 登记回填：17.7 协作式终止落地后按规则 8 删除本桩；调用点由 17.4d
// 按 Python 原位置接线（iterative.py 4 处 + continuous.py 2 处）。届时本文件
// 随包迁移至 internal/evolving/rsi/harness_rsi/single_harness/。
func RaiseIfCancelled(ctx context.Context) error {
	if err := ctx.Err(); err != nil {
		return ErrTaskCancelled
	}
	return nil
}
```

```go
// ──────────────────────────── 结构体 ────────────────────────────

// TeamEvaluator Team Skill + member harness 集合对数据集的评估门面（桩）。
//
// 对齐 Python openjiuwen/rsi/harness_rsi/evaluator/team_evaluator.py:44
// TeamEvaluator。参数与返回按 Python 真签名映射；DatasetArtifact /
// EvaluationCaseTraceRef 已在 harness_rsi/schema.go（17.1）落地。
//
// ⤴️ 17.4f 登记回填：17.6 评估 agent 章节落地真身（CaseRunner/build_judger/
// MetricsCollector 等依赖链）后，按规则 8 删除本桩（含哨兵错误与 _test.go，
// 锚点 ErrEvaluatorNotImplemented 全仓 0 命中）。
type TeamEvaluator struct {
	// config 评估器配置（对齐 Python self.config）
	config harness_rsi.EvaluatorConfig
}

// ──────────────────────────── 全局变量 ────────────────────────────

// ErrEvaluatorNotImplemented 打分桩哨兵错误：17.6 真身落地前任何评分调用立即失败暴露。
var ErrEvaluatorNotImplemented = errors.New("evaluator not implemented (stub, see 17.6)")

// ──────────────────────────── 导出函数 ────────────────────────────

// NewTeamEvaluator 构造打分桩（对齐 Python __init__）。
func NewTeamEvaluator(config harness_rsi.EvaluatorConfig) *TeamEvaluator {
	return &TeamEvaluator{config: config}
}

// Evaluate 评估数据集全部用例并返回 eval_ref.yaml 路径（桩）。
//
// Python: team_evaluator.py:66 evaluate → 转调 evaluate_batch。
// 签名对齐 Python 真签名（dataset, team_skill_ref_path, harness_refs_path,
// output_dir → str）；dataset 结构体为 17.1 已落地的最小契约。
func (te *TeamEvaluator) Evaluate(ctx context.Context, dataset harness_rsi.DatasetArtifact,
	teamSkillRefPath, harnessRefsPath, outputDir string) (string, error) {
	return te.EvaluateBatch(ctx, dataset.DatasetFiles, teamSkillRefPath, harnessRefsPath, outputDir,
		&dataset, nil, 1, nil)
}

// EvaluateBatch 按输入顺序持久化用例结果（桩）。
//
// Python: team_evaluator.py:82 evaluate_batch。case_concurrency 守卫
//（bool-as-int 与 <1 拒绝）按 Python 原始检查顺序先于桩错误（守卫顺序不得调换）。
func (te *TeamEvaluator) EvaluateBatch(ctx context.Context, cases []map[string]any,
	teamSkillRefPath, harnessRefsPath, outputDir string,
	dataset *harness_rsi.DatasetArtifact,
	onCaseStage func(map[string]any) error,
	caseConcurrency int,
	baselineContexts map[string]any) (string, error) {
	if caseConcurrency < 1 {
		return "", errors.New("case_concurrency must be a positive integer")
	}
	const logComponent = logger.ComponentAgentCore // 若与 runtime_adapters.go 重复声明则直接复用其 logComponent
	logger.Warn(logComponent).
		Str("method", "TeamEvaluator.EvaluateBatch").
		Msg("evaluator stub called; real LLM judge lands in 17.6")
	return "", ErrEvaluatorNotImplemented
}
```
