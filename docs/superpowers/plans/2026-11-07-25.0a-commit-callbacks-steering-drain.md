# 25.0a 附件提交回调与 steering 批上限 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 打通成员团队状态投递所需的三条框架通道——① 附件"先投递后 commit"的提交回调、② 一次模型调用吸收多少条 steering 积压的事件与批上限、③ `on_user_message` 的输入批次（`parts`）载体与 `_input_parts` 全链路。

**Architecture:** 按 ①→②→③ 三个批次推进，每批自成一个可提交单元。① 把 `syncPromptAttachments` 从"只 Sync"改成"返回成败 + 调用方执行 commit 回调"；② 在 `AgentCallbackEvent` 上加第 13 个事件 `before_steering_drain`，配 `SteeringDrainInputs` 与 `DrainSteering(limit *int)`；③ 新增 `UserMessageInputs` 并把 admission 逻辑收敛进 `admitUserMessage`，`_input_parts` 从 `task_loop` executor 经 `DeepAgent` 的 `Extra` 直达内层 `ctx.extra`。

**Tech Stack:** Go 1.22+、`github.com/stretchr/testify`（assert/require/suite）、`tests/integration/mockllm.MockModelClient`、`tests/integration/suite.AgentSuite`、`tests/integration/testhelpers`。

// syncPromptAttachments 把动态提示词同步进历史，并返回同步是否成功。
//
// 返回值是提交回调的执行前提：附件车道是「当前状态」通道，边界时比对
// 注册区与历史哈希，只有真正写进历史，才能把下游基线推进到"已送达"。
//
// Python: ReActAgent._sync_prompt_attachments 前半段（react_agent.py L1148-1168）。
func syncPromptAttachments(ctx context.Context, m *prompts.Manager, hc prompts.HistoryContext) bool {
	if m == nil || hc == nil {
		return true // 无附件车道：不阻断调用方，也没有回调要消费
	}
	if err := m.Sync(ctx, hc); err != nil {
		logger.Warn(logComponent).
			Str("event_type", "prompt_attachment_sync_failed").
			Err(err).
			Msg("prompt attachment sync failed")
		return false
	}
	return true
}

// runCommitCallbacks 执行并清空投递成功回调。
//
// 只在 Sync 成功之后调用：失败时调用方不会走到这里，回调留在 ctx.extra
// 等下一次边界重试。
//
// Python: ReActAgent._sync_prompt_attachments 尾段（react_agent.py L1169-1182）。
func runCommitCallbacks(cbc *interfaces.AgentCallbackContext) {
	if cbc == nil {
		return
	}
	raw, ok := cbc.DeleteExtra(prompts.CommitCallbacksExtraKey)
	if !ok {
		return
	}
	callbacks, ok := raw.([]func() error)
	if !ok {
		// Python: if not isinstance(callbacks, list): return
		return
	}
	for _, fn := range callbacks {
		runOneCommitCallback(fn)
	}
}

// runOneCommitCallback 执行单条提交回调，失败或 panic 只告警。
//
// Python 捕 Exception 继续下一条；Go 的 panic 不 recover 会掀掉整个 invoke，
// 故显式兜住（spec D-2）。
func runOneCommitCallback(fn func() error) {
	if fn == nil {
		return
	}
	defer func() {
		if r := recover(); r != nil {
			logger.Warn(logComponent).
				Str("event_type", "prompt_attachment_delivery_callback_panicked").
				Any("panic", r).
				Msg("prompt attachment delivery callback panicked")
		}
	}()
	if err := fn(); err != nil {
		logger.Warn(logComponent).
			Str("event_type", "prompt_attachment_delivery_callback_failed").
			Err(err).
			Msg("prompt attachment delivery callback failed")
	}
}
```

package attachment

// CommitCallbackSuite 覆盖 spec C1/C2：附件提交回调的执行时机与错误隔离。
type CommitCallbackSuite struct {
	suite.Suite
}

// TestCommitCallbackSuite 运行 CommitCallbackSuite。
func TestCommitCallbackSuite(t *testing.T) { suite.Run(t, new(CommitCallbackSuite)) }

// TestIT_提交回调_真Sync后才推进基线 验证回调在附件真正写进历史之后才执行。
//
// spec C1：Sync 成功 → 历史里有附件消息 且 commit 回调执行一次；
// 第二轮无变更 → 零新增、回调不再执行。
func (s *CommitCallbackSuite) TestIT_提交回调_真Sync后才推进基线() {
	ctx := context.Background()
	m := prompts.NewManager()
	mc := testhelpers.NewTestContext(s.T(), "it-commit-1")

	cbc := testhelpers.NewCallbackContextForTest(s.T(), mc.SessionID())
	commits := 0
	cbc.SetExtra(prompts.CommitCallbacksExtraKey, []func() error{
		func() error { commits++; return nil },
	})

	// 第一轮：注册 section 并同步
	_, err := m.AddSection(mc.SessionID(), "team.context", "# 团队状态 v1", prompts.AttachmentRuntime, "test.rail")
	s.NoError(err)
	m.Sync(ctx, mc)
	s.Equal(1, commits, "同步成功后必须执行一次 commit 回调")

	// 第二轮：内容不变 → manager 不写新消息 → 回调不应被触发（本次根本没注册）
	m.Sync(ctx, mc)
	s.Equal(1, commits, "无变更不得重复 commit")
}

// TestIT_提交回调_历史不可写则不推进基线 验证 Sync 失败时回调留在 extra 里。
//
// spec C1 后半：失败 → 一轮 invoke 不报错、commit 计数不变；恢复后重试成功 → 回调补执行一次。
func (s *CommitCallbackSuite) TestIT_提交回调_历史不可写则不推进基线() {
	ctx := context.Background()
	m := prompts.NewManager()
	failing := testhelpers.NewFailingHistory("it-commit-2", errors.New("history unavailable"))

	cbc := testhelpers.NewCallbackContextForTest(s.T(), "it-commit-2")
	commits := 0
	cbc.SetExtra(prompts.CommitCallbacksExtraKey, []func() error{
		func() error { commits++; return nil },
	})

	_, err := m.AddSection("it-commit-2", "team.context", "# 团队状态 v1", prompts.AttachmentRuntime, "test.rail")
	s.NoError(err)

	// 失败：sync 返回 error，回调未被取走也未执行
	s.Error(m.Sync(ctx, failing))
	s.Zero(commits, "同步失败不得 commit")

	// 恢复：可写历史，重试成功 → 回调在此执行一次
	mc := testhelpers.NewTestContext(s.T(), "it-commit-2b")
	cbc2 := testhelpers.NewCallbackContextForTest(s.T(), mc.SessionID())
	cbc2.SetExtra(prompts.CommitCallbacksExtraKey, []func() error{
		func() error { commits++; return nil },
	})
	m2 := prompts.NewManager()
	_, err = m2.AddSection(mc.SessionID(), "team.context", "# 团队状态 v1", prompts.AttachmentRuntime, "test.rail")
	s.NoError(err)
	s.NoError(m2.Sync(ctx, mc))
	s.Equal(1, commits)
}

// DrainSteering 非阻塞取走待处理 steering 消息，最旧优先。
//
// limit 为 nil 表示不限（全部取走）；非 nil 时上限为 max(1, *limit)——
// 非空队列始终至少放行 1 条，否则 HasPendingSteering 会让 ReAct 循环
// 在一个永不消费的积压上空转。未被取走的留在队列原位。
//
// Python: AgentCallbackContext.drain_steering(limit) (rail/base.py L725-785)
func (c *AgentCallbackContext) DrainSteering(limit *int) []string {
	if c.steeringQueue == nil {
		return nil
	}
	take := 0 // 0 表示不限
	if limit != nil {
		take = *limit
		if take < 1 {
			take = 1
		}
	}
	var msgs []string
	for take == 0 || len(msgs) < take {
		select {
		case msg := <-c.steeringQueue:
			msgs = append(msgs, msg)
		default:
			return msgs
		}
	}
	return msgs
}

// SteeringPendingCount 返回当前排队条数。
//
// Python: AgentCallbackContext.steering_pending_count() (rail/base.py L787-791)
func (c *AgentCallbackContext) SteeringPendingCount() int {
	if c.steeringQueue == nil {
		return 0
	}
	return len(c.steeringQueue)
}
```

// drainSteeringBatch 取本次模型调用可以吸收的那部分 steering 积压。
//
// 全部一次塞进同一轮对"两三条叮嘱"是对的，对"忙碌一段后回来看到十几条"
// 是灾难：模型被迫在一轮里同时处理所有事项。批大小因此是策略问题，由 rails
// 在**队列被取走之前**表态——事后再限制意味着先全取再把多余塞回去，而塞回去的
// 会排在期间新到的消息之后。
//
// 无队列或无积压时直接返回，不触发事件。
//
// Python: ReActAgent._drain_steering_batch (react_agent.py L964-1000)
func drainSteeringBatch(ctx context.Context, cbc *interfaces.AgentCallbackContext) ([]string, error) {
	if cbc == nil || cbc.SteeringQueue() == nil || !cbc.HasPendingSteering() {
		return nil, nil
	}
	prev := cbc.Inputs()
	inputs := &interfaces.SteeringDrainInputs{Pending: cbc.SteeringPendingCount()}
	cbc.SetInputs(inputs)
	fireErr := cbc.Fire(ctx, interfaces.CallbackBeforeSteeringDrain)
	cbc.SetInputs(prev)

	// Fire 出错仍要取走（否则积压永不消费、ReAct 循环空转），错误照抛。
	// spec D-4：上游 fire 不返回值，无此分支。
	msgs := cbc.DrainSteering(inputs.Limit)
	return msgs, fireErr
}
```

// extractUserParts 把本轮 query 归一成 ON_USER_MESSAGE 看到的输入列表。
//
// 一轮可能由多条排队输入共同驱动，调用方把它们不 join 地交下来，rails
// 才能看到接缝；其余情况下就是 query 本身这一个输入。
//
// Python: ReActAgent._extract_user_parts (react_agent.py L1002-1012)
func extractUserParts(cbc *interfaces.AgentCallbackContext, query interfaces.InvokeQuery) []string {
	if cbc != nil {
		if raw, ok := cbc.GetExtra("_input_parts"); ok {
			if parts, ok2 := raw.([]string); ok2 && len(parts) > 0 {
				return append([]string(nil), parts...)
			}
		}
	}
	if query == nil {
		return nil
	}
	return []string{query.PlainText()}
}

// admitUserMessage 把一批已消费的输入 join 成一条 UserMessage 写入历史。
//
// Agent 消费的每个输入——新一轮 query、忙碌时排队的 follow-up、一批
// steering、恢复的工作流中断——都从这里进入历史。ON_USER_MESSAGE 在
// join **之前**对列表触发，rails 因此仍能看到单条输入，把被后续条目取代的
// 整条丢掉。这是 rails 唯一能把它们当作**输入**处理的时刻：join 写入后
// 就是普通历史，受 compaction 影响，不能再按位置定位。
//
// Python: ReActAgent._admit_user_message (react_agent.py L917-955)
func (a *ReActAgent) admitUserMessage(
	ctx context.Context,
	cbc *interfaces.AgentCallbackContext,
	modelCtx ceiface.ModelContext,
	parts []string,
	source string,
	prefix string,
) error {
	prev := cbc.Inputs()
	inputs := &interfaces.UserMessageInputs{Parts: parts, Source: source}
	cbc.SetInputs(inputs)
	// 必须 defer：fire 里 panic 时 inputs 若停在 UserMessageInputs，
	// 后续 before_model_call 会拿到错误类型直接崩（Python 用 finally 等价）。
	defer cbc.SetInputs(prev)

	if err := cbc.Fire(ctx, interfaces.CallbackOnUserMessage); err != nil {
		return err
	}
	// Python: if not parts: return —— rails 可能把整批丢光
	if len(inputs.Parts) == 0 {
		return nil
	}
	if modelCtx == nil {
		return nil
	}
	// 附件先落历史，保证顺序为 [附件变更][用户消息]
	if syncPromptAttachments(ctx, a.promptAttachments, modelCtx) {
		runCommitCallbacks(cbc)
	}
	body := prefix + strings.Join(inputs.Parts, "\n")
	_, _ = modelCtx.AddMessages(ctx, []llmschema.BaseMessage{llmschema.NewUserMessage(body)})
	return nil
}
```

// extractInputParts 返回本轮输入未 join 的文本列表（多于一条时才有意义）。
//
// 一起排队的输入驱动同一轮，到达模型时是一条消息；但内层循环先把它们

// 从 query 派生。
//
// Python: TaskLoopEventExecutor._extract_input_parts
// (harness/task_loop/task_loop_event_executor.py L431-448)
func extractInputParts(event *cschema.InputEvent) []string {
	if event == nil {
		return nil
	}
	texts := make([]string, 0, len(event.InputData))
	for _, df := range event.InputData {
		if td, ok := df.(*cschema.TextDataFrame); ok && td.Text != "" {
			texts = append(texts, td.Text)
		}
	}
	if len(texts) > 1 {
		return texts
	}
	return nil
}
