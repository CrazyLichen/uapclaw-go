package runtime

import (
	"regexp"
	"strings"
)

// ──────────────────────────── 结构体 ────────────────────────────

// RuntimeAgentDefinition 一份被校验通过的声明式 root-Agent 定义。
// Python: agent_definition.py:151-238 RuntimeAgentDefinition
//
// Tools 为 "*" 时表示沿用 Runtime 已配置的工具集；显式白名单在 rail 启动后
// 于 Agent 能力边界强制。
type RuntimeAgentDefinition struct {
	// Name 名称（匹配 [A-Za-z0-9_-]{3,50}）
	Name string
	// Instructions 系统指令（校验去空白后非空，值原样保留）
	Instructions string
	// Description 可选描述
	Description *string
	// Model 可选模型标识
	Model *string
	// Tools "*" 或显式工具名元组
	Tools []string
	// ToolsWildcard 是否使用 "*"（Python tools="*"）
	ToolsWildcard bool
	// Skills 技能名元组
	Skills []string
	// MaxIterations 可选最大迭代次数（须 > 0）
	MaxIterations *int
}

// ──────────────────────────── 常量 ────────────────────────────

const (
	// agentDefinitionNamePattern 合法 agent 名
	agentDefinitionNamePattern = `^[A-Za-z0-9_-]{3,50}$`
)

// ──────────────────────────── 全局变量 ────────────────────────────

// agentDefinitionNameRegexp 编译后的名称校验正则
var agentDefinitionNameRegexp = regexp.MustCompile(agentDefinitionNamePattern)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewRuntimeAgentDefinition 校验并归一化一份定义。
// Python: agent_definition.py:167-196 __post_init__
func NewRuntimeAgentDefinition(in RuntimeAgentDefinition) (RuntimeAgentDefinition, error) {
	out := in

	// Python:170 name = _required_text("name", self.name)；校验后写回去空白值
	name := strings.TrimSpace(in.Name)
	if name == "" {
		return RuntimeAgentDefinition{}, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, "name", "name is required")
	}
	if !agentDefinitionNameRegexp.MatchString(name) {
		return RuntimeAgentDefinition{}, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, "name", "name must match [A-Za-z0-9_-]{3,50}")
	}
	out.Name = name

	// Python:173-177 instructions 保留原值，仅校验去空白后非空
	if strings.TrimSpace(in.Instructions) == "" {
		return RuntimeAgentDefinition{}, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, "instructions", "instructions is required")
	}

	// Python:187-196 max_iterations 须为正整数
	if in.MaxIterations != nil && *in.MaxIterations <= 0 {
		return RuntimeAgentDefinition{}, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, "max_iterations",
			"max_iterations must be greater than zero")
	}
	return out, nil
}
```

var (
	// ErrAgentDefinitionInvalidValue 字段值非法（Python INVALID_VALUE 对应各字段校验失败）
	ErrAgentDefinitionInvalidValue = errors.New("agent definition invalid value")
	// ErrAgentDefinitionInvalidMode 不支持的 single-Agent 模式（Python INVALID_MODE）
	ErrAgentDefinitionInvalidMode = errors.New("agent definition invalid mode")
	// ErrAgentDefinitionInvalidRequest 请求形态不合法（Python INVALID_REQUEST）
	ErrAgentDefinitionInvalidRequest = errors.New("agent definition invalid request")
	// ErrAgentDefinitionModelNotFound 配置的模型不存在（Python MODEL_NOT_FOUND）
	ErrAgentDefinitionModelNotFound = errors.New("agent definition model not found")
	// ErrAgentExecutionSessionConflict 会话已绑定另一个定义（Python SESSION_CONFLICT）
	ErrAgentExecutionSessionConflict = errors.New("agent execution session conflict")
)

// ──────────────────────────── 非导出函数 ────────────────────────────

// agentDefinitionError 带 code + field + message 的定义错误。
// Python: agent_definition.py:49-65 RuntimeAgentDefinitionError
type agentDefinitionError struct {
	sentinel error
	field    string
	message  string
}

// Error 实现 error 接口
func (e *agentDefinitionError) Error() string {
	if e.field == "" {
		return e.message
	}
	return e.field + ": " + e.message
}

// Unwrap 支持 errors.Is 判定哨兵
func (e *agentDefinitionError) Unwrap() error { return e.sentinel }

// newAgentDefinitionError 构造定义错误。
func newAgentDefinitionError(sentinel error, field, message string) error {
	return &agentDefinitionError{sentinel: sentinel, field: field, message: message}
}

// ──────────────────────────── 导出函数（追加） ────────────────────────────

// AgentDefinitionErrorField 取错误的 field（非本类型返回空串）。
// Python: agent_definition.py:54-65 的 self.field
func AgentDefinitionErrorField(err error) string {
	var target *agentDefinitionError
	if errors.As(err, &target) {
		return target.field
	}
	return ""
}

// ──────────────────────────── 常量 ────────────────────────────

const (
	// agentDefinitionSchema fingerprint 的 schema 版本标记
	// Python: agent_definition.py _DEFINITION_SCHEMA
	agentDefinitionSchema = "runtime-agent-definition/1"
)

// ──────────────────────────── 导出函数（追加） ────────────────────────────

// RuntimeAgentDefinitionToDict 返回规范的机器协议表示。
// Python: agent_definition.py:211-223 to_dict
func RuntimeAgentDefinitionToDict(d RuntimeAgentDefinition) map[string]any {
	var tools any
	if d.ToolsWildcard || d.Tools == nil {
		tools = "*"
	} else {
		tools = append([]string{}, d.Tools...)
	}
	skills := d.Skills
	if skills == nil {
		skills = []string{}
	}
	return map[string]any{
		"name":           d.Name,
		"instructions":   d.Instructions,
		"description":    d.Description,
		"model":          d.Model,
		"tools":          tools,
		"skills":         append([]string{}, skills...),
		"max_iterations": d.MaxIterations,
	}
}

// RuntimeAgentDefinitionFingerprint 返回规范的 SHA-256 指纹。
// Python: agent_definition.py:225-238 fingerprint
//
// 与上游逐字节等价的三要素：sort_keys（map 交给 json.Marshal 自动有序）、
// separators=(",",":")（json.Marshal 无空格）、ensure_ascii=False
// （SetEscapeHTML(false)）。注意 json.Encoder 会追加 '\n'，必须裁掉。
func RuntimeAgentDefinitionFingerprint(d RuntimeAgentDefinition) (string, error) {
	payload := map[string]any{
		"schema":     agentDefinitionSchema,
		"definition": RuntimeAgentDefinitionToDict(d),
	}
	var buf bytes.Buffer
	enc := json.NewEncoder(&buf)
	enc.SetEscapeHTML(false)
	if err := enc.Encode(payload); err != nil {
		return "", err
	}
	canonical := bytes.TrimSuffix(buf.Bytes(), []byte("\n"))
	sum := sha256.Sum256(canonical)
	return hex.EncodeToString(sum[:]), nil
}
```

// agentDefinitionKnownKeys 定义允许的键集合（严格映射，未知键报错）
// Python: agent_definition.py:130-149 _strict_definition_mapping
var agentDefinitionKnownKeys = map[string]struct{}{
	"name": {}, "instructions": {}, "description": {}, "model": {},
	"tools": {}, "skills": {}, "max_iterations": {},
}

// ──────────────────────────── 导出函数（追加） ────────────────────────────

// RuntimeAgentDefinitionFromMapping 解析一份严格的、传输无关的定义映射。
// Python: agent_definition.py:197-209 from_mapping + :130-149 _strict_definition_mapping
//
// 注意：本函数拒绝未知键，与项目内 pyjson 的"未知字段保留"策略相反，
// 不得复用容错解码。
func RuntimeAgentDefinitionFromMapping(value any) (RuntimeAgentDefinition, error) {
	raw, ok := value.(map[string]any)
	if !ok {
		return RuntimeAgentDefinition{}, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, "", "definition must be an object")
	}
	for key := range raw {
		if _, known := agentDefinitionKnownKeys[key]; !known {
			return RuntimeAgentDefinition{}, newAgentDefinitionError(
				ErrAgentDefinitionInvalidValue, key, "unknown definition field")
		}
	}

	out := RuntimeAgentDefinition{}
	name, err := agentDefinitionRequiredText("name", raw["name"])
	if err != nil {
		return RuntimeAgentDefinition{}, err
	}
	out.Name = name
	instructions, err := agentDefinitionRequiredTextPreserve("instructions", raw["instructions"])
	if err != nil {
		return RuntimeAgentDefinition{}, err
	}
	out.Instructions = instructions

	if v, err := agentDefinitionOptionalText("description", raw["description"]); err != nil {
		return RuntimeAgentDefinition{}, err
	} else {
		out.Description = v
	}
	if v, err := agentDefinitionOptionalText("model", raw["model"]); err != nil {
		return RuntimeAgentDefinition{}, err
	} else {
		out.Model = v
	}

	tools, wildcard, err := agentDefinitionConfiguredTools(raw["tools"])
	if err != nil {
		return RuntimeAgentDefinition{}, err
	}
	out.Tools, out.ToolsWildcard = tools, wildcard

	skills, err := agentDefinitionStringTuple("skills", raw["skills"])
	if err != nil {
		return RuntimeAgentDefinition{}, err
	}
	out.Skills = skills

	if raw["max_iterations"] != nil {
		f, ok := raw["max_iterations"].(float64)
		if !ok {
			return RuntimeAgentDefinition{}, newAgentDefinitionError(
				ErrAgentDefinitionInvalidValue, "max_iterations", "max_iterations must be an integer")
		}
		n := int(f)
		out.MaxIterations = &n
	}
	return NewRuntimeAgentDefinition(out)
}

// agentDefinitionRequiredText 必填文本，去空白后非空并返回去空白值。
// Python: agent_definition.py:84-96 _required_text(preserve=False)
func agentDefinitionRequiredText(field string, value any) (string, error) {
	s, ok := value.(string)
	if !ok {
		return "", newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, field, field+" is required")
	}
	s = strings.TrimSpace(s)
	if s == "" {
		return "", newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, field, field+" is required")
	}
	return s, nil
}

// agentDefinitionRequiredTextPreserve 必填文本，仅校验非空，值原样返回。
// Python: agent_definition.py:84-96 _required_text(preserve=True)
func agentDefinitionRequiredTextPreserve(field string, value any) (string, error) {
	s, ok := value.(string)
	if !ok || strings.TrimSpace(s) == "" {
		return "", newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, field, field+" is required")
	}
	return s, nil
}

// agentDefinitionOptionalText 可选文本，空白视为未提供。
// Python: agent_definition.py:97-102 _optional_text
func agentDefinitionOptionalText(field string, value any) (*string, error) {
	if value == nil {
		return nil, nil
	}
	s, ok := value.(string)
	if !ok {
		return nil, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, field, field+" must be a string")
	}
	s = strings.TrimSpace(s)
	if s == "" {
		return nil, nil
	}
	return &s, nil
}

// agentDefinitionConfiguredTools 解析 tools：缺省/"*" → 通配；数组 → 列表。
// Python: agent_definition.py:114-129 _configured_tools
func agentDefinitionConfiguredTools(value any) ([]string, bool, error) {
	if value == nil {
		return nil, true, nil
	}
	if s, ok := value.(string); ok {
		if strings.TrimSpace(s) == "*" {
			return nil, true, nil
		}
		return []string{s}, false, nil
	}
	list, ok := value.([]any)
	if !ok {
		return nil, false, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, "tools", "tools must be '*' or a list of names")
	}
	out := make([]string, 0, len(list))
	for _, item := range list {
		s, ok := item.(string)
		if !ok || strings.TrimSpace(s) == "" {
			return nil, false, newAgentDefinitionError(
				ErrAgentDefinitionInvalidValue, "tools", "tools entries must be non-empty strings")
		}
		out = append(out, s)
	}
	return out, false, nil
}

// agentDefinitionStringTuple 解析字符串数组。
// Python: agent_definition.py:103-113 _string_tuple
func agentDefinitionStringTuple(field string, value any) ([]string, error) {
	if value == nil {
		return nil, nil
	}
	list, ok := value.([]any)
	if !ok {
		return nil, newAgentDefinitionError(
			ErrAgentDefinitionInvalidValue, field, field+" must be a list of strings")
	}
	out := make([]string, 0, len(list))
	for _, item := range list {
		s, ok := item.(string)
		if !ok || strings.TrimSpace(s) == "" {
			return nil, newAgentDefinitionError(
				ErrAgentDefinitionInvalidValue, field, field+" entries must be non-empty strings")
		}
		out = append(out, s)
	}
	return out, nil
}

	SupportsCustomAgentDefinitions bool

// Python: mode_catalog.py:50-56 ModeCatalogResult
type ModeCatalogResult struct {
	// Modes 按产品顺序排列
	Modes []RuntimeModeDescriptor
}

// ──────────────────────────── 全局变量 ────────────────────────────

	ErrModeRequired = errors.New("mode is required")
	// ErrModeNotFound 单 Agent 模式不存在（Python code=NOT_FOUND）
	ErrModeNotFound = errors.New("single-Agent mode not found")

// Python: mode_catalog.py:59-84 _SINGLE_AGENT_MODES

	{Mode: "agent.work.normal", WorkMode: "work", IsPlan: false, SupportsCustomAgentDefinitions: true},
	{Mode: "agent.work.plan", WorkMode: "work", IsPlan: true, SupportsCustomAgentDefinitions: true},
	{Mode: "agent.code.normal", WorkMode: "code", IsPlan: false, SupportsCustomAgentDefinitions: true},
	{Mode: "agent.code.plan", WorkMode: "code", IsPlan: true, SupportsCustomAgentDefinitions: true},

func ListModeCapabilities() ModeCatalogResult {
	out := make([]RuntimeModeDescriptor, len(singleAgentModes))
	copy(out, singleAgentModes)
	return ModeCatalogResult{Modes: out}
}

// ResolveModeCapability 解析 canonical 模式，保留受支持的遗留别名。
// Python: mode_catalog.py:92-107 resolve_mode_capability
//
// 遗留别名转换（deprecate_mode）在 Go 侧由 schema.ApplyResolvedModeToRequest
// 承担；本函数只做 canonical 精确匹配，与上游 mode_catalog 的职责一致。
func ResolveModeCapability(requested any) (RuntimeModeDescriptor, error) {

	}
	for _, m := range singleAgentModes {
		if m.Mode == target {

		}
	}
	return RuntimeModeDescriptor{}, ErrModeNotFound
}

// Python: service.py:1584-1591 _agent_execution_owner_key
func (r *AgentRuntime) agentExecutionOwnerKey(req *schema.AgentRequest) (agentExecutionOwnerKey, bool) {
	sessionID := ""
	if req != nil && req.SessionID != nil {
		sessionID = strings.TrimSpace(*req.SessionID)
	}
	if sessionID == "" {
		return agentExecutionOwnerKey{}, false
	}

	if req != nil {
		if v := strings.TrimSpace(strings.ToLower(req.ChannelID)); v != "" {
			channelID = v
		}

// claimAgentExecutionOwner 把一份声明式定义钉到活跃会话上。
// Python: service.py:1593-1618 _claim_agent_execution_owner
//
// 语义：已有条目且指纹不同 → SESSION_CONFLICT；指纹相同 → 幂等返回；
// 无条目 → 写入。
func (r *AgentRuntime) claimAgentExecutionOwner(
	req *schema.AgentRequest, exec *RuntimeAgentExecution,
) error {
	key, ok := r.agentExecutionOwnerKey(req)
	if !ok || exec == nil {
		return nil
	}
	fingerprint, err := RuntimeAgentExecutionFingerprint(exec)
	if err != nil {
		return err
	}

	r.agentExecutionOwnerMu.Lock()
	defer r.agentExecutionOwnerMu.Unlock()
	current, exists := r.agentExecutionOwners[key]
	if exists {
		currentFingerprint, err := RuntimeAgentExecutionFingerprint(current)
		if err != nil {
			return err
		}
		if currentFingerprint != fingerprint {
			return newAgentDefinitionError(ErrAgentExecutionSessionConflict, "agent",
				"session is already bound to another Agent definition in this Runtime lifecycle")
		}
		return nil
	}
	r.agentExecutionOwners[key] = exec
	return nil
}

// agentExecutionOwner 查 owner 表；无条目返回 ok=false。
// Python: service.py:1620-1625 _agent_execution_owner（Python 无锁，Go 加锁见 spec D-4）
func (r *AgentRuntime) agentExecutionOwner(req *schema.AgentRequest) (*RuntimeAgentExecution, bool) {
	key, ok := r.agentExecutionOwnerKey(req)
	if !ok {
		return nil, false
	}
	r.agentExecutionOwnerMu.Lock()
	defer r.agentExecutionOwnerMu.Unlock()
	exec, exists := r.agentExecutionOwners[key]
	return exec, exists
}

// forgetAgentExecutionOwner 解绑某会话的 owner。
// Python: service.py:1627-1640 _forget_agent_execution_owner
func (r *AgentRuntime) forgetAgentExecutionOwner(channelID, sessionID string) {
	sid := strings.TrimSpace(sessionID)
	if sid == "" {
		return
	}
	cid := "default"
	if v := strings.TrimSpace(strings.ToLower(channelID)); v != "" {
		cid = v
	}
	r.agentExecutionOwnerMu.Lock()
	defer r.agentExecutionOwnerMu.Unlock()
	delete(r.agentExecutionOwners, agentExecutionOwnerKey{ChannelID: cid, SessionID: sid})
}

// bindAgentExecutionRequest 把已校验的定义绑到请求副本上。

func (r *AgentRuntime) bindAgentExecutionRequest(
	req *schema.AgentRequest, exec *RuntimeAgentExecution,
) (*schema.AgentRequest, error) {
	if req == nil || req.ReqMethod != schema.ReqMethodChatSend {
		return nil, newAgentDefinitionError(ErrAgentDefinitionInvalidRequest, "request",
			"custom Agent execution requires a chat.send request")
	}
	params := parseRequestParams(req)
	if params == nil {
		return nil, newAgentDefinitionError(ErrAgentDefinitionInvalidRequest, "request.params",
			"custom Agent execution requires object request params")
	}

	// Python:1571-1576 写入 canonical mode 与 work_mode
	params["mode"] = exec.Mode
	if strings.HasPrefix(exec.Mode, "agent.code.") {
		params["work_mode"] = "code"
	} else {
		params["work_mode"] = "work"
	}

	// Python:1577-1581 未显式给 model_name 时用定义里的模型
	if exec.Definition.Model != nil && *exec.Definition.Model != "" {
		if existing, _ := params["model_name"].(string); strings.TrimSpace(existing) == "" {
			if r.resolveModelCapability != nil {
				if desc, err := r.resolveModelCapability(*exec.Definition.Model); err == nil {
					params["model_name"] = desc.SelectionKey
				}
			}
		}
	}

	writeRequestParams(req, params)
	return req, nil
}

// Python: service.py:3024-3070 cleanup_session
//
// 四步顺序不得调整（短路顺序即语义边界）：
//  1. 关协调器执行
//  2. 遍历 manager 清运行时（单条失败只告警）
//  3. 解绑 owner（无论前面成败）
//  4. resetPlanState 为真时重置 plan
func (r *AgentRuntime) CleanupSession(
	ctx context.Context, channelID, sessionID string, resetPlanState bool,
) (bool, error) {
	r.mu.Lock()
	closed := r.closed
	r.mu.Unlock()
	if closed {
		// Python:3037 if self._closed: raise RuntimeStateError
		return false, errors.New("agent runtime is closed")
	}

	// 第 1 步：关协调器执行（Python:3052-3057）
	if r.coordinator != nil {
		if _, err := r.coordinator.CloseSession(sessionID, nil, nil); err != nil {
			return false, err
		}
	}

	// 第 2 步：清 manager 侧运行时（Python:3058-3065）
	cleaned := false
	if manager, ok := r.manager.(interface {
		CleanupSessionRuntime(context.Context, string, string) (bool, error)
	}); ok {
		result, err := manager.CleanupSessionRuntime(ctx, channelID, sessionID)
		if err != nil {
			// Python 聚合后统一 raise；Go 侧只告警（spec D-2，删除事务不该整体回滚）
			logger.Warn(logComponent).Err(err).
				Str("channel_id", channelID).Str("session_id", sessionID).
				Msg("session runtime cleanup failed")
		} else {
			cleaned = result
		}
	}

	// 第 3 步：解绑 owner（Python:3066，无条件）
	r.forgetAgentExecutionOwner(channelID, sessionID)

	// 第 4 步：按需重置 plan（Python:3070）
	if resetPlanState && r.resetPlanSession != nil {
		r.resetPlanSession(sessionID)
	}
	return cleaned, nil
}
```

// InvokeAgent 通过既有 Runtime 链执行一个声明式 root Agent。
// Python: service.py:1642-1663 invoke_agent
func (r *AgentRuntime) InvokeAgent(
	ctx context.Context, req *schema.AgentRequest, def RuntimeAgentDefinition,
) ([]*RuntimeEvent, error) {
	// 1. 启动
	if err := r.Start(); err != nil {
		return nil, err
	}
	// 2. 会话归属准入（Python:1649 _require_owned_single_agent_session）
	if err := r.requireOwnedSingleAgentSession(req); err != nil {
		return nil, err
	}
	// 3. 解析模式（Python:1654）
	params := parseRequestParams(req)
	modeDesc, err := r.ResolveModeCapability(params["mode"])
	if err != nil {
		return nil, err
	}
	// 4. 校验定义（Python:1655）
	exec, err := r.validateAgentDefinition(def, modeDesc.Mode)
	if err != nil {
		return nil, err
	}
	// 5. 绑请求（Python:1656）
	bound, err := r.bindAgentExecutionRequest(req, exec)
	if err != nil {
		return nil, err
	}
	// 6. 认领 owner（Python:1657）★R-4 写端
	if err := r.claimAgentExecutionOwner(req, exec); err != nil {
		return nil, err
	}
	// 7. 执行（Python:1658-1663，透传 _agent_execution）
	return r.invokeWithExecution(ctx, bound, exec)
}
```

`invokeWithExecution` 是 `Invoke` 的内层版本，把 `exec` 透传给 agent 调用点。同理 `StreamAgent` 对应 `streamWithExecution`。
