// 判定工具结果是否为工作流中断。
// Python: react_agent.py _is_interrupted L2515-1522。
// WorkflowOutput 分支判 State==INPUT_REQUIRED；list 分支判存在 __interaction__ 块。
func (a *ReActAgent) isInterrupted(toolResult any) bool {
	if out, ok := toolResult.(*workflow.WorkflowOutput); ok {
		return out.State == workflow.WorkflowExecutionStateInputRequired
	}
	if list, ok := toolResult.([]stream.Schema); ok {
		for _, item := range list {
			if item.SchemaType() == "__interaction__" {
				return true
			}
		}
	}
	return false
}

// extractComponentIDs 从被中断的工作流结果提取组件 ID（sorted 稳定序）。
// Python: react_agent.py _extract_component_ids L2525-2552。
// WorkflowOutput.result 扫 __interaction__ 取 payload.id；list 分支取 payload["component_id"]。
func (a *ReActAgent) extractComponentIDs(toolResult any) []string {
	if out, ok := toolResult.(*workflow.WorkflowOutput); ok {
		if list, ok := out.Result.([]stream.Schema); ok {
			ids := make([]string, 0)
			for _, item := range list {
				if item.SchemaType() != "__interaction__" {
					continue
				}
				if os, ok := item.(*stream.OutputSchema); ok {
					if p, ok := os.Payload.(interface{ GetID() string }); ok {
						ids = append(ids, p.GetID())
					} else if m, ok := os.Payload.(map[string]any); ok {
						ids = append(ids, fmt.Sprint(m["id"]))
					}
				}
			}
			sort.Strings(ids)
			return ids
		}
		return []string{}
	}
	if list, ok := toolResult.([]stream.Schema); ok {
		ids := make([]string, 0)
		for _, item := range list {
			if item.SchemaType() != "__interaction__" {
				continue
			}
			if os, ok := item.(*stream.OutputSchema); ok {
				if m, ok := os.Payload.(map[string]any); ok {
					ids = append(ids, fmt.Sprint(m["component_id"]))
				}
			}
		}
		sort.Strings(ids)
		return ids
	}
	return []string{}
}

// extractWorkflowID 经 ability_manager 按 tool_call.name 反查 workflow_id。
// Python: react_agent.py _extract_workflow_id L2531-1536。miss fallback tool_call.name。
func (a *ReActAgent) extractWorkflowID(toolCall *llmschema.ToolCall) string {
	for _, ability := range a.AbilityManager().List() {
		if ability.Name == toolCall.Name {
			return ability.ID
		}
	}
	return toolCall.Name
}
```

注意：`AbilityManager()` 为 nil 时（测试直构 `&ReActAgent{}`）`extractWorkflowID` 会 panic——测试只覆盖 `isInterrupted/extractComponentIDs`；`extractWorkflowID` 的测试用注入了 `ability.NewAbilityManager(runner.GetResourceMgr())` 的真 agent 或置 fallback 断言（`ability.Name` 字段名实施时以 ability 包 schema 实际字段为准核对）。

// afterExecuteToolCall 工具执行后收集全部被中断工作流，构建 InterruptionState。
// Python: react_agent.py _after_execute_tool_call L2554-1596（workflow 版，与
// AfterExecuteToolCallForHITL 并存；pending 取第一个被中断项）。
func (a *ReActAgent) afterExecuteToolCall(
	results []any,
	toolCalls []*llmschema.ToolCall,
	aiMessage *llmschema.AssistantMessage,
	iteration int,
	originalQuery string,
) *saschema.InterruptionState {
	interrupted := make(map[string]*saschema.WorkflowInterruptEntry)
	firstWorkflowID := ""
	firstComponentID := ""
	for i, toolResult := range results {
		if !a.isInterrupted(toolResult) {
			continue
		}
		workflowID := a.extractWorkflowID(toolCalls[i])
		componentIDs := a.extractComponentIDs(toolResult)
		interrupted[workflowID] = &saschema.WorkflowInterruptEntry{
			ToolCall:               toolCalls[i],
			ComponentIDs:           componentIDs,
			WorkflowExecutionState: toolResult,
		}
		if firstWorkflowID == "" {
			firstWorkflowID = workflowID
			if len(componentIDs) > 0 {
				firstComponentID = componentIDs[0]
			}
		}
	}
	if len(interrupted) == 0 {
		return nil
	}
	return &saschema.InterruptionState{
		BaseInterruptionState: saschema.BaseInterruptionState{
			AIMessage:     aiMessage,
			Iteration:     iteration,
			OriginalQuery: originalQuery,
		},
		InterruptedWorkflows: interrupted,
		PendingWorkflowID:    firstWorkflowID,
		PendingComponentID:   firstComponentID,
	}
}

// saveInterruptionState 持久化工作流中断态（Python _save_interruption_state L2611-1613）。
func (a *ReActAgent) saveInterruptionState(st *saschema.InterruptionState, sess sessioninterfaces.SessionFacade) {
	if sess == nil {
		return
	}
	sess.UpdateState(map[string]any{saschema.InterruptionKey: st})
}

// loadInterruptionState 加载工作流中断态（Python _load_interruption_state L2615-1618）。
func (a *ReActAgent) loadInterruptionState(sess sessioninterfaces.SessionFacade) *saschema.InterruptionState {
	if sess == nil {
		return nil
	}
	v, err := sess.GetState(saschema.InterruptionKey)
	if err != nil || v == nil {
		return nil
	}
	st, _ := v.(*saschema.InterruptionState)
	return st
}

// clearInterruptionState 清除工作流中断态（Python _clear_interruption_state L2620-1623）。
func (a *ReActAgent) clearInterruptionState(sess sessioninterfaces.SessionFacade) {
	if sess == nil {
		return
	}
	sess.UpdateState(map[string]any{saschema.InterruptionKey: nil})
}
```

注意：`GetState` 键类型为 `state.StateKey`——若 `InterruptionKey` 不是该类型需 `state.StateKey(saschema.InterruptionKey)` 转换，以 `interrupt/handler.go:85` 既有用法为准照抄。`GetState` 返回 `(any, error)` 中已存在键值为 nil 的区分按既有 handler 惯例处理。

// CommitWorkflowInterrupt 提交工作流中断（D-26-1 双入口：与既有 CommitInterrupt 并存）。

func (a *ReActAgent) CommitWorkflowInterrupt(
	ctx context.Context,
	st *saschema.InterruptionState,
	modelCtx ceiface.ModelContext,
	sess sessioninterfaces.SessionFacade,
	invokeInputs *interfaces.InvokeInputs,
) (map[string]any, error) {
	pendingEntry := st.InterruptedWorkflows[st.PendingWorkflowID]

		return nil, err
	}
	if err := a.contextEngine.SaveContexts(sess); err != nil {

	invokeInputs.Result = result

func (e *ReActExecutable) streamEntry(ctx context.Context, inputs map[string]any, sess *session.NodeSessionFacade, _ iface.ModelContext) (<-chan graphexe.StreamChunk[any], error) {
	// 新建 agent 会话（Python create_agent_session(session.get_session_id(), card)）：

	ch, err := e.reactAgent.Stream(ctx, inputs, interfaces.WithSession(agentSess))
	if err != nil {
		return nil, err
	}
	out := make(chan graphexe.StreamChunk[any])
	go func() {
		defer close(out)
		for chunk := range ch {
			if os, ok := chunk.(*stream.OutputSchema); ok {
				// OutputSchema 解包（Python L70-183 三态）
				out <- graphexe.StreamChunk[any]{Frame: unpackChunk(os)}
			} else {
				out <- graphexe.StreamChunk[any]{Frame: chunk}
			}
		}
		// Python except Exception → yield {"type": "error", "payload": {...}}（结构带外层 type 键，逐字）
	}()
	return out, nil

// unpackChunk OutputSchema 解包三态（Python react_executable.py L70-183）：
// llm_output 且含 content → {"output": content}；其余 OutputSchema → payload；非 → 原样。
func unpackChunk(chunk any) any {

	if !ok {
		return chunk
	}
	if os.Type == "llm_output" {
		if m, ok := os.Payload.(map[string]any); ok {
			if content, exists := m["content"]; exists {
				return map[string]any{"output": content}
			}
		}
	}
	return os.Payload
