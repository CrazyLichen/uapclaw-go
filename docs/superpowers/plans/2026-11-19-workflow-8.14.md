// BaseComponentComposable 组合面基座（Python ComponentComposable mixin 的默认实现承载者）。
type BaseComponentComposable struct {
	// toExecutableFn 槽承载「基座回调外层」唯一边（Python isinstance(self, Executable)→return self）
	toExecutableFn func() executable.GeneralExecutor
}

// AddComponent 默认实现一比一（component.py:248-260）。
func (m *BaseComponentComposable) AddComponent(g *graph.PregelGraph, nodeID string, waitForAll bool) error {
	if m.toExecutableFn == nil { // ≙ Python to_executable NotImplementedError 分支
		return exception.BuildError(exception.StatusWorkflowInnerOrchestrationError,
			exception.WithParam("reason", "Component is missing required method: to_executable()"))
	}
	opts := []graph.AddNodeOption{}
	if waitForAll {
		opts = append(opts, graph.WithWaitForAll())
	}
	g.AddNode(nodeID, m.toExecutableFn(), opts...)
	return nil
}

// ToExecutable 基座默认抛错语义由 nil 槽在 AddComponent 中承载；
// WorkflowComponent 嵌入后由构造函数接线，自身即满足 GeneralExecutor（编译期由 Start/End 断言兜住）。
func (m *BaseComponentComposable) ToExecutable() executable.GeneralExecutor {
	return m.toExecutableFn()
}

// validateSchemas 批/流 schema 键冲突检测（_workflow.py:545-569）。
func (w *BaseWorkflow) validateSchemas(compID string, inputsSchema, outputsSchema,
	streamInputsSchema, streamOutputsSchema any) error {
	inMap, _ := inputsSchema.(map[string]any)
	sInMap, _ := streamInputsSchema.(map[string]any)
	if inMap != nil && sInMap != nil {
		fi := flattenDict(inMap)
		fs := flattenDict(sInMap)
		for k := range fi {
			if _, dup := fs[k]; dup {
				return exception.BuildError(exception.StatusWorkflowComponentSchemaInvalid,
					exception.WithParam("comp_id", compID),
					exception.WithParam("reason", "duplicate key both exist in inputs_schema with stream_inputs_schema, key="+k),
					exception.WithParam("workflow", w.workflowConfig.Card.ToStr()))
			}
		}
	}
	outMap, _ := outputsSchema.(map[string]any)
	sOutMap, _ := streamOutputsSchema.(map[string]any)
	if outMap != nil && sOutMap != nil {
		fo := flattenDict(outMap)
		fs := flattenDict(sOutMap)
		for k := range fo {
			if _, dup := fs[k]; dup {
				return exception.BuildError(exception.StatusWorkflowComponentSchemaInvalid,
					exception.WithParam("reason", "duplicate key both exist in outputs_schema with stream_outputs_schema, key="+k))

		}
	}
	return nil
}

// flattenDict 递归展平嵌套 dict，键以 "." 连接（Python dict_utils.flatten_dict）。
func flattenDict(d map[string]any) map[string]any {
	out := map[string]any{}
	var walk func(prefix string, m map[string]any)
	walk = func(prefix string, m map[string]any) {
		for k, v := range m {
			key := k
			if prefix != "" {
				key = prefix + "." + k
			}
			if sub, ok := v.(map[string]any); ok {
				walk(key, sub)
			} else {
				out[key] = v
			}
		}
	}
	walk("", d)
	return out
}

func (w *BaseWorkflow) validateEdge(srcCompID any, targetCompID string, code exception.StatusCode) error {
	toStr := func() string { return w.workflowConfig.Card.ToStr() }
	switch v := srcCompID.(type) {
	case string:
		if v == "" {
			return exception.BuildError(code,
				exception.WithParam("src_comp_id", v), exception.WithParam("target_comp_id", targetCompID),
				exception.WithParam("reason", "src_comp_id cannot be empty or None"),
				exception.WithParam("workflow", toStr()))
		}
	case []string:
		for _, id := range v {
			if id == "" {
				return exception.BuildError(code,
					exception.WithParam("src_comp_id", strings.Join(v, ",")), exception.WithParam("target_comp_id", targetCompID),
					exception.WithParam("reason", "src_comp_id list contains empty or None value"),
					exception.WithParam("workflow", toStr()))
			}
		}
	default:
		return exception.BuildError(code,
			exception.WithParam("target_comp_id", targetCompID),
			exception.WithParam("reason", "src_comp_id must be a string or list[string]"),
			exception.WithParam("workflow", toStr()))
	}
	return nil
}

func (w *BaseWorkflow) AddWorkflowComp(
	compID string,
	workflowComp components.ComponentComposable,
	waitForAll *bool,
	inputsSchema, outputsSchema any,
	streamInputsSchema, streamOutputsSchema any,
	compAbility []components.ComponentAbility,
	maxRetries int,
	timeout float64,
	exceptionConfig *ExceptionConfig,
	name *string,
) error {
	if err := w.validateCompID(compID); err != nil {
		return err
	}
	if err := w.validateSchemas(compID, inputsSchema, outputsSchema, streamInputsSchema, streamOutputsSchema); err != nil {
		return err
	}
	if err := w.validateCompAbility(compID, compAbility, waitForAll); err != nil {
		return err
	}
	nodeSpec, err := NewNodeSpec(
		WithIOConfigs(inputsSchema, outputsSchema),
		WithStreamIOConfigs(streamInputsSchema, streamOutputsSchema),
		WithAbilities(compAbility),
		WithMaxRetries(maxRetries),
		WithTimeout(timeout),
		WithExceptionConfig(exceptionConfig),
		WithSpecName(name),
	)
	if err != nil {
		return err
	}
	w.workflowSpec.CompConfigs[compID] = nodeSpec
	wf := false
	if waitForAll == nil { // ≙ Python wait_for_all is None → False（_workflow.py:196-197）

	return nil

// AddConnection 常规边（_workflow.py:225-243；spec.Edges 单/多源双分支一比一）。
func (w *BaseWorkflow) AddConnection(srcCompID any, targetCompID string) error {
	if err := w.validateEdge(srcCompID, targetCompID, exception.StatusWorkflowEdgeInvalid); err != nil {
		return err
	}
	switch v := srcCompID.(type) {
	case string:
		w.graph.AddEdge([]string{v}, targetCompID)
		w.appendEdge(v, targetCompID)
	case []string:
		w.graph.AddEdge(v, targetCompID)
		for _, src := range v {
			w.appendEdge(src, targetCompID)
		}
	}
	return nil
}

// AddStreamConnection 流式边（_workflow.py:245-257；目标可执行体注册为流消费者）。
func (w *BaseWorkflow) AddStreamConnection(srcCompID string, targetCompID string) error {
	if err := w.validateEdge(srcCompID, targetCompID, exception.StatusWorkflowStreamEdgeInvalid); err != nil {
		return err
	}
	w.graph.AddEdge([]string{srcCompID}, targetCompID)
	nodes := w.graph.GetNodes()
	if tgt, ok := nodes[targetCompID]; ok {
		w.streamActor.AddStreamConsumer(tgt, targetCompID) // Python stream_executables[target_comp_id]
	}
	w.appendStreamEdge(srcCompID, targetCompID)
	return nil
}

// AddConditionalConnection 条件边（_workflow.py:259-295 四重校验 + BranchRouter/普通 router 双分支）。
func (w *BaseWorkflow) AddConditionalConnection(srcCompID any, router any) error {
	toStr := func() string { return w.workflowConfig.Card.ToStr() }
	src, ok := srcCompID.(string)
	if !ok || src == "" {
		return exception.BuildError(exception.StatusWorkflowConditionEdgeInvalid,
			exception.WithParam("reason", "src_comp_id cannot be empty or None"),
			exception.WithParam("workflow", toStr()))
	}
	if router == nil {
		return exception.BuildError(exception.StatusWorkflowConditionEdgeInvalid,
			exception.WithParam("src_comp_id", src),
			exception.WithParam("reason", "router function is required for conditional edges"),
			exception.WithParam("workflow", toStr()))
	}
	switch r := router.(type) {
	case *components.BranchRouter:
		r.SetSession(w.session)
		w.graph.AddConditionalEdges(src, r)
		allTargets := r.AllTargets()
		if len(allTargets) > 1 { // CNF barrier 解析登记（_workflow.py:279-282）
			w.graph.RegisterBranchTargets(src, allTargets)
		}
	case pregel.Selector: // 普通 router：签名嗅探 session 形参后包装（_workflow.py:284-291）
		w.graph.AddConditionalEdges(src, wrapRouterWithSession(r, w.session))
	default:
		return exception.BuildError(exception.StatusWorkflowConditionEdgeInvalid,
			exception.WithParam("src_comp_id", src),
			exception.WithParam("reason", "router must be a callable function"),
			exception.WithParam("workflow", toStr()))
	}
	return nil
}

// validateEdgeNodes 步骤2（_workflow.py:363-376；Python raise → Go panic 构造期纪律）。
func (w *BaseWorkflow) validateEdgeNodes(topo *EdgeTopology) {
	registered := map[string]struct{}{}
	for id := range w.workflowSpec.CompConfigs { registered[id] = struct{}{} }
	var missing []string
	for id := range topo.allEdgeNodes() {
		if _, ok := registered[id]; !ok { missing = append(missing, id) }
	}
	if len(missing) == 0 { return }
	sort.Strings(missing)
	panic(exception.BuildError(exception.StatusWorkflowCompileError,
		exception.WithParam("reason", "Component ID mismatch: nodes "+strings.Join(missing, ",")+
			" are referenced in edges but not registered via add_workflow_comp/set_start_comp/set_end_comp.\n"+
			w.collectProblematicEdges(topo, missing)))
}

// collectProblematicEdges 明细拼接（_workflow.py:378-388；ConnectionType 仅此处消费）。
func (w *BaseWorkflow) collectProblematicEdges(topo *EdgeTopology, missing []string) string {
	miss := map[string]struct{}{}
	for _, m := range missing { miss[m] = struct{}{} }
	var lines []string
	for _, pair := range []struct{ label string; m map[string][]string }{
		{"connection", topo.SourceMap}, {"stream_connection", topo.SourceStreamMap},
	} {
		for src, ts := range pair.m {
			for _, tgt := range ts {
				if _, bad := miss[src]; bad { lines = append(lines, "  - "+pair.label+": '"+src+"' -> '"+tgt+"'") }
				if _, bad := miss[tgt]; bad { lines = append(lines, "  - "+pair.label+": '"+src+"' -> '"+tgt+"'") }
			}
		}
	}
	return strings.Join(lines, "\n")
}

// completeStreamNodeAbilities 步骤4b（_workflow.py:413-435；三分支含 elif 非 start 补 STREAM）。
func (w *BaseWorkflow) completeStreamNodeAbilities(topo *EdgeTopology, userProvided map[string]bool) {
	for node := range topo.SourceStreamMap {
		if userProvided[node] { continue }
		if _, ok := topo.TargetMap[node]; ok { w.addAbilityToNode(node, components.AbilityStream) }
		if _, ok := topo.TargetStreamMap[node]; ok {
			w.addAbilityToNode(node, components.AbilityTransform)
		} else if !w.isStartNode(node) { // when branch target node with stream edge
			w.addAbilityToNode(node, components.AbilityStream)
		}
	}
	for node := range topo.TargetStreamMap {
		if userProvided[node] { continue }
		if _, ok := topo.SourceMap[node]; ok { w.addAbilityToNode(node, components.AbilityCollect) }
	}
}

func (w *Workflow) executeWithTimeout(ctx context.Context, fn func(context.Context) (any, error),
	timeout time.Duration) (any, error) {
	logC := logger.ComponentAgentCore
	task, _ := utils.CreateBackgroundTask(ctx, fn, "workflow_exec_timeout:"+w.card.ID, "workflow")
	waitCtx := ctx
	var cancel context.CancelFunc
	if timeout > 0 { // ≙ Python wait_for(..., timeout if (timeout and timeout>0) else None)
		waitCtx, cancel = context.WithTimeout(ctx, timeout)
		defer cancel()
	}
	errCh := make(chan error, 1)
	var result any
	go func() {
		r, e := task.WaitWithContext(waitCtx) // BackgroundTask.Wait 语义 + waitCtx 取消感知
		result, errCh <- r, e
	}()
	select {
	case <-waitCtx.Done():
		if errors.Is(waitCtx.Err(), context.DeadlineExceeded) {
			logger.Error(logC).Str("workflow_id", w.card.ID).Str("workflow_name", w.card.Name).
				Err(waitCtx.Err()).Str("event_type", "WORKFLOW_EXECUTE_ERROR").
				Float64("timeout", timeout.Seconds()).Msg("Workflow execution timeout")
			if !taskDone(task) { _ = task.Stop(time.Second) } // finally: 未完则取消
			return nil, exception.BuildError(exception.StatusWorkflowExecutionTimeout,
				exception.WithParam("timeout", timeout.String()),

		}

			Err(waitCtx.Err()).Str("event_type", "WORKFLOW_EXECUTE_ERROR").
			Msg("Workflow execution cancelled")
		if !taskDone(task) { _ = task.Stop(time.Second) }
		return nil, waitCtx.Err()
	case e := <-errCh:
		if !taskDone(task) { _ = task.Stop(time.Second) }
		if e == nil {
			return result, nil
		}
		var re pregel.RecursionError
		if errors.As(e, &re) {

// Stream 流式执行门面（workflow.py:410-469 一比一；A' 签名：启动错误走第二返回值）。

	}

// streamInternal 内部流循环（workflow.py:501-592 步骤一比一；主循环中转禁直透——spec §10）。
func (w *Workflow) streamInternal(ctx context.Context, inputs any, ws *workflowSession.WorkflowSession,
	opts ...InvokeOption) <-chan workflow.WorkflowChunk {
	out := make(chan workflow.WorkflowChunk)
	go func() { // ② 转发协程（spec §10 注记 3）
		defer close(out)
		tracer.TracerWorkflowUtils.TraceWorkflowStart(ctx, ws, inputs)
		trigger(WorkflowStarted, w.card.ID, w.card.Name, inputs)

		timeout := envDuration(ws, constants.WorkflowExecuteTimeoutEnvKey)
		frameTimeout := envDuration(ws, constants.WorkflowStreamFrameTimeoutEnvKey)
		if timeout > 0 && timeout <= frameTimeout { frameTimeout = timeout }      // workflow.py:516-517
		ws.Config().SetEnvs(map[string]any{constants.WorkflowStreamFrameTimeoutEnvKey: frameTimeout})
		firstFrameTimeout := envDuration(ws, constants.WorkflowStreamFirstFrameTimeoutEnvKey)
		if timeout > 0 && timeout <= firstFrameTimeout { firstFrameTimeout = timeout } // :520-521
		ws.Config().SetEnvs(map[string]any{constants.WorkflowStreamFirstFrameTimeoutEnvKey: firstFrameTimeout})

		// 后台图执行（① stream_process；workflow.py:524-537）
		task, terr := utils.CreateBackgroundTask(ctx, func(ctx context.Context) error {
			compiled, err := w.internal.Compile(ws, nil)
			if err != nil {
				return err
			}
			defer func() { // ≙ finally + asyncio.shield（D-c）
				outputs := ws.State().GetOutputs(w.endCompID)
				tracer.TracerWorkflowUtils.TraceWorkflowDone(ctx, ws, outputs)
				_ = ws.StreamWriterManager().StreamEmitter().Close(ctx)
			}()
			_, err = compiled.Invoke(ctx, map[string]any{graphexe.InputsKey: inputs, graphexe.ConfigKey: nil}, ws)
			return err
		}, "workflow_stream_process:"+w.card.ID, "workflow")
		if terr != nil {
			out <- chunkOf(terr)
			return
		}
		var chunks []workflow.WorkflowChunk
		idx := 0
		emit := ws.StreamWriterManager().StreamOutput(
			stream.WithFirstFrameTimeout(firstFrameTimeout),
			stream.WithTimeout(frameTimeout))
		for {
			select {
			case <-ctx.Done(): // 主循环取消（workflow.py:567-580）
				trigger(WorkflowCancelled, w.card.ID, w.card.Name)
				logger.Warn(logger.ComponentAgentCore).
					Str("workflow_id", w.card.ID).Str("workflow_name", w.card.Name).
					Str("event_type", "WORKFLOW_EXECUTE_ERROR").Msg("Canecel stream output task")
				if !taskDone(task) { _ = task.Stop(time.Second) }
				w.cleanup(ws)
				return
			case chunk, ok := <-emit:
				if !ok { goto drained }
				logger.Debug(logger.ComponentAgentCore).
					Str("workflow_id", w.card.ID).Str("workflow_name", w.card.Name).
					Int("chunk_idx", idx).Str("event_type", "WORKFLOW_OUTPUT_CHUNK").
					Msg(fmt.Sprintf("Output workflow chunk[%d]", idx))
				out <- chunk
				chunks = append(chunks, chunk)
				idx++
			}
		}
	drained:
		if err := waitTask(ctx, task); err != nil { // task.Wait 处取消（workflow.py:559-566）
			if errors.Is(err, context.Canceled) {
				logger.Warn(logger.ComponentAgentCore).
					Str("workflow_id", w.card.ID).Str("workflow_name", w.card.Name).
					Str("event_type", "WORKFLOW_EXECUTE_ERROR").
					Msg("Workflow stream output be cancelled")
				w.cleanup(ws)
				return
			}
			trigger(WorkflowError, w.card.ID, w.card.Name, err) // BaseError（:581-587）
			w.cleanup(ws)
			return
		}
		results := ws.State().GetOutputs(w.endCompID)
		trigger(WorkflowFinished, w.card.ID, w.card.Name, results) // workflow.py:552-556
		if results != nil && len(results) > 0 {
			out <- workflow.WorkflowChunk(stream.OutputSchema{ // workflow.py:557-558
				Type: "workflow_final", Index: 0, Payload: results,
			})
		}
		w.cleanup(ws) // finally：cancel 未完任务 + shield(session.Close) + reset
	}()
	return out
}

// cleanup 收尾（workflow.py:588-592 finally 一比一；D-c defer 承载 shield）。
func (w *Workflow) cleanup(ws *workflowSession.WorkflowSession) {
	_ = ws.Close()
	_ = w.internal.Reset(context.Background())
}
