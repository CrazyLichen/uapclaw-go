```go
package schema

// ──────────────────────────── 结构体 ────────────────────────────

// ReasoningConfig 厂商中立的思考/推理请求意图。
// Python: schema/config.py:59-66 ReasoningConfig（model_config extra="forbid"，不接受额外键）
type ReasoningConfig struct {
	// Mode 思考模式，缺省 auto；取值 auto/enabled/disabled
	Mode string `json:"mode"`
	// Effort 强度档位（厂商自定义取值，如 low/medium/high/xhigh/max）
	Effort *string `json:"effort,omitempty"`
	// BudgetTokens 思考预算（token），nil 表示未设置
	BudgetTokens *int `json:"budget_tokens,omitempty"`
}

// ReasoningIntent 区分「未显式传 reasoning」与「显式传了 reasoning（值可为 nil）」。
// Python: reasoning.py:33 UNSET_REASONING 哨兵 + reasoning.py:300-306 _explicit_reasoning_value
type ReasoningIntent struct {
	// Set 是否显式传入 reasoning
	Set bool
	// Value 传入的值；Set=true 且 Value=nil 表示显式传了 null
	Value any
}

// ──────────────────────────── 常量 ────────────────────────────

// reasoningModeAuto ReasoningConfig.Mode 的缺省值。
// Python: schema/config.py:62 Field(default="auto")
const reasoningModeAuto = "auto"

// newReasoningKeys 新式 reasoning 配置允许的键集合。
// Python: reasoning.py:35 _NEW_REASONING_KEYS
var newReasoningKeys = map[string]struct{}{
	"mode":          {},
	"effort":        {},
	"budget_tokens": {},
}

// ──────────────────────────── 导出函数 ────────────────────────────

// NewReasoningConfig 创建带缺省值的 ReasoningConfig。
// Go 的 Mode 零值是 ""，而 Python 默认 "auto"；不修正会让 resolve_reasoning_plan
// 的 `mode == "auto" && 无 effort && 无 budget` 三条件早返回失效。
func NewReasoningConfig() *ReasoningConfig {
	return &ReasoningConfig{Mode: reasoningModeAuto}
}

// CoerceReasoningConfig 判断 reasoning 值属于新式配置还是遗留裸 dict。
// 返回 (cfg, nil) 表示新式；(nil, raw) 表示遗留，需原样透传；(nil, nil) 表示未设置。
// 判定时机与 Python 一致：每次 resolve 时判定，不在反序列化时定死。
// Python: reasoning.py:255-265 _coerce_reasoning_config
func CoerceReasoningConfig(value any) (cfg *ReasoningConfig, raw map[string]any) {
	if value == nil {
		return nil, nil
	}
	if c, ok := value.(*ReasoningConfig); ok {
		return c, nil
	}
	m, ok := value.(map[string]any)
	if !ok {
		return nil, nil
	}
	// 键集必须全部落在 {mode, effort, budget_tokens} 内才是新式
	for key := range m {
		if _, known := newReasoningKeys[key]; !known {
			return nil, m
		}
	}
	return reasoningConfigFromMap(m), nil
}

// IsReasoningConfigIntent 判定是否为「新式配置意图」。
// Python: reasoning.py:268-270 is_reasoning_config_intent
func IsReasoningConfigIntent(value any) bool {
	cfg, _ := CoerceReasoningConfig(value)
	return cfg != nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// reasoningConfigFromMap 由 map 构造 ReasoningConfig（仅用于键集已受限的输入）。
// Python: reasoning.py:264 ReasoningConfig(**raw)
func reasoningConfigFromMap(m map[string]any) *ReasoningConfig {
	out := &ReasoningConfig{Mode: reasoningModeAuto}
	if v, ok := m["mode"].(string); ok {
		out.Mode = v
	}
	if v, ok := m["effort"].(string); ok {
		out.Effort = &v
	}
	if v, ok := m["budget_tokens"]; ok {
		out.BudgetTokens = toIntPtr(v)
	}
	return out
}

// toIntPtr JSON 数字反序列化后是 float64，需转 int；非数字返回 nil。
func toIntPtr(v any) *int {
	switch n := v.(type) {
	case int:
		return &n
	case int64:
		i := int(n)
		return &i
	case float64:
		i := int(n)
		return &i
	}
	return nil
}
```

```go
func NormalizeModelClientConfig(config *llmschema.ModelClientConfig) *llmschema.ModelClientConfig {
	if config == nil {
		return nil
	}
	// 浅拷贝（Extensions/Extra 深拷贝，避免共享）
	out := *config
	out.Extra = cloneAnyMap(config.Extra)
	out.Extensions = cloneExtensions(config.Extensions)

	provider := config.ClientProvider
	alias, ok := legacyProviderAliases[provider]
	if !ok {
		return &out
	}

	// 协议为 anthropic_messages 且 provider 在 anthropic 兼容集合内 → 改用 Anthropic 客户端
	// Python: endpoint_profiles.py:203-205

		if _, compat := anthropicCompatibleProviders[provider]; compat {
			out.ClientProvider = "Anthropic"
			setLegacyProvider(&out, provider)
			return &out
		}
	}

	out.ClientProvider = alias.ClientProvider
	// 显式 endpoint_profile 优先，别名只补空缺
	if out.EndpointProfile == "" && alias.EndpointProfile != "" {
		out.EndpointProfile = alias.EndpointProfile
	}
	// OpenAIAccount 的 auth_mode=api_key / api_mode="" 是打印出来的默认值，不算用户意图
	// Python: endpoint_profiles.py:183-190 _is_implicit_openai_account_default

		out.AuthMode == llmschema.AuthModeAPIKey) {

	}

		}

	}

	return &out
}

		return ""

// isImplicitOpenAIAccountDefault 判断是否是 OpenAIAccount 隐式默认值。
// Python: endpoint_profiles.py:183-190 _is_implicit_openai_account_default
func isImplicitOpenAIAccountDefault(provider, key string, current any) bool {
	if provider != "OpenAIAccount" {
		return false
	}
	switch key {
	case "api_mode":
		return current == "" || current == nil
	case "auth_mode":
		return current == "api_key" || current == llmschema.AuthModeAPIKey
	}
	return false
}

// cloneAnyMap 浅拷贝 map。
func cloneAnyMap(in map[string]any) map[string]any {
	if in == nil {
		return nil
	}
	out := make(map[string]any, len(in))
	for k, v := range in {
		out[k] = v
	}
	return out
}

// cloneExtensions 深拷贝扩展配置。

		return nil

		KVCache:          in.KVCache,
		ResponseFields:   cloneAnyMap(in.ResponseFields),
		RequestExtraBody: cloneAnyMap(in.RequestExtraBody),
	}
	return out
}
```

```go
// FindReasoningProfile 按 provider + 模型名查找档案；provider 作用域优先于模型名 fallback。
// Python: reasoning_profiles.py:367-383 find_reasoning_profile
func FindReasoningProfile(provider, model string) *ReasoningProfile {
	providerKey := strings.ReplaceAll(strings.ToLower(strings.TrimSpace(provider)), "_", "-")
	modelKey := strings.ToLower(strings.TrimSpace(model))

	for i := range reasoningProfiles {
		p := &reasoningProfiles[i]
		if p.Provider != providerKey {
			continue
		}
		if matchAny(modelKey, p.ModelPatterns) {
			return p
		}
	}

	// fallback：同时用完整 modelKey 与去掉 vendor 前缀后的裸名匹配
	candidates := []string{modelKey}
	if idx := strings.LastIndex(modelKey, "/"); idx >= 0 {
		candidates = append(candidates, modelKey[idx+1:])
	}
	for i := range modelReasoningFallbacks {
		p := &modelReasoningFallbacks[i]
		for _, c := range candidates {
			if matchAny(c, p.ModelPatterns) {
				return p
			}
		}
	}
	return nil
}

// StandardReasoningCapability 返回协议默认能力。
// Python: reasoning_profiles.py:386-389 standard_reasoning_capability
func StandardReasoningCapability(protocol string) ReasoningCapability {
	if strings.ToLower(strings.TrimSpace(protocol)) == "anthropic_messages" {
		return standardAnthropicCapability
	}
	return standardOpenAICapability
}

// ProfileReasoningCapability 返回档案在指定协议下的能力。
// Python: reasoning_profiles.py:391-395 profile_reasoning_capability
func ProfileReasoningCapability(profile *ReasoningProfile, protocol string) ReasoningCapability {
	if profile == nil {
		return StandardReasoningCapability(protocol)
	}
	if strings.ToLower(strings.TrimSpace(protocol)) == "anthropic_messages" && profile.AnthropicCapability != nil {
		return *profile.AnthropicCapability
	}
	return profile.Capability
}

// ProviderKey 由 provider 名 / endpoint_profile / api_base 推断生效的 provider key。
// 顺序：非通用 endpoint_profile 直接用它 → api_base host 推断 → provider 名。
// Python: reasoning.py:113-131 _provider_key_from_values
func ProviderKey(provider, endpointProfile, apiBase string) string {
	profile := strings.ReplaceAll(strings.ToLower(strings.TrimSpace(endpointProfile)), "_", "-")
	if _, generic := genericEndpointProfiles[profile]; !generic {
		return profile
	}
	if inferred := providerFromAPIBase(apiBase); inferred != "" {
		return inferred
	}
	if profile != "" {
		return profile
	}
	return strings.ReplaceAll(strings.ToLower(strings.TrimSpace(provider)), "_", "-")
}

// ProviderFromAPIBase 由 api_base host 推断 provider；无法推断返回空串。
// Python: reasoning.py:122-131 _provider_from_api_base

// GetReasoningCapability 解析 UI 可用的思考档位，不构造客户端。
// 查询是建议性的：表内模型返回精确能力，未列模型继承所选协议的默认能力。
// Python: reasoning.py:150-177 get_reasoning_capability
func GetReasoningCapability(provider, model, protocol, apiBase, endpointProfile string) ReasoningCapability {
	protocolKey := strings.ToLower(strings.TrimSpace(protocol))
	switch protocolKey {
	case "anthropic", "messages", "anthropic-messages":
		protocolKey = protocolAnthropicMessages
	case "openai", "chat", "chat-completions":
		protocolKey = protocolChatCompletions
	}
	providerKey := ProviderKey(provider, endpointProfile, apiBase)
	if profile := FindReasoningProfile(providerKey, model); profile != nil {
		return ProfileReasoningCapability(profile, protocolKey)
	}
	return StandardReasoningCapability(protocolKey)
}

// GetProviderReasoningRules 返回该 provider 的 pattern 规则，供前端能力匹配。
// 顺序敏感：调用方须取第一条命中项，与 FindReasoningProfile 一致。
// Python: reasoning.py:180-215 get_provider_reasoning_rules
func GetProviderReasoningRules(provider, apiBase, endpointProfile string) []map[string]any {
	providerKey := ProviderKey(provider, endpointProfile, apiBase)
	out := make([]map[string]any, 0, len(reasoningProfiles))
	for i := range reasoningProfiles {
		p := &reasoningProfiles[i]
		if p.Provider != providerKey {
			continue
		}
		out = append(out, map[string]any{
			"patterns": p.ModelPatterns,
			"capabilities": map[string]any{
				"openai":    capabilityToMap(ProfileReasoningCapability(p, protocolChatCompletions)),
				"anthropic": capabilityToMap(ProfileReasoningCapability(p, protocolAnthropicMessages)),
			},
		})
	}
	return out
}
```

```go
// ResolveReasoningProtocol 判定客户端使用哪种协议。
// Python: reasoning.py:134-147
func ResolveReasoningProtocol(clientConfig *llmschema.ModelClientConfig) string {
	if clientConfig == nil {
		return protocolChatCompletions
	}
	provider := strings.ReplaceAll(strings.ToLower(strings.TrimSpace(ProviderIdentity(clientConfig))), "_", "")
	if clientConfig.APIMode == protocolAnthropicMessages || provider == "anthropic" {
		return protocolAnthropicMessages
	}
	if clientConfig.APIMode == protocolResponses ||
		clientConfig.AuthMode == llmschema.AuthModeOpenAIAccountOAuth ||
		provider == "openaiaccount" {
		return protocolResponses
	}
	return protocolChatCompletions
}

// ResolveReasoningPlan 把厂商中立的 reasoning 意图翻译成协议相关的线上参数。
// Python: reasoning.py:309-351
func ResolveReasoningPlan(
	clientConfig *llmschema.ModelClientConfig,
	modelConfig *llmschema.ModelRequestConfig,
	requestModel string,
	explicit llmschema.ReasoningIntent,
) ReasoningPlan {
	protocol := ResolveReasoningProtocol(clientConfig)
	model := modelName(modelConfig, requestModel)
	provider := providerKey(clientConfig)

	// 1. 显式 kwargs 里的 reasoning 优先，其次 model_config.reasoning
	reasoningValue := explicitReasoningValue(modelConfig, explicit)
	currentValues := mergeForLegacyDetection(modelConfig, explicit)
	cfg, raw := llmschema.CoerceReasoningConfig(reasoningValue)

	// 2. 遗留裸 dict：Chat Completions / Responses 原样透传
	if raw != nil {
		if protocol == protocolChatCompletions || protocol == protocolResponses {
			return ReasoningPlan{SDKParams: map[string]any{"reasoning": raw}}
		}
		return ReasoningPlan{}
	}

	// 3. 未设置
	if cfg == nil {
		return ReasoningPlan{}
	}

	// 4. 遗留控制已存在 → 告警并放弃
	legacyValues := currentValues
	if explicit.Set {
		legacyValues = explicitKWVals(explicit)
	}
	if hasLegacyReasoningControls(legacyValues) {
		return ReasoningPlan{Warnings: []string{
			"reasoning config ignored because explicit legacy reasoning controls are already set",
		}}
	}

	// 5. 按协议分发
	switch protocol {
	case protocolAnthropicMessages:
		return anthropicReasoningPlan(cfg, clientConfig, model)
	case protocolResponses:
		return openAIResponsesReasoningPlan(cfg)
	default:
		return openAIChatReasoningPlan(provider, model, cfg, clientConfig)
	}
}

// ApplyReasoningPlan 把 plan 写入请求参数。
// Python: reasoning.py:935-962
func ApplyReasoningPlan(params map[string]any, plan ReasoningPlan, override bool) {
	for _, w := range plan.Warnings {
		logger.Warn(logComponent).Str("warning", w).Msg("Reasoning config resolution")
	}
	for k, v := range plan.SDKParams {
		if _, exists := params[k]; override || !exists {
			params[k] = v
		}
	}
	if len(plan.ExtraBody) == 0 {
		return
	}
	existing, exists := params["extra_body"]
	if !exists || existing == nil {
		merged := make(map[string]any, len(plan.ExtraBody))
		for k, v := range plan.ExtraBody {
			merged[k] = v
		}
		params["extra_body"] = merged
		return
	}
	m, ok := existing.(map[string]any)
	if !ok {
		logger.Warn(logComponent).Msg("Reasoning extra_body dropped: request extra_body is not a mapping")
		return
	}
	if override {
		for k, v := range plan.ExtraBody {
			m[k] = v
		}
		return
	}
	// 非 override：已有值优先，只补空缺
	for k, v := range plan.ExtraBody {
		if _, has := m[k]; !has {
			m[k] = v
		}
	}
}

// ReasoningRequestControls 从最终请求中提取可安全日志输出的 wire 控制字段。
// Python: reasoning.py:965-981
func ReasoningRequestControls(params map[string]any) map[string]any {
	controls := make(map[string]any)
	for _, k := range reasoningWireKeys {
		if v, ok := params[k]; ok {
			controls[k] = v
		}
	}
	if eb, ok := params["extra_body"].(map[string]any); ok {
		extra := make(map[string]any)
		for _, k := range reasoningWireKeys {
			if v, ok := eb[k]; ok {
				extra[k] = v
			}
		}
		if len(extra) > 0 {
			controls["extra_body"] = extra
		}
	}
	return controls
}
```

```go
import "strings"

// ──────────────────────────── 结构体 ────────────────────────────

// ModelParamRule 按模型名注入 extra_body 字段的规则。
// Python: openai_model_client.py:80-85 ModelParamRule
type ModelParamRule struct {
	// Name 规则名（用于日志与排错）
	Name string
	// Predicate 命中判定
	Predicate func(model string) bool
	// ExtraBodyFields 命中后合并进 extra_body 的字段
	ExtraBodyFields map[string]any
}

// ──────────────────────────── 全局变量 ────────────────────────────

// defaultModelParamRules 默认规则集（上游只有 MiniMax 一条）。
// Python: openai_model_client.py:87-93 _DEFAULT_MODEL_PARAM_RULES
var defaultModelParamRules = []ModelParamRule{
	{
		Name: "minimax_reasoning_split",
		// 大小写敏感：上游用 startswith("MiniMax-M")，小写不匹配
		Predicate:       func(m string) bool { return strings.HasPrefix(m, "MiniMax-M") },
		ExtraBodyFields: map[string]any{"reasoning_split": true},
	},
}

// ──────────────────────────── 导出函数 ────────────────────────────

// ApplyModelSpecificParams 按模型名把规则的 extra_body 字段就地合并进 params。
// 已有键保留；多条规则命中时后者覆盖前者同名键。
// Python: openai_model_client.py:486-503 _apply_model_specific_params
func ApplyModelSpecificParams(rules []ModelParamRule, model string, params map[string]any) {
	if model == "" {
		return
	}
	for _, rule := range rules {
		if rule.Predicate == nil || !rule.Predicate(model) || len(rule.ExtraBodyFields) == 0 {
			continue
		}
		extraBody, _ := params["extra_body"].(map[string]any)
		if extraBody == nil {
			extraBody = make(map[string]any, len(rule.ExtraBodyFields))
		}
		for k, v := range rule.ExtraBodyFields {
			extraBody[k] = v
		}
		params["extra_body"] = extraBody
	}
}
```

```go
		if !ok {
			continue
		}
		kept := make([]any, 0, len(calls))
		for _, c := range calls {
			if !isOrphanToolCall(c) {
				kept = append(kept, c)
			}
		}
		if len(kept) == len(calls) {
			continue
		}
		dropped := len(calls) - len(kept)
		repaired := make(map[string]any, len(msg))
		for k, v := range msg {
			repaired[k] = v
		}
		if len(kept) == 0 {
			delete(repaired, "tool_calls")
		} else {
			repaired["tool_calls"] = kept
		}
		messages[pos] = repaired
		logger.Warn(logComponent).
			Int("dropped", dropped).
			Int("position", pos).
			Msg("Dropped tool call(s) without id and name from history message.")
	}
	return messages
}

// isOrphanToolCall 判定 tool_call 是否既无 id 又无 function.name。
// Python: llm_provider_compat_patch.py:142-145 _is_orphan_tool_call
func isOrphanToolCall(call any) bool {
	m, ok := call.(map[string]any)
	if !ok {
		return false
	}
	id, _ := m["id"].(string)
	if id != "" {
		return false
	}
	fn, _ := m["function"].(map[string]any)
	if fn == nil {
		return true
	}
	name, _ := fn["name"].(string)
	return name == ""
}
```
