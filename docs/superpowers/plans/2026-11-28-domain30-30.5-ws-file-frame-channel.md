# 30.5 WS 文件分帧通道 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为知识库管理面新增 kb.* WS 文件分帧通道：上传（控制面/数据面分离→流式落盘→hash 校验）与反向流式下载，gateway 薄映射复用既有 file.* 协议，止步安全落盘不触发解析（spec：`docs/superpowers/specs/2026-11-28-domain30-30.5-ws-file-frame-channel-design.md`）。

**Architecture:** B1 双层复用——前端↔gateway 用 kb.* 方法族；gateway 把数据面 `kb.file.upload_chunk` 映射为既有 `file.upload_chunk` 协议转发（其余控制面原样转发）；server 侧 kbadmin DocService 承接 upload/complete/cancel/download，落盘复用 `attachments` 惯例（消毒/去重槽位），落盘根目录置于 `pathutil.WorkspaceDir()/kb_uploads/`（保证既有 `isPathWithinUserDirs` 边界校验放行）。上传槽位内存登记（TTL 清理 + 串行守卫），complete 时复算 sha256 比对。

**Tech Stack:** Go 1.x、GORM（既有 Store）、gorilla WS（既有通道）、`schema.ReqMethod` 注册表。

```go
package kbadmin

import (
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"io"
	"os"
	"path/filepath"
	"strings"
	"sync"
	"time"

	"github.com/uapclaw/uapclaw-go/internal/common/utils/path"
	"github.com/uapclaw/uapclaw-go/internal/swarm/server/runtime/attachments"
)

// ──────────────────────────── 全局变量 ────────────────────────────

// ErrFileTooLarge 超过单文件上限
var ErrFileTooLarge = errors.New("file too large")

// ErrUploadChannelBusy 同 upload_id 并发帧/重复发起
var ErrUploadChannelBusy = errors.New("upload channel busy")

// ErrKBUploadNotFound upload_id 不存在（超时清理后引用）
var ErrKBUploadNotFound = errors.New("kb upload not found")

// ──────────────────────────── 常量 ────────────────────────────

// MaxFileBytes KB 单文件字节上限（30.5 固定 50MiB；KB config 可调 ⤵️30.2，D-30-5-1）
const MaxFileBytes = attachments.MaxFileBytes

// uploadTTLDefault 槽位默认存活时长
const uploadTTLDefault = 30 * time.Minute

// ──────────────────────────── 结构体 ────────────────────────────

// UploadSlot 一次上传的槽位
type UploadSlot struct {
	UploadID     string    `json:"upload_id"`
	KBID         string    `json:"kb_id"`
	FileName     string    `json:"file_name"`
	TotalSize    int64     `json:"total_size"`
	ExpectedHash string    `json:"expected_hash"`
	ResolvedPath string    `json:"resolved_path"`
	CreatedAt    time.Time `json:"created_at"`
}

// UploadStore KB 上传槽位登记表（内存单点，全局一份）
type UploadStore struct {
	mu    sync.Mutex
	slots map[string]*UploadSlot
	root  string
	ttl   time.Duration
}

// ──────────────────────────── 导出函数 ────────────────────────────

// NewUploadStore 构造槽位表；root 为落盘根目录（默认 WorkspaceDir()/kb_uploads）
func NewUploadStore(root string) *UploadStore {
	return &UploadStore{slots: map[string]*UploadSlot{}, root: root, ttl: uploadTTLDefault}
}

// InitUpload 登记槽位并认领去重落盘路径（O_EXCL 首块认领语义由调用方在首帧走
// AtomicWriteUnique 时落地；此处只建目录与登记）。
func (s *UploadStore) InitUpload(kbID, fileName string, totalSize int64, expectedHash string) (*UploadSlot, error) {
	if totalSize > MaxFileBytes {
		return nil, ErrFileTooLarge
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, exists := s.slots[kbID+"/"+fileName]; exists {
		// 同 KB 同名在途上传 → busy（串行守卫）
		return nil, ErrUploadChannelBusy
	}
	id, err := newUploadID()
	if err != nil {
		return nil, err
	}
	if err := os.MkdirAll(s.root, 0o755); err != nil {
		return nil, err
	}
	dir := filepath.Join(s.root, id)
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return nil, err
	}
	safe := attachments.SafeUploadFilename(fileName, "upload")
	target := attachments.UniqueUploadPath(filepath.Join(dir, safe))
	slot := &UploadSlot{
		UploadID: id, KBID: kbID, FileName: safe,
		TotalSize: totalSize, ExpectedHash: strings.ToLower(expectedHash),
		ResolvedPath: target, CreatedAt: time.Now(),
	}
	s.slots[id] = slot
	return slot, nil
}

// Get 取槽位（不存在返 ErrKBUploadNotFound）
func (s *UploadStore) Get(uploadID string) (*UploadSlot, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	slot, ok := s.slots[uploadID]
	if !ok {
		return nil, ErrKBUploadNotFound
	}
	return slot, nil
}

// Complete 收尾：复算 sha256 比对；一致则摘除槽位并返回落盘路径。
// 调用方负责把文件移交 KB 暂存目录（30.5 X 边界：不触发解析）。
func (s *UploadStore) Complete(uploadID string) (*UploadSlot, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	slot, ok := s.slots[uploadID]
	if !ok {
		return nil, ErrKBUploadNotFound
	}
	sum, err := fileSha256(slot.ResolvedPath)
	if err != nil {
		return nil, err
	}
	if slot.ExpectedHash != "" && sum != slot.ExpectedHash {
		_ = os.RemoveAll(filepath.Dir(slot.ResolvedPath))
		delete(s.slots, uploadID)
		return nil, ErrHashMismatch
	}
	delete(s.slots, uploadID)
	return slot, nil
}

// Cancel 取消并清理分片目录
func (s *UploadStore) Cancel(uploadID string) error {
	s.mu.Lock()
	slot, ok := s.slots[uploadID]
	if ok {
		delete(s.slots, uploadID)
	}
	s.mu.Unlock()
	if !ok {
		return ErrKBUploadNotFound
	}
	return os.RemoveAll(filepath.Dir(slot.ResolvedPath))
}

// CleanupExpired 清理超时未 complete 的槽位（启动方挂后台 goroutine 周期调用）
func (s *UploadStore) CleanupExpired() {
	s.mu.Lock()
	var expired []*UploadSlot
	now := time.Now()
	for id, slot := range s.slots {
		if now.Sub(slot.CreatedAt) > s.ttl {
			expired = append(expired, slot)
			delete(s.slots, id)
		}
	}
	s.mu.Unlock()
	for _, slot := range expired {
		_ = os.RemoveAll(filepath.Dir(slot.ResolvedPath))
	}
}

// FileSha256Hash 对外暴露单文件 sha256（供集成测试比对）——见导出函数区
func FileSha256(p string) (string, error) { return fileSha256(p) }

// ──────────────────────────── 非导出函数 ────────────────────────────

// ErrHashMismatch 收尾 hash 比对不符
var ErrHashMismatch = errors.New("upload hash mismatch")

func newUploadID() (string, error) {
	b := make([]byte, 16)
	if _, err := io.ReadFull(randReader, b); err != nil {
		return "", err
	}
	return hex.EncodeToString(b), nil
}

func fileSha256(p string) (string, error) {
	f, err := os.Open(p)
	if err != nil {
		return "", err
	}
	defer func() { _ = f.Close() }()
	h := sha256.New()
	if _, err := io.Copy(h, f); err != nil {
		return "", err
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}
```

注意：`randReader` 用 `crypto/rand.Reader`——在导入区加 `"crypto/rand"` 并把 `newUploadID` 里 `randReader` 换成 `rand.Reader`（上面代码为省 import 别名，落地时直接用 `rand.Reader`，删除 `randReader` 引用）。

package kbadmin

import (
	"errors"

// ──────────────────────────── 结构体 ────────────────────────────

// UploadedFile upload 收尾结果（X 边界：不含解析任务字段，⤵️30.2 加挂点）
type UploadedFile struct {
	UploadID  string `json:"upload_id"`
	KBID      string `json:"kb_id"`
	FileName  string `json:"file_name"`
	Size      int64  `json:"size"`
	FinalPath string `json:"final_path"`
}

// ──────────────────────────── 导出函数 ────────────────────────────

// SetUploadStore 注入槽位表（Domain 组装期调用；UploadStore 生命周期独立于 SQLite）
func (s *DocService) SetUploadStore(ups *UploadStore) { s.uploads = ups }

// InitUpload kb.doc.upload：校验 KB 存在 → 登记槽位
func (s *DocService) InitUpload(kbID, fileName string, totalSize int64, expectedHash string) (*UploadSlot, error) {
	if _, err := s.store.KB().GetByID(kbID); err != nil {
		return nil, ErrKBNotFound
	}
	return s.uploads.InitUpload(kbID, fileName, totalSize, expectedHash)
}

// CompleteUpload kb.file.upload_complete：hash 复算一致后把文件移入 KB 暂存区。
// **不触发解析**（30.5 X 边界拍板，⤵️30.2 在此加挂点接管建任务）。
func (s *DocService) CompleteUpload(uploadID string) (*UploadedFile, error) {
	slot, err := s.uploads.Complete(uploadID)
	if err != nil {
		return nil, err
	}
	finalPath := filepath.Join(filepath.Dir(slot.ResolvedPath), "..", slot.FileName)
	if err := os.Rename(slot.ResolvedPath, filepath.Clean(finalPath)); err != nil {
		return nil, err
	}
	fi, err := os.Stat(filepath.Clean(finalPath))
	if err != nil {
		return nil, err
	}
	return &UploadedFile{
		UploadID: slot.UploadID, KBID: slot.KBID, FileName: slot.FileName,
		Size: fi.Size(), FinalPath: filepath.Clean(finalPath),
	}, nil
}

// CancelUpload kb.doc.cancel_upload：清理分片
func (s *DocService) CancelUpload(uploadID string) error { return s.uploads.Cancel(uploadID) }

// DownloadChunk kb.doc.download 数据面：有界分块读 KB 暂存文件（反向流式）
func (s *DocService) DownloadChunk(path string, offset, limit int) ([]byte, bool, error) {
	if limit <= 0 {
		limit = 512 * 1024
	}
	f, err := os.Open(filepath.Clean(path))
	if err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return nil, false, ErrKBUploadNotFound
		}
		return nil, false, err
	}
	defer func() { _ = f.Close() }()
	buf := make([]byte, limit)
	n, err := f.ReadAt(buf, int64(offset))
	if err != nil && n == 0 {
		return nil, false, ErrKBUploadNotFound
	}
	eof := offset+n >= fileSizeOf(path)
	return buf[:n], eof, nil
}

// fileSizeOf 单文件大小（download eof 判定）
func fileSizeOf(p string) int64 {
	fi, err := os.Stat(p)
	if err != nil {
		return 0
	}
	return fi.Size()
}
