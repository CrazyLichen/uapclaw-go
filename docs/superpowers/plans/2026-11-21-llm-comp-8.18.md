// validateTemplate 模板校验（llm_comp.py:443-472 一比一）
func validateTemplate(templateContent []any, systemPromptTemplate *llmschema.SystemMessage, userPromptTemplate *llmschema.UserMessage) error {
	// Python: system/user 模板任一非 None 或模板内容为空 → 直接返回
	if systemPromptTemplate != nil || userPromptTemplate != nil || len(templateContent) == 0 {
		return nil
	}
	// 第一遍：校验所有 system 元素（Pydantic SystemMessage.model_validate 等价：
	// role 必须为 "system" 且 content 必填非空）
	for _, element := range templateContent {
		dict, ok := element.(map[string]any)
		if !ok {
			continue // Python dict.get 对非 dict 会 AttributeError，本仓模板约定为 map；保留宽松断言登记于偏差
		}
		if roleString(dict) == "system" && !validSystemElement(dict) {
			return exception.NewStatusCodeError(exception.StatusComponentLLMTemplateConfigError, "system message is invalid")
		}
	}
	// 第二遍：user 必须存在，且 user 之后不得再出现 system
	ifContainUserMessage := false
	for _, element := range templateContent {
		dict := element.(map[string]any)
		if roleString(dict) == "user" {
			if !validUserElement(dict) {
				return exception.NewStatusCodeError(exception.StatusComponentLLMTemplateConfigError, "user message is invalid")
			}
			ifContainUserMessage = true
		}
		if ifContainUserMessage && roleString(dict) == "system" {
			return exception.NewStatusCodeError(exception.StatusComponentLLMTemplateConfigError,
				"system message must be before user message")
		}
	}
	if !ifContainUserMessage {
		return exception.NewStatusCodeError(exception.StatusComponentLLMTemplateConfigError, "user message is required")
	}
	return nil
}

// validateResponseFormat 响应格式校验（llm_comp.py:507-522）
var responseFormatPattern = regexp.MustCompile(`^(text|markdown|json)$`)

func validateResponseFormat(responseFormat map[string]any, outputConfig map[string]any) error {
	typeStr, _ := responseFormat["type"].(string)
	if !responseFormatPattern.MatchString(typeStr) {
		return exception.NewStatusCodeError(exception.StatusComponentLLMResponseConfigInvalid,
			fmt.Sprintf("response format %v is invalid", responseFormat))
	}
	if (typeStr == "text" || typeStr == "markdown") && len(outputConfig) != 1 {
		return exception.NewStatusCodeError(exception.StatusComponentLLMResponseConfigInvalid,
			"output config must contain exactly one parameter for text or markdown response type")
	}
	return nil
}

// validateOutputConfig 输出配置校验（llm_comp.py:475-504）
func validateOutputConfig(outputConfig map[string]any) error {
	if len(outputConfig) == 0 {
		return exception.NewStatusCodeError(exception.StatusComponentLLMConfigError, "output config is empty")
	}
	if t, ok := outputConfig["type"].(string); ok && t == "object" {
		return nil // object 型整体 schema 直通
	}
	for param, value := range outputConfig {
		if param == "" {
			return exception.NewStatusCodeError(exception.StatusComponentLLMConfigError,
				fmt.Sprintf("output config parameter %s is empty", param))
		}
		if !validOutputParam(value) {
			if security.IsSensitive() {
				return exception.NewStatusCodeError(exception.StatusComponentLLMConfigError,
					"output config parameter's config value is invalid")
			}
			return exception.NewStatusCodeError(exception.StatusComponentLLMConfigError,
				fmt.Sprintf("output config parameter's config %v is invalid", value))
		}
	}
	return nil
}
