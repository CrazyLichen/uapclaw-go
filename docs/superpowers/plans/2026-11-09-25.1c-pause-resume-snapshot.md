# 25.1c + 25.1e Implementation Plan（pause/resume、快照回滚与 turn 身份）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让团队成员的“暂停/继续”就地生效且可回滚：快照轨跟踪相位并在干净边界协作停止，`pause` 停在最近边界、`resume` 原地续跑，`abort(immediate)` 回滚上下文，`round_aborted` 标记作废半轮输出；同时补上成员 turn 身份、冷恢复标记、生命周期与网关接线。

**Architecture:** 在 `agent_teams/harness` 内新增相位/快照轨（`snapshot_rail.go`）与 turn 身份（`turn.go`），扩 `ActiveRound` 承载快照与暂停标志；`NativeHarness` 的 supervisor 成为 pause/resume/abort 的唯一决策者，轮协程只负责提交轮次、等待完成、投递无应答的 `cmdRoundFinished`。上层按 `StreamController → CoordinationKernel → TeamAgent → TeamManager → AgentServer → 网关` 逐级接线，冷恢复标记落在 per-team session 命名空间。

**Tech Stack:** Go 1.22、`testify`、`go test -tags "sqlite_fts5 test"`、集成套件 `tests/integration/agentteams/harness` + `mockllm`。

**Spec:** `docs/superpowers/specs/2026-11-09-25.1c-pause-resume-snapshot-design.md`

```go
import (
	"github.com/google/uuid"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/session/state"
	"github.com/uapclaw/uapclaw-go/internal/common/logger"

// ──────────────────────────── 结构体 ────────────────────────────

// MemberTurn 团队成员一次工作的轨迹身份。
// Python: MemberTurn (agent_teams/harness/turn.py L52-94)
type MemberTurn struct {
	// TurnID 该 turn 内所有 span 共享的稳定标识
	TurnID string
	// TurnNumber 成员会话内的 1-based 轮次号
	TurnNumber int
}

// TurnStateStore turn 身份所需的最小会话状态面。
// Python: TurnStateStore Protocol (turn.py L40-49)
type TurnStateStore interface {
	// GetState 读取状态
	GetState(key state.StateKey) (any, error)
	// UpdateState 浅合并状态
	UpdateState(data map[string]any)
}

// ──────────────────────────── 常量 ────────────────────────────

// memberTurnStateKey 成员 turn 在 session state 中的键。
// 与 deepagent blob 分开命名：pause/abort 回滚 DeepAgentState 时不能把计数器一起回退。
// Python: MEMBER_TURN_STATE_KEY = "trajectory_member_turn"（turn.py L34）
const memberTurnStateKey = "trajectory_member_turn"

const (
	turnIDField     = "turn_id"
	turnNumberField = "turn_number"
)

// ──────────────────────────── 导出函数 ────────────────────────────

// LoadMemberTurn 读取成员最近一次 turn；任何异常/畸形数据视为不存在。
// Python: load_member_turn (turn.py L97-112)
func LoadMemberTurn(store TurnStateStore) (MemberTurn, bool) {
	if store == nil {
		return MemberTurn{}, false
	}
	raw, err := store.GetState(state.StringKey(memberTurnStateKey))
	if err != nil {
		logger.Debug(logComponent).Err(err).Msg("member turn state read failed")
		return MemberTurn{}, false
	}
	return memberTurnFromDict(raw)
}

// ResolveMemberTurn 解析本轮所属 turn。
// continuesTurn=true 且已有记录时续用；否则新开一个并推进计数器。
// 第二个返回值表示本轮是否新开（只有新开需要持久化 commit）。
// Python: resolve_member_turn (turn.py L115-142)
func ResolveMemberTurn(store TurnStateStore, continuesTurn bool) (MemberTurn, bool) {
	known, hasKnown := LoadMemberTurn(store)
	if continuesTurn && hasKnown {
		return known, false
	}
	previous := 0
	if hasKnown {
		previous = known.TurnNumber
	}
	opened := MemberTurn{TurnID: uuid.NewString(), TurnNumber: previous + 1}
	if store != nil {
		store.UpdateState(map[string]any{memberTurnStateKey: opened.toDict()})
	}
	return opened, true
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// toDict 序列化为会话状态字典。
// Python: MemberTurn.to_dict (turn.py L65-74)
func (t MemberTurn) toDict() map[string]any {
	return map[string]any{
		turnIDField:     t.TurnID,
		turnNumberField: t.TurnNumber,
	}
}

// memberTurnFromDict 从会话状态恢复；畸形数据返回不存在。
// Python: MemberTurn.from_dict (turn.py L76-94)
func memberTurnFromDict(data any) (MemberTurn, bool) {
	m, ok := data.(map[string]any)
	if !ok {
		return MemberTurn{}, false
	}
	id, _ := m[turnIDField].(string)
	if id == "" {
		return MemberTurn{}, false
	}
	var number int
	switch v := m[turnNumberField].(type) {
	case int:
		number = v
	case int64:
		number = int(v)
	default:
		return MemberTurn{}, false
	}
	if number < 1 {
		return MemberTurn{}, false
	}
	return MemberTurn{TurnID: id, TurnNumber: number}, true
}
```

```go
	LastSafeSnapshot *SafeStateSnapshot

	// IterPhase 当前子相位（atomic.Value 存 RoundPhase）
	IterPhase atomic.Value
	// ModelCallInFlight 模型调用在飞（唯一可安全硬取消的窗口）

	// ToolStarted 当前迭代已开始执行工具

	// PauseRequested 协作停止已武装（正确性权威）
	PauseRequested atomic.Bool
	// GracefulAbort 优雅中止：跑完本轮后回 IDLE
	GracefulAbort atomic.Bool
	// FailureRetry 本轮是否为崩溃后的一次性重试
	FailureRetry bool
	// PauseAck 工具相位暂停的延迟应答
	PauseAck chan error
	// mode 本轮启动方式（同包可见）
	mode startMode
}

// Phase 返回当前子相位。
func (r *ActiveRound) Phase() RoundPhase {
	if v, ok := r.IterPhase.Load().(RoundPhase); ok {
		return v
	}
	return RoundPhaseBoundary
}

func (r *ActiveRound) SetPhase(p RoundPhase) { r.IterPhase.Store(p) }
```

```go
// roundOptions 选项收集结果
type roundOptions struct {
	// taskID 调用方指定的调度任务标识
	taskID string
	// turnNumber 本轮所属成员 turn 号（0 表示未设置）
	turnNumber int
	// resumeContinuation 本轮是否在既有上下文上续跑
	resumeContinuation bool
}

// WithTurnNumber 设置本轮成员 turn 号。
// Python: submit_round(turn_number=...)
func WithTurnNumber(n int) RoundOption {
```

```go
// captureSnapshot 捕获当前轮上下文消息 + DeepAgentState。
// Python: capture_snapshot (snapshot_rail.py L48-92)
func captureSnapshot(
	agent *agentcore.DeepAgent,
	sess interfaces.SessionFacade,
	previous *SafeStateSnapshot,
	index *int,
) *SafeStateSnapshot {
	var messages []llmschema.BaseMessage
	if agent != nil && sess != nil {
		if react := agent.ReactAgent(); react != nil {
			if ce := react.ContextEngine(); ce != nil {
				if mc := ce.GetContext(reactContextID, sess.GetSessionID()); mc != nil {
					if msgs, err := mc.GetMessages(0, false); err == nil {
						messages = cloneSnapshotMessages(msgs)
					}
				} else {
					logger.Warn(logComponent).
						Str("event_type", "HARNESS_SNAPSHOT_NO_CONTEXT").
						Str("session_id", sess.GetSessionID()).
						Msg("snapshot context not found; capture degraded to empty")
				}
			}
		}
	}
	state := map[string]any{}
	if agent != nil && sess != nil {
		state = agent.LoadState(sess).ToSessionDict()
	}
	return &SafeStateSnapshot{
		ContextMessages: messages,
		DeepAgentState:  state,
		IterationIndex:  nextIterationIndex(previous, index),
	}
}

// cloneSnapshotMessages 复制消息切片，避免快照与上下文共享底层数组。
func cloneSnapshotMessages(msgs []llmschema.BaseMessage) []llmschema.BaseMessage {
	return append([]llmschema.BaseMessage(nil), msgs...)
}

// nextIterationIndex 计算快照的迭代序号：显式优先，其次链式 +1，默认 1。
// Python: capture_snapshot 的 index 逻辑（snapshot_rail.py L86-87）
func nextIterationIndex(previous *SafeStateSnapshot, index *int) int {
	if index != nil {
		return *index
	}
	if previous != nil {
		return previous.IterationIndex + 1
	}
	return 1
}
```

```go
// Python: _run_round (native_harness.py L1361-1447)
func (h *NativeHarness) runRound(ctx context.Context, active *ActiveRound) {
	defer close(active.Done)

	h.commitOpenedTurn(ctx, active)

	inv := &agentinterfaces.InvokeInputs{
		Query:          toInvokeQuery(active.OriginalQuery),
		ConversationID: h.SessionID(),
		RunContext:     active.RunContext,
	}
	cbc := agentinterfaces.NewAgentCallbackContext(h.agent, inv, h.sess)

	var result map[string]any
	var runErr error
	lifeErr := cbc.FireLifecycle(ctx,
		agentinterfaces.CallbackBeforeInvoke,
		agentinterfaces.CallbackAfterInvoke,
		func() error {
			if h.controller == nil {
				runErr = fmt.Errorf("harness: 无任务循环控制器")
				return runErr
			}
			query := queryText(active.OriginalQuery)
			err := h.controller.SubmitRound(ctx, h.sess, query,
				active.mode == startModeFollowUp, true,
				agentinterfaces.RunKindNormal, active.RunContext,
				task_loop.WithTaskID(active.TaskID),
				task_loop.WithTurnNumber(active.Turn.TurnNumber),
				task_loop.WithResumeContinuation(active.mode == startModeResumeContinuation),
			)
			if err != nil {
				runErr = err
				return err
			}
			result = h.controller.WaitRoundCompletion(ctx, nil)
			if !isControlResult(result) {
				h.agent.WriteRoundResultToStream(ctx, result, h.sess)
				inv.Result = result
			}
			return nil
		})
	if lifeErr != nil && runErr == nil {
		runErr = lifeErr
	}

	select {
	case h.cmd <- &cmdRoundFinished{roundID: active.RoundID, result: result, err: runErr}:
	case <-ctx.Done():
	}
}
```

```go
// toInvokeQuery 把查询转成 InvokeQuery。
func toInvokeQuery(q any) agentinterfaces.InvokeQuery {
	switch v := q.(type) {
	case string:
		return agentinterfaces.InvokeQueryString(v)
	case *interaction.InteractiveInput:
		return v
	case []string:
		return agentinterfaces.InvokeQueryString(strings.Join(v, "\n"))
	default:
		return agentinterfaces.InvokeQueryString(fmt.Sprintf("%v", v))
	}
}

// truncateForLog 截断日志字段到 n 个 rune。
func truncateForLog(v any, n int) string {
	s := fmt.Sprintf("%v", v)
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n])
}
```

```go
// onAbort 中断当前轮；两种模式都回到 IDLE。
// Python: _on_abort (native_harness.py L928-980)
func (h *NativeHarness) onAbort(cmd *cmdAbort) error {
	phase := h.st.Phase()
	if phase == HarnessStateIdle {
		return nil
	}
	if phase == HarnessStatePaused {
		h.st.ClearPausedQuery()
		h.transition(context.Background(), HarnessStateIdle)
		return nil
	}
	active := h.st.Active()
	if active == nil {
		h.transition(context.Background(), HarnessStateIdle)
		return nil
	}
	if active.PauseRequested.Load() {
		active.PauseRequested.Store(false)
		h.resolvePauseAck(active)
	}
	if cmd.immediate {
		h.hardCancelRound(active)
		h.rollback(firstNonNilSnapshot(active.LastIterSnapshot, active.PreRoundSnapshot))
		h.resetCoordinator()
		h.emitAborted(active.RoundID, "abort")
		h.events.emitRound(context.Background(), HarnessRoundEvent{Kind: "aborted", RoundID: active.RoundID})
		h.st.SetActive(nil)
		h.transition(context.Background(), HarnessStateIdle)
		return nil
	}
	active.GracefulAbort.Store(true)
	if phase == HarnessStatePausing {
		h.transition(context.Background(), HarnessStateRunning)
	}
	return nil
}
```

```go
func (h *NativeHarness) rollback(snapshot *SafeStateSnapshot) {
	if h.sess == nil {
		return
	}
	react := h.agent.ReactAgent()
	if react == nil {
		return
	}
	ce := react.ContextEngine()
	if ce == nil {
		react.ClearContextMessages(context.Background(), h.sess)
		return
	}
	mc := ce.GetContext(reactContextID, h.sess.GetSessionID())
	if snapshot != nil && mc != nil {
		mc.SetMessages(snapshot.ContextMessages, false)
		restored := hschema.DeepAgentState{}.FromSessionDict(snapshot.DeepAgentState)
		h.agent.SaveState(h.sess, &restored)
		logger.Info(logComponent).
			Int("iteration", snapshot.IterationIndex).
			Int("msgs", len(snapshot.ContextMessages)).
			Msg("rolled back to snapshot")
		return
	}
	react.ClearContextMessages(context.Background(), h.sess)
	logger.Info(logComponent).Msg("no snapshot; cleared current-round messages")
}
```

```go
// popInboxBatch 取队首批输入：连续字符串合批，遇非字符串停止。
func (s *HarnessInternalState) popInboxBatch() (any, bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if len(s.queue) == 0 {
		return nil, false
	}
	first := s.queue[0]
	if _, isStr := first.Content.(string); !isStr {
		s.queue = s.queue[1:]
		return first.Content, true
	}
	batch := []string{first.Content.(string)}
	s.queue = s.queue[1:]
	for len(s.queue) > 0 {
		next := s.queue[0]
		text, isStr := next.Content.(string)
		if !isStr {
			break
		}
		batch = append(batch, text)
		s.queue = s.queue[1:]
	}
	if len(batch) == 1 {
		return batch[0], true
	}
	return batch, true
}

// clearInbox 清空待驱动队列。

	s.mu.Lock()
	defer s.mu.Unlock()
```

```go
// settleRoundDone 结算一轮：发事件、清活动轮、按决策优先级决定下一轮。
// Python: _settle_round_done (native_harness.py L1098-1237)
func (h *NativeHarness) settleRoundDone(ctx context.Context, cmd *cmdRoundFinished) {
	active := h.st.Active()
	if active == nil || active.RoundID != cmd.roundID {
		return
	}
	wasGraceful := active.GracefulAbort.Load()
	_, isResume := active.OriginalQuery.(*interaction.InteractiveInput)

	resultError := ""
	if cmd.result != nil {
		if e, ok := cmd.result["error"].(string); ok {
			resultError = e
		}
	}
	taskFailed := resultError != "" && resultError != "cancelled" && resultError != "no active round"
	runtimeCrashed := cmd.err != nil
	h.st.SetActive(nil)

	// 协作暂停：轮在干净边界自然收尾，不推进 coordinator、不丢 follow-up
	if active.PauseRequested.Load() && !runtimeCrashed && !taskFailed {
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "paused", RoundID: cmd.roundID})
		h.transition(ctx, HarnessStatePaused)
		h.resolvePauseAck(active)
		return
	}
	defer h.resolvePauseAck(active)

	switch {
	case cmd.err != nil:
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "failed", RoundID: cmd.roundID, Result: cmd.result})
	case taskFailed:
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "failed", RoundID: cmd.roundID, Result: cmd.result})
	default:
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "finished", RoundID: cmd.roundID, Result: cmd.result})
	}

	if h.sess == nil || h.coordinator == nil {
		h.transition(ctx, HarnessStateIdle)
		return
	}

	// 推进 coordinator + 持久化停止条件（对齐 _run_task_loop 每轮尾部）
	h.coordinator.IncrementIteration()
	if cmd.result != nil {
		h.coordinator.SetLastResult(cmd.result)
	}
	st := h.agent.LoadState(h.sess)
	exported := h.coordinator.ExportState()
	st.StopConditionState = map[string]any{
		"iteration":        exported.Iteration,
		"token_usage":      exported.TokenUsage,
		"stop_reason":      exported.StopReason,
		"evaluator_states": exported.EvaluatorStates,
	}
	h.agent.SaveState(h.sess, st)

	// 优雅中止：本轮已跑完，丢弃排队 follow-up 后 IDLE
	if wasGraceful {
		h.drainFollowUpsDiscard(ctx)
		h.transition(ctx, HarnessStateIdle)
		return
	}

	// 崩溃重试：重置 coordinator；只重试一次；InteractiveInput 不可重放
	if runtimeCrashed {
		h.resetCoordinator()
		replayable := !isResume
		if replayable && !active.FailureRetry {
			logger.Warn(logComponent).Err(cmd.err).Int("round_id", cmd.roundID).
				Msg("round died abnormally; retrying its query once")
			nxt := h.startRound(ctx, active.OriginalQuery, startModeFailureRetry, active.RunContext)
			h.events.emitRound(ctx, HarnessRoundEvent{Kind: "started", RoundID: nxt.RoundID})
			return
		}
		logger.Warn(logComponent).Err(cmd.err).Int("round_id", cmd.roundID).
			Msg("round died abnormally; giving up on its query and going idle")
		h.transition(ctx, HarnessStateIdle)
		return
	}

	resultType := ""
	if cmd.result != nil {
		if rt, ok := cmd.result["result_type"].(string); ok {
			resultType = rt
		}
	}
	if resultType == "interrupt" || h.coordinator.IsAborted() {
		h.transition(ctx, HarnessStateIdle)
		return
	}

	if content, ok := h.st.popInboxBatch(); ok {
		nxt := h.startRound(ctx, content, startModeFollowUp, nil)
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "started", RoundID: nxt.RoundID})
		return
	}
	if batch := h.drainPendingFollowUps(); len(batch) > 0 {
		var q any = batch
		if len(batch) == 1 {
			q = batch[0]
		}
		nxt := h.startRound(ctx, q, startModeFollowUp, nil)
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "started", RoundID: nxt.RoundID})
		return
	}
	if isResume {
		h.transition(ctx, HarnessStateIdle)
		return
	}
	if h.hasRemainingTasks() && queryText(active.OriginalQuery) != "" {
		nxt := h.startRound(ctx, active.OriginalQuery, startModeContinuesTurn, nil)
		h.events.emitRound(ctx, HarnessRoundEvent{Kind: "started", RoundID: nxt.RoundID})
		return
	}
	h.transition(ctx, HarnessStateIdle)
}
```

```go
// cancelStreamsForPausedSession 团队暂停后取消同 session 的流消费者。
// Python: _cancel_streams_for_paused_session (message_handler.py L1309-1370)
func (mh *MessageHandler) cancelStreamsForPausedSession(
	_ context.Context, msg *schema.Message, sessionMode string,
) {
	sessionID := msg.SessionID
	if sessionID == "" {
		return
	}

	mh.streamMu.Lock()
	var victims []string
	emitVictims := make(map[string]bool)
	for rid, sid := range mh.streamSessions {
		if sid != sessionID {
			continue
		}
		if !schema.IsTeamMode(mh.streamModes[rid]) {
			continue
		}
		entry, ok := mh.streamTasks[rid]
		if !ok || entry == nil {
			continue
		}
		victims = append(victims, rid)
		emitVictims[rid] = mh.streamEmitsProcessingStatus[rid]
	}
	mh.streamMu.Unlock()

	if len(victims) == 0 {
		if schema.IsTeamMode(sessionMode) {
			mh.sendProcessingStatus("", sessionID, "", false)
		}
		return
	}

	logger.Info(logComponent).Str("event_type", "pause_stream_governance").
		Str("session_id", sessionID).Strs("cancelled_rids", victims).
		Msg("team pause: cancelling same-session stream consumers")

	for _, rid := range victims {
		mh.cancelStreamTask(rid)
	}

	anyEmit := false
	for _, emit := range emitVictims {
		if emit {
			anyEmit = true
			break
		}
	}
	if !anyEmit && schema.IsTeamMode(sessionMode) {
		mh.sendProcessingStatus("", sessionID, "", false)
	}
}
```

```go
func TestPauseResumeSuite(t *testing.T) {
	suite.Run(t, new(PauseResumeSuite))
}

// newStartedHarness 构造并 Start 一个带 MockLLM 的 NativeHarness。
func (s *PauseResumeSuite) newStartedHarness() (*teamharness.NativeHarness, *session.Session) {
	spec, err := s.NewSpecWithMockLLM(4)
	s.Require().NoError(err)
	h, err := teamharness.NewNativeHarness(spec, nil)
	s.Require().NoError(err)
	sess := session.CreateAgentSession("pr-sess", h.DeepAgent().Card(), nil)
	s.Require().NoError(sess.PreRun(s.Ctx, map[string]any{}))
	s.Require().NoError(h.Start(s.Ctx, sess))
	return h, sess
}

// countContent 统计上下文中内容完全匹配的消息条数。
func countContent(msgs []llmschema.BaseMessage, want string) int {
	n := 0
	for _, m := range msgs {
		if m.GetContent().String() == want {
			n++
		}
	}
	return n
}

// TestI01_模型相位暂停后恢复不重复用户消息
func (s *PauseResumeSuite) TestI01_模型相位暂停后恢复不重复用户消息() {
	h, sess := s.newStartedHarness()
	defer func() { _ = h.Stop(context.Background()) }()

	// 首帧卡死 5s：确保暂停落在 MODEL 相位
	s.MockLLM.WithStreamScripts(mockllm.StreamChunkScript{
		StallBefore: 5 * time.Second,
		Chunks:      []*llmschema.AssistantMessageChunk{llmschema.NewAssistantMessageChunk("完成")},
	})

	s.Require().NoError(h.Send(s.Ctx, "原始任务", false))
	s.Require().Eventually(func() bool { return h.State() == teamharness.HarnessStateRunning },
		5*time.Second, 20*time.Millisecond)
	s.Require().NoError(h.Pause(s.Ctx))
	s.Require().Equal(teamharness.HarnessStatePaused, h.State())

	// 清空卡死脚本，恢复后立刻完成
	s.MockLLM.WithStreamScripts()
	s.Require().NoError(h.Resume(s.Ctx, nil))
	s.Require().Eventually(func() bool { return h.State() == teamharness.HarnessStateIdle },
		10*time.Second, 50*time.Millisecond)

	mc := h.DeepAgent().ReactAgent().ContextEngine().GetContext("default_context", sess.GetSessionID())
	s.Require().NotNil(mc)
	msgs, err := mc.GetMessages(0, true)
	s.Require().NoError(err)
	s.Equal(1, countContent(msgs, "原始任务"), "续跑轮不得重复追加 user turn")
}

// TestI03_立即中止回滚并投递marker
func (s *PauseResumeSuite) TestI03_立即中止回滚并投递marker() {
	h, sess := s.newStartedHarness()
	defer func() { _ = h.Stop(context.Background()) }()

	s.MockLLM.WithStreamScripts(mockllm.StreamChunkScript{
		StallBefore: 5 * time.Second,
		Chunks:      []*llmschema.AssistantMessageChunk{llmschema.NewAssistantMessageChunk("不该出现")},
	})
	s.Require().NoError(h.Send(s.Ctx, "长任务", false))
	s.Require().Eventually(func() bool { return h.State() == teamharness.HarnessStateRunning },
		5*time.Second, 20*time.Millisecond)

	s.Require().NoError(h.Abort(s.Ctx, true))
	s.Require().Equal(teamharness.HarnessStateIdle, h.State())

	// marker 必须到达输出（单消费者，本用例是唯一消费者）
	readCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	sawMarker := false
	for {
		chunk, ok := h.Outputs().Next(readCtx)
		if !ok {
			break
		}
		if out, isOut := chunk.(stream.OutputSchema); isOut && out.Type == "round_aborted" {
			sawMarker = true
		}
	}
	s.True(sawMarker, "immediate abort 必须投递 round_aborted 标记")

	mc := h.DeepAgent().ReactAgent().ContextEngine().GetContext("default_context", sess.GetSessionID())
	s.Require().NotNil(mc)
	msgs, err := mc.GetMessages(0, false)
	s.Require().NoError(err)
	s.Equal(0, countContent(msgs, "不该出现"), "半轮 assistant 消息必须被回滚")
}

// TestI05_冷恢复续跑不追加用户消息
func (s *PauseResumeSuite) TestI05_冷恢复续跑不追加用户消息() {
	h, sess := s.newStartedHarness()
	s.Require().NoError(h.Stop(context.Background()))

	// 同一会话重建 harness：模拟 pause → stop → start
	spec, err := s.NewSpecWithMockLLM(4)
	s.Require().NoError(err)
	h2, err := teamharness.NewNativeHarness(spec, nil)
	s.Require().NoError(err)
	s.Require().NoError(h2.Start(s.Ctx, sess))
	defer func() { _ = h2.Stop(context.Background()) }()

	s.MockLLM.SetResponses(mockllm.CreateTextResponse("续跑完成"))
	s.Require().NoError(h2.Resume(s.Ctx, "原始任务"))
	s.Require().Eventually(func() bool { return h2.State() == teamharness.HarnessStateIdle },
		10*time.Second, 50*time.Millisecond)

	mc := h2.DeepAgent().ReactAgent().ContextEngine().GetContext("default_context", sess.GetSessionID())
	s.Require().NotNil(mc)
	msgs, err := mc.GetMessages(0, true)
	s.Require().NoError(err)
	s.Equal(0, countContent(msgs, "原始任务"), "冷恢复轮不得追加 user turn")
}
```
