// UsesSystemAttachmentRole 判断当前路由是否把附件投递为 SystemMessage。
// Python: prompt_attachment_manager.py:172-188 _uses_system_attachment_role(model_client_config)。
// 仅百炼/DashScope 需要；Anthropic 客户端把 System 提到顶层 system 字段，故保持 UserMessage。
// 归一前原名（legacy_client_provider）经 endpointprofiles 取，覆盖 DashScope 别名未归一的调用场景。
func UsesSystemAttachmentRole(cfg *llmschema.ModelClientConfig) bool {
	if cfg == nil {
		return false
	}
	provider := normalizeConfigValue(cfg.ClientProvider)
	apiMode := normalizeConfigValue(string(cfg.APIMode))
	legacyProvider := normalizeConfigValue(endpointprofiles.LegacyProvider(cfg))
	endpointProfile := normalizeConfigValue(cfg.EndpointProfile)

	if provider == "anthropic" || anthropicAPIModes[apiMode] {
		return false
	}
	backend := endpointProfile
	if genericEndpointProfiles[backend] {
		if legacyProvider != "" {
			backend = legacyProvider
		} else {
			backend = provider
		}
	}
	return systemAttachmentRoleProfiles[backend] || systemAttachmentRoleProviders[backend]
}

// normalizeConfigValue 对齐 Python _normalized_config_value：小写 + trim。
func normalizeConfigValue(s string) string { return strings.ToLower(strings.TrimSpace(s)) }

// 三个集合对齐 Python prompt_attachment_manager.py:77-80 的 frozenset。
var (
	anthropicAPIModes             = map[string]bool{"anthropic": true, "anthropic-messages": true, "messages": true}
	genericEndpointProfiles       = map[string]bool{"": true, "openai": true, "openai-compatible": true}
	systemAttachmentRoleProviders = map[string]bool{"bailian": true, "dashscope": true}
	systemAttachmentRoleProfiles  = map[string]bool{"bailian": true, "dashscope": true}
)

// statusInTextRe 从错误文本推断 HTTP 状态码。
// Python: provider_error.py:23 _STATUS_IN_TEXT_RE
var statusInTextRe = regexp.MustCompile(`(?i)\b(?:HTTP|Status(?:\s*Code)?|status)\D{0,10}([1-5]\d{2})\b`)

// FormatProviderException 格式化 provider 错误：空消息回退类型名，非空走摘要。
// Python: provider_error.py:30-44 format_provider_exception。
// includeExcType=true 时输出 "TypeName: 摘要"；Go 侧 TypeName 取 fmt.Sprintf("%T") 短名。
func FormatProviderException(err error, includeExcType bool, statusCode int) string {
	if err == nil {
		return ""
	}
	raw := strings.TrimSpace(err.Error())
	if raw == "" {
		return shortTypeName(err)
	}
	summarized := SummarizeProviderErrorText(raw, statusCode)
	if !includeExcType {
		return summarized
	}
	return shortTypeName(err) + ": " + summarized
}

// SummarizeProviderErrorText 返回面向用户的错误体，折叠 HTML/XML 页面。
// Python: provider_error.py:47-59 summarize_provider_error_text。
func SummarizeProviderErrorText(text string, statusCode int) string {
	raw := text
	if raw == "" {
		return ""
	}
	if markup, ok := markupPayload(raw); ok {
		inferred := statusCode
		if inferred == 0 {
			inferred = statusFromText(raw)
		}
		return summarizeMarkup(markup, inferred)
	}
	if len(raw) > MaxProviderErrorChars {
		overflow := len(raw) - MaxProviderErrorChars
		return fmt.Sprintf("%s... [truncated %d chars]", raw[:MaxProviderErrorChars], overflow)
	}
	return raw
}

// Python: provider_error.py:75-84 _as_http_status。
func asHTTPStatus(value int) int {

	code, err := strconv.Atoi(m[1])

package providererror

import (
	"regexp"
	"strconv"
	"strings"
)

// markupStartRe 匹配 HTML/XML 开头。Python: _markup_payload 内联判定。
var (
	markupStartRe = regexp.MustCompile(`(?is)^\s*<(!DOCTYPE|html|xml|body|head)\b`)
	htmlTagRe     = regexp.MustCompile(`(?s)<[^>]*>`)
	headingRe     = regexp.MustCompile(`(?is)<h[1-6][^>]*>(.*?)</h[1-6]>`)
	titleRe       = regexp.MustCompile(`(?is)<title[^>]*>(.*?)</title>`)
	whitespaceRe  = regexp.MustCompile(`\s+`)
)

// markupPayload 判定文本是否为错误页面 markup，返回剥壳后的文本。
// Python: provider_error.py:94-105 _markup_payload + :107-117 _is_error_page。
func markupPayload(text string) (string, bool) {
	trimmed := strings.TrimSpace(text)
	if trimmed == "" || !strings.HasPrefix(trimmed, "<") {
		return "", false
	}
	if !markupStartRe.MatchString(trimmed) && !strings.Contains(strings.ToLower(trimmed), "<html") {
		return "", false
	}
	return trimmed, true
}

// summarizeMarkup 把 markup 折叠为一行摘要。
// Python: provider_error.py:119-134 _summarize_markup。
func summarizeMarkup(markup string, statusCode int) string {
	heading := extractHeading(markup)
	if statusCode == 0 {
		statusCode = statusFromHeading(heading)
	}
	if statusCode == 0 {
		statusCode = statusFromText(stripTags(markup))
	}
	plain := plainText(stripTags(markup))
	parts := make([]string, 0, 3)
	if statusCode != 0 {
		parts = append(parts, "HTTP "+strconv.Itoa(statusCode))
	}
	if heading != "" {
		parts = append(parts, heading)
	}
	if plain != "" {
		parts = append(parts, plain)
	}
	if len(parts) == 0 {
		return "provider returned an error page"
	}
	return strings.Join(parts, ": ")
}

// looksLikeHtml 判断文本是否含 HTML 标签。Python: provider_error.py:136-139。
func looksLikeHtml(text string) bool {
	return htmlTagRe.MatchString(text)
}

// extractHeading 提取首个 h1-h6 或 title 文本。Python: provider_error.py:141-150。
func extractHeading(markup string) string {
	for _, re := range []*regexp.Regexp{headingRe, titleRe} {
		if m := re.FindStringSubmatch(markup); m != nil {
			return plainText(m[1])
		}
	}
	return ""
}

// statusFromHeading 从 heading（如 "502 Bad Gateway"）推断状态码。Python: provider_error.py:152-159。
func statusFromHeading(heading string) int {
	m := regexp.MustCompile(`\b([1-5]\d{2})\b`).FindStringSubmatch(heading)
	if m == nil {
		return 0
	}
	code, err := strconv.Atoi(m[1])
	if err != nil {
		return 0
	}
	return asHTTPStatus(code)
}

// plainText 压缩空白成单行。Python: provider_error.py:161-163 _plain_text。
func plainText(s string) string {
	return strings.TrimSpace(whitespaceRe.ReplaceAllString(s, " "))
}

// stripTags 剥掉全部标签。
func stripTags(s string) string { return htmlTagRe.ReplaceAllString(s, " ") }

// RecoverModelFields 修复持久化模型条目的类型错乱，返回是否发生非空修复。
// Python: jiuwenswarm/common/config.py:1643-1669 _recover_model_fields(entry)。
func RecoverModelFields(entry map[string]any) bool {
	recovered := false
	for _, key := range []string{"model_client_config", "model_config_obj", "model_detail"} {
		value, exists := entry[key]
		if exists {
			if _, ok := value.(map[string]any); !ok {
				entry[key] = map[string]any{}
				recovered = recovered || value != nil
			}
		}
	}
	client, ok := entry["model_client_config"].(map[string]any)
	if ok {
		for _, key := range []string{"model_name", "api_base", "api_key", "client_provider"} {
			value, exists := client[key]
			if exists && value != nil {
				if _, isStr := value.(string); !isStr {
					client[key] = ""
					recovered = true
				}
			}
		}
		if headers, exists := client["custom_headers"]; exists && headers != nil {
			switch headers.(type) {
			case string, map[string]any:
			default:
				client["custom_headers"] = map[string]any{}
				recovered = true
			}
		}
	}
	if detail, ok := entry["model_detail"].(map[string]any); ok {
		for _, key := range []string{"fallback_tag", "model_description"} {
			value, exists := detail[key]
			if exists && value != nil {
				if _, isStr := value.(string); !isStr {
					detail[key] = ""
					recovered = true
				}
			}
		}
	}
	return recovered
}
