// Package graph 内。Python: vertex.py SUB_WORKFLOW_COMPONENT 常量（vertex.py:30）
const subWorkflowComponent = "sub_workflow"

// Vertex 工作流图节点执行器；双链模型——批入侧（INVOKE/STREAM 同步）+ 流入侧（COLLECT/TRANSFORM 异步）。
// Python: Vertex(AsyncAtomicNode, StreamConsumer)（vertex.py:65），单结构体承载双基类（8.10 D1 先例）。
// 并发映射零锁依据（spec §1 不变式）：计数器/streamDone 槽位的跨 goroutine 访问对
// 全部被引擎屏障定序；-race 并发用例（Task 12）锁定，若定序被破坏 race 当场报出。
type Vertex struct {
	// AtomicNode 承载 Python AsyncAtomicNode 基类（构造期注入 atomicInvokeFunc）
	AtomicNode
	// executable Python: self._executable: Executable
	executable GeneralExecutor
	// nodeID Python: self._node_id
	nodeID string
	// ctx Python: self._context（init 时从 kwargs 注入）
	ctx context.Context
	// nodeSession Python: self._session: NodeSession
	nodeSession *internal.NodeSession // 以实际 session 包别名为准（见 Step 4 说明）
	// streamCalledTimeout Python: self._stream_called_timeout = 10
	streamCalledTimeout float64
	// streamDone 当轮流侧结果容器（Python: asyncio.Future，每轮 streamCall 新建；缓冲 1
	// 对齐 Future.set_result 不阻塞语义；屏障定序保证读写窗口无并发写——spec §4.1）
	streamDone chan error
	// callCount Python: self._call_count
	callCount atomic.Int64
	// streamCallCount Python: self._stream_call_count
	streamCallCount atomic.Int64
	// isEndNode Python: self.is_end_node
	isEndNode bool
	// isStarted trace 去重（Python: asyncio.Event；trace 埋点合并协议 spec §4.3-3）
	isStarted atomic.Bool
	// isCallStarted 批入侧已记输入标记（Python: asyncio.Event）
	isCallStarted atomic.Bool
	// nodeConfig Python: self._node_config
	nodeConfig *workflow.NodeConfig
	// componentAbility Python: self._component_ability
	componentAbility []components.ComponentAbility
	// hasStreamCall / hasCall Python: self._has_stream_call / self._has_call
	hasStreamCall bool
	hasCall       bool
	// sourceID Python: self._source_id
	sourceID []string
	// isFirstInit Python: self._is_first_init
	isFirstInit atomic.Bool
}

// NewVertex 创建 Vertex（Python __init__，vertex.py:67-85 字段一比一）
func NewVertex(nodeID string, executable GeneralExecutor) *Vertex {
	return &Vertex{
		nodeID:              nodeID,
		executable:          executable,
		streamCalledTimeout: 10,
		streamDone:          make(chan error, 1),
		isFirstInit:         *new(atomic.Bool), // 零值 false，下方 Store(true) 显式置位
	}
}

// collectRefSourceIDs 递归收集 schema 中 ${comp_id...} 引用的组件 ID 集合。
// Python: _collect_ref_source_ids（vertex.py:33-62，模块级函数→包级函数）
func collectRefSourceIDs(schema map[string]any) map[string]bool {
	ids := map[string]bool{}
	var walk func(obj any)
	walk = func(obj any) {
		switch v := obj.(type) {
		case map[string]any:
			for _, val := range v {
				walk(val)
			}
		case []any:
			for _, item := range v {
				walk(item)
			}
		case string:
			if utils.IsRefPath(v) {
				origin := utils.ExtractOriginKey(v)
				if origin != "" {
					ids[strings.SplitN(origin, ".", 2)[0]] = true
				}
			}
		}
	}
	walk(schema)
	return ids
}
