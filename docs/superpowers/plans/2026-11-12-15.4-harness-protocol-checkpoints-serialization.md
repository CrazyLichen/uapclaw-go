package harness_protocol

// 本文件对应 Python openjiuwen/harness_protocol/checkpoints.py（111 行）：
// 第三方 harness 的带版本 checkpoint 契约。
//
// checkpoint 是协议层的第四个通道（与 observation / interaction / hooks 并列）：
// 宿主把 provider 的不透明状态持久化，供 REQUIRE_RESUME 时续跑。
// provider 才可解释 data 与 schema_version，宿主只搬运完整信封。

// ──────────────────────────── 结构体 ────────────────────────────

// HarnessCheckpoint 不透明的 provider 状态，作用域为单个 agent + 宿主会话。
//
// 只有 Provider 字段命名的 provider 可解释 Data 与 SchemaVersion；
// 宿主持久化完整信封、不做解释。
// Python: checkpoints.py:28-45 HarnessCheckpoint
type HarnessCheckpoint struct {
	// Provider 拥有该 checkpoint 的 provider 名
	Provider string
	// SchemaVersion provider 私有 schema 版本
	SchemaVersion string
	// AgentID agent 标识
	AgentID string
	// HostSessionID 宿主会话标识
	HostSessionID string
	// CheckpointID checkpoint 幂等标识
	CheckpointID string
	// Sequence 同一 provider/agent/会话作用域内单调递增序号
	Sequence int
	// Data provider 私有数据（已冻结）
	Data JsonObject
	// ProviderSessionID provider 原生会话标识（nil = 未提供；空串非法）
	ProviderSessionID *string
	// Revision 宿主存储修订号（nil = 未提供；空串非法）
	Revision *string
}

	CheckpointID string
	// Sequence 单调序号

	// Data provider 私有数据

// Python: checkpoints.py:66-73 CheckpointSaveReceipt

	// CheckpointID checkpoint 标识
	CheckpointID string
	// Sequence 单调序号
	Sequence int
	// StorageRevision 宿主存储修订号
	StorageRevision string

// HarnessCheckpointSink 宿主侧主动发布 checkpoint 的持久化目标。
//
// 三条必须遵守的语义（Python: checkpoints.py:83-103 的 docstring）：
//  1. Sequence 在「provider / agent / host-session」三元组作用域内单调递增。
//  2. 重试同一 CheckpointID 幂等，返回原 receipt。
//  3. 对「已存在或更低 sequence」的不同写入、或 ExpectedStorageRevision 比对失败，
//     须返回 CheckpointConflictError，不得覆盖新状态。

type HarnessCheckpointSink interface {
	// Save 持久化 checkpoint 并返回存储修订号

		ctx context.Context,

const (

	CheckpointReasonSessionActivated CheckpointReason = "session_activated"

	CheckpointReasonStateChanged CheckpointReason = "state_changed"

	CheckpointReasonProviderRequested CheckpointReason = "provider_requested"

// NewHarnessCheckpoint 创建 checkpoint 信封，校验顺序对齐上游 __post_init__。
// Python: checkpoints.py:46-65 HarnessCheckpoint.__post_init__（6 步）
func NewHarnessCheckpoint(opts CheckpointOptions) (HarnessCheckpoint, error) {
	// 步骤 1：5 个必填字段逐项非空
	// Python: checkpoints.py:47-56
	required := []struct {
		name  string
		value string
	}{
		{"provider", opts.Provider},
		{"schema_version", opts.SchemaVersion},
		{"agent_id", opts.AgentID},
		{"host_session_id", opts.HostSessionID},
		{"checkpoint_id", opts.CheckpointID},
	}
	for _, f := range required {
		if f.value == "" {
			return HarnessCheckpoint{}, &HarnessProtocolError{
				Msg: "checkpoint " + f.name + " must not be empty",
			}
		}
	}
	return HarnessCheckpoint{}, nil
}

func TestNewHarnessCheckpoint_步骤5尺寸(t *testing.T) {

	t.Run("恰好边界内", func(t *testing.T) {

		require.NoError(t, err, "恰好 4MiB 应通过")

	t.Run("超限 1 字节", func(t *testing.T) {
		_, err := NewHarnessCheckpoint(mkOpts(strings.Repeat("x", MaxCheckpointBytes-overhead+1)))

	t.Run("全 ASCII 紧凑编码字节数", func(t *testing.T) {
		// {"payload":"xxx"} 紧凑编码 = 17 字节（与 Python ensure_ascii=False 一致）
		n, err := checkpointDataSize(map[string]any{"payload": "xxx"})

	})

	t.Run("HTML 不转义", func(t *testing.T) {

//
// 编码口径（spec Q12 决策 B）：SetEscapeHTML(false) 对齐 Python 的
// ensure_ascii=False——仓内 github.py:408、gitcode.py:522 均用该口径，
// 属上游主流。checkpoints.py:62 未传 ensure_ascii=False 是其自身疏漏，
// 中文场景两侧差 33 vs 39 字节已登记为 spec D5。
//
// json.Encoder.Encode 会追加 "\n"，必须裁掉，否则每条 checkpoint 多算 1 字节。
// Python: checkpoints.py:62
func checkpointDataSize(data JsonObject) (int, error) {
	// 步骤 1：归一化为标准 JSON 容器
	// Python: checkpoints.py:62 json_value_to_builtin(data)
	builtin := JsonValueToBuiltin(data)

	// 步骤 2：紧凑编码，不转义 HTML
	var buf bytes.Buffer
	enc := json.NewEncoder(&buf)
	enc.SetEscapeHTML(false)
	if err := enc.Encode(builtin); err != nil {
		return 0, err
	}

	// 步骤 3：裁掉 Encoder 追加的换行
	raw := buf.Bytes()
	if n := len(raw); n > 0 && raw[n-1] == '\n' {
		raw = raw[:n-1]
	}
	return len(raw), nil
}

// asInteger 校验整数，四步：
//  1. 拒绝 bool（Python 的 isinstance(value, bool) 优先排除）
//  2. 整数族直接转 int
//  3. float64 必须整值、有限、且在 int64 范围内（spec D4 放宽）
//  4. 其余报错
//
// 步骤 3 的范围检查不可省：实测 float64(1e300) 满足 v == Trunc(v)，
// 直接 int(v) 会溢出。JSON 里写 1e300 完全合法，json.Unmarshal 原样解出，
// 因此这是必现路径。
// Python: serialization.py:502-506 _integer
func asInteger(value any, fieldName string) (int, error) {
	// 步骤 1：拒绝 bool
	if _, isBool := value.(bool); isBool {
		return 0, &HarnessProtocolError{Msg: fieldName + " must be an integer"}
	}
	// 步骤 2：整数族
	switch v := value.(type) {
	case int:
		return v, nil
	case int8:
		return int(v), nil
	case int16:
		return int(v), nil
	case int32:
		return int(v), nil
	case int64:
		return int(v), nil
	case uint:
		return int(v), nil
	case uint8:
		return int(v), nil
	case uint16:
		return int(v), nil
	case uint32:
		return int(v), nil
	case uint64:
		return int(v), nil
	}
	// 步骤 3：整值 float64
	if f, ok := value.(float64); ok {
		if math.IsNaN(f) || math.IsInf(f, 0) || f != math.Trunc(f) {
			return 0, &HarnessProtocolError{Msg: fieldName + " must be an integer"}
		}
		if f < math.MinInt64 || f > math.MaxInt64 {
			return 0, &HarnessProtocolError{Msg: fieldName + " must be an integer"}
		}
		return int(f), nil
	}
	// 步骤 4：其余
	return 0, &HarnessProtocolError{Msg: fieldName + " must be an integer"}
}

// asNumber 校验浮点数，三步：拒 bool、整数族转 float64、float64 原值。

// Python 侧是 type(event).__name__。
func payloadTypeName(event HarnessEventPayload) string {
	if event == nil {

	rv := reflect.ValueOf(event)

		if err != nil {

package harness_providers

import (
	"fmt"
	"math"
	"reflect"
	"strings"
)

// 本文件对应 Python openjiuwen/harness_providers/jsonsafe.py（55 行）：
// 把厂商 SDK 对象转换成协议安全的 JSON 值。
//
// 这是**序列化前的类型归一化**，与 harness_protocol.FreezeJsonValue（严格准入校验）
// 串联两道工序：
//
//	provider SDK 对象
//	  → ToJsonSafe       归一化（NaN→nil、枚举→字符串值、对象→字符串）
//	  → FreezeJsonValue  递归校验 + 冻结（拒绝残留 NaN / 非 string key）
//	  → 进 HarnessEvent / HarnessCheckpoint
//
// 两道不可合并：FreezeJsonValue 是严格拒绝的（NaN 抛错），而 provider 数据里
// NaN 常见（SDK 的 usage、耗时字段）。没有本文件的清洗，provider 吐一个 NaN
// 整条事件构造就失败。
//
// 这也解释了目录归属：本文件是 provider 侧的脏数据清洗（harness_providers/），
// FreezeJsonValue 是协议侧的准入校验（harness_protocol/）。

// ──────────────────────────── 导出函数 ────────────────────────────

// ToJsonSafe 返回 value 的 JSON 兼容副本，可被 FreezeJsonValue 接受。
//
// 分支顺序对齐上游（jsonsafe.py:24-43），其中两项顺序不可调：
//   - []byte 必须在 slice 之前判断（Kind() 都是 Slice，否则会展开成数字数组）
//   - 非有限 float 必须在普通 float 之前清洗成 nil
//
// 与上游的三处偏差（spec D1/D2/D3）：
//   - os.PathLike：Go 无对应抽象，仓内路径一律 string，走兜底
//   - dataclass：Go 无对应物，用「有字段的结构体」启发式会误伤 time.Time
//     等（Python 对它们走兜底），故不实现
//   - model_dump：用可选接口 ModelDump 探测，本批无实现者
//
// Python: jsonsafe.py:15-43 to_json_safe
func ToJsonSafe(value any) any {
	// 步骤 1：None（Go 的 nil）原样
	// Python: jsonsafe.py:24
	if value == nil {
		return nil
	}

	// 步骤 2：基础类型原样；非有限 float 洗成 nil
	// Python: jsonsafe.py:24-27
	switch v := value.(type) {
	case string, bool,
		int, int8, int16, int32, int64,
		uint, uint8, uint16, uint32, uint64:
		return v
	case float64:
		if math.IsNaN(v) || math.IsInf(v, 0) {
			return nil
		}
		return v
	case float32:
		f := float64(v)
		if math.IsNaN(f) || math.IsInf(f, 0) {
			return nil
		}
		return v
	}

	// 步骤 3：bytes 必须先于 slice（Kind() 都是 Slice）
	// Python: jsonsafe.py:32-33（decode("utf-8", errors="replace")）
	// 用 strings.ToValidUTF8 复刻 errors="replace"（仓内先例：
	// internal/agentcore/harness/tools/shell/output.go:108-110）
	if b, ok := value.([]byte); ok {
		return strings.ToValidUTF8(string(b), "\ufffd")
	}

	// 步骤 4：model_dump 可选接口（spec D3 留缝，本批无实现者）
	// Python: jsonsafe.py:36-38
	if dumper, ok := value.(interface {
		ModelDump(mode string, byAlias bool) (any, error)
	}); ok {
		if dumped, err := dumper.ModelDump("json", true); err == nil {
			return ToJsonSafe(dumped)
		}
	}

	// 步骤 5：容器递归
	// Python: jsonsafe.py:39-42
	rv := reflect.ValueOf(value)
	switch rv.Kind() {
	case reflect.Map:
		// Python: jsonsafe.py:39-40（key 转 str）
		out := make(map[string]any, rv.Len())
		iter := rv.MapRange()
		for iter.Next() {
			out[fmt.Sprint(iter.Key().Interface())] = ToJsonSafe(iter.Value().Interface())
		}
		return out
	case reflect.Slice, reflect.Array:
		// Python: jsonsafe.py:41-42（list/tuple/set/frozenset → list）
		out := make([]any, rv.Len())
		for i := 0; i < rv.Len(); i++ {
			out[i] = ToJsonSafe(rv.Index(i).Interface())
		}
		return out
	case reflect.String:
		// spec Q3：具名 string 类型（本仓全部枚举都是 type X string）
		// 取字符串值，等价于 Python str,Enum 取 .value。
		// 注：不能靠 Type().Name() 判枚举——实测 reflect 对预声明 string
		// 也返回 "string" 而非空串，该判据不成立。三者 Kind() 都是 String，
		// 返回原字符串本身，语义与 Python 完全一致。
		return rv.String()
	case reflect.Ptr, reflect.Interface:
		// Python: 指针无对应概念，解引用后递归；nil 保持 nil
		if rv.IsNil() {
			return nil
		}
		return ToJsonSafe(rv.Elem().Interface())
	}

	// 步骤 6：兜底转字符串
	// 这是本函数的灵魂：provider payload 绝不把 live SDK 对象泄漏进事件流。
	// Python: jsonsafe.py:43
	return fmt.Sprint(value)
}

// ToJsonObject 把 value 转成 JSON 对象，非 mapping 值包成 {"value": converted}。
// Python: jsonsafe.py:46-52 to_json_object
func ToJsonObject(value any) map[string]any {
	// 步骤 1：归一化
	converted := ToJsonSafe(value)
	// 步骤 2：是对象则直接返回，否则包一层
	if obj, ok := converted.(map[string]any); ok {
		return obj
	}
	return map[string]any{"value": converted}
}

// externalCheckpointStateKey 成员会话里存放外部 runtime 状态的键。

// sessionCheckpointSink 把 checkpoint 落进真实 AgentSession。
//
// 这是 agent_teams/external/member_runtime.py:124-152
// _MemberSessionCheckpointSink 的 Go 等价物，同时补上该实现未做的
// 幂等与 compare-and-set（上游把这两项留在契约里、由生产实现自行保证）。
type sessionCheckpointSink struct {
	sess *session.Session

	mu       sync.Mutex
	receipts map[string]harness_protocol.CheckpointSaveReceipt
	latest   int
	revision string

// Save 持久化 checkpoint 并返回回执。
// 三条语义对齐 Python: checkpoints.py:83-103。
func (s *sessionCheckpointSink) Save(
	_ context.Context,
	checkpoint harness_protocol.HarnessCheckpoint,
	_ harness_protocol.CheckpointReason,
	expectedStorageRevision *string,
) (harness_protocol.CheckpointSaveReceipt, error) {
	s.mu.Lock()
	defer s.mu.Unlock()

	// 语义 2：同一 checkpoint_id 幂等，返回原 receipt
	if receipt, ok := s.receipts[checkpoint.CheckpointID]; ok {
		return receipt, nil
	}
	// 语义 3a：更低的 sequence 视为 stale
	if checkpoint.Sequence <= s.latest {
		return harness_protocol.CheckpointSaveReceipt{},
			&harness_protocol.CheckpointConflictError{Msg: "stale checkpoint sequence"}
	}
	// 语义 3b：CAS 比对失败
	if expectedStorageRevision != nil && *expectedStorageRevision != s.revision {
		return harness_protocol.CheckpointSaveReceipt{},
			&harness_protocol.CheckpointConflictError{Msg: "checkpoint compare-and-set failed"}
	}

	// 落盘到真实 session
	// 注：GetState 的第二个返回值恒为 nil（见 agent.go:383-385），
	// 无需判断 ErrStateNotFound；首次写入前该键不存在属正常路径。
	s.sess.UpdateState(map[string]any{
		externalCheckpointStateKey: map[string]any{
			"backend":    checkpoint.Provider,
			"checkpoint": checkpointToDict(checkpoint),
		},
	})
	if err := s.sess.Commit(context.Background()); err != nil {
		return harness_protocol.CheckpointSaveReceipt{}, err
	}

	s.latest = checkpoint.Sequence
	s.revision = "storage-" + checkpoint.CheckpointID
	receipt, err := harness_protocol.NewCheckpointSaveReceipt(
		checkpoint.CheckpointID, checkpoint.Sequence, s.revision)
	if err != nil {
		return harness_protocol.CheckpointSaveReceipt{}, err
	}
	s.receipts[checkpoint.CheckpointID] = receipt
	return receipt, nil
}

// checkpointToDict 把 checkpoint 信封编码成纯 JSON 数据。
// 对应 Python: member_runtime.py:155-167 checkpoint_to_dict。
func checkpointToDict(cp harness_protocol.HarnessCheckpoint) map[string]any {
	return map[string]any{
		"provider":            cp.Provider,
		"schema_version":      cp.SchemaVersion,
		"agent_id":            cp.AgentID,
		"host_session_id":     cp.HostSessionID,
		"checkpoint_id":       cp.CheckpointID,
		"sequence":            cp.Sequence,
		"data":                harness_protocol.JsonValueToBuiltin(cp.Data),
		"provider_session_id": cp.ProviderSessionID,
		"revision":            cp.Revision,
	}
}

// checkpointFromDict 从纯 JSON 数据还原 checkpoint（走构造器校验）。
// 对应 Python: member_runtime.py:170-190 checkpoint_from_dict。
func checkpointFromDict(value any) (*harness_protocol.HarnessCheckpoint, error) {
	obj, ok := value.(map[string]any)
	if !ok {
		return nil, errors.New("checkpoint 数据不是对象")
	}
	seq, _ := obj["sequence"].(int)
	cp, err := harness_protocol.NewHarnessCheckpoint(harness_protocol.CheckpointOptions{
		Provider:      stringField(obj, "provider"),
		SchemaVersion: stringField(obj, "schema_version"),
		AgentID:       stringField(obj, "agent_id"),
		HostSessionID: stringField(obj, "host_session_id"),
		CheckpointID:  stringField(obj, "checkpoint_id"),
		Sequence:      seq,
		Data:          objectField(obj, "data"),
	})
	if err != nil {
		return nil, err
	}
	return &cp, nil
}

// stringField 取字符串字段，缺失返回空串。
func stringField(obj map[string]any, key string) string {
	s, _ := obj[key].(string)
	return s
}

// objectField 取对象字段，缺失返回空 map。
func objectField(obj map[string]any, key string) map[string]any {
	m, ok := obj[key].(map[string]any)
	if !ok {
		return map[string]any{}
	}
	return m
}
