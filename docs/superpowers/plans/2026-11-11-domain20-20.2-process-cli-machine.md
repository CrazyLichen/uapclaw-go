```go
// FlushHistoryQueue 等待 history 异步写入队列刷盘。
// timeoutSeconds 为等待上限；返回是否在超时前全部写完。
// Python: session_history.py:1384 flush_pending_writes(timeout: float = 10) -> bool
func FlushHistoryQueue(timeoutSeconds float64) bool {
	// 如果 worker 还没启动，无需等待
	if historyWriteQueue == nil {
		return true
	}
	// 发送哨兵项，worker 处理后会向 flushDone 发送确认
	flushDone := make(chan struct{})
	historyWriteQueue <- historyWriteItem{sessionID: "", record: nil, flushDone: flushDone}
	// 等待 worker 确认，确保所有之前的写入已完成
	timer := time.NewTimer(time.Duration(timeoutSeconds * float64(time.Second)))
	defer timer.Stop()
	select {
	case <-flushDone:
		return true
	case <-timer.C:
		logger.Warn(logComponent).Float64("timeout_seconds", timeoutSeconds).
			Msg("FlushHistoryQueue timed out")
		return false
	}
}
```

```go
// Python: service.py:874-909 create_or_resume_session
//
// 偏差 D-12：上游 resolve_manager(request) 按租户分槽，Go 侧只有 AgentRuntimeDeps.Manager
// 单槽（无多租户池，IterManagers 仅用于 host 事件遍历），故此处固定使用单槽。
func (r *AgentRuntime) CreateOrResumeSession(channelID, sessionID string) (string, error) {
	// 1. Python: self._require_started()
	if err := r.requireStarted(); err != nil {
		return "", err
	}
	requested := strings.TrimSpace(sessionID)

	// 2. Python: if requested: ...（校验 ID + 渠道归属）
	if requested != "" {
		// 2a. Python: if not is_valid_session_id(requested): raise ValueError
		if !common.IsValidSessionID(requested) {
			return "", errors.New("invalid session_id")
		}
		// 2b. Python: descriptor = await self.describe_session(session_id=requested)
		descriptor, err := r.DescribeSession(requested)
		if err != nil {
			return "", err
		}
		// 2c. Python: 渠道不一致 -> SessionCatalogError("session not found")
		//     不泄露「存在但属于别人」
		if descriptor != nil {
			requestedChannel := normalizedChannelOrDefault(channelID)
			persistedChannel := normalizedChannelOrDefault(descriptor.ChannelID)
			if persistedChannel != requestedChannel {
				return "", ErrSessionNotFound
			}
		}
	}

	// 3. Python: manager = await self.resolve_manager(request)（Go 单槽，见偏差 D-12）
	if r.manager == nil {
		return "", errors.New("agent runtime manager is nil")
	}

	// 4. Python: resolved = await manager.create_session(channel_id, session_id=requested or None)
	createSession, ok := r.manager.(sessionCreator)
	if !ok {
		return "", errors.New("agent manager does not implement CreateSession")
	}
	resolved, err := createSession.CreateSession(channelID, requested)
	if err != nil {
		return "", err
	}

	// 5. Python: await self._register_session(session_id=resolved, channel_id=channel_id)
	if _, err := r.RegisterSession(resolved, channelID, sessionpkg.SessionPersistencePolicyPersistent); err != nil {
		return "", err
	}
	return resolved, nil
}

// Python: server/runtime/agent_manager.py AgentManager.create_session

// Python: str(channel_id or "default").strip().lower()
func normalizedChannelOrDefault(channelID string) string {
	v := strings.ToLower(strings.TrimSpace(channelID))
	if v == "" {
		return "default"
	}
	return v
}
```

```go
import (
	"context"
```

```go
	agentinterfaces "github.com/uapclaw/uapclaw-go/internal/agentcore/single_agent/interfaces"
```

```go
// ──────────────────────────── 常量 ────────────────────────────

const (
	// processCliIterationLimitAfterPriority AFTER_REACT_ITERATION 上最后执行：
	// 先看别的 rail 是否已请求 finish，再认领「迭代已完成」。
	processCliIterationLimitAfterPriority = -1000
	// processCliIterationLimitBeforePriority BEFORE_MODEL_CALL 上最先执行：
	// 在普通模型钩子之前拒掉多余的那一轮模型调用。
	processCliIterationLimitBeforePriority = 1000
	// processCliIterationLimitOutput 超限终止时给模型的输出去向说明。
	processCliIterationLimitOutput = "Agent reached its iteration limit before completion."
	// processCliIterationLimitErrorCode 超限终止的错误码。
	processCliIterationLimitErrorCode = "MAX_ITERATIONS_REACHED"
)

// ──────────────────────────── 结构体 ────────────────────────────

// ProcessCliIterationLimitRail 在配置的迭代边界停下，并把原因保留在结果里。
// Python: jiuwenswarm/runtime/iteration_limit.py ProcessCliIterationLimitRail
//
// 内层循环被配置为多跑一轮（configure_process_cli_run 里 max_iterations = limit + 1），
// 让最后一个被中断的工具能在下一个模型边界报告上限之前恢复；那一轮永远不会调模型。
// 成功的最终回答与更早的 force_finish 请求保持原结果。
type ProcessCliIterationLimitRail struct {
	*agentinterfaces.BaseRail
	// maxIterations 配置的迭代上限
	maxIterations int
}

// ──────────────────────────── 导出函数 ────────────────────────────

// NewProcessCliIterationLimitRail 构造迭代上限 rail。
// Python: ProcessCliIterationLimitRail(max_iterations)
func NewProcessCliIterationLimitRail(maxIterations int) *ProcessCliIterationLimitRail {
	return &ProcessCliIterationLimitRail{
		BaseRail:      agentinterfaces.NewBaseRail(),
		maxIterations: maxIterations,
	}
}

// CallbackPriorityFor 按事件返回分档优先级。
// Python: iteration_limit.py:26-29 callback_priority
func (r *ProcessCliIterationLimitRail) CallbackPriorityFor(event agentinterfaces.AgentCallbackEvent) int {
	if event == agentinterfaces.CallbackAfterReactIteration {
		return processCliIterationLimitAfterPriority
	}
	return processCliIterationLimitBeforePriority
}

// BeforeModelCall 超过上限时拒掉本轮模型调用。
// Python: iteration_limit.py:31-34 before_model_call
func (r *ProcessCliIterationLimitRail) BeforeModelCall(ctx context.Context, cbc *agentinterfaces.AgentCallbackContext) error {
	inputs, ok := cbc.Inputs().(*agentinterfaces.ModelCallInputs)
	if !ok {
		return nil
	}
	r.finish(cbc, inputs.ReactIteration > r.maxIterations)
	return nil
}

// AfterReactIteration 达到上限时请求终止。
// Python: iteration_limit.py:36-39 after_react_iteration
func (r *ProcessCliIterationLimitRail) AfterReactIteration(ctx context.Context, cbc *agentinterfaces.AgentCallbackContext) error {
	inputs, ok := cbc.Inputs().(*agentinterfaces.ModelCallInputs)
	if !ok {
		return nil
	}
	r.finish(cbc, inputs.ReactIteration >= r.maxIterations)
	return nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// finish 已耗尽且无人请求终止且无异常时，请求 force_finish。
// Python: iteration_limit.py:41-48 _finish
func (r *ProcessCliIterationLimitRail) finish(cbc *agentinterfaces.AgentCallbackContext, exhausted bool) {
	if !exhausted || cbc.HasForceFinishRequest() || cbc.Exception() != nil {
		return
	}
	cbc.RequestForceFinish(map[string]any{
		"output":      processCliIterationLimitOutput,
		"result_type": "error",
		"code":        processCliIterationLimitErrorCode,
	})
}
```

```go
// configureProcessCliRun ConfigureProcessCliRun 的实现（内部调用）。
func (d *DeepAdapter) configureProcessCliRun(ctx context.Context, maxTurns *int, hostTools []any) error {
	// 步骤 1: 渠道守卫
	// Python: :12430 if self._channel_id != "process_cli": raise ValueError
	if d.channelID != "process_cli" {
		return errors.New("process CLI run options require a Process CLI Agent")
	}
	// 步骤 2: 根是 router → 暂存选项，等 child 消费
	// Python: :12433-12437
	if !d.isSessionScoped {
		d.processCliRunOptions = &processCliRunOptions{maxTurns: maxTurns, hostTools: hostTools}
		return nil
	}
	// 步骤 3: child 直接应用
	// Python: :12438-12440 if self._instance is None: raise
	if d.instance == nil {
		return errors.New("process CLI session Agent is not ready")
	}
	// 步骤 4: max_turns → 改 react config + 注册 rail
	// Python: :12442-12456
	if maxTurns != nil {
		if err := d.applyMaxTurns(*maxTurns); err != nil {
			return err
		}
	}
	// 步骤 5: host_tools → add_ability（重名报错）
	// Python: :12457-12467
	// ⤵️ 回填 20.6（客户端工具暂停-回填）
	if len(hostTools) > 0 {
		return errors.New("host tools are not implemented yet (see 20.6)")
	}
	return nil
}

// applyMaxTurns 把 react 配置的 max_iterations 限死为 limit+1 并注册迭代上限 rail。
// Python: interface_deep.py:12444-12456
func (d *DeepAdapter) applyMaxTurns(maxTurns int) error {
	react := d.instance.ReactAgent()
	if react == nil {
		return errors.New("Agent model loop is unavailable")
	}
	// Python: config = react.config.model_copy()（深拷贝）
	// Q5：Go 的 ReActAgent.Config() 返回内部指针，ReActAgentConfig 无 Clone，
	//     必须手写深拷贝，否则原地污染共享实例。
	cfg, ok := react.Config().(*saconfig.ReActAgentConfig)
	if !ok {
		return fmt.Errorf("unexpected react config type %T", react.Config())
	}
	copied := deepCopyReActAgentConfig(cfg)

	existing := copied.MaxIterations
	// Python: iteration_limit = min(existing, max_turns) if existing else max_turns
	limit := maxTurns
	if existing > 0 && existing < maxTurns {
		limit = existing
	}
	// Python: 多一轮给「恢复的最终工具」用，那一轮由 rail 阻止调模型
	copied.MaxIterations = limit + 1
	if err := react.Configure(context.Background(), copied); err != nil {
		return err
	}
	d.instance.AddRail(harnessrails.NewProcessCliIterationLimitRail(limit))
	return nil
}

// takeProcessCliRunOptions 取出一次性的 run 选项并立即清空。
// Python: interface_deep.py:18989-18992（取用前先置 None，失败运行不泄漏回调）
func (d *DeepAdapter) takeProcessCliRunOptions() *processCliRunOptions {
	opts := d.processCliRunOptions
	d.processCliRunOptions = nil
	return opts
}

// deepCopyReActAgentConfig 深拷贝 ReActAgent 配置。
// Python: ReActAgentConfig.model_copy()（pydantic 深拷贝）
```

```go
var SupportedSchemaVersions = []string{CurrentSchemaVersion}

// IsSchemaVersionSupported 该版本是否为精确实现的机器协议版本。

func IsSchemaVersionSupported(version any) bool {
	s, ok := version.(string)
	if !ok {
		return false
	}
	for _, v := range SupportedSchemaVersions {
		if v == s {
			return true
		}
	}
	return false
}

// RequireSupportedSchemaVersion 校验版本，不做常驻连接协商。
// Python: version.py:22-30 require_supported_schema_version
func RequireSupportedSchemaVersion(version any) (string, error) {
	s, ok := version.(string)
	if !ok {
		// Python: raise TypeError("schema_version must be a string")
		return "", fmt.Errorf("schema_version must be a string")
	}
	if !IsSchemaVersionSupported(s) {
		// Python: raise ValueError(f"unsupported schema_version: {version}")
		return "", fmt.Errorf("unsupported schema_version: %s", s)
	}
	return s, nil
}
```

```go
func freezeJson(name string, value any, ancestors map[uintptr]bool) (any, error) {
	switch v := value.(type) {
	case nil, string, bool, int, int64:
		return v, nil
	case float64:
		// Python: model.py:143-146 if not math.isfinite(value): raise ValueError
		if math.IsNaN(v) || math.IsInf(v, 0) {
			return nil, fmt.Errorf("%s must contain only finite numbers", name)
		}
		return v, nil
	case map[string]any:
		marker := reflectPointer(v)
		// Python: model.py:149-151 if marker in active: raise ValueError("reference cycles")
		if ancestors[marker] {
			return nil, fmt.Errorf("%s must not contain reference cycles", name)
		}
		ancestors[marker] = true
		defer delete(ancestors, marker)
		out := make(map[string]any, len(v))
		for k, item := range v {
			frozen, err := freezeJson(name+"."+k, item, ancestors)
			if err != nil {
				return nil, err
			}
			out[k] = frozen
		}
		return out, nil
	case []any:
		marker := reflectPointer(v)
		if ancestors[marker] {
			return nil, fmt.Errorf("%s must not contain reference cycles", name)
		}
		ancestors[marker] = true
		defer delete(ancestors, marker)
		out := make([]any, len(v))
		for i, item := range v {
			frozen, err := freezeJson(fmt.Sprintf("%s[%d]", name, i), item, ancestors)
			if err != nil {
				return nil, err
			}
			out[i] = frozen
		}
		return out, nil
	default:
		return nil, fmt.Errorf("%s must contain only JSON-compatible values", name)
	}
}
```

```go
	}
```

```go
	enc := json.NewEncoder(&buf)
	// Python: ensure_ascii=False → 不转义非 ASCII；SetEscapeHTML(false) 防 < > & 变 \u003c
	enc.SetEscapeHTML(false)

	return buf.String(), nil
```

```go
package protocol

import (
	_ "embed"
	"encoding/json"
	"fmt"
	"sync"
	"sync/atomic"

	"github.com/santhosh-tekuri/jsonschema/v6"
	"github.com/santhosh-tekuri/jsonschema/v6/kind"
)

// ──────────────────────────── 全局变量 ────────────────────────────

//go:embed schema.json
var schemaJSON []byte

var (
	// schemaOnce 解析 schema.json 一次
	schemaOnce sync.Once
	// schemaData 解析结果（只读）
	schemaData map[string]any
	// schemaErr 解析错误
	schemaErr error
)

// validators 按 kind 缓存的校验器。
// Python: schema.py:23-26 @lru_cache(maxsize=3) _validator
var validators sync.Map

// ──────────────────────────── 导出函数 ────────────────────────────

// ProtocolSchema 返回打包的完整协议 schema 的新副本。
// Python: schema.py:15-18 protocol_schema()
func ProtocolSchema() map[string]any {
	loadSchema()
	if schemaData == nil {
		return map[string]any{}
	}
	// 深拷贝，避免调用方修改污染内部
	return ThawJson(schemaData).(map[string]any)
}

// InputErrorDetails 定位被拒输入，不回显值或敏感 key。
// Python: schema.py:38-63 input_error_details
//
// DTO 校验仍是权威；schema 诊断只在其失败时求值。
func InputErrorDetails(value any, schemaKind string) map[string]any {
	v, err := validatorFor(schemaKind)
	if err != nil || v == nil {
		return map[string]any{"field": "/", "reason": "invalid_value"}
	}
	verr := v.Validate(value)
	if verr == nil {
		return map[string]any{"field": "/", "reason": "invalid_value"}
	}
	ve, ok := verr.(*jsonschema.ValidationError)
	if !ok {
		return map[string]any{"field": "/", "reason": "invalid_value"}
	}
	// Python: :52-58 while error.context: 取 absolute_path 最长、schema != {"type":"null"} 的子错误
	deepest := deepestValidationError(ve)
	path := append([]string{}, deepest.InstanceLocation...)
	// Python: :53-56 validator == "required" 时把缺失的第一个 key 追加到 path
	if req, ok := deepest.ErrorKind.(*kind.Required); ok && len(req.Missing) > 0 {
		path = append(path, req.Missing[0])
	}
	// Python: :57-59 JSON Pointer 转义
	field := "/"
	for _, seg := range path {
		field += escapeJSONPointer(seg) + "/"
	}
	field = strings.TrimSuffix(field, "/")
	if field == "" {
		field = "/"
	}
	// Python: :60-62 reason 映射
	out := map[string]any{"field": field, "reason": reasonFor(deepest.ErrorKind)}
	// Python: :63 validator == "type" 时追加 expected
	if t, ok := deepest.ErrorKind.(*kind.Type); ok {
		out["expected"] = t.Want
	}
	return out
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// loadSchema 解析 schema.json（一次）。
func loadSchema() {
	schemaOnce.Do(func() {
		schemaErr = json.Unmarshal(schemaJSON, &schemaData)
	})
}

// validatorFor 取指定 $defs 项的校验器。
// Python: schema.py:23-26 _validator(kind) = Draft202012Validator({"$ref": f"#/$defs/{kind}", "$defs": ...})
func validatorFor(schemaKind string) (*jsonschema.Schema, error) {
	if cached, ok := validators.Load(schemaKind); ok {
		return cached.(*jsonschema.Schema), nil
	}
	loadSchema()
	if schemaErr != nil {
		return nil, schemaErr
	}
	defs, _ := schemaData["$defs"].(map[string]any)
	if defs == nil {
		return nil, fmt.Errorf("schema.json missing $defs")
	}
	doc := map[string]any{
		"$ref":  "#/$defs/" + schemaKind,
		"$defs": defs,
	}
	raw, err := json.Marshal(doc)
	if err != nil {
		return nil, err
	}
	compiler := jsonschema.NewCompiler()
	decoded, err := jsonschema.UnmarshalJSON(bytes.NewReader(raw))
	if err != nil {
		return nil, err
	}
	if err := compiler.AddResource("mem://inline", decoded); err != nil {
		return nil, err
	}
	s, err := compiler.Compile("mem://inline")
	if err != nil {
		return nil, err
	}
	validators.Store(schemaKind, s)
	return s, nil
}

// deepestValidationError 取 absolute_path 最深、且 schema 不等于 {"type":"null"} 的错误。
// Python: schema.py:52-58
func deepestValidationError(e *jsonschema.ValidationError) *jsonschema.ValidationError {
	cur := e
	for len(cur.Causes) > 0 {
		best := cur.Causes[0]
		for _, c := range cur.Causes[1:] {
			if len(c.InstanceLocation) > len(best.InstanceLocation) {
				best = c
			} else if len(c.InstanceLocation) == len(best.InstanceLocation) &&
				!isNullOnlySchema(c) {
				best = c
			}
		}
		cur = best
	}
	return cur
}

// isNullOnlySchema 该错误的 schema 是否仅为 {"type": "null"}。
func isNullOnlySchema(e *jsonschema.ValidationError) bool {
	if t, ok := e.ErrorKind.(*kind.Type); ok {
		return len(t.Want) == 1 && t.Want[0] == "null"
	}
	return false
}

// reasonFor 把 ErrorKind 映射为稳定的 reason 字符串。
// Python: schema.py:60-62 reasons 映射
func reasonFor(k any) string {
	switch k.(type) {
	case *kind.Type:
		return "invalid_type"
	case *kind.Required:
		return "required"
	case *kind.AdditionalProperties:
		return "unknown_field"
	default:
		return "invalid_value"
	}
}

// escapeJSONPointer JSON Pointer 段转义。
// Python: schema.py:58-59 replace("~","~0").replace("/","~1")
func escapeJSONPointer(seg string) string {
	seg = strings.ReplaceAll(seg, "~", "~0")
	return strings.ReplaceAll(seg, "/", "~1")
}
```

```go
//go:build !windows

package processcli

import (
	"os"
	"syscall"
)

// dupStdoutToStderr POSIX 实现：dup 一份原 fd1 给 writer，再把 fd1 dup2 到 stderr。
// Python: machine_entry.py:66-80（os.dup + os.dup2）
func dupStdoutToStderr() (func() error, *os.File, error) {
	stdoutFD := int(os.Stdout.Fd())
	stderrFD := int(os.Stderr.Fd())

	// Python: output_fd = os.dup(stdout_fd)
	dataFD, err := syscall.Dup(stdoutFD)
	if err != nil {
		return nil, nil, err
	}
	data := os.NewFile(uintptr(dataFD), "protocol-stdout")

	// Python: os.dup2(stderr_fd, stdout_fd)
	if err := syscall.Dup2(stderrFD, stdoutFD); err != nil {
		data.Close()
		return nil, nil, err
	}

	restore := func() error {
		// Python: :75-80 用 dup 出的 fd 恢复，不 close 后重开
		restoreErr := syscall.Dup2(dataFD, stdoutFD)
		// Python: with contextlib.suppress(OSError): output.close()
		_ = data.Close()
		return restoreErr
	}
	return restore, data, nil
}
```

`protocol_stdout_windows.go`：

```go
//go:build windows

package processcli

import (
	"os"
	"syscall"
)

// dupStdoutToStderr Windows 实现：DuplicateHandle 出可写副本，再 SetStdHandle 改 fd1。
// Python: machine_entry.py:66-80（os.dup + os.dup2）
//
// 平台绑定代码：按项目规范 3.2 第 3 条用 build tag 隔离，不纳入非 Windows 覆盖率基线。
func dupStdoutToStderr() (func() error, *os.File, error) {
	stdoutHandle := syscall.Handle(os.Stdout.Fd())
	stderrHandle := syscall.Handle(os.Stderr.Fd())
	proc := syscall.GetCurrentProcess()

	// Python: output_fd = os.dup(stdout_fd)
	var dataHandle syscall.Handle
	if err := syscall.DuplicateHandle(proc, stdoutHandle, proc, &dataHandle,
		0, false, syscall.DUPLICATE_SAME_ACCESS); err != nil {
		return nil, nil, err
	}
	data := os.NewFile(uintptr(dataHandle), "protocol-stdout")

	// Python: os.dup2(stderr_fd, stdout_fd)
	if err := syscall.SetStdHandle(syscall.STD_OUTPUT_HANDLE, stderrHandle); err != nil {
		data.Close()
		return nil, nil, err
	}

	restore := func() error {
		restoreErr := syscall.SetStdHandle(syscall.STD_OUTPUT_HANDLE, stdoutHandle)
		_ = data.Close()
		return restoreErr
	}
	return restore, data, nil
}
```

**注意**：Windows 的 `syscall.DuplicateHandle` / `SetStdHandle` 签名需按实际 Go 版本核对。若 `syscall` 包不提供，用 `golang.org/x/sys/windows`。
