### 0.1 本计划的三个"反直觉"事实（先读，否则会做错方向）

**① `BuildContext` 必须放 core，不能放 `agent_teams`。**
上游 `openjiuwen/harness/schema/build_context.py`（169 行）是真相来源，`openjiuwen/agent_teams/schema/build_context.py` 只有 25 行纯转发。本仓 `internal/agentcore/harness/schema/build_context.go` 必须**是叶子包**——只 import 标准库和 `internal/agentcore/harness/workspace`，**绝不能** import `internal/agent_teams/**`。团队句柄一律走 `Extras map[string]any`，键常量定义在 `agent_teams/rails` 侧。

**② `TeardownTools` 在 Go 侧不是"照抄上游"，需要先补三个配套。**
核查结论（务必当真）：

| 项 | 上游 | Go 现状 |
|---|---|---|
| `AbilityManager._owner_id` | 有（构造注入） | **完全没有该字段** |
| `ToolCard.stateless` | 有（区分有状态/共享单例） | **`tool.ToolCard` 无此字段** |
| `qualify_tool_id(card, owner_id)` | 有（`add_ability` 里改写 `card.id = f"{name}_{owner}"`） | **无**，`Add()` 直接 `am.tools[a.Name] = a` |
| `teardown_tools()` | 有（`ability_manager.py:860-883`） | 无 |

所以 Task 12 的顺序是：先加 `Stateless` 字段 → 再加 `ownerID` + `QualifyToolID` → 最后 `TeardownTools`。**不要跳过前两步直接写 teardown**，那样它永远匹配不到任何工具（`card.ID != name+"_"+ownerID` 恒成立），表现为"调用成功但什么都没删"的静默失效。

**③ `FirstIterationGate` 要删，不是要改。**
上游 `kernel.py:709` 原文："the old FirstIterationGate wait is gone with the single-supervisor model"。它存在的历史前提是"controller 自己驱动轮次时，已启动与首轮已开始之间有窗口"，25.1a 改 supervisor 后该窗口在结构上消失。删除是**对齐**，不是功能倒退。

创建 `internal/agentcore/harness/schema/build_context.go`：

```go
package schema

import (
	"context"
	"fmt"
	"sync"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/harness/workspace"
)

// ──────────────────────────── 结构体 ────────────────────────────

// BuildContextInterface 构建上下文接口。
//
// 用接口而非直接传 *BuildContext 的唯一原因是：上游 BuildContext.derive 用
// copy.copy 保留子类（jiuwenswarm 的 CodeBuildContext 靠这个活），Go 无 copy.copy，
// 只能把 Derive 的返回类型做成接口，让子类自己实现以保类型（偏差 D-1）。
type BuildContextInterface interface {
	// GetLanguage 返回解析后的语言码（如 "cn"/"en"）
	GetLanguage() string
	// GetMemberName 返回团队成员名（非团队成员为空串）
	GetMemberName() string
	// GetRole 返回团队角色值（如 "leader"/"teammate"）
	GetRole() string
	// GetWorkspace 返回活的 Workspace（可能为 nil）
	GetWorkspace() *workspace.Workspace
	// GetMemberCardID 返回解析后的 card id（工具 id 以此做命名空间）
	GetMemberCardID() string
	// GetProjectDir 返回项目根目录（LSP 等 provider 以此定根，可能为空）
	GetProjectDir() string
	// GetTeamOutputsDir 返回共享最终产物目录（无项目成员才有值）
	GetTeamOutputsDir() string
	// GetExtras 返回逃逸口袋，团队活句柄都在这（键见 agent_teams 侧 TeamHandleKey）
	GetExtras() map[string]any
	// GetMoa 返回仅对根 agent 生效的调用策略（可能为 nil）
	GetMoa() any
	// Derive 返回带 overrides 的浅拷贝；实现方必须保留自身具体类型。
	Derive(o BuildContextOverrides) BuildContextInterface
}

// BuildContextOverrides Derive 的覆盖项。
//
// 用指针字段区分"未指定"（nil，继承原值）与"显式置空"（指向空值）。
// Workspace 本身不可空，故另设 HasWorkspace 位表达"显式清空"。
type BuildContextOverrides struct {
	// Language 覆盖语言，nil 表示继承
	Language *string
	// MemberName 覆盖成员名，nil 表示继承
	MemberName *string
	// Role 覆盖角色，nil 表示继承
	Role *string
	// Workspace 覆盖工作空间；仅在 HasWorkspace 为真时生效
	Workspace *workspace.Workspace
	// HasWorkspace 是否应用 Workspace 覆盖（含显式置 nil 清空）
	HasWorkspace bool
	// MemberCardID 覆盖 card id，nil 表示继承
	MemberCardID *string
	// ProjectDir 覆盖项目目录，nil 表示继承
	ProjectDir *string
	// TeamOutputsDir 覆盖产物目录，nil 表示继承
	TeamOutputsDir *string
	// Moa 覆盖调用策略；nil 表示不动
	Moa any
}

// BuildContext 交给能力 provider 的不透明运行时载体。
//
// 它只在"建造那一下"有用：spec 解析时 provider 从它取语言/工作空间/团队句柄，
// 造完就基本不用了（除 NativeHarness 留存给用量归属）。
//
// Python: BuildContext (openjiuwen/harness/schema/build_context.py:52-169)
type BuildContext struct {
	// Language 解析后的语言码
	Language string
	// MemberName 团队成员名
	MemberName string
	// Role 团队角色值
	Role string
	// Workspace 活的工作空间
	Workspace *workspace.Workspace
	// MemberCardID 解析后的 card id
	MemberCardID string
	// ProjectDir 项目根目录
	ProjectDir string
	// TeamOutputsDir 共享最终产物目录
	TeamOutputsDir string
	// Extras 逃逸口袋（团队活句柄）
	Extras map[string]any
	// Moa 仅对根 agent 生效的调用策略
	Moa any
}

// BuildContextFactory 从可序列化种子重建活上下文的工厂。
//
// 跨序列化边界（子进程 spawn / 冷恢复）时 BuildContext 本身不可序列化，
// 平台注册本工厂，spec 携带种子，接收侧用本地句柄重建。
// Python: register_build_context_factory (build_context.py:147-169)
type BuildContextFactory func(seed map[string]any) (BuildContextInterface, error)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// buildContextFactoryMu 保护 buildContextFactory（读写都可能有并发）
	buildContextFactoryMu sync.RWMutex
	// buildContextFactory 进程级唯一工厂；未注册时 BuildContextFromSeed 报错
	buildContextFactory BuildContextFactory
)

// ──────────────────────────── 导出函数 ────────────────────────────

// StringPtr 返回字符串指针（供 BuildContextOverrides 表达"显式指定"）。
func StringPtr(s string) *string { return &s }

// RegisterBuildContextFactory 注册种子重建工厂（同名覆盖，幂等）。
// Python: register_build_context_factory (build_context.py:147-169)
func RegisterBuildContextFactory(f BuildContextFactory) {
	buildContextFactoryMu.Lock()
	defer buildContextFactoryMu.Unlock()
	buildContextFactory = f
}

// ResetBuildContextFactoryForTest 清空工厂（仅测试用）。
//
// 生产不得调用：清空会让跨序列化边界的成员重建失败。
func ResetBuildContextFactoryForTest() {
	buildContextFactoryMu.Lock()
	defer buildContextFactoryMu.Unlock()
	buildContextFactory = nil
}

// BuildContextFromSeed 从可序列化种子重建活上下文。
// Python: build_context_from_seed (build_context.py)
func BuildContextFromSeed(seed map[string]any) (BuildContextInterface, error) {
	buildContextFactoryMu.RLock()
	f := buildContextFactory
	buildContextFactoryMu.RUnlock()
	if f == nil {
		return nil, fmt.Errorf("build context factory not registered")
	}
	return f(seed)
}

// Derive 返回带 overrides 的浅拷贝。
//
// Extras 深拷贝一层：对齐上游 member_build_context.extras = dict(...)（
// agent_configurator.py:867），保证成员之间永不共享句柄字典。
//
// Python: BuildContext.derive (**overrides) (build_context.py:106-118)
func (c *BuildContext) Derive(o BuildContextOverrides) BuildContextInterface {
	clone := *c
	if o.Language != nil {
		clone.Language = *o.Language
	}
	if o.MemberName != nil {
		clone.MemberName = *o.MemberName
	}
	if o.Role != nil {
		clone.Role = *o.Role
	}
	if o.HasWorkspace {
		clone.Workspace = o.Workspace
	}
	if o.MemberCardID != nil {
		clone.MemberCardID = *o.MemberCardID
	}
	if o.ProjectDir != nil {
		clone.ProjectDir = *o.ProjectDir
	}
	if o.TeamOutputsDir != nil {
		clone.TeamOutputsDir = *o.TeamOutputsDir
	}
	if o.Moa != nil {
		clone.Moa = o.Moa
	}
	// Extras 深拷贝一层（值为句柄指针，不递归）
	if c.Extras != nil {
		clone.Extras = make(map[string]any, len(c.Extras))
		for k, v := range c.Extras {
			clone.Extras[k] = v
		}
	}
	return &clone
}
```

```go
package schema

import "fmt"

// ──────────────────────────── 结构体 ────────────────────────────

// ConstructionInput harness 元素的"双源构造输入"契约。
//
// 一个 rail / tool 的构造参数来自两个命名空间：
//   - params：spec 里写的可序列化配置（能被 checkpoint 持久化）
//   - context：运行时活句柄与解析后的事实（不可序列化）
//
// 上游用 pydantic 的 model_fields + json_schema_extra 反射式取值；
// Go 无此能力，改为 struct tag 声明来源 + 手写 Resolve（偏差 D-3）。
//
// Python: ConstructionInput (harness/manifest/inputs.py:98-130)
type ConstructionInput interface {
	// Resolve 从 params + ctx 取齐所有字段并做类型校验。
	Resolve(params map[string]any, ctx BuildContextInterface) error
}

// ──────────────────────────── 导出函数 ────────────────────────────

// ParamString 从 params 取字符串；缺失或为 nil 时返回 def。
func ParamString(params map[string]any, key, def string) string {
	if params == nil {
		return def
	}
	raw, ok := params[key]
	if !ok || raw == nil {
		return def
	}
	s, ok := raw.(string)
	if !ok {
		return def
	}
	return s
}

// ParamInt 从 params 取整数；缺失或为 nil 时返回 def。
//
// 接受 int / int64 / float64（JSON 反序列化会产出 float64）三种形态。
// 类型不符时返回 error（不静默回落默认——静默会让配置错误无声消失）。
func ParamInt(params map[string]any, key string, def int) (int, error) {
	if params == nil {
		return def, nil
	}
	raw, ok := params[key]
	if !ok || raw == nil {
		return def, nil
	}
	switch v := raw.(type) {
	case int:
		return v, nil
	case int64:
		return int(v), nil
	case float64:
		// JSON 数字统一解成 float64；非整值视为配置错误
		if v != float64(int(v)) {
			return def, fmt.Errorf("param %q: expected integer, got float %v", key, v)
		}
		return int(v), nil
	default:
		return def, fmt.Errorf("param %q: expected integer, got %T", key, raw)
	}
}

// ParamBool 从 params 取布尔；缺失或为 nil 时返回 def。
func ParamBool(params map[string]any, key string, def bool) bool {
	if params == nil {
		return def
	}
	raw, ok := params[key]
	if !ok || raw == nil {
		return def
	}
	b, ok := raw.(bool)
	if !ok {
		return def
	}
	return b
}

// ParamStringSlice 从 params 取字符串切片；缺失或为 nil 时返回 nil。
//
// 接受 []string 与 []any 两种形态（JSON 反序列化产出 []any）。
func ParamStringSlice(params map[string]any, key string) []string {
	if params == nil {
		return nil
	}
	raw, ok := params[key]
	if !ok || raw == nil {
		return nil
	}
	switch v := raw.(type) {
	case []string:
		return v
	case []any:
		out := make([]string, 0, len(v))
		for _, item := range v {
			if s, ok := item.(string); ok {
				out = append(out, s)
			}
		}
		return out
	default:
		return nil
	}
}

// ParamStringPtr 从 params 取字符串指针；缺失或为 nil 时返回 nil。
//
// 与 ParamString 的区别：nil 表示"调用方未提供"，空串表示"显式给了空串"。
func ParamStringPtr(params map[string]any, key string) *string {
	if params == nil {
		return nil
	}
	raw, ok := params[key]
	if !ok || raw == nil {
		return nil
	}
	s, ok := raw.(string)
	if !ok {
		return nil
	}
	return &s
}

// ParamMap 从 params 取子映射；缺失或类型不符时返回 nil。
func ParamMap(params map[string]any, key string) map[string]any {
	if params == nil {
		return nil
	}
	raw, ok := params[key]
	if !ok || raw == nil {
		return nil
	}
	m, ok := raw.(map[string]any)
	if !ok {
		return nil
	}
	return m
}

// ContextString 从 context 取字符串字段；取不到或为空时返回 def。
//
// attr 是 BuildContextInterface 上的 getter 去掉 Get 前缀后的名字，
// 例如 "Role" 对应 GetRole()。ctx 为 nil 时直接返回 def。
func ContextString(ctx BuildContextInterface, attr, def string) string {
	if ctx == nil {
		return def
	}
	var v string
	switch attr {
	case "Language":
		v = ctx.GetLanguage()
	case "MemberName":
		v = ctx.GetMemberName()
	case "Role":
		v = ctx.GetRole()
	case "MemberCardID":
		v = ctx.GetMemberCardID()
	case "ProjectDir":
		v = ctx.GetProjectDir()
	case "TeamOutputsDir":
		v = ctx.GetTeamOutputsDir()
	default:
		// 未知属性视同取不到：不 panic，回落默认
		return def
	}
	if v == "" {
		return def
	}
	return v
}

// ContextExtras 返回 context 的 Extras（ctx 或 Extras 为 nil 时返回空 map）。
func ContextExtras(ctx BuildContextInterface) map[string]any {
	if ctx == nil {
		return map[string]any{}
	}
	extras := ctx.GetExtras()
	if extras == nil {
		return map[string]any{}
	}
	return extras
}
```

```go
package rails

import (
	"sync"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/harness/schema"
)

// ──────────────────────────── 结构体 ────────────────────────────

// RailProvider 把一个 RailSpec 的 type+params 解析成活 rail 实例。
//
// 返回 any 的原因：一个 provider 可能返回单个 rail、rail 列表或 nil
// （对齐上游 "A rail, a list of rails, or None; callers flatten the result"）。
//
// ctx 用具体接口类型而非 any：注册表本身不碰团队类型，
// BuildContext.Extras 是唯一的团队句柄通道，因此不引入 any 债。
//
// Python: rail provider 的 (params, context) -> rail 契约
// (harness/schema/deep_agent_spec.py:71-78)
type RailProvider func(params map[string]any, ctx schema.BuildContextInterface) (any, error)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// railRegistryMu 保护 railRegistry。
	// 高频读（每次构造成员）+ 低频写（注册在 init/首次构建），RWMutex 优于 sync.Map。
	railRegistryMu sync.RWMutex
	// railRegistry type 名 → provider
	railRegistry = make(map[string]RailProvider)
)

// ──────────────────────────── 导出函数 ────────────────────────────

// RegisterRailProvider 注册一个 rail provider（同名覆盖，幂等）。
//
// Python: register_rail_provider (harness/schema/deep_agent_spec.py:71-78)
func RegisterRailProvider(name string, f RailProvider) {
	railRegistryMu.Lock()
	defer railRegistryMu.Unlock()
	railRegistry[name] = f
}

// LookupRailProvider 查表；未注册返回 (nil, false)。
func LookupRailProvider(name string) (RailProvider, bool) {
	railRegistryMu.RLock()
	defer railRegistryMu.RUnlock()
	provider, ok := railRegistry[name]
	return provider, ok
}

// RegisteredRailNames 返回已注册的全部 type 名（测试与诊断用）。
func RegisteredRailNames() []string {
	railRegistryMu.RLock()
	defer railRegistryMu.RUnlock()
	names := make([]string, 0, len(railRegistry))
	for name := range railRegistry {
		names = append(names, name)
	}
	return names
}

// ResetRailRegistryForTest 清空注册表（仅测试用）。
//
// 生产不得调用：清空会让所有 RailSpec 解析失败（走"未知 type"分支跳过）。
func ResetRailRegistryForTest() {
```

```go
package schema

import "sync"

// ──────────────────────────── 结构体 ────────────────────────────

// RailProviderFunc rail provider 的函数形态（与 rails.RailProvider 同签名）。
//
// 在这里再声明一次是为了打破 schema ↔ rails 的循环依赖：
// rails 包 import schema（用 BuildContextInterface），若 schema 反过来
// import rails 就会成环。此处的函数类型是两侧的共同契约，
// rails 包在 init 阶段把自己注进来。
type RailProviderFunc func(params map[string]any, ctx BuildContextInterface) (any, error)

// ──────────────────────────── 全局变量 ────────────────────────────

var (
	// railLookupMu 保护 railLookup
	railLookupMu sync.RWMutex
	// railLookup type 名 → provider（由 rails 包注入）
	railLookup = make(map[string]RailProviderFunc)
)

// ──────────────────────────── 导出函数 ────────────────────────────

// RegisterRailProviderFunc 注册一个 rail provider（供 rails 包转发调用）。
//
// 生产调用方只有一个：rails.RegisterRailProvider 内部同时写自己表和这里。
func RegisterRailProviderFunc(name string, f RailProviderFunc) {
	railLookupMu.Lock()
	defer railLookupMu.Unlock()
	railLookup[name] = f
}

// LookupRailProviderFunc 查表；未注册返回 (nil,false)。
func LookupRailProviderFunc(name string) (RailProviderFunc, bool) {
	railLookupMu.RLock()
	defer railLookupMu.RUnlock()
	f, ok := railLookup[name]
	return f, ok
}

// ResetRailProviderFuncsForTest 清空注入表（仅测试用）。
func ResetRailProviderFuncsForTest() {
	railLookupMu.Lock()
	defer railLookupMu.Unlock()
	railLookup = make(map[string]RailProviderFunc)
}

// lookupRailProviderHook 内部查表入口。
func lookupRailProviderHook(name string) (RailProviderFunc, bool) {
	return LookupRailProviderFunc(name)
}
```

然后修改 `internal/agentcore/harness/rails/registry.go` 的 `RegisterRailProvider` 与 `LookupRailProvider`，让它们**同时**写/读 schema 侧的注入表：

```go
// RegisterRailProvider 注册一个 rail provider（同名覆盖，幂等）。
//
// 同时写入本包注册表与 schema 侧的注入表：后者让 schema.RailSpec.Build
// 能查到 provider 而不必 import 本包（打破 schema ↔ rails 循环依赖）。
//
// Python: register_rail_provider (harness/schema/deep_agent_spec.py:71-78)
func RegisterRailProvider(name string, f RailProvider) {
	railRegistryMu.Lock()
	railRegistry[name] = f
	railRegistryMu.Unlock()

}
```

```go
type DeepAgentParts struct {

			agentschema.WithAgentDescription("DeepAgent instance"),

func ApplyDeepAgentParts(ctx context.Context, parts *DeepAgentParts) (*DeepAgent, error) {

		return nil, fmt.Errorf("apply deep agent parts: parts or config is nil")
	}
	card := parts.Config.Card

	// 步骤 8：实例化
	agent := NewDeepAgent(card)

		return nil, fmt.Errorf("配置 DeepAgent 失败: %w", err)
```

```go
// ResolveParts 把 spec 解析成 DeepAgent 的装配产物（不创建实例）。
//
// 步骤严格对齐上游：
//  1. 模型 / 语言 / 视觉 / 音频 / workspace / card id 逐个解析
//  2. 衍生 build_ctx（回填 workspace + card id + moa=nil），ctx 为 nil 则自建
//  3. build_ctx.Extras["_parent_model"] = 模型（子代理工厂要读）
//  4. sys_operation 解析——**必须在 rails / subagents 之前**：不带 sys_operation
//     的子代理会拿到一个全新的 LOCAL 实例，既逃出本 agent 的沙箱，又把
//     restrict_to_sandbox 对着自己更窄的 workspace 打开
//  5. rails：逐个 RailSpec.Build(language, workspace, build_ctx) 并展平
//  6. subagents：逐个 build（⤵️ 领域十五，本版直接透传）
//
// 返回的 parts 交给 ApplyDeepAgentParts 物化。
//
// Python: DeepAgentSpec.resolve_parts (harness/schema/deep_agent_spec.py:524-634)
func (s *DeepAgentSpec) ResolveParts(ctx BuildContextInterface) (*DeepAgentParts, error) {
	// 步骤 1：逐项解析
	var llmModel *llm.Model
	if s.Model != nil {
		cfg := s.Model.ModelClientConfig
		m, err := llm.NewModel(&cfg, s.Model.ModelRequestConfig)
		if err != nil {
			logger.Warn(logComponent).Err(err).
				Str("model_name", cfg.ModelName).
				Msg("resolve parts: model build failed, agent will have no model")
		} else {
			llmModel = m
		}
	}
	language := hprompts.ResolveLanguage(s.Language)

	var workspaceObj *workspace.Workspace
	if s.Workspace != nil {
		workspaceObj = &workspace.Workspace{
			RootPath: s.Workspace.RootPath,
			Language: language,
		}
	}
	memberCardID := ""
	if s.Card != nil {
		memberCardID = s.Card.GetID()
	}

	// 步骤 2：衍生 build_ctx
	var buildCtx BuildContextInterface
	if ctx != nil {
		buildCtx = ctx.Derive(BuildContextOverrides{
			Language:     StringPtr(language),
			MemberCardID: StringPtr(memberCardID),
			HasWorkspace: true,
			Workspace:    workspaceObj,
		})
	} else {
		buildCtx = &BuildContext{
			Language:     language,
			Workspace:    workspaceObj,
			MemberCardID: memberCardID,
		}
	}

	// 步骤 3：父模型进 extras（键必须存在，值可为 nil）
	if buildCtx.GetExtras() != nil {
		buildCtx.GetExtras()["_parent_model"] = llmModel
	}

	// 步骤 4：sys_operation（先于 rails / subagents）
	var sysOp sysop.SysOperation
	if s.SysOperation != nil {
		resolved, err := resolveSysOperationSpec(s.SysOperation)
		if err != nil {
			return nil, fmt.Errorf("resolve parts: sys_operation failed: %w", err)
		}
		sysOp = resolved
	}
	buildCtx.GetExtras()[ParentSysOperationExtrasKey] = sysOp

	// 步骤 5：rails 经 provider 解析
	rails := make([]agentinterfaces.AgentRail, 0, len(s.Rails))
	for _, railSpec := range s.Rails {
		built, err := railSpec.Build(language, workspaceObj, buildCtx)
		if err != nil {
			return nil, fmt.Errorf("resolve parts: rail %q failed: %w", railSpec.Type, err)
		}
		if built == nil {
			continue
		}
		rails = append(rails, flattenRails(built)...)
	}

	// 步骤 6：组装 CreateDeepAgentParams（复用既有 configure 路径）
	params := s.toCreateParams(llmModel, language, workspaceObj, sysOp, rails)

	// 步骤 7：委托同包的 config 组装（schema 包内的私有版本）
	//
	// 命名说明：schema 包内是私有 resolveDeepAgentPartsFromParams，
	// 它导出的包装是 ResolveDeepAgentPartsFromParams（供 harness 包调用）。
	// 两个名字指同一逻辑，切勿在本包内调大写版本（会绕过私有细节）。
	return resolveDeepAgentPartsFromParams(&params)
}

// flattenRails 把 provider 返回的 any 展平成 rail 列表。
//
// provider 可能返回：单个 AgentRail、[]AgentRail、[]any、nil。
func flattenRails(built any) []agentinterfaces.AgentRail {
	switch v := built.(type) {
	case nil:
		return nil
	case agentinterfaces.AgentRail:
		return []agentinterfaces.AgentRail{v}
	case []agentinterfaces.AgentRail:
		return v
	case []any:
		out := make([]agentinterfaces.AgentRail, 0, len(v))
		for _, item := range v {
			if r, ok := item.(agentinterfaces.AgentRail); ok {
				out = append(out, r)
			}
		}
		return out
	default:
		logger.Warn(logComponent).
			Str("built_type", fmt.Sprintf("%T", built)).
			Msg("resolve parts: rail provider returned unexpected type, skipping")
		return nil
	}
}
```

```go
// resolveSysOperationSpec 把 SysOperationSpec 解析成活实例。
//
// 上游是 get-or-create（同一稳定 id 复用同一实例）；Go 侧按 ID 去重由
// 资源管理器负责，这里只负责构造。
//
// WorkConfig / GatewayConfig 在 Go 侧仍是 any 占位（见 SysOperationSpec 注释），
// 因此按 mode 分支取用：Local 模式看 WorkConfig，Sandbox 模式看 GatewayConfig。
//
// Python: SysOperationSpec.resolve
func resolveSysOperationSpec(spec *SysOperationSpec) (sysop.SysOperation, error) {
	if spec == nil {
		return nil, nil
	}
	var mode sysop.OperationMode
	switch spec.Mode {
	case "local", "":
		mode = sysop.OperationModeLocal
	case "sandbox":
		mode = sysop.OperationModeSandbox
	default:
		return nil, fmt.Errorf("unknown sys_operation mode %q", spec.Mode)
	}

	card := sysop.NewSysOperationCardWithMode(mode)
	if spec.ID != "" {
		card.SetID(spec.ID)
	}

	// 运行配置透传：Local 用 WorkConfig，Sandbox 用 GatewayConfig
	switch mode {
	case sysop.OperationModeLocal:
		if cfg, ok := spec.WorkConfig.(*sysop.LocalWorkConfig); ok {
			card.SetRunConfig(cfg)
		}
	case sysop.OperationModeSandbox:
		if cfg, ok := spec.GatewayConfig.(*sysop.SandboxGatewayConfig); ok {
			card.SetRunConfig(cfg)
		}
	}
	return sysop.NewSysOperation(card)
}
```

```go
// TeardownTools 把本 owner 限定过的有状态工具从资源管理器里摘掉。
//
// 轮末拆除调用：停掉的 agent 不该把逐 agent（有状态）的工具实例留在
// 进程级资源管理器里。否则下一个运行周期重建一个全新 agent 时会用同一个
// id 重新注册，而 Add 的重复检测会为每个残留 id 记一条告警。
//
// 只移除本管理器用自己的 owner id 限定过的工具（id == "{name}_{ownerID}"）；
// 无状态共享单例保持不动（别的 agent 还在用）。
//
// Python: AbilityManager.teardown_tools (ability_manager.py:860-883)
func (am *AbilityManager) TeardownTools() {
	if am.ownerID == "" {
		return
	}

	am.mu.Lock()
	names := make([]string, 0, len(am.tools))
	ids := make([]string, 0, len(am.tools))
	for name, card := range am.tools {
		if card == nil || card.Stateless {
			continue
		}
		// 只摘本 owner 限定过的（防御：ownerID 变更或卡片来自别处时不误删）
		if card.ID != name+"_"+am.ownerID {
			continue
		}
		names = append(names, name)
		ids = append(ids, card.ID)
	}
	// 先从本 manager 摘除（持锁）
	for _, name := range names {
		delete(am.tools, name)
	}
	if len(names) > 0 {
		am.registryRevision++
	}
	am.mu.Unlock()

	// 再摘资源管理器（在锁外，避免 RemoveTool 内部可能的锁顺序问题）
	if len(ids) > 0 && am.resourceMgr != nil {
		_, _ = am.resourceMgr.RemoveTool(ids)
	}
}
```

```go
// AnnounceTaskRevisions 立即投递成员的待发任务编辑通知（F-129）。
//
// 通知的唯一来源是成员的任务通知器（与 policy rail 借用的是同一实例）。
// 运行周期外没有会话可认领，什么都不认领——rail 会在下次输入时告诉成员。

//
// Python: TeamHarness.announce_task_revisions (team_harness.py:291-304)
func (h *TeamHarness) AnnounceTaskRevisions(ctx context.Context) error {
	if h.taskEventTracker == nil {
		return nil
	}
	cur := h.CurrentSession()
	if cur == nil {
		return nil
	}
	_, err := h.taskEventTracker.Announce(cur, func(text string) error {

		return h.Send(ctx, text, true)

	return err
```

```go
// Python: TaskBoardHandler.on_task_updated (task_board.py:332-360)
func (h *TaskBoardHandler) OnTaskUpdated(ctx context.Context, event types.CoordinationEvent) {
	memberName := h.blueprint.DispatcherMemberName()
	if memberName == "" {
		return
	}
	if event.IsInner() {
		return
	}

	payloadMember, _ := event.Transport.Payload["member_name"].(string)
	if payloadMember != memberName {
		// 别人的编辑：走通用任务板逻辑（leader 观察；编辑不扩大可认领池）
		h.OnTaskBoardEvent(ctx, event)
		return

	// 是自己的任务：恢复轮询 + 让通知器立刻说话
	h.poll.ResumePolls(ctx)

		}
```
