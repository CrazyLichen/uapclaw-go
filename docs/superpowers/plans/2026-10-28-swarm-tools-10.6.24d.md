func IsReasoningEvent(eventType any, payload map[string]any) bool {
	eventName, _ := eventType.(string)
	if eventName == "chat.reasoning" {
		return true
	}
	scType, _ := payload["source_chunk_type"].(string)
	peType, _ := payload["event_type"].(string)
	return scType == "llm_reasoning" || peType == "chat.reasoning"
}

// BuildAcpToolDescriptor 构造 ACP toolCall descriptor。
// Python: build_acp_tool_descriptor (L118-147)
func BuildAcpToolDescriptor(toolName string, arguments any, opts ...AcpToolDescriptorOption) map[string]any {
	cfg := &acpToolDescriptorConfig{}
	for _, o := range opts {
		o(cfg)
	}
	rawInput := normalizeArguments(arguments)
	normalized := NormalizeToolName(toolName)
	resolvedKind := strings.TrimSpace(cfg.Kind)
	if resolvedKind == "" {
		resolvedKind = inferToolKind(normalized)
	}
	if resolvedKind == "" {
		resolvedKind = "other"
	}
	resolvedTitle := strings.TrimSpace(cfg.Title)
	if resolvedTitle == "" {
		resolvedTitle = buildToolTitle(normalized, rawInput)
	}

	descriptor := map[string]any{
		"toolCallId": strings.TrimSpace(cfg.ToolCallID),
		"title":      resolvedTitle,
		"kind":       resolvedKind,
	}
	if cfg.Status != "" {
		descriptor["status"] = cfg.Status
	}
	if len(rawInput) > 0 {
		descriptor["rawInput"] = rawInput
	}
	locations := extractLocations(rawInput)
	if len(locations) > 0 {
		descriptor["locations"] = locations
	}
	if cfg.RawOutput != nil && !isListLikeToolName(normalized) {
		descriptor["rawOutput"] = cfg.RawOutput
	}
	return descriptor
}

// BuildAcpToolCallUpdate 构造 tool_call session/update。
// Python: build_acp_tool_call_update (L150-189)
func BuildAcpToolCallUpdate(payload map[string]any, cache map[string]map[string]any) map[string]any {
	toolCall, _ := payload["tool_call"].(map[string]any)
	if toolCall == nil {
		return nil
	}
	toolCallID := resolveToolCallID(toolCall)
	if toolCallID == "" {
		return nil
	}
	originalName, _ := toolCall["name"].(string)
	arguments := toolCall["arguments"]
	if arguments == nil {
		arguments = map[string]any{}
	}
	status, _ := toolCall["status"].(string)
	if status == "" {
		status, _ = payload["status"].(string)
	}
	if status == "" {
		status = "pending"
	}
	title := firstNonEmptyStr(toolCall["title"], payload["title"])
	kind := firstNonEmptyStr(toolCall["kind"], payload["kind"])
	descriptor := BuildAcpToolDescriptor(originalName, arguments,
		WithToolCallID(toolCallID), WithStatus(status),
		WithTitle(title), WithKind(kind))

	update := map[string]any{
		"sessionUpdate": "tool_call",
		"toolCall": map[string]any{
			"id":        toolCallID,
			"name":      originalName,
			"arguments": legacyArguments(arguments),
		},
	}
	for k, v := range descriptor {
		update[k] = v
	}
	if cache != nil {
		cached := map[string]any{"toolName": originalName}
		for k, v := range descriptor {
			cached[k] = v
		}
		cache[toolCallID] = cached
	}
	return update
}

// BuildAcpToolResultUpdate 构造 tool_call_update session/update。

func BuildAcpToolResultUpdate(payload map[string]any, cache map[string]map[string]any) map[string]any {
	result := payload["result"]
	if result == nil {
		result = payload["content"]
		if result == nil {
			result = ""
		}
	}
	toolCallID := resolveToolCallID(payload)
	var cached map[string]any
	if cache != nil && toolCallID != "" {
		cached = cache[toolCallID]
		if cached == nil {
			cached = map[string]any{}
		}
	}
	toolName := firstNonEmptyStr(payload["tool_name"], payload["name"])
	if toolName == "" {
		if tc, ok := payload["toolCall"].(map[string]any); ok {
			toolName, _ = tc["name"].(string)
		}
	}
	if toolName == "" {
		toolName, _ = cached["toolName"].(string)
	}
	arguments := firstNonNil(payload["arguments"], payload["rawInput"])
	if arguments == nil || arguments == "" {
		if tc, ok := payload["toolCall"].(map[string]any); ok {
			arguments = tc["arguments"]
		}
	}
	rawOutput := firstNonNil(payload["rawOutput"], payload["raw_output"])
	terminalID := extractTerminalID(rawOutput)

	update := map[string]any{
		"sessionUpdate": "tool_call_update",
		"toolCallId":    toolCallID,
	}
	if result != nil && result != "" && terminalID == "" {
		update["result"] = result
	}
	if toolName != "" {
		update["toolName"] = toolName
	}
	for _, key := range []string{"title", "kind", "rawInput", "locations"} {
		if v := cached[key]; v != nil && v != "" {
			update[key] = v
		}
	}
	title := firstNonEmptyStr(payload["title"], strAny(update["title"]))
	kind := firstNonEmptyStr(payload["kind"], strAny(update["kind"]))
	descriptor := BuildAcpToolDescriptor(toolName,
		firstNonNil(arguments, update["rawInput"], map[string]any{}),
		WithToolCallID(toolCallID),
		WithStatus(resolveToolResultStatus(payload, result, toolName, rawOutput)),
		WithRawOutput(rawOutput), WithTitle(title), WithKind(kind))
	for _, key := range []string{"title", "kind", "status", "rawOutput"} {
		if v := descriptor[key]; v != nil && v != "" {
			update[key] = v
		}
	}
	content := buildTerminalContentBlocks(terminalID)
	if len(content) == 0 {
		content = buildContentBlocks(result)
	}
	if len(content) > 0 {
		update["content"] = content
	}
	if cache != nil && toolCallID != "" {
		merged := map[string]any{}
		for k, v := range cached {
			merged[k] = v
		}
		for _, key := range []string{"title", "kind", "rawInput", "locations", "rawOutput"} {
			if v := descriptor[key]; v != nil {
				merged[key] = v
			}
		}
		merged["toolName"] = toolName
		if merged["toolName"] == "" {
			merged["toolName"] = strAny(cached["toolName"])

	path := summarizePath(firstStringStrict(arguments, pathKeys...))
	query := firstStringStrict(arguments, "pattern", "query", "term", "text")

	if containsSet(executeToolNames, normalized) {

		}
		if command != "" {

		}
		if path != "" {

		}

	}

		return "Fetching " + url
	}
	if containsSet(searchToolNames, normalized) {
		if query != "" && path != "" {

		}

			return "Searching " + query
		}
		if path != "" {
			return "Searching in " + path
		}

	if containsSet(editToolNames, normalized) {

	}

// summarizePath 路径摘要（末 3 段，Python _summarize_path L455-465）。
func summarizePath(path string) string {
	if path == "" {
		return ""
	}
	trimmed := strings.Trim(path, "/\\")
	parts := strings.Split(trimmed, "/")
	// Windows 反斜杠路径归一
	if len(parts) == 1 && strings.Contains(parts[0], "\\") {
		parts = strings.Split(trimmed, "\\")
	}
	if len(parts) == 0 {
		return path
	}
	if len(parts) > 3 {
		parts = parts[len(parts)-3:]
	}
	return strings.Join(parts, "/")
}

			continue

// consumeOneJSON 从缓冲头解析一个完整 JSON 值；不完整返回 (nil, 原缓冲)。
// Python: _consume_one_json (stdio_client.py L65-81)
func consumeOneJSON(buf []byte) (any, []byte) {
	s := bytes.TrimLeft(buf, " \t\r\n")
	if len(s) == 0 {
		return nil, buf
	}
	if len(s) > maxStdoutBuffer() {
		panic(fmt.Sprintf("ACP stdout parse buffer exceeds %d bytes", maxStdoutBuffer()))
	}
	dec := json.NewDecoder(bytes.NewReader(s))
	var obj any
	if err := dec.Decode(&obj); err != nil {
		return nil, buf
	}
	// 消费长度 = 总长 - Buffered 剩余
	consumed := len(s) - dec.Buffered()
	rest := bytes.TrimLeft(s[consumed:], " \t\r\n")
	return obj, rest
}

// isPeerJSONRPCRequest 判定消息是否为对端 JSON-RPC 请求（期待我们回响应）。
// Python: _is_peer_jsonrpc_request (stdio_client.py L84-97)
func isPeerJSONRPCRequest(msg map[string]any) bool {
	if msg["jsonrpc"] != "2.0" {
		return false
	}
	if _, ok := msg["method"].(string); !ok {
		return false
	}
	if msg["id"] == nil {
		return false
	}
	if _, hasResult := msg["result"]; hasResult {
		return false
	}
	if _, hasError := msg["error"]; hasError {
		return false
	}
	return true
}

// extractSessionUpdateText 提取 session/update 通知的文本内容。

func extractSessionUpdateText(msg map[string]any) string {
	if msg["method"] != "session/update" {
		return ""
	}
	params, _ := msg["params"].(map[string]any)
	if params == nil {
		return ""
	}
	update, _ := params["update"].(map[string]any)
	if update == nil {
		return ""
	}
	content, _ := update["content"].(map[string]any)
	if content == nil {
		return ""
	}
	if content["type"] == "text" {
		return strOf(content["text"])
	}

// jsonrpcErrorMessage 人类可读的 JSON-RPC 错误描述。
// Python: _jsonrpc_error_message (stdio_client.py L48-62)
func jsonrpcErrorMessage(err any) string {
	m, ok := err.(map[string]any)
	if !ok {
		return fmt.Sprintf("ACP JSON-RPC error: %v", err)
	}
	var parts []string
	if code, exists := m["code"]; exists && code != nil {
		parts = append(parts, strOf(code))
	}
	if message, exists := m["message"]; exists && message != nil {
		if s, ok := message.(string); ok && s != "" {
			parts = append(parts, s)
		}
	}
	if data, exists := m["data"]; exists && data != nil {
		parts = append(parts, fmt.Sprintf("data=%v", data))
	}
	if len(parts) == 0 {
		return fmt.Sprintf("ACP JSON-RPC error: %v", err)
	}
	return "ACP JSON-RPC error: " + strings.Join(parts, " ")
}

			return v

// NewAcpStdioClient 构造客户端（env 为 profile env map，内部经 buildAcpSubprocessEnv 合成）。
func NewAcpStdioClient(command string, args []string, cwd string, env map[string]any) *AcpStdioClient {
	c := &AcpStdioClient{
		command: strings.TrimSpace(command),
		args:    append([]string{}, args...),
		cwd:     cwd,
		pending: make(map[int64]chan *rpcResult),
	}
	if cwd == "" {
		c.cwd = ""
	}
	envMap := BuildAcpSubprocessEnv(env)
	for k, v := range envMap {
		c.env = append(c.env, k+"="+v)
	}
	return c
}

// SessionID 返回当前会话 ID。
func (c *AcpStdioClient) SessionID() string {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.sessionID
}

// IsConnected 是否已连接且子进程存活。
func (c *AcpStdioClient) IsConnected() bool {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.connected && c.cmd != nil && c.cmd.ProcessState == nil
}

// Connect 启动子进程并完成 initialize + session/new 握手。
// Python: connect() (L488-543)
func (c *AcpStdioClient) Connect(ctx context.Context) error {
	c.mu.Lock()
	if c.connected && c.cmd != nil {
		c.mu.Unlock()
		return nil
	}
	if c.closeInvoked() {
		c.mu.Unlock()
		return errors.New("AcpStdioClient is closed")
	}
	if c.command == "" {
		c.mu.Unlock()
		return errors.New("ACP agent command is empty")
	}
	c.stdoutBuf = nil
	cmd := exec.CommandContext(ctx, c.command, c.args...)
	cmd.Env = c.env
	if c.cwd != "" {
		cmd.Dir = c.cwd
	}
	if killProcessGroup() && runtime.GOOS != "windows" {
		cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	}
	stdin, err := cmd.StdinPipe()
	if err != nil {
		c.mu.Unlock()
		return err
	}
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		c.mu.Unlock()
		return err
	}
	stderr, err := cmd.StderrPipe()
	if err != nil {
		c.mu.Unlock()
		return err
	}
	if err := cmd.Start(); err != nil {
		c.mu.Unlock()
		return err
	}
	c.cmd = cmd
	c.stdin = stdin
	c.connected = true
	stderrDone := make(chan struct{})
	c.stderrDone = stderrDone
	c.mu.Unlock()

	// stderr drain goroutine（Python _drain_stderr L396-409）
	go func() {
		defer close(stderrDone)
		scanner := bufio.NewScanner(stderr)
		scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
		for scanner.Scan() {
			if txt := scanner.Text(); txt != "" {
				logger.Debug(logComponent).Str("line", txt).Msg("[acp-agent stderr]")
			}
		}
	}()

	// reader goroutine：解析 stdout 并分发（Python _read_one_message 缓冲循环）
	go c.readerLoop(stdout)

	// initialize（120s）
	_, err = c.rpcCall(ctx, "initialize", map[string]any{
		"protocolVersion": 1,
		"clientCapabilities": map[string]any{
			"fs":       map[string]any{"readTextFile": true, "writeTextFile": true},
			"terminal": false,
		},
		"clientInfo": map[string]any{"name": "jiuwenswarm", "version": "0.1"},
	}, 120*time.Second)
	if err != nil {
		_ = c.Close()
		return err
	}

	// session/new（60s）
	sessionCwd := c.cwd
	if sessionCwd == "" {
		sessionCwd, _ = os.Getwd()
	}
	result, err := c.rpcCall(ctx, "session/new", map[string]any{
		"cwd": sessionCwd, "mcpServers": []any{},
	}, 60*time.Second)
	if err != nil {
		_ = c.Close()
		return err
	}
	resultMap, ok := result.(map[string]any)
	if !ok {
		_ = c.Close()
		return fmt.Errorf("session/new: unexpected result: %v", result)
	}
	sid := strings.TrimSpace(strOf(firstNonNil(resultMap["sessionId"], resultMap["session_id"])))
	if sid == "" || sid == "<nil>" {
		_ = c.Close()
		return fmt.Errorf("session/new: missing sessionId in %v", result)
	}
	c.mu.Lock()
	c.sessionID = sid
	c.mu.Unlock()
	logger.Info(logComponent).Str("session_id", sid).Msg("[AcpStdioClient] connected")
	return nil
}

// Chat 发送 prompt 并聚合文本回复。
// Python: chat() (L545-590)
func (c *AcpStdioClient) Chat(ctx context.Context, message string, opts ...ChatOption) (string, error) {
	cfg := chatConfig{timeout: defaultChatTimeout()}
	for _, o := range opts {
		o(&cfg)
	}
	c.mu.Lock()
	if !c.connected || c.sessionID == "" {
		c.mu.Unlock()
		return "", errors.New("not connected; call connect() first")
	}
	sid := c.sessionID
	c.mu.Unlock()

	rid := c.allocID()
	params := map[string]any{
		"sessionId": sid,
		"prompt":    []any{map[string]any{"type": "text", "text": message}},
	}
	if err := c.writeRequest(rid, "session/prompt", params); err != nil {
		return "", err
	}

	waitCh, err := c.registerPending(rid)
	if err != nil {
		return "", err
	}
	defer c.removePending(rid)

	var parts []string
	deadline := time.After(cfg.timeout)
	updates := c.updates()
	for {
		select {
		case <-ctx.Done():
			return "", ctx.Err()
		case <-deadline:
			return "", errors.New("timed out or EOF waiting for session/prompt completion")
		case text := <-updates:
			if text != "" {
				parts = append(parts, text)
			}
		case res := <-waitCh:
			if res.Err != nil {
				return "", res.Err
			}
			joined := strings.TrimSpace(strings.Join(parts, ""))
			if joined != "" {
				return joined, nil
			}
			return resultTextFallback(res.Msg["result"]), nil
		}
	}
}

**updates 通道设计**（与 spec §3.2 一致）：readerLoop 解析出无 id 通知（含 session/update）→ 提取文本 → 非阻塞投递 `updates chan string`（缓冲 256；满时丢弃并 Debug 日志——对齐 Python chat 循环里非本请求消息被 continue 的语义，缓冲满丢最旧不可取，直接丢弃新项防阻塞 reader）。

**peer request 处理**：readerLoop 识别 isPeerJSONRPCRequest → `go c.handlePeerRequest(msg)`（回写响应，见 spec §3.4 四分支——request_permission 的 outcome 结构、fs 读写防穿越、-32601 兜底）。

**rpcCall**（Python L454-486）：写请求 → select 等 pending[rid] → 返回 result；**注意 Python `_rpc_call` 内 session/update 是 continue（丢弃）**——Go 里 readerLoop 统一投递 updates chan，rpcCall 不消费 updates（等价：rpcCall 只关心自己的响应）。

```go
// Close 幂等三段优雅关闭。
// Python: close() (L594-660)
func (c *AcpStdioClient) Close() error {
	c.closeOnce.Do(func() {
		c.mu.Lock()
		c.connected = false
		c.sessionID = ""
		cmd := c.cmd
		stdin := c.stdin
		stderrDone := c.stderrDone
		c.cmd = nil
		c.stdin = nil
		c.stderrDone = nil
		c.mu.Unlock()
		c.markClosed()

		if cmd == nil {
			if stderrDone != nil {
				<-stderrDone
			}
			return
		}

		// 1. stdin 关闭（等 ACP_CLOSE_STDIN_WAIT_S）
		if stdin != nil {
			_ = stdin.Close()
			waitProc(cmd, closeStdinWaitS())
		}

		// 2. TERM（进程组，等 ACP_CLOSE_TERM_WAIT_S）
		if cmd.ProcessState == nil {
			signalProcessLeader(cmd, false)
			waitProc(cmd, closeTermWaitS())
		}

		// 3. KILL（等 ACP_CLOSE_KILL_WAIT_S）
		if cmd.ProcessState == nil {
			signalProcessLeader(cmd, true)
			waitProc(cmd, closeKillWaitS())
		}

		// 4. stderr join（≤ ACP_CLOSE_STDERR_JOIN_S）
		if stderrDone != nil {
			select {
			case <-stderrDone:
			case <-time.After(closeStderrJoinS()):
			}
		}

		// 5. 失败 pending 投递错误
		c.failAllPending(errors.New("AcpStdioClient is closed"))
	})
	return c.closeErr
}
```

辅助：`waitProc(cmd, d)` = `done := make(chan struct{}); go func(){ _ = cmd.Wait(); close(done) }(); select { <-done, <-time.After(d) }`；`signalProcessLeader(cmd, brutal)` = Windows 用 Kill/Terminate，非 Windows killpg 或 单进程信号（`syscall.Kill(-pid, sig)` 进程组 / `cmd.Process.Signal(sig)`）；各 wait 值经 env 读取（默认 4/5/8/3s）。

// sliceFileLines 行切片（line<1 钳 1；limit>0 生效否则到尾）。
// Python: _slice_file_lines (L155-168)
func sliceFileLines(text string, lineRaw, limitRaw any) string {
	lines := strings.Split(text, "\n")
	// splitlines(True) 保留换行符
	for i, l := range lines {
		if i < len(lines)-1 {
			lines[i] = l + "\n"
		}
	}
	lineNo := 1
	if v, ok := toInt(lineRaw); ok && v > 1 {
		lineNo = v
	}
	start := lineNo - 1
	if start > len(lines) {
		start = len(lines)
	}
	if lim, ok := toInt(limitRaw); ok && lim > 0 {
		end := start + lim
		if end > len(lines) {
			end = len(lines)
		}
		return strings.Join(lines[start:end], "")
	}
	return strings.Join(lines[start:], "")
}

		return p, nil

		return p, err
	}

}

// readerLoop 读 stdout、解析 JSON-RPC 消息并分发（Python _read_one_message L411-452 的常驻化）。
func (c *AcpStdioClient) readerLoop(stdout io.ReadCloser) {
	buf := make([]byte, stdoutReadChunk())
	for {
		n, err := stdout.Read(buf)
		if n > 0 {
			c.updateMu.Lock()
			c.stdoutBuf = append(c.stdoutBuf, buf[:n]...)
			// 解析循环
			for {
				obj, rest, perr := consumeOneJSONErr(c.stdoutBuf)
				if perr != nil {
					logger.Error(logComponent).Err(perr).Msg("stdout parse buffer exceeds limit")
					c.failAllPending(perr)
					return
				}
				if obj == nil {
					break
				}
				c.stdoutBuf = rest
				msg, ok := obj.(map[string]any)
				if !ok {
					logger.Warn(logComponent).Msg("[AcpStdioClient] dropping non-object JSON-RPC message")
					continue
				}
				c.dispatchMessage(msg)
			}
			c.updateMu.Unlock()
		}
		if err != nil {
			if err != io.EOF && !errors.Is(err, os.ErrClosed) {
				logger.Debug(logComponent).Err(err).Msg("stdout reader stopped")
			}
			c.updateMu.Lock()
			if len(bytes.TrimSpace(c.stdoutBuf)) > 0 {
				logger.Warn(logComponent).Int("bytes", len(c.stdoutBuf)).
					Msg("[AcpStdioClient] EOF while partial JSON buffered")
			}
			c.updateMu.Unlock()
			c.failAllPending(errors.New("ACP agent closed stream"))
			return
		}
	}
}

// dispatchMessage 按消息形态分发（响应/对端请求/通知）。
func (c *AcpStdioClient) dispatchMessage(msg map[string]any) {
	// 响应：有 id 且有 result 或 error
	if idAny := msg["id"]; idAny != nil {
		if _, hasResult := msg["result"]; hasResult {
			c.deliverPending(idAny, &rpcResult{Msg: msg})
			return
		}
		if _, hasError := msg["error"]; hasError {
			errMsg := jsonrpcErrorMessage(msg["error"])
			c.deliverPending(idAny, &rpcResult{Msg: msg, Err: errors.New(errMsg)})
			return
		}
	}
	// 对端请求
	if isPeerJSONRPCRequest(msg) {
		go c.handlePeerRequest(msg)
		return
	}
	// 通知：session/update 文本投递
	if text := extractSessionUpdateText(msg); text != "" {
		select {
		case c.updatesCh <- text:
		default:
			logger.Debug(logComponent).Str("text", text).
				Msg("[AcpStdioClient] updates channel full, dropping session/update")
		}
	}
}

// deliverPending 向等待者非阻塞投递结果。
func (c *AcpStdioClient) deliverPending(idAny any, res *rpcResult) {
	rid, ok := toInt(idAny)
	if !ok {
		return
	}
	c.pendingMu.Lock()
	ch := c.pending[int64(rid)]
	delete(c.pending, int64(rid))
	c.pendingMu.Unlock()
	if ch != nil {
		select {
		case ch <- res:
		default:
		}
	}
}

// Python: send_jsonrpc_request (L136-237)
func (m *AcpOutputManager) SendJsonrpcRequest(ctx context.Context, method string,
	params map[string]any, opts ...AcpRequestOption) (map[string]any, error) {
	cfg := acpRequestConfig{
		ChannelID: "acp",
		Timeout:   acpRequestTimeoutSeconds,
	}
	for _, o := range opts {
		o(&cfg)
	}
	logger.Info(logComponent).Str("method", method).Strs("params_keys", mapKeys(params)).
		Msg("[AcpOutput] send_jsonrpc_request called")

	m.mu.Lock()
	if m.sendPushFunc == nil {
		m.mu.Unlock()
		logger.Error(logComponent).Msg("[AcpOutput] send_push callback is None!")
		return nil, fmt.Errorf("ACP output send_push callback not set")
	}
	pushFn := m.sendPushFunc
	m.jsonrpcCounter++
	jsonrpcID := fmt.Sprintf("%d", m.jsonrpcCounter)
	requestID := fmt.Sprintf("acp_out_%s", randomHex12())
	done := make(chan AcpOutputResult, 1)
	acpReq := &AcpOutputRequest{
		JsonrpcID: jsonrpcID,
		Method:    method,
		Params:    params,
		Done:      done,
		RequestID: requestID,
	}
	m.pending[jsonrpcID] = acpReq
	m.mu.Unlock()
	logger.Info(logComponent).Str("jsonrpc_id", jsonrpcID).
		Strs("pending_keys", m.pendingKeys()).
		Msg("[AcpOutput] added to pending")

	// E2AResponse 构造（Python L175-210）
	bodyParams := params
	if cfg.SessionID != "" {
		merged := make(map[string]any, len(params)+1)
		for k, v := range params {
			merged[k] = v
		}
		merged["sessionId"] = cfg.SessionID
		bodyParams = merged
	}
	resp := e2a.NewE2AResponse()
	resp.ResponseID = fmt.Sprintf("acp_out_resp_%s", randomHex12())
	resp.RequestID = requestID
	resp.Sequence = 0
	resp.IsFinal = false
	resp.ResponseKind = e2a.E2AResponseKindACPOutputRequest
	resp.Timestamp = e2a.UTCNowISO()
	resp.Provenance = e2a.E2AProvenance{
		SourceProtocol: e2a.E2ASourceProtocolE2A,
		Converter: "jiuwenswarm.agents.harness.common.tools.acp_output_tools:send_jsonrpc_request",
		ConvertedAt:   e2a.UTCNowISO(),
		Details:       map[string]any{"kind": "acp_output_request", "acp_method": method},
	}
	resp.Body = map[string]any{
		"jsonrpc": "2.0",
		"id":      jsonrpcID,
		"method":  method,
		"params":  bodyParams,
	}
	resp.JSONRPCID = jsonrpcID
	resp.SessionID = cfg.SessionID
	resp.Channel = cfg.ChannelID
	resp.Metadata = map[string]any{e2a.E2AWireServerPushKey: true}

	pushMsg := resp.ToMap()
	// Python L207-210: to_dict 后处理双 key 兼容
	if _, exists := pushMsg["channel_id"]; !exists {
		if ch, ok := pushMsg["channel"]; ok {
			pushMsg["channel_id"] = ch
		}
	}
	if _, exists := pushMsg["payload"]; !exists {
		if body, ok := pushMsg["body"]; ok {
			pushMsg["payload"] = body
		}
	}

	if err := pushFn(ctx, pushMsg); err != nil {
		m.mu.Lock()
		delete(m.pending, jsonrpcID)
		m.mu.Unlock()
		return nil, fmt.Errorf("failed to send ACP output request: %w", err)
	}
	logger.Info(logComponent).Str("jsonrpc_id", jsonrpcID).Str("method", method).
		Str("request_id", requestID).Msg("[AcpOutput] sent E2A request")

	select {
	case res := <-done:
		if res.Err != nil {
			return nil, res.Err
		}
		return res.Response, nil
	case <-ctx.Done():
		m.mu.Lock()
		delete(m.pending, jsonrpcID)
		m.mu.Unlock()
		return nil, ctx.Err()
	case <-time.After(cfg.Timeout):
		m.mu.Lock()
		delete(m.pending, jsonrpcID)
		m.mu.Unlock()
		logger.Warn(logComponent).Str("jsonrpc_id", jsonrpcID).Str("method", method).
			Float64("timeout", cfg.Timeout.Seconds()).
			Msg("[AcpOutput] request timed out")
		return nil, fmt.Errorf("acp output request timed out after %.1fs: %s", cfg.Timeout.Seconds(), method)
	}
}

func waitForTerminalExit(ctx context.Context, mgr *AcpOutputManager,

	resp, err := sendAcpToolRequest(ctx, mgr, "terminal/wait_for_exit",

				"exitCode": nil, "signal": nil,
				"timedOut": true, "running": true, "shouldRetry": true,
			}, nil
		}
		return nil, err
	}
	result := copyMapAny(resp)
	if _, exists := result["timedOut"]; !exists {
		result["timedOut"] = false
	}
	if _, exists := result["running"]; !exists {
		if result["exitCode"] != nil || result["signal"] != nil {
			result["running"] = false
		}

	return result, nil
}

// normalizeJsonrpcID jsonrpc ID 归一化（string/number → trim 字符串）。
// Python: str(jsonrpc_id or "").strip() (L94-95)
func normalizeJsonrpcID(id any) string {
	switch v := id.(type) {
	case string:
		return strings.TrimSpace(v)
	case float64:
		return strconv.FormatFloat(v, 'f', -1, 64)
	case int:
		return strconv.Itoa(v)
	case int64:
		return strconv.FormatInt(v, 10)
	case json.Number:
		return strings.TrimSpace(v.String())
	default:
		return ""
	}
}

// acquireProfileLock 获取 profile 锁（Python _profile_lock L22-25）。
func acquireProfileLock(name string) *profileLock {
	locksMu.Lock()
	defer locksMu.Unlock()
	l := locks[name]
	if l == nil {
		l = &profileLock{}
		locks[name] = l
	}
	l.holders++
	return l
}

// releaseProfileLock 释放并清理空条目（Go 侧增强：防 map 膨胀）。
func releaseProfileLock(name string, l *profileLock) {
	locksMu.Lock()
	defer locksMu.Unlock()
	l.holders--
	if l.holders <= 0 {
		delete(locks, name)
	}
	l.mu.Unlock()
}

// loadProfile 读取 acp_agents 下 profile 规格（Python _load_profile L28-38）。
func loadProfile(configBase map[string]any, agentName string) (map[string]any, error) {
	if configBase == nil {
		return nil, fmt.Errorf("config has no acp_agents mapping")
	}
	agents, ok := configBase["acp_agents"].(map[string]any)
	if !ok {
		return nil, fmt.Errorf("config has no acp_agents mapping")
	}
	spec, ok := agents[strings.TrimSpace(agentName)].(map[string]any)
	if !ok {
		return nil, fmt.Errorf(
			"unknown acp_agents profile %q; define it under acp_agents in config.yaml", agentName)
	}
	return spec, nil
}

// buildClientSpec 解析 spawn 规格（Python _get_or_create_client L62-87 的解析段）。
func buildClientSpec(spec map[string]any) (*clientSpec, error) {
	command := ""
	if c, ok := spec["command"].(string); ok {
		command = strings.TrimSpace(c)
	}
	if command == "" {
		return nil, fmt.Errorf("acp_agents profile needs non-empty command")
	}
	var args []string
	switch raw := spec["args"].(type) {
	case nil:
		args = []string{}
	case []any:
		for _, item := range raw {
			args = append(args, fmt.Sprintf("%v", item))
		}
	default:
		return nil, fmt.Errorf("acp_agents.args must be a list of strings")
	}
	cwd := ""
	if c, ok := spec["cwd"].(string); ok {
		trimmed := strings.TrimSpace(c)
		if trimmed != "" {
			cwd = trimmed
		}
	}
	if spec["env"] != nil {
		if _, ok := spec["env"].(map[string]any); !ok {
			return nil, fmt.Errorf("acp_agents.env must be a map of strings")
		}
	}
	env, _ := spec["env"].(map[string]any)
	return &clientSpec{command: command, args: args, cwd: cwd, env: env}, nil
}

// closeProfileUnlocked 关闭并移除 profile client（Python _close_profile_unlocked L41-47）。
func closeProfileUnlocked(profile string) {
	clientsMu.Lock()
	client := clients[profile]
	delete(clients, profile)
	clientsMu.Unlock()
	if client != nil {
		if err := client.Close(); err != nil {
			logger.Warn(logComponent).Str("profile", profile).Err(err).
				Msg("[acp_chat] close profile failed")
		}
	}
}

// getOrCreateClient 取/建 profile client（Python _get_or_create_client L50-91）。
func getOrCreateClient(ctx context.Context, configBase map[string]any,
	profile string, newSession bool) (*swarmacp.AcpStdioClient, error) {
	spec, err := loadProfile(configBase, profile)
	if err != nil {
		return nil, err
	}
	if newSession {
		closeProfileUnlocked(profile)
	}
	clientsMu.Lock()
	existing := clients[profile]
	clientsMu.Unlock()
	if existing != nil {
		if existing.IsConnected() {
			return existing, nil
		}
		closeProfileUnlocked(profile)
	}
	cs, err := buildClientSpec(spec)
	if err != nil {
		return nil, err
	}
	client := swarmacp.NewAcpStdioClient(cs.command, cs.args, cs.cwd, cs.env)
	if err := client.Connect(ctx); err != nil {
		return nil, err
	}
	clientsMu.Lock()
	clients[profile] = client
	clientsMu.Unlock()
	return client, nil
}

	clientsMu.Lock()
	defer clientsMu.Unlock()

		_ = client.Close()
		delete(clients, profile)
	}
}
