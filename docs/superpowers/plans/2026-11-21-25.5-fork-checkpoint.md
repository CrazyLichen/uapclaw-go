// Python: _named_checkpoints 值形态 {count, description, created_by}。
type CheckpointRecord struct {

// ReadTeamCheckpoints 读取 per-team 命名空间中的 checkpoint 索引；缺失/形态不符返回不存在。
// Python: read_team_checkpoints (metadata.py L129-157)
func ReadTeamCheckpoints(sess interfaces.SessionFacade, teamName string) (map[string]CheckpointRecord, bool) {
	bucket := ReadTeamNamespace(sess, teamName)
	if bucket == nil {
		return nil, false
	}
	raw, ok := bucket[TeamCheckpointsKey].(map[string]any)
	if !ok {
		return nil, false
	}
	out := make(map[string]CheckpointRecord, len(raw))
	for name, v := range raw {
		rec, err := decodeCheckpointRecord(v)
		if err != nil {
			return nil, false
		}
		out[name] = rec
	}
	return out, true
}

// MergeTeamCheckpoints 记录/覆盖 checkpoint 索引。
// Python: merge_team_checkpoints (metadata.py L158-160)
func MergeTeamCheckpoints(sess interfaces.SessionFacade, teamName string, mapping map[string]CheckpointRecord) {
	values := make(map[string]any, len(mapping))
	for name, rec := range mapping {
		values[name] = rec
	}
	MergeTeamNamespace(sess, teamName, map[string]any{TeamCheckpointsKey: values})
}
```

// ForkContext 可序列化的 fork 上下文快照（跨进程 json-native）。
// Python: ForkContext dataclass (fork.py:32-46)
type ForkContext struct {
	// Messages 编码后的消息列表（json 原生 dict 形态）
	Messages []map[string]any `json:"messages"`
	// CompactSplit 非 nil 时 compact_context 压缩 compact_direction 的对侧
	CompactSplit *int `json:"compact_split,omitempty"`
	// CompactDirection "before"（默认，压尾部）| "after"（压头部）
	CompactDirection string `json:"compact_direction,omitempty"`
}

// ForkContextFromMessages 从消息列表抓快照：剥 SystemMessage → checkpoint 截断。
// Python: ForkContext.from_agent (fork.py:52-117) 的截断半区；
// agent 读取半区（GetCtxMessages）见 Task 6 resolveForkNative。
// before：checkpoint 落在工具调用 assistant 与其 ToolMessage 之间（下标=当时消息数），
//   闭合 ToolMessage 块跨边界携带，避免注入上下文出现悬挂 tool call。
// after：窗口起点的孤儿 ToolMessage 剥除（其 assistant 未被继承）。
// Python: _trim_leading_orphan_tool_messages (fork.py:119-131)
func ForkContextFromMessages(msgs []llmschema.BaseMessage, checkpoint *int, keep string) *ForkContext {
	filtered := make([]llmschema.BaseMessage, 0, len(msgs))
	for _, m := range msgs {
		if _, isSystem := m.(*llmschema.SystemMessage); isSystem {
			continue // Python: [m for m in msgs if not isinstance(m, SystemMessage)]
		}
		filtered = append(filtered, m)
	}
	if checkpoint != nil && *checkpoint >= 0 && *checkpoint < len(filtered) {
		if keep == "after" {
			filtered = trimLeadingOrphanToolMessages(filtered[*checkpoint:])
		} else {
			truncated := append([]llmschema.BaseMessage(nil), filtered[:*checkpoint]...)
			// before：末元素是带 tool_calls 的 assistant 且下一条是 ToolMessage →
			// 连续 ToolMessage 块整体跨过边界。Python: fork.py:95-105
			if last := lastAssistantWithToolCalls(truncated); last != nil && isToolMessage(filtered[*checkpoint]) {
				for i := *checkpoint; i < len(filtered) && isToolMessage(filtered[i]); i++ {
					truncated = append(truncated, filtered[i])
				}
			}
			filtered = truncated
		}
	}
	return &ForkContext{Messages: encodeMessages(filtered), CompactDirection: "before"}
}
```

外加：`TrimLeadingOrphanToolMessages`（导出供测试）、`encodeMessages`（走消息 schema 的 json tag 直 marshal）、`ToForkMessages`（decode 回 BaseMessage 列表，Python `to_messages`）、`IsEmpty`、`NormalizeMessages`（Python `normalize_messages` fork.py:150-175：BaseMessage 直收 / dict decode / 其他→nil）。
