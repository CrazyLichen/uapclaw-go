# 11.9 GatewayServer 多路由 WS 服务器 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 补齐 Python `app_gateway.py` GatewayServer 类本体（L240-720）——多路由 WebSocket 服务器监听 19001，含 RouteConfig 路由解析、双客户端路由表、res/event 帧协议、InboundGatewayServer 双层 chan 入站链、ExternalChannel 出站装配。

```go
package gateway

import (
	"strings"

	"github.com/gorilla/websocket"
	"github.com/uapclaw/uapclaw-go/internal/swarm/gateway/routing"
)

// ──────────────────────────── 结构体 ────────────────────────────

// RouteConfig 单条路由的配置（/acp、/tui 等）。
//
// Python: app_gateway.py RouteConfig dataclass（L208-219）。
type RouteConfig struct {
	// Path 路由路径（如 "/acp"）
	Path string
	// ChannelID 渠道标识（如 "acp"/"tui"）
	ChannelID string
	// ForwardMethods 转发到 MessageHandler 的 method 集合
	ForwardMethods map[string]bool
	// ForwardNoLocalHandlerMethods 仅转发、无本地 handler 的 method 集合
	ForwardNoLocalHandlerMethods map[string]bool
	// LocalHandlers 本地 handler（method → handler）
	LocalHandlers map[string]LocalHandlerFunc
	// InboundInterceptor 入站拦截器（可 nil）
	InboundInterceptor GatewayInterceptorFunc
	// OutboundInterceptor 出站拦截器（可 nil）
	OutboundInterceptor GatewayInterceptorFunc
	// CleanupHandler 断连清理钩子（可 nil，DisconnectHandler 缺省时的兜底）
	CleanupHandler CleanupFunc
	// DisconnectHandler 断连回调（可 nil，带 staleSessionKeys）
	DisconnectHandler DisconnectFunc
}

// GatewayServerConfig GatewayServer 配置。
//
// Python: app_gateway.py GatewayServerConfig dataclass（L223-238）。
type GatewayServerConfig struct {
	// Enabled 是否启用（默认 true）
	Enabled bool
	// Host 监听地址（默认 127.0.0.1）
	Host string
	// Port 监听端口（默认 19001）
	Port int
	// Routes 路由表（path → RouteConfig）
	Routes map[string]*RouteConfig
	// Path 单路由便捷字段（兜底构造用）
	Path string
	// ChannelID 单路由便捷字段（兜底构造用）
	ChannelID string
}

// ──────────────────────────── 枚举 ────────────────────────────

// ──────────────────────────── 常量 ────────────────────────────

// ──────────────────────────── 全局变量 ────────────────────────────

// ──────────────────────────── 导出函数 ────────────────────────────

// NewGatewayServerConfig 创建默认配置。
//
// 默认值对齐 Python GatewayServerConfig：enabled=true, host="127.0.0.1", port=19001。
func NewGatewayServerConfig() *GatewayServerConfig {
	return &GatewayServerConfig{
		Enabled: true,
		Host:    "127.0.0.1",
		Port:    19001,
		Routes:  make(map[string]*RouteConfig),
	}
}

// BuildACPRouteBinding 从转发集合构建 ACP 路由绑定。
//
// Python: _build_acp_route_binding（L723-736）。
// 注意：Python 的 on_message_cb 参数未存入 binding（死参数），Go 侧不保留。
// forwardMethods/forwardNoLocal 由调用方直接放进 RouteConfig（见 BuildRouteConfigMap 注释），
// 本函数只装 binding 本体字段。
// A1 空路由形态下暂无调用方，11.25 接线时使用。
func BuildACPRouteBinding(path, channelID string) *routing.RouteBinding {
	return &routing.RouteBinding{
		Path:      path,
		ChannelID: channelID,
	}
}

// BuildRouteConfigMap 将 RouteBinding 列表转为 path 索引的 RouteConfig map。
//
// Python: _build_route_config_map（L739-752），8 字段透传。
// Go 侧两处类型差异（对齐既有骨架决策）：
//  1. ForwardMethods/ForwardNoLocalHandlerMethods：routing.RouteBinding 已删除这两个字段
//     （运行时路由用 web 包集合变量），调用方（11.25/11.15）在 binding 之外
//     直接对返回的 RouteConfig 赋值
//  2. InboundInterceptor/OutboundInterceptor：routing.InterceptorFunc 是 byte 层签名
//     func(channel any, data []byte) ([]byte, error)，本包 GatewayInterceptorFunc 是
//     帧层签名 func(ws *websocket.Conn, data map[string]any) bool——转换逻辑随 ACP 桥
//     实现（⤵️ 11.25），本函数置 nil
//  3. DisconnectHandler：routing.DisconnectFunc 签名 func(channel any, sessionKeys []string)
//     与本包 DisconnectFunc func(ws *websocket.Conn, staleSessionKeys [][2]string) 不同，
//     同样随桥/路由装配时转换（⤵️ 11.25/11.15），本函数置 nil
func BuildRouteConfigMap(bindings []*routing.RouteBinding) map[string]*RouteConfig {
	m := make(map[string]*RouteConfig)
	for _, b := range bindings {
		m[b.Path] = &RouteConfig{
			Path:           b.Path,
			ChannelID:      b.ChannelID,
			LocalHandlers:  make(map[string]LocalHandlerFunc),
			// InboundInterceptor/OutboundInterceptor/DisconnectHandler/CleanupHandler
			// ⤵️ 11.25/11.15：byte 层→帧层转换随 ACP 桥/TUI 装配实现
		}
	}
	return m
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// LocalHandlerFunc 本地 handler（对齐 Python Callable[..., Awaitable[None]]，
// Python 签名 (ws, req_id, params, session_id)）。
type LocalHandlerFunc func(ws *websocket.Conn, reqID string, params map[string]any, sessionID string) error

// GatewayInterceptorFunc 入/出站拦截器（对齐 Python Awaitable[bool]，返回 true 短路）。
type GatewayInterceptorFunc func(ws *websocket.Conn, data map[string]any) bool

// CleanupFunc 断连清理（对齐 cleanup_handler，无 disconnect_handler 时的兜底）。
type CleanupFunc func(ws *websocket.Conn)

// DisconnectFunc 断连回调（对齐 disconnect_handler，带 staleSessionKeys）。
type DisconnectFunc func(ws *websocket.Conn, staleSessionKeys [][2]string)

// applyConfigFallback 兜底构造单路由（对齐 Python __post_init__）。
//
// 无 routes 且 Path+ChannelID 齐全时自动造一条 RouteConfig 塞入 Routes。
func applyConfigFallback(cfg *GatewayServerConfig) {
	if len(cfg.Routes) > 0 {
		return
	}
	path := strings.TrimSpace(cfg.Path)
	channelID := strings.TrimSpace(cfg.ChannelID)
	if path != "" && channelID != "" {
		cfg.Routes[path] = &RouteConfig{
			Path:                          path,
			ChannelID:                     channelID,
			ForwardMethods:                make(map[string]bool),
			ForwardNoLocalHandlerMethods:  make(map[string]bool),
			LocalHandlers:                 make(map[string]LocalHandlerFunc),
		}
	}
}
```

**⚠️ 类型定义位置注意**：`LocalHandlerFunc`/`GatewayInterceptorFunc`/`CleanupFunc`/`DisconnectFunc` 四个类型是**导出类型**，按 Go 规范应放"结构体区块"（类型定义归结构体区，接口/结构体/函数类型声明统一在结构体区）。写码时把这四个 type 声明移到文件头部"结构体"分隔注释下（RouteConfig 之前），非导出函数区只留 `applyConfigFallback`。上面代码块为阅读完整性保留了顺序示意，落盘时按规范调整。

```go
// RegisterExternalChannel 登记一个已由外部完成入站装配的渠道实例。
//
// 不做存活检查、不设 onMessage，仅入 externalChannels 表供出站派发。
// Python: ChannelManager.register_external_channel()（channel_manager.py L86-88）。
func (cm *ChannelManager) RegisterExternalChannel(channelID string, ch ExternalChannel) {
	cm.mu.Lock()
	defer cm.mu.Unlock()

	if _, exists := cm.externalChannels[channelID]; exists {
		logger.Warn(logComponent).
			Str("channel_id", channelID).
			Msg("external channel re-registered, overwriting")
	}
```

```go
import (
	"context"
```

```go
// ──────────────────────────── 结构体 ────────────────────────────

// GatewayServer 通用多路由 WebSocket Gateway Server。
//
// 对齐 Python app_gateway.py GatewayServer 类（L240-720）：
// 支持多个路径（如 /acp、/tui），每条路径有独立 channel_id 和本地 handler；
// 本地 handler 优先处理请求，未处理或无匹配则 forward 到 MessageHandler。
//
// acpBridge ⤵️ 11.25：AcpGatewayBridge 注入点（构造注入 dispatchOnMessage/
// bindAcpSessionClient + _install_default_route_hooks + handleRawMessage/send
// 的 acp 分支），A1 空路由形态下不建模。
type GatewayServer struct {
	// config 服务器配置（路由表/监听地址）
	config *GatewayServerConfig
	// bus 对齐 Python router/_DummyBus 占位（Go 暂未消费，保留字段）
	bus any
	// httpServer 独立 HTTP 服务器（对齐 Python websockets.serve）
	httpServer *http.Server
	// upgrader gorilla WebSocket 升级器
	upgrader websocket.Upgrader
	// running 运行状态
	running atomic.Bool
	// onMessageCb 入站消息回调（对齐 _on_message_cb）
	onMessageCb func(ctx context.Context, msg *schema.Message) error
	// clientsMu 保护以下三张表
	clientsMu sync.RWMutex
	// clients 在线连接集合（对齐 _clients set）
	clients map[*websocket.Conn]struct{}
	// requestToClient 请求路由表：(channel, reqID) → ws
	requestToClient map[[2]string]*websocket.Conn
	// sessionToClient 会话路由表：(channel, sessionID) → ws
	sessionToClient map[[2]string]*websocket.Conn
	// wg 连接 handler 生命周期
	wg sync.WaitGroup
	// listenErr 启动错误回传 channel
	listenErr chan error
}

// gwReadLimit WebSocket 消息大小限制（8MB，对齐 Python ws_max_size 与 web_connect.go 先例）
const gwReadLimit = 8 * 1024 * 1024

// ──────────────────────────── 全局变量 ────────────────────────────

// ──────────────────────────── 导出函数 ────────────────────────────

// NewGatewayServer 创建多路由 WS 服务器实例。
//
// Python: GatewayServer(config, router)——router 在 Python 侧仅作占位传递
// （_DummyBus），Go 侧以 bus 字段 any 类型保留语义，不消费。
func NewGatewayServer(config *GatewayServerConfig) *GatewayServer {
	if config == nil {
		config = NewGatewayServerConfig()
	}
	applyConfigFallback(config)
	return &GatewayServer{
		config: config,
		bus:    struct{}{},
		upgrader: websocket.Upgrader{
			CheckOrigin: wsorigin.GorillaCheckOrigin(),
		},
		clients:         make(map[*websocket.Conn]struct{}),
		requestToClient: make(map[[2]string]*websocket.Conn),
		sessionToClient: make(map[[2]string]*websocket.Conn),
		listenErr:       make(chan error, 1),
	}
}

// Start 启动 19001 监听。
//
// Python: GatewayServer.start()（L371-393）——已运行或未启用则直接返回。
func (s *GatewayServer) Start(ctx context.Context) error {
	if s.running.Load() || !s.config.Enabled {
		return nil
	}
	mux := http.NewServeMux()
	mux.HandleFunc("/", s.HandleConnection)
	s.httpServer = &http.Server{
		Addr:    fmt.Sprintf("%s:%d", s.config.Host, s.config.Port),
		Handler: mux,
	}
	go func() {
		if err := s.httpServer.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			select {
			case s.listenErr <- err:
			default:
			}
		}
	}()
	s.running.Store(true)

	paths := make([]string, 0, len(s.config.Routes))
	for p := range s.config.Routes {
		paths = append(paths, p)
	}
	logger.Info(logComponent).
		Str("addr", s.httpServer.Addr).
		Str("paths", strings.Join(paths, ", ")).
		Msg("Gateway server started")
	return nil
}

// Stop 停机：关连接 → 清表 → 关监听。
//
// Python: GatewayServer.stop()（L395-410）。
func (s *GatewayServer) Stop() error {
	s.running.Store(false)

	s.clientsMu.Lock()
	clients := make([]*websocket.Conn, 0, len(s.clients))
	for c := range s.clients {
		clients = append(clients, c)
		// 对齐 Python close(code=1001, reason="server shutdown")
		_ = c.WriteControl(websocket.CloseMessage,
			websocket.FormatCloseMessage(websocket.CloseGoingAway, "server shutdown"),
			time.Now().Add(time.Second))
	}
	s.clients = make(map[*websocket.Conn]struct{})
	s.requestToClient = make(map[[2]string]*websocket.Conn)
	s.sessionToClient = make(map[[2]string]*websocket.Conn)
	s.clientsMu.Unlock()

	for _, c := range clients {
		_ = c.Close()
	}
	if s.httpServer != nil {
		if err := s.httpServer.Close(); err != nil {
			logger.Error(logComponent).Err(err).Msg("failed to close gateway http server")
		}
	}
	s.wg.Wait()
	logger.Info(logComponent).Msg("Gateway server stopped")
	return nil
}

func (s *GatewayServer) OnMessage(cb func(ctx context.Context, msg *schema.Message) error) {

// HandleConnection HTTP 处理入口：升级 WS + 路由分流 + 读循环。
//
// Python: GatewayServer._connection_handler()（L501-559）。
func (s *GatewayServer) HandleConnection(w http.ResponseWriter, r *http.Request) {
	route, matchedPath := s.resolveRoute(r.URL.Path)
	if route == nil {
		// 未注册路径：HTTP 层直接拒绝（对齐 Python close(1008, unsupported path)）
		http.Error(w, fmt.Sprintf("unsupported path: %s", r.URL.Path), http.StatusNotFound)
		return
	}

	conn, err := s.upgrader.Upgrade(w, r, nil)
	if err != nil {
		logger.Error(logComponent).Err(err).Msg("WebSocket upgrade failed")
		return
	}
	conn.SetReadLimit(gwReadLimit)

	s.clientsMu.Lock()
	s.clients[conn] = struct{}{}
	s.clientsMu.Unlock()

	// connection.ack 首帧（对齐 L513-525）
	ack := buildConnectionAckFrame(route.ChannelID)
	if err := conn.WriteMessage(websocket.TextMessage, ack); err != nil {
		logger.Debug(logComponent).Err(err).Msg("failed to send connection.ack, discarding client")
		s.clientsMu.Lock()
		delete(s.clients, conn)
		s.clientsMu.Unlock()
		return
	}

	s.wg.Add(1)
	defer s.wg.Done()
	defer func() {
		if r := recover(); r != nil {
			logger.Error(logComponent).Any("panic", r).Msg("connection handler panicked")
		}
	}()

	// 读循环（对齐 L527-530）
	for {
		_, raw, err := conn.ReadMessage()
		if err != nil {
			// 对齐 ConnectionClosedError → logger.info
			logger.Info(logComponent).
				Str("channel", route.ChannelID).
				Msg("WebSocket connection closed")
			break
		}
		s.handleRawMessage(context.Background(), conn, raw, matchedPath, route)
	}

	// finally 语义（对齐 L532-559）：清集合 + 清 stale 路由键 + 断连回调
	s.clientsMu.Lock()
	delete(s.clients, conn)
	var staleSessionKeys [][2]string
	for key, c := range s.sessionToClient {
		if c == conn {
			staleSessionKeys = append(staleSessionKeys, key)
			delete(s.sessionToClient, key)
		}
	}
	for key, c := range s.requestToClient {
		if c == conn {
			delete(s.requestToClient, key)
		}
	}
	s.clientsMu.Unlock()

	if route.DisconnectHandler != nil {
		func() {
			defer func() {
				if r := recover(); r != nil {
					logger.Warn(logComponent).Any("panic", r).Msg("GatewayServer disconnect handler failed")
				}
			}()
			route.DisconnectHandler(conn, staleSessionKeys)
		}()
	} else if route.CleanupHandler != nil {
		func() {
			defer func() {
				if r := recover(); r != nil {
					logger.Warn(logComponent).Any("panic", r).Msg("GatewayServer cleanup handler failed")
				}
			}()
			route.CleanupHandler(conn)
		}()
	}
}

// Send 出站消息派发（实现 ExternalChannel 接口）。
//
// Python: GatewayServer.send()（L412-499）——四段路由：
// request 定向 → session 定向 → 广播 fallback → 帧发送。
// acp 分支 ⤵️ 11.25。
func (s *GatewayServer) Send(ctx context.Context, msg *schema.Message) error {
	var ws *websocket.Conn
	requestKey, ok := clientRouteKey(msg.ChannelID, msg.ID)
	if ok {
		s.clientsMu.RLock()
		ws = s.requestToClient[requestKey]
		s.clientsMu.RUnlock()
	}
	if ws == nil {
		sessionKey, ok := clientRouteKey(msg.ChannelID, msg.SessionID)
		if ok {
			s.clientsMu.RLock()
			ws = s.sessionToClient[sessionKey]
			s.clientsMu.RUnlock()
		}
	}

	if ws == nil {
		// 广播 fallback（对齐 L423-439）
		if msg.ChannelID != "" && msg.ChannelID != "acp" && msg.Type != schema.MessageTypeRes {
			clients := s.findChannelClients(msg.ChannelID)
			if len(clients) > 0 {
				frame := buildEventFrame(msg)
				data, _ := json.Marshal(frame)
				logger.Info(logComponent).
					Str("channel_id", msg.ChannelID).
					Int("clients", len(clients)).
					Str("id", msg.ID).
					Msg("broadcast fallback")
				s.clientsMu.RLock()
				for _, c := range clients {
					if err := c.WriteMessage(websocket.TextMessage, data); err != nil {
						logger.Debug(logComponent).Err(err).Msg("broadcast send failed (client disconnected?)")
					}
				}
				s.clientsMu.RUnlock()
				return nil
			}
		}
		logger.Warn(logComponent).
			Str("channel_id", msg.ChannelID).
			Str("session_id", msg.SessionID).
			Str("id", msg.ID).
			Msg("message dropped: no WebSocket client found")
		return nil
	}

	// 出站拦截器（对齐 L443-467：只处理第一条 channel_id 匹配的路由）
	s.clientsMu.RLock()
	for _, route := range s.config.Routes {
		if route.ChannelID == msg.ChannelID && route.OutboundInterceptor != nil {
			handled := route.OutboundInterceptor(ws, map[string]any{"raw": msg})
			s.clientsMu.RUnlock()
			if handled {
				return nil
			}
			break
		}
	}
	// ⚠️ 上面 RLock 循环内 return/break 的解锁路径需在实现时仔细处理
	//（建议改为：先在锁内拷贝 interceptor 指针，锁外调用）
	s.clientsMu.RUnlock()

	frame := buildOutboundFrame(msg)
	data, _ := json.Marshal(frame)
	if err := ws.WriteMessage(websocket.TextMessage, data); err != nil {
		logger.Debug(logComponent).Err(err).Msg("gateway send failed (client disconnected?)")
	}
	return nil
}

// resolveRoute 按路径解析路由，支持尾斜杠变体。
//
// Python: _resolve_route（L304-316）四分支。
func (s *GatewayServer) resolveRoute(requestPath string) (*RouteConfig, string) {
	p := strings.TrimSpace(requestPath)
	if p == "" {
		return nil, requestPath
	}
	if rc, ok := s.config.Routes[p]; ok {
		return rc, p
	}
	trimmed := strings.TrimSuffix(p, "/")
	if p != "/" && trimmed != p {
		if rc, ok := s.config.Routes[trimmed]; ok {
			return rc, trimmed
		}
	}
	if !strings.HasSuffix(p, "/") {
		withSlash := p + "/"
		if rc, ok := s.config.Routes[withSlash]; ok {
			return rc, withSlash
		}
	}
	return nil, p
}

// clientRouteKey 构造 (channel, scopedID) 路由键。
//
// Python: _client_route_key（L265-271）——任一空值返回不可用。
func clientRouteKey(channelID, scopedID string) ([2]string, bool) {
	channel := strings.TrimSpace(channelID)
	scope := strings.TrimSpace(scopedID)
	if channel == "" || scope == "" {
		return [2]string{}, false
	}
	return [2]string{channel, scope}, true
}

// findChannelClients 找到指定渠道的全部在线客户端。
//
// Python: _find_channel_clients（L273-278）。
// ⚠️ 有意偏差（spec §7-3）：不判 closed，直接返回，发送失败由调用方忽略。
func (s *GatewayServer) findChannelClients(channelID string) []*websocket.Conn {
	s.clientsMu.RLock()
	defer s.clientsMu.RUnlock()
	var out []*websocket.Conn
	for key, c := range s.sessionToClient {
		if key[0] == channelID {
			out = append(out, c)
		}
	}
	return out
}

// handleRawMessage 单帧入站处理。
//
// Python: _handle_raw_message（L561-720）三层处理。
// acp 分支 ⤵️ 11.25（L573-579 的 acp_bridge.handle_jsonrpc_request/is_jsonrpc_request）。
func (s *GatewayServer) handleRawMessage(ctx context.Context, ws *websocket.Conn, raw []byte, requestPath string, route *RouteConfig) {
	var data map[string]any
	if err := json.Unmarshal(raw, &data); err != nil {
		s.sendResponse(ws, "", false, nil, "invalid json", "")
		return
	}

	// 入站拦截器（对齐 L582-594）
	if route.InboundInterceptor != nil {
		handled := func() (h bool) {
			defer func() {
				if r := recover(); r != nil {
					logger.Warn(logComponent).Any("panic", r).Str("path", requestPath).Msg("GatewayServer inbound interceptor failed")
					h = false
				}
			}()
			return route.InboundInterceptor(ws, data)
		}()
		if handled {
			return
		}
	}

	if data["type"] != "req" {
		s.sendResponse(ws, "", false, nil, "invalid request", "")
		return
	}
	reqID := strings.TrimSpace(strVal(data["id"]))
	method := strings.TrimSpace(strVal(data["method"]))
	if reqID == "" || method == "" {
		s.sendResponse(ws, reqID, false, nil, "invalid request", "")
		return
	}
	params, _ := data["params"].(map[string]any)
	if params == nil {
		params = map[string]any{}
	}
	sessionID := strings.TrimSpace(strVal(params["session_id"]))
	if sessionID == "" {
		sessionID = reqID // 对齐 L619 回退
	}

	// 1. forward 集合（对齐 L621-671）
	if route.ForwardMethods[method] {
		reqMethod, err := schema.ParseReqMethod(method)
		if err != nil {
			s.sendResponse(ws, reqID, false, nil, fmt.Sprintf("unknown method: %s", method), "")
			return
		}

		// 注册路由键（对齐 L634-640）
		s.clientsMu.Lock()
		if key, ok := clientRouteKey(route.ChannelID, reqID); ok {
			s.requestToClient[key] = ws
		}
		if key, ok := clientRouteKey(route.ChannelID, sessionID); ok {
			s.sessionToClient[key] = ws
		}
		s.clientsMu.Unlock()

		// default_mode：tui → code.normal，其余 → agent.plan（对齐 L645）
		defaultMode := schema.ModeAgentPlan
		if route.ChannelID == "tui" {
			defaultMode = schema.ModeCodeNormal
		}
		mode := schema.ParseMode(strVal(params["mode"]), defaultMode)

		// params 拷贝 + setdefault("mode")（对齐 L649-651）
		paramsCopy := make(map[string]any, len(params))
		for k, v := range params {
			paramsCopy[k] = v
		}
		if _, has := paramsCopy["mode"]; !has {
			paramsCopy["mode"] = mode.String()
		}

		// metadata：method + 可选 cwd（对齐 L653-657）
		metadata := map[string]any{"method": method}
		if cwd, ok := params["cwd"].(string); ok && strings.TrimSpace(cwd) != "" {
			metadata["cwd"] = strings.TrimSpace(cwd)
		}

		isStream, _ := data["is_stream"].(bool)
		paramsJSON, _ := json.Marshal(paramsCopy)

		msg := &schema.Message{
			ID:        reqID,
			Type:      schema.MessageTypeReq,
			ChannelID: route.ChannelID,
			SessionID: sessionID,
			Params:    paramsJSON,
			Timestamp: schema.NowTimestamp(),
			OK:        true,
			ReqMethod: reqMethod,
			Mode:      mode,
			Metadata:  metadata,
			IsStream:  isStream,
		}

		if s.onMessageCb != nil {
			if err := s.onMessageCb(ctx, msg); err != nil {
				logger.Warn(logComponent).Err(err).Str("msg_id", msg.ID).Msg("gateway on_message callback failed")
			}
		}

		// acp early-return ⤵️ 11.25（对齐 L665-668）

		// 仅转发集合 → 无本地 ack（对齐 L670-671）
		if route.ForwardNoLocalHandlerMethods[method] {
			return
		}
	}

	// 2. 本地 handler（对齐 L674-707）
	if handler, ok := route.LocalHandlers[method]; ok {
		if err := handler(ws, reqID, params, sessionID); err != nil {
			// ws 已关判定（写探针对齐 getattr(ws,"closed")）
			if !s.probeWritable(ws) {
				logger.Warn(logComponent).
					Str("method", method).Err(err).
					Msg("GatewayServer local handler aborted on closed ws")
				return
			}
			logger.Error(logComponent).
				Str("method", method).Err(err).
				Msg("GatewayServer local handler error")
			s.sendResponse(ws, reqID, false, nil, fmt.Sprintf("handler error: %s", err), "INTERNAL_ERROR")
		}
		return
	}

	// 3. 无 forward 无本地（对齐 L710-717）
	s.sendResponse(ws, reqID, false, nil, fmt.Sprintf("unknown method: %s", method), "")
}

// probeWritable 写探针：判断 ws 是否仍可写（对齐 Python getattr(ws,"closed")）。
func (s *GatewayServer) probeWritable(ws *websocket.Conn) bool {
	_ = ws.SetWriteDeadline(time.Now().Add(50 * time.Millisecond))
	defer func() { _ = ws.SetWriteDeadline(time.Time{}) }()
	return ws.WriteMessage(websocket.PingMessage, nil) == nil
}

// buildEventFrame 构建出站 event 帧（广播 fallback 与定向共用）。
//
// Python: send() 尾段（L478-499）+ _build_event_frame。
func buildEventFrame(msg *schema.Message) map[string]any {
	eventName := "chat.final"
	if msg.EventType != "" {
		eventName = msg.EventType.String()
	}
	var payload map[string]any
	if msg.Payload != nil {
		payload = make(map[string]any, len(msg.Payload))
		for k, v := range msg.Payload {
			payload[k] = v
		}
		if _, has := payload["session_id"]; !has {
			payload["session_id"] = msg.SessionID
		}
	} else {
		payload = map[string]any{"session_id": msg.SessionID}
	}
	return map[string]any{
		"type":    "event",
		"event":   eventName,
		"payload": payload,
	}
}

// buildOutboundFrame 构建定向出站帧（res 或 event）。
//
// Python: send() L468-499 尾段。
func buildOutboundFrame(msg *schema.Message) map[string]any {
	if msg.Type == schema.MessageTypeRes {
		payload := map[string]any{}
		if msg.Payload != nil {
			for k, v := range msg.Payload {
				payload[k] = v
			}
		}
		frame := map[string]any{
			"type":    "res",
			"id":      msg.ID,
			"ok":      msg.OK,
			"payload": payload,
		}
		if !msg.OK {
			errText, _ := payload["error"].(string)
			if errText == "" {
				errText = "request failed"
			}
			frame["error"] = errText
		}
		return frame
	}
	return buildEventFrame(msg)
}

	s, _ := v.(string)
	return s
```

```go
// ──────────────────────────── 结构体 ────────────────────────────

// InboundGatewayServer Gateway 内部入站服务：渠道消息经队列转发到 MessageHandler。
//
// 对齐 Python _InboundGatewayServer（app_gateway.py L132-173，
// asyncio.Queue + 独立 task 的双层队列结构）。
//
// 满队语义为有意偏差（spec §7-1）：Python asyncio.Queue 无界；
// Go 用容量 256 的 chan + 满时 Warn 丢弃，与下游
// MessageHandler.userMessages（256 有界 chan）链路容量一致。
type InboundGatewayServer struct {
	// inboundHandler serveLoop 消费的处理器
	inboundHandler func(ctx context.Context, msg *schema.Message)
	// queue 消息队列（容量 256）
	queue chan *schema.Message
	// taskDone serveLoop 退出信号（对齐 _task cancel/await）
	taskDone chan struct{}
	// running 运行状态
	running atomic.Bool
	// cancel serveLoop context 取消函数（对齐 task.cancel()）
	cancel context.CancelFunc
}

// ──────────────────────────── 常量 ────────────────────────────

// inboundQueueCapacity 入站队列容量（对齐 MessageHandler.userMessages 链路容量 256）
const inboundQueueCapacity = 256

// ──────────────────────────── 全局变量 ────────────────────────────

// ──────────────────────────── 导出函数 ────────────────────────────

// NewInboundGatewayServer 创建入站服务实例。
//
// Python: _InboundGatewayServer(inbound_handler)。
func NewInboundGatewayServer(inboundHandler func(ctx context.Context, msg *schema.Message)) *InboundGatewayServer {
	return &InboundGatewayServer{
		inboundHandler: inboundHandler,
		queue:          make(chan *schema.Message, inboundQueueCapacity),
		taskDone:       make(chan struct{}),
	}
}

// Start 启动 serveLoop（幂等：已启动则直接返回）。
//
// Python: _InboundGatewayServer.start()——if self._task is not None: return。
func (s *InboundGatewayServer) Start(ctx context.Context) error {
	if s.running.Load() {
		return nil
	}
	s.running.Store(true)
	serveCtx, cancel := context.WithCancel(ctx)
	s.cancel = cancel
	go s.serveLoop(serveCtx)
	return nil
}

// Stop 停止 serveLoop 并等待退出。
//
// Python: _InboundGatewayServer.stop()——cancel + await + CancelledError 吞掉。
func (s *InboundGatewayServer) Stop() {
	s.running.Store(false)
	if s.cancel != nil {
		s.cancel()
	}
	if s.taskDone != nil {
		<-s.taskDone
	}
}

// HandleMessage 入队一条消息。
//
// Python: handle_message——await queue.put(msg); return True。
// 满队时 Warn 丢弃（有意偏差，见类型注释）。返回值恒 nil，
// 签名匹配 GatewayServer.OnMessage 回调类型。
func (s *InboundGatewayServer) HandleMessage(ctx context.Context, msg *schema.Message) error {
	select {
	case s.queue <- msg:
	default:
		logger.Warn(logComponent).
			Str("msg_id", msg.ID).
			Msg("gateway inbound queue full, dropping message")
	}
	return nil
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// serveLoop 消费循环。
//
// Python: _serve_loop（L162-173）——每条消息独立异常隔离。
func (s *InboundGatewayServer) serveLoop(ctx context.Context) {
	defer close(s.taskDone)
	for s.running.Load() {
		select {
		case <-ctx.Done():
			return
		case msg := <-s.queue:
			func() {
				defer func() {
					if r := recover(); r != nil {
						logger.Error(logComponent).
							Any("panic", r).
							Msg("Gateway inbound handling failed")
					}
				}()
				s.inboundHandler(ctx, msg)
			}()
		}
	}
}
```
