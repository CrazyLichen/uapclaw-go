```go
// Package tools 提供 harness 工具层公共契约：ToolOutput 与字段渲染。
// Python: openjiuwen/harness/tools/base_tool.py + openjiuwen/core/foundation/tool/schema.ToolOutput
package tools

import (
	"encoding/json"
	"strings"
)

// ToolOutput 演进工具统一返回契约（Python ToolOutput(success, data, error) 三字段）。
// 裁决 2026-12-24 方案 A：独立 struct；D-15.21-4 map 暂代形态随本批回收。
type ToolOutput struct {
	// Success 是否成功
	Success bool `json:"success"`
	// Data 结构化结果负载
	Data map[string]any `json:"data,omitempty"`
	// Error 失败信息（partial 提交时为通用错误，明细在 Data）
	Error string `json:"error,omitempty"`
}

// ToMap 转为 foundation/tool.Invoke 返回的 map 形态。
// Data 原样透传（禁止 json.Marshal→Unmarshal 往返：int 会漂移为 float64）。
func (o ToolOutput) ToMap() map[string]any {
	m := map[string]any{"success": o.Success}
	if o.Data != nil {
		m["data"] = o.Data
	}
	if o.Error != "" {
		m["error"] = o.Error
	}
	return m
}

// RenderFields 将扁平记录渲染为 "key: value" 行序列（模型可见文本）。
// Python: render_fields (harness/tools/base_tool.py:10-30)——None/空串跳过，
// 字符串原样，其余 JSON 编码（default=str 语义：Go 用 fmt.Sprintf 兜底）。
func RenderFields(fields map[string]any) string {
	pairs := make([]string, 0, len(fields))
	for key, value := range fields {
		if value == nil || value == "" {
			continue
		}
		if text, ok := value.(string); ok {
			pairs = append(pairs, key+": "+text)
			continue
		}
		encoded, err := json.Marshal(value)
		if err != nil {
			encoded = []byte(fmt.Sprintf("%v", value))
		}
		pairs = append(pairs, key+": "+string(encoded))
	}
	return strings.Join(pairs, "\n")
}
```

（`fmt` 导入按需补；分隔符参数 Python 默认 `\n`，Go 单分隔符即可，调用方无自定义点。）
