func TestRepoLock_并发串行化(t *testing.T) {
	root := t.TempDir()
	release1, err := RepoLock(root, 120.0, true)()
	if err != nil { t.Fatalf("首次获取失败: %v", err) }
	// 第二次获取（同 repo）必须等第一次释放：用 goroutine + 超时验证互斥
	done := make(chan error, 1)
	go func() {
		r2, err := RepoLock(root, 0.2, true)()
		if err == nil { defer r2() }
		done <- err
	}()
	_ = release1() // 释放后第二个锁才能拿到
	if err := <-done; err != nil { t.Fatalf("释放后应能获取: %v", err) }
}

func TestRepoLock_超时返回WorktreeLockTimeout(t *testing.T) {
	root := t.TempDir()
	release, err := RepoLock(root, 10.0, true)()
	if err != nil { t.Fatal(err) }
	defer release()
	_, err = RepoLock(root, 0.1, true)()
	var te *WorktreeLockTimeout
	if !errors.As(err, &te) { t.Fatalf("期望 WorktreeLockTimeout, got %v", err) }
}

func TestRepoLock_禁用时直通(t *testing.T) {
	release, err := RepoLock(t.TempDir(), 0, false)()
	if err != nil { t.Fatal(err) }
	release() // 无文件锁副作用即通过
}
