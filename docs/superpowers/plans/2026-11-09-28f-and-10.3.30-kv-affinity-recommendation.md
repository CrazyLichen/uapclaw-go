> 2026-11-10 用户拍板：**「按 Python 严格对齐、连带改签名」** + **「本轮一并改成无 error，但 Python 有抛异常时返回 error 是对的」**。
> 本章是逐条核实的仲裁结果，实施时以本表为准，不得自行决定某个签名带不带 error。

### 仲裁方法

对每个「本轮新增/改动」的函数，先在上游 Python 源码里查三件事：① 返回标注（`-> None` / `-> bool` / 无标注）；② 函数体内是否有 `raise`；③ 调用方是否检查返回值或 `except`。**只有①无返回且②有 raise 时才保留 `error`**；②③都吞掉的一律无 error。Go 侧既有契约（非本轮引入）单列。

### 逐条结论

		s, ok := v.(string)

	}

	}

// Python: kv_cache_metadata.py:88 first_changed_index(old: Sequence[T], new: Sequence[T])
// ★ 严格对齐：Python 是泛型序列逐元素比较（old_item != new_item），
//   Go 1.22 用 [T comparable] 一比一，不得改成 (len,len,equal) 闭包形态。
func FirstChangedIndex[T comparable](old, new []T) *int {
	n := len(old)
	if len(new) < n {
		n = len(new)
	}
	for i := 0; i < n; i++ {
		if old[i] != new[i] {
			idx := i
			return &idx
		}
	}
	if len(new) < len(old) {
		idx := len(new)
		return &idx
	}
	return nil
}

	f.mu.Lock()
	call, ok := f.turns[sessionID]
	if !ok {
		f.mu.Unlock()
		return
	}
	delete(f.turns, sessionID)
	if opts.Status == "success" && call != nil && !call.AnswerHasToolCalls {

		return

	cfg *kv_cache.KVCacheAffinityConfig,
	sess kv_cache.LineageSession,

// runFork 重放父调用并解析建议；落库与推帧经窄接口，未注入则跳过。
// Python: prompt_suggestion.py:447 _fork
func (f *PromptSuggestionForks) runFork(
	ctx context.Context, sessionID, requestID string, call *MainCall,
) {
	defer func() {
		if v := recover(); v != nil {
			logger.Debug(logComponent).Any("panic", v).Msg("prompt suggestion fork recovered")
		}
		f.mu.Lock()
		delete(f.tasks, sessionID)
		f.mu.Unlock()
	}()
	epoch := ConsentEpoch(nil)
	if epoch == 0 {
		return // 推荐关闭：不生成也不记录（对齐 :386-388）
	}
	maxIter := call.MaxIterations
	if maxIter <= 0 {
		maxIter = 1
	}
	messages := append([]llmschema.BaseMessage{}, call.Messages...)
	messages = append(messages, call.Answer,
		llmschema.NewUserMessage(SuggestionPrompt))
	for i := 0; i < maxIter; i++ {
		if ConsentEpoch(nil) != epoch {
			return // :466 同意 epoch 变化即退出
		}
		if ctx.Err() != nil {
			return
		}
		reply, err := call.LLM.Invoke(ctx, messages,
			model_clients.WithTools(call.Tools...),
			model_clients.WithInvokeExtra(call.Parameters))
		if err != nil {
			logger.Debug(logComponent).Err(err).
				Str("session_id", sessionID).Msg("prompt suggestion fork invoke failed")
			return
		}
		messages = append(messages, reply)
	}
	text := extractSuggestion(messages[len(messages)-1])
	if text == "" {
		return
	}
	result := ForkResult{
		SuggestionID: newSuggestionID(), SessionID: sessionID,
		RequestID: requestID, Text: text,
	}
	f.mu.Lock()
	pub, pusher := f.publisher, f.readyPusher
	f.mu.Unlock()
	if pub == nil {
		logger.Debug(logComponent).Str("session_id", sessionID).
			Msg("fork result discarded: no publisher")
		return
	}
	// Python _record(:326)/deliver(:743) 均不抛异常；实现方内部 Warn，不向上抛。
	pub.PublishForkResult(ctx, result)
	if pusher != nil {
		pusher.PushReady(ctx, SuggestionReadyEvent{
			SessionID: sessionID, RequestID: requestID,
			SuggestionID: result.SuggestionID, Text: text,
		})
	}
}

func (a *sessionOpsAdapter) CopySessionContext(
	ctx context.Context, agent swruntime.SessionAgent, sourceID, targetID string, forceHistory bool,
) bool {
	deep := a.liveDeepAgent(agent, sourceID)
	if deep == nil {

		return false

	}

}
