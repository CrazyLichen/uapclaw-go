```go
package retriever

import (
	"context"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/retrieval/common"
)

// ──────────────────────────── 接口 ────────────────────────────

// Retriever 检索器抽象接口。
//
// Python: openjiuwen/core/retrieval/retriever/base.py (Retriever)。
// retrieve/batch_retrieve 为 Python 抽象方法；retrieve_search_results/close
// 在 Python 基类是非抽象"可选 API"（空方法体，裸调用返回 None），Go 接口
// 无默认实现，四方法全进接口。AgenticRetriever(13.25)/GraphRetriever(13.23)
// 届时自行决定是否返回 nil,nil 对齐 Python None 语义。
type Retriever interface {
	// Retrieve 检索文档，返回归一化的 RetrievalResult 列表。
	// mode 零值 "" 时由实现回落各自默认模式（spec §8-1 已知偏差：
	// Python 显式传空串走 155200，Go 空串等同未设）。
	Retrieve(ctx context.Context, query string, topK int, scoreThreshold *float64, mode RetrievalMode, opts ...RetrieveOption) ([]common.RetrievalResult, error)
	// BatchRetrieve 批量并发检索。裸 errgroup 首错即返回（对齐 Python
	// 裸 asyncio.gather：全启动、首错传播、其余 goroutine 不取消跑完结果丢弃）。
	BatchRetrieve(ctx context.Context, queries []string, topK int, opts ...RetrieveOption) ([][]common.RetrievalResult, error)
	// RetrieveSearchResults 检索原始 SearchResult（不归一化、不阈值过滤）。
	RetrieveSearchResults(ctx context.Context, query string, topK int, mode RetrievalMode, opts ...RetrieveOption) ([]common.SearchResult, error)
	// Close 关闭检索器并释放底层资源。
	Close()
}

// ──────────────────────────── 枚举 ────────────────────────────

// RetrievalMode 检索模式，对齐 Python Literal["vector", "sparse", "hybrid"]。
type RetrievalMode string

const (
	// RetrievalModeVector 向量检索
	RetrievalModeVector RetrievalMode = "vector"
	// RetrievalModeSparse 稀疏检索（BM25）
	RetrievalModeSparse RetrievalMode = "sparse"
	// RetrievalModeHybrid 混合检索
	RetrievalModeHybrid RetrievalMode = "hybrid"
)

// ──────────────────────────── 结构体 ────────────────────────────

// RetrieveOptions 检索选项（Python **kwargs 收纳容器）。
//
// 本节 Vector/Sparse 收而不用（对齐 Python 死参数：kwargs 收了不读）；
// 13.21 HybridRetriever 消费 alpha 键；13.27 SimpleKnowledgeBase 写入 filters 键。
type RetrieveOptions struct {
	// Extra 透传给实现的额外参数
	Extra map[string]any
}

// RetrieveOption RetrieveOptions 的函数选项。
type RetrieveOption func(*RetrieveOptions)

// ──────────────────────────── 导出函数 ────────────────────────────

// NewRetrieveOptions 从可变参数构建 RetrieveOptions。
func NewRetrieveOptions(opts ...RetrieveOption) RetrieveOptions {
	o := RetrieveOptions{}
	for _, opt := range opts {
		opt(&o)
	}
	return o
}

// WithRetrieveExtra 设置 Extra 单键的 RetrieveOption。
func WithRetrieveExtra(key string, value any) RetrieveOption {
	return func(o *RetrieveOptions) {
		if o.Extra == nil {
			o.Extra = make(map[string]any)
		}
		o.Extra[key] = value
	}
}
```

```go
package retriever

import (
	"context"
	"fmt"

	"golang.org/x/sync/errgroup"

	"github.com/uapclaw/uapclaw-go/internal/agentcore/foundation/store/embedding"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/retrieval/common"
	"github.com/uapclaw/uapclaw-go/internal/agentcore/retrieval/vector_store"
	"github.com/uapclaw/uapclaw-go/internal/common/exception"
)

// ──────────────────────────── 结构体 ────────────────────────────

// VectorRetriever 向量检索器实现。
//
// Python: openjiuwen/core/retrieval/retriever/vector_retriever.py
type VectorRetriever struct {
	// vectorStore 底层向量存储
	vectorStore vector_store.VectorStore
	// embedModel 嵌入模型（nil = Python None；向量检索必需，
	// 缺失时 retrieve 报 155202）
	embedModel embedding.BaseEmbedding
}

// ──────────────────────────── 导出函数 ────────────────────────────

// NewVectorRetriever 创建向量检索器。
//
// 对齐 Python VectorRetriever.__init__：构造零校验，
// embedModel 为 nil 的 155202 在 Retrieve 时触发。
func NewVectorRetriever(vectorStore vector_store.VectorStore, embedModel embedding.BaseEmbedding) *VectorRetriever {
	return &VectorRetriever{
		vectorStore: vectorStore,
		embedModel:  embedModel,
	}
}

// Retrieve 向量检索。
//
// 对齐 Python VectorRetriever.retrieve：mode 校验 → embed → search →
// 空结果 BM25 兜底 → 归一化（chunk_id 取 metadata["chunk_id"]）。
// 注意：Python 此处不向 store.search 传 filters（kwargs 有 filters 也不透传，字面复刻）。
func (r *VectorRetriever) Retrieve(ctx context.Context, query string, topK int, scoreThreshold *float64, mode RetrievalMode, _ ...RetrieveOption) ([]common.RetrievalResult, error) {
	// mode 零值回落子类默认（spec §8-1：Python 签名默认 "vector"）
	if mode == "" {
		mode = RetrievalModeVector
	}
	if mode != RetrievalModeVector {
		return nil, exception.BuildError(
			exception.StatusRetrievalRetrieverModeNotSupport,
			exception.WithParam("error_msg", fmt.Sprintf("VectorRetriever only supports 'vector' mode, got %s", mode)),
		)
	}
	// Python L65-69 的 155201（threshold+mode≠vector）为死分支（前置 mode 校验
	// 先抛 155200），跳过归档 spec §8-2；155201 留 13.21 HybridRetriever 真实消费

	if r.embedModel == nil {
		return nil, exception.BuildError(
			exception.StatusRetrievalRetrieverEmbedModelNotFound,
			exception.WithParam("error_msg", "embed_model is required for vector search"),
		)
	}

	queryVector, err := r.embedModel.EmbedQuery(ctx, query)
	if err != nil {
		return nil, err // Python await 直接冒泡，不包装
	}
	searchResults, err := r.vectorStore.Search(ctx, queryVector, topK, nil)
	if err != nil {
		return nil, err
	}
	// 向量检索空结果时兜底 BM25（对齐 Python fallback to sparse）
	if len(searchResults) == 0 {
		searchResults, err = r.vectorStore.SparseSearch(ctx, query, topK, nil)
		if err != nil {
			return nil, err
		}
	}

	retrievalResults := make([]common.RetrievalResult, 0, len(searchResults))
	for _, result := range searchResults {
		// 阈值过滤（Python result.score is not None 的 nil 检查在 Go 无对应物，spec §8-4）
		if scoreThreshold != nil && result.Score < *scoreThreshold {
			continue
		}
		retrievalResults = append(retrievalResults, common.RetrievalResult{
			Text:     result.Text,
			Score:    result.Score,
			Metadata: result.Metadata,
			DocID:    metadataString(result.Metadata, "doc_id"),
			ChunkID:  metadataString(result.Metadata, "chunk_id"), // Vector 从 metadata 取（spec 决策 7）
		})
	}
	return retrievalResults, nil
}

// BatchRetrieve 批量并发检索。
//
// 对齐 Python 裸 asyncio.gather：全启动、首错传播、其余不取消。
// 裸 errgroup（不派生 ctx、不 cancel）——字面一致。
func (r *VectorRetriever) BatchRetrieve(ctx context.Context, queries []string, topK int, opts ...RetrieveOption) ([][]common.RetrievalResult, error) {
	results := make([][]common.RetrievalResult, len(queries))
	var g errgroup.Group
	for i, q := range queries {
		g.Go(func() error {
			// Python batch_retrieve 不传 mode/score_threshold → 子类默认/None
			res, err := r.Retrieve(ctx, q, topK, nil, "", opts...)
			if err != nil {
				return err
			}
			results[i] = res
			return nil
		})
	}
	if err := g.Wait(); err != nil {
		return nil, err
	}
	return results, nil
}

// RetrieveSearchResults 向量检索（原始 SearchResult，不归一化）。
//
// 对齐 Python VectorRetriever.retrieve_search_results：仅 embedModel 校验，
// 无 mode 校验（Python 两作者不对称，字面复刻 spec §8-8）。
func (r *VectorRetriever) RetrieveSearchResults(ctx context.Context, query string, topK int, _ RetrievalMode, _ ...RetrieveOption) ([]common.SearchResult, error) {
	if r.embedModel == nil {
		return nil, exception.BuildError(
			exception.StatusRetrievalRetrieverEmbedModelNotFound,
			exception.WithParam("error_msg", "embed_model is required for vector search"),
		)
	}
	queryVector, err := r.embedModel.EmbedQuery(ctx, query)
	if err != nil {
		return nil, err
	}
	searchResults, err := r.vectorStore.Search(ctx, queryVector, topK, nil)
	if err != nil {
		return nil, err
	}
	if len(searchResults) == 0 {
		searchResults, err = r.vectorStore.SparseSearch(ctx, query, topK, nil)
		if err != nil {
			return nil, err
		}
	}
	return searchResults, nil
}

// Close 关闭检索器。
//
// 对齐 Python close()：if self.vector_store 判空后关底层。
// Python 的 getattr+iscoroutinefunction 鸭子探测在 Go 消解为
// 接口静态 Close() 直调（spec §8-7 语言特性级等价物）。
func (r *VectorRetriever) Close() {
	if r.vectorStore != nil {
		r.vectorStore.Close()
	}
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// metadataString 从 metadata 取字符串值，断言失败或缺失返回空串。
//
// 对齐 Python metadata.get(key) + pydantic lax str 强转：正常链路
// （写入侧 13.5 mergeChunkMetadata 恒写 uuid 字符串）不受影响；
// non-str 值 Python 强转 str(123)="123" vs Go 空串，归档 spec §8-3。
func metadataString(m map[string]any, key string) string {
	if m == nil {
		return ""
	}
	if v, ok := m[key]; ok {
		if s, ok := v.(string); ok {
			return s
		}
	}
	return ""
}
```

```go
// Retrieve 稀疏检索（BM25）。
//
// 对齐 Python SparseRetriever.retrieve：仅 mode 校验——
// 无 threshold 校验（收而不用，静默忽略，spec §8-5）、
// 无 embedModel 校验（BM25 不需要嵌入）。
// 归一化 chunk_id 取 result.ID（与 Vector 的 metadata 来源不同，
// Python 三作者不一致，字面复刻 spec 决策 7）。
func (r *SparseRetriever) Retrieve(ctx context.Context, query string, topK int, _ *float64, mode RetrievalMode, _ ...RetrieveOption) ([]common.RetrievalResult, error) {
	// mode 零值回落子类默认（spec §8-1：Python 签名默认 "sparse"）
	if mode == "" {
		mode = RetrievalModeSparse
	}
	if mode != RetrievalModeSparse {
		return nil, exception.BuildError(
			exception.StatusRetrievalRetrieverModeNotSupport,
			exception.WithParam("error_msg", fmt.Sprintf("SparseRetriever only supports 'sparse' mode, got %s", mode)),
		)
	}

	searchResults, err := r.vectorStore.SparseSearch(ctx, query, topK, nil)
	if err != nil {
		return nil, err
	}

	retrievalResults := make([]common.RetrievalResult, 0, len(searchResults))
	for _, result := range searchResults {
		retrievalResults = append(retrievalResults, common.RetrievalResult{
			Text:     result.Text,
			Score:    result.Score,
			Metadata: result.Metadata,
			DocID:    metadataString(result.Metadata, "doc_id"),
			ChunkID:  result.ID, // Sparse 从主键取（spec 决策 7）
		})
	}
	return retrievalResults, nil
}

// BatchRetrieve 批量并发检索（裸 errgroup，同 VectorRetriever 语义）。
func (r *SparseRetriever) BatchRetrieve(ctx context.Context, queries []string, topK int, opts ...RetrieveOption) ([][]common.RetrievalResult, error) {

	for i, q := range queries {
		g.Go(func() error {
			res, err := r.Retrieve(ctx, q, topK, nil, "", opts...)
			if err != nil {
				return err
			}
			results[i] = res
			return nil
		})
	}
	if err := g.Wait(); err != nil {
		return nil, err
	}
	return results, nil
}

}
```

```go
// StoreCreateOptions 向量存储构造选项（收纳 Python 构造 **kwargs）。

// CreateVectorStore 根据配置动态创建向量存储实例。
//
// Python: openjiuwen/core/retrieval/vector_store/store.py (create_vector_store)。
// milvus/chroma 分支组装既有构造器（含 str→强类型转换，对齐 Python
// isinstance 分支）；pgvector 分支返回 155402 not-implemented（⤵️ PGVectorStore
// 依赖 pgvector-go 拆后交付，届时只改本分支内部）；default 分支 155402
// 错误文案逐字符对齐 Python（含逗号+空格拼接）。
func CreateVectorStore(config common.VectorStoreConfig, opts ...StoreCreateOption) (VectorStore, error) {
	o := NewStoreCreateOptions(opts...)

	switch config.StoreProvider {
	case common.StoreTypeMilvus:
		vectorField := o.MilvusVectorField
		if vectorField == nil {
			// 对齐 Python: isinstance(vector_field, str) → MilvusAUTO(vector_field=...)
			name := o.VectorFieldName
			if name == "" {
				name = defaultVectorFieldName
			}
			vectorField = vector_fields.NewMilvusAUTO(name)
		}
		return NewMilvusVectorStore(
			config, o.MilvusURI, o.MilvusToken,
			defaultIfEmpty(o.TextField, defaultTextField), vectorField,
			defaultIfEmpty(o.SparseVectorField, defaultSparseVectorField),
			defaultIfEmpty(o.MetadataField, defaultMetadataField),
			defaultIfEmpty(o.DocIDField, defaultDocIDField),
		)

	case common.StoreTypeChroma:
		vectorField := o.ChromaVectorField
		if vectorField == nil {
			// 对齐 Python: isinstance(vector_field, str) → ChromaVectorField(vector_field=...)
			name := o.VectorFieldName
			if name == "" {
				name = defaultVectorFieldName
			}
			vectorField = vector_fields.NewChromaVectorFieldFromName(name)
		}
		return NewChromaVectorStore(
			config, o.ChromaPath,
			defaultIfEmpty(o.TextField, defaultTextField), vectorField,
			defaultIfEmpty(o.SparseVectorField, defaultSparseVectorField),
			defaultIfEmpty(o.MetadataField, defaultMetadataField),
			defaultIfEmpty(o.DocIDField, defaultDocIDField),
		)

	case common.StoreTypePGVector:
		// ⤵️ PGVectorStore（pgvector-go 新依赖）拆后交付——届时只改本分支内部
		return nil, exception.BuildError(
			exception.StatusRetrievalVectorStoreProviderInvalid,
			exception.WithParam("error_msg", "pgvector store is not implemented yet, planned for later delivery"),
		)

	default:
		// 对齐 Python: "unavailable vector store provider: {provider},"
		//   f"and available providers are: {', '.join(...)}" 两行拼接（逗号+空格）
		return nil, exception.BuildError(
			exception.StatusRetrievalVectorStoreProviderInvalid,
			exception.WithParam("error_msg", fmt.Sprintf(
				"unavailable vector store provider: %s, and available providers are: milvus, chroma, pgvector",
				config.StoreProvider)),
		)
	}
}

// ──────────────────────────── 非导出函数 ────────────────────────────

// defaultIfEmpty 空串回落默认值（Python 构造参数默认值语义）。
func defaultIfEmpty(v, def string) string {
	if v == "" {
		return def
	}
	return v
}
```
