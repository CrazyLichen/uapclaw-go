```go
package resources

import (
	"encoding/json"
	"fmt"
	"os"
	"path"
	"path/filepath"
	"strings"
)

// AgentTemplateCard 一个 AgentTemplate 包的展示信息（只读 manifest.json，不加载包体）。
// Python: openjiuwen/harness/resources/template_card.py AgentTemplateCard（dataclass frozen，D-2）
// 卡片刻意宽松：只要求 package_type 正确；能否运行由全量 loader 决定。
type AgentTemplateCard struct {
	// Name manifest.name；缺失为空串
	Name string `json:"name"`
	// Description manifest.description；缺失为空串
	Description string `json:"description"`
	// DisplayName 展示名：字符串或 {语言:文本}（D-1：union 判别，string 或 map[string]string）
	DisplayName any `json:"display_name"`
	// DisplayDescription 展示描述：同 DisplayName 形状
	DisplayDescription any `json:"display_description"`
	// Category 分类；非字符串读空
	Category string `json:"category"`
	// Tags 标签（原样：字符串或本地化 map）
	Tags []any `json:"tags"`
	// Version 版本
	Version string `json:"version"`
	// Avatar 包相对头像路径（原样）
	Avatar string `json:"avatar"`
	// Runtime manifest 声明 runtime 为 "external"，否则 "native"
	Runtime string `json:"runtime"`
	// RuntimeProvider 外部 runtime 的 manifest.runtime.provider_name
	RuntimeProvider *string `json:"runtime_provider,omitempty"`
	// RuntimeConfig manifest.runtime.config 原样（宿主交给 provider 的配置）
	RuntimeConfig map[string]any `json:"runtime_config"`
	// Skills 技能目录名（manifest 顺序，去重）
	Skills []string `json:"skills"`
	// Connectors 宿主托管 MCP connector 名（{"connector": name} 项）
	Connectors []string `json:"connectors"`
}

// DisplayLabel 指定语言的展示名，回退到 name。
func (c AgentTemplateCard) DisplayLabel(language string) string {
	return orString(localizedText(c.DisplayName, language), c.Name)
}

// DisplayText 指定语言的展示描述，回退到 description。
func (c AgentTemplateCard) DisplayText(language string) string {
	return orString(localizedText(c.DisplayDescription, language), c.Description)
}

// TagLabels 标签按语言渲染，空标签丢弃。
func (c AgentTemplateCard) TagLabels(language string) []string {
	labels := make([]string, 0, len(c.Tags))
	for _, tag := range c.Tags {
		if label := localizedText(tag, language); label != "" {
			labels = append(labels, label)
		}
	}
	return labels
}

// languageKeys 团队/harness 语言码与 manifest 键映射。
// Python: template_card.py:30-34 _LANGUAGE_KEYS
var languageKeys = map[string][]string{
	"cn": {"cn", "zh"},
	"zh": {"zh", "cn"},
	"en": {"en"},
}

// localizedText 按 language 取值（cn 兼读 zh），再英文，再任意文本。
// Python: template_card.py:79-87 localized_text
func localizedText(value any, language string) string {
	switch v := value.(type) {
	case string:
		return strings.TrimSpace(v)
	case map[string]string:
		keys := languageKeys[language]
		if keys == nil {
			keys = []string{language}
		}
		for _, key := range append(append([]string{}, keys...), "en", "zh", "cn") {
			if text := strings.TrimSpace(v[key]); text != "" {
				return text
			}
		}
		for _, text := range v {
			if text = strings.TrimSpace(text); text != "" {
				return text
			}
		}
		return ""
	default:
		return ""
	}
}

// ParseAgentTemplateCard 从已解析的 manifest 构造卡片（纯函数）。
// Python: template_card.py:90-119 parse_agent_template_card
func ParseAgentTemplateCard(manifestJSON []byte) (AgentTemplateCard, error) {
	var payload map[string]any
	if err := json.Unmarshal(manifestJSON, &payload); err != nil {
		return AgentTemplateCard{}, fmt.Errorf("manifest must contain a mapping")
	}
	if payload["package_type"] != "agent_template" {
		return AgentTemplateCard{}, fmt.Errorf(
			"manifest declares package_type=%q, expected 'agent_template'", payload["package_type"])
	}
	runtime, _ := payload["runtime"].(map[string]any)
	var provider *string
	if raw, ok := runtime["provider_name"].(string); ok && strings.TrimSpace(raw) != "" {
		p := strings.TrimSpace(raw)
		provider = &p
	}
	runtimeName := "native"
	if runtime != nil {
		runtimeName = "external"
	}
	return AgentTemplateCard{
		Name:           manifestText(payload["name"]),
		Description:    manifestText(payload["description"]),
		DisplayName:    normalizeLocalized(payload["display_name"]),
		DisplayDescription: normalizeLocalized(payload["display_description"]),
		Category:       manifestText(payload["category"]), // 非 str→空（manifestText 语义一致）
		Tags:           asListFiltered(payload["tags"]),
		Version:        manifestText(payload["version"]),
		Avatar:         asString(payload["avatar"]),
		Runtime:        runtimeName,
		RuntimeProvider: provider,
		RuntimeConfig:  runtimeConfigOf(runtime),
		Skills:         skillNames(payload["skills"]),
		Connectors:     connectorNames(payload["mcps"]),
	}, nil
}

// ReadAgentTemplateCard LoadAgentTemplateCard 的异步语义入口（D-3：D-15b 同步实现）。
func ReadAgentTemplateCard(manifestPath string) (AgentTemplateCard, error) {
	return LoadAgentTemplateCard(manifestPath)
}

// LoadAgentTemplateCard 只读 manifest.json 构造卡片。
// Python: template_card.py:127-137 load_agent_template_card
func LoadAgentTemplateCard(manifestPath string) (AgentTemplateCard, error) {
	payload, err := readManifestMapping(manifestPath)
	if err != nil {
		return AgentTemplateCard{}, err
	}
	raw, err := json.Marshal(payload)
	if err != nil {
		return AgentTemplateCard{}, err
	}
	return ParseAgentTemplateCard(raw)
}

// readManifestMapping 读 JSON manifest，读/解析失败统一映射 error。
// Python: template_card.py:140-152 read_manifest_mapping
func readManifestMapping(pathStr string) (map[string]any, error) {
	expanded, err := filepath.Abs(strings.TrimPrefix(pathStr, "~"))
	if err != nil {
		return nil, fmt.Errorf("manifest not readable: %s", pathStr)
	}
	text, err := os.ReadFile(expanded)
	if err != nil {
		return nil, fmt.Errorf("manifest not readable: %s", expanded)
	}
	var payload map[string]any
	if err := json.Unmarshal(text, &payload); err != nil {
		return nil, fmt.Errorf("manifest is not valid JSON: %s", expanded)
	}
	if payload == nil {
		return nil, fmt.Errorf("manifest must contain a JSON object: %s", expanded)
	}
	return payload, nil
}

// manifestText manifest 文本字段 strip；非字符串读空。
// Python: template_card.py:155-157 manifest_text
func manifestText(value any) string {
	if s, ok := value.(string); ok {
		return strings.TrimSpace(s)
	}
	return ""
}

// normalizeLocalized 归一化展示值：保留字符串，map 值字符串化，其余清空。
// Python: template_card.py:160-166 normalize_localized
func normalizeLocalized(value any) any {
	switch v := value.(type) {
	case string:
		return v
	case map[string]any:
		out := make(map[string]string, len(v))
		for key, text := range v {
			if text != nil {
				out[fmt.Sprintf("%v", key)] = fmt.Sprintf("%v", text)
			}
		}
		return out
	default:
		return ""
	}
}

// asList 非列表读空。Python: template_card.py:169-170 _as_list
func asList(value any) []any {
	if list, ok := value.([]any); ok {
		return list
	}
	return nil
}

// asListFiltered tags 过滤：仅保留字符串或 map 项。Python: template_card.py:109-111
func asListFiltered(value any) []any {
	var out []any
	for _, item := range asList(value) {
		switch item.(type) {
		case string, map[string]any:
			out = append(out, item)
		}
	}
	return out
}

// skillNames 技能目录名（str 或 {"dir":...} 项），path.Base 取名 + 去重（D-4）。
// Python: template_card.py:173-180 _skill_names
func skillNames(items any) []string {
	var names []string
	for _, item := range asList(items) {
		var raw string
		switch v := item.(type) {
		case string:
			raw = v
		case map[string]any:
			if d, ok := v["dir"].(string); ok {
				raw = d
			}
		}
		name := ""
		if raw != "" {
			name = path.Base(raw)
		}
		if name != "" && !containsString(names, name) {
			names = append(names, name)
		}
	}
	return names
}

// connectorNames connector 名去重。Python: template_card.py:183-190 _connector_names
func connectorNames(items any) []string {
	var names []string
	for _, item := range asList(items) {
		m, ok := item.(map[string]any)
		if !ok {
			continue
		}
		if c, ok := m["connector"].(string); ok {
			name := strings.TrimSpace(c)
			if name != "" && !containsString(names, name) {
				names = append(names, name)
			}
		}
	}
	return names
}

// runtimeConfigOf 提取 runtime.config（非 map→空 map）。
func runtimeConfigOf(runtime map[string]any) map[string]any {
	out := map[string]any{}
	if runtime == nil {
		return out
	}
	if cfg, ok := runtime["config"].(map[string]any); ok && cfg != nil {
		for k, v := range cfg {
			out[k] = v
		}
	}
	return out
}

func asString(v any) string {
	if s, ok := v.(string); ok {
		return s
	}
	return ""
}

func containsString(list []string, s string) bool {
	for _, item := range list {
		if item == s {
			return true
		}
	}
	return false
}

func orString(a, b string) string {
	if a != "" {
		return a
	}
	return b
}
```

```go
package experts

import (
	"path"

	hschema "github.com/uapclaw/uapclaw-go/internal/agentcore/harness/schema"
)

// defaultLanguage persona 渲染默认语言。
// Python: template_member.py:34 _DEFAULT_LANGUAGE
const defaultLanguage = "cn"

// TemplateMember 一个模板装配成的 Team 成员。
// Python: template_member.py:99-106 TemplateMember（dataclass frozen，D-2）
type TemplateMember struct {
	// Spec base spec + 能力快照
	Spec hschema.DeepAgentSpec
	// Persona 成员私有 prompt 的渲染产物
	Persona string
}

// RenderTemplatePersona 把模板 prompt_sections 渲染拍平成一段人格文本。
// 按 priority 排序；每段以 language 渲染（{{ language }}/{{ workspace }} +
// 段 render_params，后者覆盖前者）；空段丢弃；其余空行拼接。
// Python: template_member.py:37-66 render_template_persona
func RenderTemplatePersona(template hschema.AgentTemplateSpec, language string, workspace string) string {
	resolved := language
	if trimmed := trimSpace(resolved); trimmed == "" {
		resolved = defaultLanguage
	} else {
		resolved = trimmed
	}
	// base_params：language/workspace 派生值在前（render_params 覆盖在后）
	baseParams := map[string]any{
		"language":  resolved,
		"workspace": workspace,
	}
	// 按 priority 升序排序（Python: sorted(..., key=priority)）
	sections := append([]hschema.PromptSectionSpec(nil), template.PromptSections...)
	for i := 1; i < len(sections); i++ { // 插入排序，稳定
		for j := i; j > 0 && sections[j].Priority < sections[j-1].Priority; j-- {
			sections[j], sections[j-1] = sections[j-1], sections[j]
		}
	}
	var rendered []string
	for _, section := range sections {
		params := map[string]any{}
		for k, v := range baseParams {
			params[k] = v
		}
		for k, v := range section.RenderParams {
			params[k] = v
		}
		text := trimSpace(hres_RenderPromptContent(section.Content, resolved, params))
		if text != "" {
			rendered = append(rendered, text)
		}
	}
	return joinBlankLines(rendered)
}

// TemplateSnapshot 能力快照：去掉 prompt_sections 的模板，JSON 数据形态。
// Python: template_member.py:69-71 template_snapshot
func TemplateSnapshot(template hschema.AgentTemplateSpec) map[string]any {
	template.PromptSections = nil
	return template.ModelDump()
}

// TemplateSkillNames 快照的技能目录名（顺序去重）。
// Python: template_member.py:74-82 template_skill_names
func TemplateSkillNames(snapshot map[string]any) []string {
	var names []string
	rawSkills, _ := snapshot["skills"].([]any)
	for _, item := range rawSkills {
		m, ok := item.(map[string]any)
		if !ok {
			continue
		}
		dir, _ := m["dir"].(string)
		name := ""
		if dir != "" {
			name = path.Base(dir)
		}
		if name != "" && !containsString(names, name) {
			names = append(names, name)
		}
	}
	return names
}

// ApplyTemplateSnapshot 把能力快照挂到 base spec 并种子 Skill 可见性。
// 返回 base 的拷贝：agent_template_spec=snapshot，skills=base.skills ∪ 快照技能目录名（保序去重）。
// Python: template_member.py:85-96 apply_template_snapshot
func ApplyTemplateSnapshot(baseSpec hschema.DeepAgentSpec, snapshot map[string]any) hschema.DeepAgentSpec {
	skills := append([]string(nil), baseSpec.Skills...)
	for _, name := range TemplateSkillNames(snapshot) {
		if name != "" && !containsString(skills, name) {
			skills = append(skills, name)
		}
	}
	baseSpec.AgentTemplateSpec = snapshot
	baseSpec.Skills = skills
	return baseSpec
}

// ComposeTemplateMember 在 base spec 上装配模板（persona + 能力快照）。
// language 缺省取 baseSpec.Language；workspace 取 baseSpec.workspace 根路径。
// Python: template_member.py:109-130 compose_template_member
func ComposeTemplateMember(baseSpec hschema.DeepAgentSpec, template hschema.AgentTemplateSpec, language string) TemplateMember {
	var workspace string
	if baseSpec.Workspace != nil {
		workspace = baseSpec.Workspace.RootPath
	}
	lang := language
	if lang == "" {
		lang = baseSpec.Language
	}
	return TemplateMember{
		Spec:    ApplyTemplateSnapshot(baseSpec, TemplateSnapshot(template)),
		Persona: RenderTemplatePersona(template, lang, workspace),
	}
}

// hres_RenderPromptContent 转调 resources.RenderPromptContent（避免 import 别名漂移，集中一处）。
func hres_RenderPromptContent(content map[string]string, language string, params map[string]any) string {
	return hres.RenderPromptContent(content, language, params)
}

func trimSpace(s string) string { return strings.TrimSpace(s) }

func joinBlankLines(parts []string) string { return strings.Join(parts, "\n\n") }

func containsString(list []string, s string) bool {
	for _, item := range list {
		if item == s {
			return true
		}
	}
	return false
}
```
