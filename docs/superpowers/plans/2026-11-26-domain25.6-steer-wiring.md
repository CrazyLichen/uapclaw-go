// Python: session_input.py QueuedSessionInput
type QueuedSessionInput struct {
	// Query 用户输入文本
	Query string
	// RequestID 请求信封 id
	RequestID string
	// DisplayContent 前端展示文本（图片-only 补充时与 Query 可能不同）
	DisplayContent string
	// MediaItems steer 图片引用列表
	MediaItems []map[string]any

	CrossSession map[string]any
	// MessageRoute 跨会话路由五键
	MessageRoute map[string]any
	// BoundaryReady 边界就绪信号（BoundaryGuard 消费）
	BoundaryReady chan struct{}
	// BoundaryError 入队失败原因（nil=成功）
	BoundaryError error
}

// EnqueueBoundSessionInput 同步校验并入队活跃轮的 steering queue。
// Python: session_input.py enqueue_bound_session_input（:102-160）
//
// 铁律：目标校验与入队之间不得插入任何可让渡点（含锁获取）；否则可能错开新轮。
// 入口前置条件由 deliver 链保证：instance 非 nil、bound_round 已在准入时捕获。
func EnqueueBoundSessionInput(instance any, boundRound any, request *schema.AgentRequest, sdkRequest any) (*QueuedSessionInput, error) {
	// ① execution 复核（expected_execution_id 非空时）
	//    Python: expected_id → runtime.get_session_execution → 终态/取消/异会话 → TARGET_ERROR
	// ② boundRound 复核：instance.activeRound is not boundRound → TARGET_ERROR
	// ③ steering queue 存在性：nil → RuntimeError("active execution has no steering queue; supplemental input was not sent")
	// ④ 构造 entry：Query/DisplayContent/MediaItems/CrossSession/MessageRoute
	//    message_route 五键：session_id/request_id/user_id/chain_id/parent_message_id/hop_count
	// ⑤ controller.enqueue_steer(entry) —— 与②③同一临界区
	return entry, nil
}

// Python: session_input.py SessionInputBoundaryGuard (priority=1000)
type SessionInputBoundaryGuard struct{ *rail.BaseRail }
func (g *SessionInputBoundaryGuard) Priority() int { return 1000 }
func (g *SessionInputBoundaryGuard) OnUserMessage(ctx *rail.InputContext) error {
	admitted := make([]any, 0, len(ctx.Inputs.Parts))
	for _, part := range ctx.Inputs.Parts {
		if entry, ok := part.(*runtime.QueuedSessionInput); ok {
			<-entry.BoundaryReady // 等待边界结果
			if entry.BoundaryError != nil {
				continue
			}
		}
		admitted = append(admitted, part)
	}
	ctx.Inputs.Parts = admitted
	return nil
}
